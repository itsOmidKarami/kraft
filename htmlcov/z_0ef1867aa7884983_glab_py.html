32#: glab's pipeline vocabulary, from `glab ci list --help` (glab 1.116.0). 

33#: 'skipped' is deliberately not success: nothing proved the branch green, and 

34#: the next node is merge. Anything unrecognised falls through to 'failed' for 

35#: the same reason — guessing in the direction of merging is the one guess that 

36#: cannot be walked back. 'canceled' is a wait, not red: a cancelled pipeline 

37#: is never a verdict (Kraft-zn8me). `set_labels` re-creates the MR pipeline 

38#: and auto-cancel of redundant pipelines cancels the old one; a pipeline a

40def _bot_feedback(reviewer, notes: list[dict], discussions: list[dict], since) -> ReviewerFeedback: 

41 """One bot's part of the merge request's notes and discussions.""" 

42 bot = reviewer.bot 

43 

44 def mine(note: dict) -> bool: 

45 return mr_ops.same_login(str((note.get("author") or {}).get("username", "")), bot) 

46 

47 summaries: list[dict] = [] 

48 seen: list[datetime | None] = [] 

49 for n in notes: 

50 # An approval, or its withdrawal, is activity, dated by its system 

51 # note: GitLab's approvals list says who approved, not when. 

52 if n.get("system") and mine(n) and n.get("body") in _APPROVAL_NOTES: 

53 seen.append(mr_ops.forge_time(n.get("updated_at"))) 

54 continue 

55 # A system note ("added 1 commit") and an inline one (a discussion's, 

56 # read below) are not the bot's summary. 

57 if n.get("system") or n.get("position") or n.get("type") == "DiffNote" or not mine(n): 

58 continue 

59 at = mr_ops.forge_time(n.get("updated_at")) 

60 seen.append(at) 

61 if at is not None and at > since: 

62 summaries.append( 

63 { 

64 "kind": "comment", 

65 "state": None, 

66 "body": n.get("body") or "", 

67 "updated_at": n["updated_at"], 

68 } 

69 ) 

70 threads: list[dict] = [] 

71 for d in discussions: 

72 thread = d.get("notes") or [] 

73 seen += [mr_ops.forge_time(n.get("updated_at")) for n in thread if mine(n)] 

74 resolvable = [n for n in thread if n.get("resolvable")] 

75 if ( 

76 thread 

77 and mine(thread[0]) 

78 and resolvable 

79 and not all(n.get("resolved") for n in resolvable) 

80 ): 

81 first = thread[0] 

82 where = first.get("position") or {} 

83 threads.append( 

84 { 

85 "id": d["id"], 

86 "file": where.get("new_path") or where.get("old_path"), 

87 "line": where.get("new_line") or where.get("old_line"), 

88 "body": "\n".join( 

89 [ 

90 first.get("body") or "", 

91 *( 

92 f"{(n.get('author') or {}).get('username')}: {n.get('body')}" 

93 for n in thread[1:] 

94 ), 

95 ] 

96 ), 

97 } 

98 ) 

99 return ReviewerFeedback(reviewer, mr_ops.newest(seen), tuple(summaries), tuple(threads))

250 async def _merge_view(self, repo: Path, branch: str = "") -> tuple[str, dict]: 

251 """The merge request's own view of whether it can merge, and the 

252 `glab mr view` object it came from (`{}` when there is none to give: 

253 a payload that is not an object, or the `find_mr` fallback below). 

254 `ci_status` reads the object's `head_pipeline` and `iid` too, so the 

255 merge request is asked once per read. 

256 

257 `glab mr view <branch> -F json` is the call `open_mr` already makes. 

258 `detailed_merge_status` says *why*; `merge_status` is the older, 

259 coarser field, kept as a fallback for an older GitLab. Anything else 

260 — including a payload that is not an object — is "the forge did not 

261 say", which `mr.mergeable` reads as undecided. 

262 

263 A merge request already merged out-of-band -- a person merges it in 

264 the GitLab UI while this node is still polling -- reads `"state": 

265 "merged"` here, checked ahead of either merge-status field: a merged 

266 MR may not carry either (Kraft-v6ci). And if `glab mr view` itself 

267 raises -- the reported failure shape, the branch resolving to nothing 

268 when GitLab is asked for its current view -- fall back to `find_mr`, 

269 the same `--all` lookup `merge`'s own "already merged" shortcut 

270 already trusts (Kraft-xron), before giving up: this only reads as 

271 "merged" when `find_mr` confirms it, so a genuine outage or auth 

272 failure still raises rather than being read as "someone merged it". 

273 """ 

274 try: 

275 raw = await git.run_git( 

276 repo, ["glab", "mr", "view", *mr_ops.target(0, branch), "-F", "json"] 

277 ) 

278 data = mr_ops.parse_json(raw, "glab mr view") 

279 except ForgeError: 

280 ref = await self.find_mr(repo=repo, branch=branch) if branch else None 

281 if ref is not None and ref.state == "merged": 

282 return "merged", {} 

283 raise 

284 if not isinstance(data, dict): 

285 return "", {} 

286 if data.get("state") == "merged": 

287 return "merged", data 

288 return str(data.get("detailed_merge_status") or data.get("merge_status") or ""), data

290 async def ci_status( 

291 self, *, repo: Path, mr: MR, branch: str = "", pipeline_id: str = "" 

292 ) -> CIStatus: 

293 # The merge request's own state first: a conflict fails the check node 

294 # whatever colour the pipeline is, and no wait should sit out a 

295 # pipeline to learn it (Kraft-ejj9). Only meaningful pre-merge, on the 

296 # checked-out feature branch that actually has an open MR -- see 

297 # `branch_ci_status` for the post-merge, no-MR read. 

298 detail, view = await self._merge_view(repo, branch) 

299 # Merged out-of-band is a result, not a conflict and not a pipeline to 

300 # read: a deleted source branch's pipeline list is not a fact this 

301 # node needs once the MR has already landed (Kraft-v6ci). `_run_one`'s 

302 # `case "ci_poll"` already treats state == "success" as status == 

303 # "done" -- no handler-side change needed there. 

304 if detail == "merged": 

305 return CIStatus( 

306 state="success", 

307 url="", 

308 jobs=("merge request already merged",), 

309 mergeable=True, 

310 merge_detail="merged", 

311 ) 

312 mergeable = mr_ops.mergeable(detail) 

313 block_reason = mr_ops.classify_block_reason(detail) 

314 expected_sha = await git._head_sha(repo, branch) 

315 pipeline = await self._pipeline_on_ref(repo, branch, pipeline_id, expected_sha, view) 

316 # `no_checks=False`: before the merge a missing pipeline stays a plain 

317 # wait; only `branch_ci_status` reports it (Kraft-rbjo3). 

318 return replace( 

319 pipeline, 

320 mergeable=mergeable, 

321 merge_detail=detail, 

322 block_reason=block_reason, 

323 no_checks=False, 

324 )

343 async def _pipeline_on_ref( 

344 self, 

345 repo: Path, 

346 branch: str, 

347 pipeline_id: str, 

348 expected_sha: str, 

349 view: dict | None = None, 

350 ) -> CIStatus: 

351 """Read the newest pipeline for `expected_sha`, pinned or not -- 

352 shared by `ci_status` and `branch_ci_status`, which differ only in 

353 whether `mergeable`/`merge_detail` get filled in on top of what this 

354 returns. `view` is `ci_status`'s merge request (`_merge_view`): its 

355 `head_pipeline` is a candidate, and its `iid` lets a merged results 

356 pipeline count (`_answered_for`). `branch_ci_status` has none, so no 

357 merge request's pipeline ever answers for a target branch. 

358 """ 

359 view = view or {} 

360 iid = str(view.get("iid") or "") 

361 if pipeline_id: 

362 # Pinned to a pipeline the caller already resolved against the 

363 # current head (Kraft-ivh1): no re-resolving here, unlike 

364 # `glab ci list` below, which can answer with the previous 

365 # commit's pipeline for a few seconds after a push. 

366 try: 

367 raw = await git.run_git( 

368 repo, ["glab", "ci", "get", "--pipeline-id", pipeline_id, "-F", "json"] 

369 ) 

370 top = mr_ops.parse_json(raw, "glab ci get") 

371 except ForgeError: 

372 top = None 

373 # A cancelled pin has a successor to find, so it falls through too. 

374 # So does one this head no longer answers for: returning it would 

375 # have `ci_poll` pin it again on every poll. 

376 answered = None 

377 if isinstance(top, dict) and top and top.get("status") != "canceled": 

378 answered, _ = await self._answered_for(repo, top, expected_sha, iid) 

379 if isinstance(top, dict) and answered is not None: 

380 raw_state = str(top.get("status", "")) 

381 state: CIState = _GLAB_STATES.get(raw_state, "failed") 

382 url = str(top.get("web_url", "")) 

383 pipeline_ref = str(top.get("id", pipeline_id)) 

384 jobs: tuple[str, ...] = (f"pipeline {pipeline_ref}: {raw_state}",) 

385 failed_jobs: tuple[FailedJob, ...] = () 

386 if state == "failed": 

387 detail_lines, failed_jobs = await self._failure_detail(repo, pipeline_ref) 

388 jobs += detail_lines 

389 return CIStatus( 

390 state=state, 

391 url=url, 

392 jobs=jobs, 

393 sha=answered, 

394 failed_jobs=failed_jobs, 

395 pipeline_ref=pipeline_ref, 

396 ) 

397 # Pinned id unreadable (deleted, transient CLI error) -- fall 

398 # through to the ordinary "newest for this head" path below 

399 # rather than erroring the whole poll. 

400 # --ref, or this returns the newest pipeline in the whole project: a 

401 # green run on main would pass the gate for a red branch. 

402 ref = ["--ref", branch] if branch else [] 

403 raw = await git.run_git(repo, ["glab", "ci", "list", "-F", "json", "-P", "1", *ref]) 

404 rows = mr_ops.parse_json(raw, "glab ci list") 

405 row = rows[0] if rows and isinstance(rows[0], dict) else None 

406 # Two candidates, the merge request's head pipeline first so the list 

407 # row wins a tie. Both, because neither is enough alone: GitLab moves 

408 # `head_pipeline` in a background worker after a pipeline is created, 

409 # so right after `set_labels` or a description re-run starts one it can 

410 # still name the old red pipeline for this same head, which the list 

411 # (gitlab.com's `?ref=` includes merge request pipelines) already has 

412 # a newer one than; and a GitLab whose `?ref=` leaves merge request 

413 # pipelines out of the list has only the payload to offer. 

414 # shortcut: on such a GitLab, that lag can still hand back the old 

415 # pipeline for a few seconds after a re-create; `set_labels` could 

416 # return the id it created for the caller to pin instead. 

417 candidates: dict[str, dict] = {} 

418 head_pipeline = view.get("head_pipeline") 

419 if isinstance(head_pipeline, dict) and head_pipeline.get("id"): 

420 candidates[str(head_pipeline["id"])] = head_pipeline 

421 if row is not None: 

422 candidates[str(row.get("id", ""))] = row 

423 accepted: list[tuple[dict, str]] = [] 

424 refused: list[str] = [] 

425 for top in candidates.values(): 

426 answered, why = await self._answered_for(repo, top, expected_sha, iid) 

427 if answered is not None: 

428 accepted.append((top, answered)) 

429 elif why: 

430 refused.append(why) 

431 if accepted: 

432 # The newest pipeline for the head, which is GitLab's own rule for 

433 # a merge request's head pipeline. 

434 top, answered = max(accepted, key=lambda pair: _pipeline_number(pair[0])) 

435 return await self._read_pipeline(repo, top, answered) 

436 # No pipeline yet is not a green one. For a few seconds after a push 

437 # the list still answers with the *previous* commit's pipeline -- green, 

438 # for code the branch no longer has. ci_poll pushes now (Kraft-bxj8), so 

439 # that window is on the hot path, and a pipeline that is not for this 

440 # head is not a result: the same answer as no pipeline at all. 

441 # `expected_sha` is the caller's own reference commit -- `ci_status` 

442 # passes the item branch's local tip, `branch_ci_status` the upstream 

443 # head it fetched, never this checkout's possibly-stale local HEAD. 

444 jobs = ("no pipeline yet",) 

445 if candidates: 

446 jobs = (f"no pipeline for {expected_sha[:7]} yet", *refused) 

447 return CIStatus( 

448 state="pending", 

449 url="", 

450 jobs=jobs, 

451 sha=str(row.get("sha", "")) if row is not None else "", 

452 no_checks=True, 

453 )

455 async def _read_pipeline(self, repo: Path, top: dict, sha: str) -> CIStatus: 

456 """One pipeline that answers for `sha` (`_answered_for`), as a 

457 `CIStatus`: `sha` is the head it tested, never a merge commit's own.""" 

458 raw_state = str(top.get("status", "")) 

459 state: CIState = _GLAB_STATES.get(raw_state, "failed") 

460 # Never pin a cancelled pipeline: its successor is the one to read. 

461 pipeline_ref = "" if raw_state == "canceled" else str(top.get("id", "")) 

462 jobs: tuple[str, ...] = (f"pipeline {top.get('id')}: {raw_state}",) 

463 cancelled_at = "" 

464 if raw_state == "canceled": # when, for `render_ci` (Kraft-kbqmk) 

465 cancelled_at = str(top.get("updated_at") or "") 

466 failed_jobs: tuple[FailedJob, ...] = () 

467 if state == "failed": 

468 detail_lines, failed_jobs = await self._failure_detail(repo, pipeline_ref) 

469 jobs += detail_lines 

470 return CIStatus( 

471 state=state, 

472 url=str(top.get("web_url", "")), 

473 jobs=jobs, 

474 sha=sha, 

475 failed_jobs=failed_jobs, 

476 pipeline_ref=pipeline_ref, 

477 cancelled_at=cancelled_at, 

478 )

480 async def _answered_for( 

481 self, repo: Path, top: dict, expected: str, iid: str 

482 ) -> tuple[str | None, str]: 

483 """The head pipeline `top` tested, when that is `expected`, and why 

484 not otherwise. 

485 

486 A pipeline on the head itself (a branch pipeline, a detached merge 

487 request pipeline) answers for its own `sha`; with no `sha` or no 

488 `expected` there is nothing to compare, and it answers for its own as 

489 it always has. A merged results pipeline's `sha` is a merge commit, 

490 built on `_MERGE_REF` for this merge request's `iid` only: it answers 

491 for `expected` when `expected` is that commit's second parent, the 

492 head merged in. Every merged results pipeline of one merge request 

493 shares that ref, and `head_pipeline` can still name the previous 

494 head's for a moment after a push, so the ref alone never proves it. 

495 `(None, why)` is "not for this head", a wait; `why` is "" when it is 

496 plainly another commit's, and names the pipeline when a merge commit 

497 was asked about, so a wait that times out says what it saw. 

498 """ 

499 sha = str(top.get("sha") or "") 

500 if not sha or not expected or sha == expected: 

501 return sha, "" 

502 ref = str(top.get("ref") or "") 

503 if not iid or ref != _MERGE_REF.format(iid=iid): 

504 return None, "" 

505 # The merge ref lives in the pipeline's own project, which `:id` (the 

506 # checkout's remote) may not be. 

507 project = str(top.get("project_id") or ":id") 

508 name = f"pipeline {top.get('id')} on {ref}" 

509 try: 

510 raw = await git.run_git( 

511 repo, ["glab", "api", f"projects/{project}/repository/commits/{sha}"] 

512 ) 

513 commit = mr_ops.parse_json(raw, "glab api commit") 

514 except ForgeError as exc: 

515 return None, f"{name}: could not read merge commit {sha[:7]}: {exc}" 

516 parents = commit.get("parent_ids") if isinstance(commit, dict) else None 

517 # The second parent; a slice, so a commit with one parent (or none) 

518 # reads as having tested nothing rather than raising. 

519 tested = (parents[1:2] or [""])[0] if isinstance(parents, list) else "" 

520 if tested != expected: 

521 return None, f"{name} tested {str(tested)[:7] or 'nothing'}, not {expected[:7]}" 

522 return expected, ""

524 async def _failure_detail( 

525 self, repo: Path, pipeline_id: str 

526 ) -> tuple[tuple[str, ...], tuple[FailedJob, ...]]: 

527 """The failed jobs of a red pipeline, each job's own `failure_reason` 

528 (Kraft-ddxn), and the trace tail for each -- MR !89 was red because a 

529 job wanted a `release::` label, and that sentence existed only in the 

530 job's trace (Kraft-xh0q). A pipeline with no jobs at all reports the 

531 pipeline's own `yaml_errors` instead (Kraft-h81i, Kraft-s8ul, 

532 Kraft-ddxn): a config error stops anything from being created, so 

533 there is no job to ask. 

534 

535 Best effort throughout: a diagnosis that cannot be fetched must not 

536 turn a pipeline result that *was* read into a node failure. But 

537 "could not fetch the diagnosis" and "the forge confirmed there is no 

538 diagnosis to fetch" are not the same fact, and returning the same 

539 `((), ())` for both was a real bug caught on review of this plan: a 

540 genuine `script_failure` whose `glab ci get` call happens to fail (a 

541 transient CLI error, a malformed response) was indistinguishable from 

542 a pipeline the forge itself reports as having zero jobs, and 

543 `ci.is_infra_red` reads an empty `failed_jobs` tuple as infra-shaped 

544 either way -- so an ordinary code failure would self-retry twice and 

545 stop at `needs_human` with no fix loop ever run, the exact outcome 

546 this spec exists to remove. The three "could not read" branches below 

547 (`pipeline_id` missing, `ForgeError`, unparseable JSON) return 

548 `_UNREADABLE_JOBS` instead of `()`; only a *successful* read that 

549 confirms no job actually ran (empty `jobs`, no `yaml_errors`) returns 

550 the real empty tuple `is_infra_red` treats as infra. 

551 """ 

552 if not pipeline_id: 

553 return (), _UNREADABLE_JOBS 

554 try: 

555 raw = await git.run_git( 

556 repo, ["glab", "ci", "get", "--pipeline-id", pipeline_id, "-F", "json"] 

557 ) 

558 data = mr_ops.parse_json(raw, "glab ci get") 

559 except ForgeError: 

560 return (), _UNREADABLE_JOBS 

561 if not isinstance(data, dict): 

562 return (), _UNREADABLE_JOBS 

563 jobs_raw = [j for j in (data.get("jobs") or []) if isinstance(j, dict)] 

564 failed_raw = [j for j in jobs_raw if str(j.get("status", "")) == "failed"] 

565 if not failed_raw: 

566 yaml_errors = data.get("yaml_errors") 

567 if yaml_errors: 

568 return (f"pipeline config error: {yaml_errors}",), ( 

569 FailedJob("(pipeline)", "failed", "config_error"), 

570 ) 

571 return (), () 

572 detail: tuple[str, ...] = () 

573 failed_jobs: list[FailedJob] = [] 

574 # ponytail: first two failed jobs, last 15 trace lines each -- widen 

575 # if a real pipeline needs it (unchanged cap from before this task). 

576 for j in failed_raw[:2]: 

577 name = str(j.get("name", "")) 

578 reason = j.get("failure_reason") 

579 failed_jobs.append(FailedJob(name, "failed", str(reason) if reason else None)) 

580 detail += (f"job {name}: failed" + (f" ({reason})" if reason else ""),) 

581 try: 

582 trace = await git.run_git( 

583 repo, ["glab", "ci", "trace", name, "--pipeline-id", pipeline_id] 

584 ) 

585 except ForgeError: 

586 continue 

587 detail += tuple(f" {line}" for line in trace.strip().splitlines()[-15:]) 

588 # Any failed job past the first two still counts for classification, 

589 # even though its trace is not fetched. 

590 failed_jobs += [ 

591 FailedJob( 

592 str(j.get("name", "")), 

593 "failed", 

594 (str(j["failure_reason"]) if j.get("failure_reason") else None), 

595 ) 

596 for j in failed_raw[2:] 

597 ] 

598 return detail, tuple(failed_jobs)

669 async def _check_review(self, repo: Path, branch: str, check: str) -> ReviewResult:

739 async def set_labels(self, *, repo: Path, branch: str, mr: MR, labels: tuple[str, ...]) -> None: 

740 """Label the merge request, then start a pipeline that can see it. 

741 

742 `CI_MERGE_REQUEST_LABELS` is fixed when a pipeline is created, so a job 

743 that failed for a missing label re-reads the old, empty value on retry. 

744 Labelling alone therefore leaves the merge request looking fixed and 

745 still red -- the re-create is part of the capability, not the caller's 

746 homework (Kraft-xh0q). 

747 

748 GitLab's free tier does not enforce a scoped label's exclusivity 

749 server-side, so `--label` alone only adds: a second repair pass leaves 

750 both `release::minor` and `release::patch` on the MR, and 

751 `next_tag.py` now raises on more than one `release::` label rather 

752 than picking one (Kraft-zfdu8). So any current label sharing a new 

753 label's `scope::` prefix is dropped in the same call. 

754 """ 

755 if not labels: 

756 return 

757 target = mr_ops.target(mr.number, branch) 

758 # Also the read that used to be sentinel-only (`mr.number <= 0`, to 

759 # resolve the number for the pipeline re-create below): every caller 

760 # now needs the MR's current labels too, so one `mr view` serves both. 

761 raw = await git.run_git(repo, ["glab", "mr", "view", *target, "-F", "json"]) 

762 data = mr_ops.parse_json(raw, "glab mr view") 

763 number = mr.number if mr.number > 0 else int(data["iid"]) 

764 current = [str(label) for label in data.get("labels") or []] 

765 drop = mr_ops.same_scope_labels(current, labels) 

766 args = ["glab", "mr", "update", *target, "--label", ",".join(labels)] 

767 for label in drop: 

768 args += ["--unlabel", label] 

769 await git.run_git(repo, args) 

770 await self._new_mr_pipeline(repo, number)

790 async def merge(self, *, repo: Path, branch: str, mr: MR, head: str) -> None: 

791 """Ask GitLab to merge, at `head` only (`--sha`): GitLab answers 409 to 

792 a merge request whose source branch is anywhere else, now or when its 

793 pipeline succeeds. On a merge train project the merge request is 

794 added to the train. `glab mr merge` -- which on GitLab before 19.1 can 

795 merge round a train -- runs only when `projects/:id` says plainly there 

796 is none (`_train_enabled`); a read that failed raises, as a failed 

797 `glab mr merge` does.""" 

798 if not head: 

799 raise ForgeError("cannot merge without the head to merge") 

800 await git._assert_pushed(repo, branch) 

801 try: 

802 if await self._train_enabled(repo): 

803 await self._add_to_train(repo, branch, mr, head) 

804 return 

805 # number 0 is "not known": `run_task` does not thread the MR between 

806 # nodes, so the branch names it. Passing a literal 0 would target a 

807 # merge request that does not exist. 

808 target = mr_ops.target(mr.number, branch) 

809 await git.run_git(repo, ["glab", "mr", "merge", *target, "--yes", "--sha", head]) 

810 except ForgeError as exc: 

811 if _HEAD_MOVED in str(exc).lower(): 

812 raise HeadMoved(str(exc), await self._forge_head(repo, branch)) from exc 

813 raise

900 async def merge_queue(self, *, repo: Path, branch: str, base: str) -> QueueRead: 

901 """Where this branch's merge request stands on the project's merge 

902 train. `base` is unused: GitLab trains are per project, not per branch.""" 

903 if not await self._train_enabled(repo): 

904 return QueueRead() 

905 detail, view = await self._merge_view(repo, branch) 

906 if detail == "merged": 

907 return QueueRead(kind="train", state="merged", url=str(view.get("web_url") or "")) 

908 iid = view.get("iid") 

909 if not iid: 

910 return QueueRead(kind="train") 

911 try: 

912 car = await self._json( 

913 repo, ["api", f"projects/:id/merge_trains/merge_requests/{iid}"], "glab api car" 

914 ) 

915 except ForgeError as exc: 

916 status = _http_status(exc) 

917 if status == 403: 

918 return QueueRead() 

919 if status != 404: 

920 raise 

921 # GitLab destroys a car when it drops the merge request -- and when 

922 # it merges it: `locked` is a merge in progress, so a wait. 

923 if str(view.get("state")) == "locked": 

924 return QueueRead(kind="train", state="merging", url=str(view.get("web_url") or "")) 

925 if str(view.get("state")) != "opened": 

926 return QueueRead(kind="train", url=str(view.get("web_url") or "")) 

927 return await self._dropped(repo, view, str(iid)) 

928 if not isinstance(car, dict): 

929 return QueueRead(kind="train", url=str(view.get("web_url") or "")) 

930 state = _TRAIN_CAR_STATES.get(str(car.get("status")), "queued") 

931 pipeline = car.get("pipeline") 

932 url = str( 

933 (pipeline.get("web_url") if isinstance(pipeline, dict) else "") 

934 or view.get("web_url") 

935 or "" 

936 ) 

937 return QueueRead(kind="train", state=state, url=url)

939 async def _dropped(self, repo: Path, view: dict, iid: str) -> QueueRead: 

940 """A merge request GitLab took off the train: why, in GitLab's words, 

941 and the train pipeline that failed when there is one. The train 

942 pipeline stays the merge request's `head_pipeline`.""" 

943 failure = None 

944 head_pipeline = view.get("head_pipeline") 

945 if ( 

946 isinstance(head_pipeline, dict) 

947 and head_pipeline.get("ref") == _TRAIN_REF.format(iid=iid) 

948 and head_pipeline.get("sha") 

949 ): 

950 failure = await self._read_pipeline(repo, head_pipeline, str(head_pipeline["sha"])) 

951 note = await self._removal_note(repo, iid) 

952 cause: DropCause = "other" 

953 # No note is no reason yet: GitLab writes it a moment after the car 

954 # goes, and `run._merge_dropped` reads again next poll rather than 

955 # reading it as a person's removal. 

956 reason = "" 

957 if note.startswith(_ABORTED_NOTE): 

958 cause, reason = "not_added", note[len(_ABORTED_NOTE) :] 

959 elif note.startswith(_REMOVED_NOTE + " because "): 

960 reason = note[len(_REMOVED_NOTE + " because ") :] 

961 # GitLab's reason may only say "not mergeable"; the merge request 

962 # itself says whether that is a conflict. 

963 if ( 

964 "conflict" in reason.lower() 

965 or view.get("has_conflicts") is True 

966 or view.get("detailed_merge_status") == "conflict" 

967 ): 

968 cause = "conflict" 

969 elif failure is not None and failure.state == "failed": 

970 cause = "code" 

971 elif note.rstrip(".") in (_REMOVED_NOTE, _CANCELLED_ADD_NOTE): 

972 reason = note.rstrip(".") 

973 url = (failure.url if failure is not None else "") or str(view.get("web_url") or "") 

974 return QueueRead( 

975 kind="train", 

976 state="dropped", 

977 cause=cause, 

978 reason=reason, 

979 failure=failure, 

980 url=url, 

981 )

983 async def _removal_note(self, repo: Path, iid: str) -> str: 

984 """The newest system note that says the merge request left the train, 

985 else "". A note that cannot be read is no note: the drop itself was 

986 already established. Newest first, a note that it joined the train 

987 reached before any leaving note is this entry's: GitLab has not 

988 written why it left yet, and an older leaving note is an earlier 

989 drop's, so no note either.""" 

990 try: 

991 notes = await self._json( 

992 repo, 

993 [ 

994 "api", 

995 f"projects/:id/merge_requests/{iid}/notes" 

996 "?sort=desc&order_by=created_at&per_page=20", 

997 ], 

998 "glab api notes", 

999 ) 

1000 except ForgeError: 

1001 return "" 

1002 for note in notes if isinstance(notes, list) else []: 

1003 if not isinstance(note, dict) or not note.get("system"): 

1004 continue 

1005 body = str(note.get("body") or "").strip() 

1006 if body.startswith((_ABORTED_NOTE, _REMOVED_NOTE, _CANCELLED_ADD_NOTE)): 

1007 return body 

1008 if body.startswith(_ENQUEUE_NOTES): 

1009 return "" 

1010 return ""

1053def _http_status(exc: ForgeError) -> int | None: 

1054 """The 403 or 404 a failed `glab api` call reports, else None. 

1055 

1056 Only what the CLI said, after `run_git`'s `<argv> failed: `: the argv holds 

1057 the merge request's iid, which can contain "404" itself, and a timeout has 

1058 no detail at all. A 500, a network error or a timeout is not a status here. 

1059 """ 

1060 _, sep, detail = str(exc).partition(" failed: ") 

1061 if not sep: 

1062 return None 

1063 found = _HTTP_STATUS.search(detail) 

1064 return int(next(g for g in found.groups() if g)[:3]) if found else None
