103def _carry_severity( 

104 found: list[_findings.Finding], previous: list[_findings.Finding], *, unchanged_tree: bool 

105) -> list[_findings.Finding]: 

106 """`found` with each repeat's severity floored at its previous rating, but 

107 only when no fix cycle ran between the two measurements (Kraft-s7c04.3). 

108 

109 `policy.loop_severities` is a hard cliff: below it a finding is dropped from 

110 the loop entirely, so one inconsistent re-rating ends a loop that has not 

111 converged. 43717ee6 priced the same forge/run.py defect minor -> absent -> 

112 important across three reviews of a byte-identical tree, and the round that 

113 said `minor` is what let the loop believe it was converging. 

114 

115 `unchanged_tree` is the gate: this round's head is the head the previous 

116 measurement was taken at, so nothing the reviewer is looking at has moved 

117 and a different answer is the reviewer disagreeing with itself. Flooring 

118 unconditionally would turn a genuine partial fix -- `important` reduced to 

119 `minor` *because the fix worked* -- into `prints == previous_prints` and 

120 park the item at "stuck: 1 finding(s) unchanged", reporting real progress as 

121 being stuck. 

122 

123 Deliberately the head and not `fix_ran`: a fix cycle *starting* says nothing 

124 about whether it changed anything, and in a normal loop one always does, so 

125 `fix_ran` would disable this entirely. `dispatch_node` commits stragglers 

126 after every agent task (Kraft-7fip), so a fix that wrote anything has moved 

127 the head by the time the next measurement is dispatched. 

128 

129 An upgrade is always taken as given. A severity neither side's payload 

130 validated (`from_payload` accepts whatever it is handed) floors nothing 

131 rather than raising inside the loop. 

132 """ 

133 if not unchanged_tree or not previous: 

134 return found 

135 rank = {s: i for i, s in enumerate(_findings.SEVERITIES)} # critical = 0 

136 unknown = len(_findings.SEVERITIES) 

137 was = {f.fingerprint: f.severity for f in previous} 

138 return [ 

139 replace(f, severity=was[f.fingerprint]) 

140 if f.fingerprint in was 

141 and rank.get(was[f.fingerprint], unknown) < rank.get(f.severity, unknown) 

142 else f 

143 for f in found 

144 ]

166async def recover_node( 

167 db, 

168 run_dirs, 

169 work_item_id: str, 

170 node: dict, 

171 row, 

172 registry: Registry, 

173 worktree, 

174 *, 

175 failed: list[str], 

176 steer: Steer | None, 

177 launch: LaunchContext | None, 

178 budget: _policy.Budget, 

179 measured_round: int, 

180 round: int = 1, 

181 loop_severities: frozenset[str] = _policy.DEFAULT_LOOP_SEVERITIES, 

182) -> tuple[str, list[str], list[BaseException]]: 

183 """One repair pass over a node whose tasks failed (Kraft-rv6i). 

184 

185 Runs the node's `on_failure` tasks, then lets the node measure itself 

186 again, and reports that second measurement. The re-measure is the point: 

187 a node's contract is its own tasks passing, so a repair is believed only 

188 when they do -- `on.ci.poll` going green, not a remediator claiming it 

189 labelled something. 

190 

191 Once per entry into the node, not a loop. A repair that did not take is a 

192 blocker Kraft does not understand, and the honest move is to stop for a 

193 human rather than to keep pulling the same lever. 

194 

195 `round` keys the repair hook's dispatch and its re-measure in 

196 `sessions_for_round` (`dispatch.collect_findings`, 

197 `dispatch.needs_context_question`) -- it must not collide with any other 

198 round the *same node* writes sessions under. The no-fix-loop branch's call 

199 below needs nothing special: a node with `on_failure` and no `fix_loop` 

200 never runs `walk_node`'s paid-cycle loop at all, so `round=1` (the 

201 default) can never alias anything else for that node. The fix-loop 

202 branch, added in a later task, passes its own reserved round instead, 

203 because for a node with *both* `fix_loop` and `on_failure` (Task 6 allows 

204 this now; it used to be forbidden precisely to dodge this collision) 

205 `round=1` already means "the measurement after the first paid fix cycle" 

206 to that same node's `collect_findings`/`needs_context_question` calls. 

207 """ 

208 hooks = list(node["on_failure"]) 

209 await db.write( 

210 lambda c: events.append( 

211 c, 

212 work_item_id, 

213 "node_recovery_started", 

214 {"node_id": node["id"], "failed_tasks": failed, "tasks": hooks}, 

215 ) 

216 ) 

217 found, _reported = dispatch.collect_findings(db, work_item_id, node, measured_round, registry) 

218 note = _failure_note(node, failed) 

219 seeded = prompts.seeded_findings_note(found) if found else None 

220 context = f"{note}\n\n{seeded}" if seeded else note 

221 if steer is not None and steer: 

222 # A person's own instruction leads and keeps its own template; 

223 # `_SEEDED_FINDINGS_STEER`'s lead-in sentence ("Findings the last 

224 # review of this node left unresolved:") is what stops the bullets 

225 # below it reading as something that person wrote. `.take()` because 

226 # the incoming note is folded into the one replacing it -- leaving it 

227 # undelivered here would deliver it twice (Kraft-s7c04.58 covers the 

228 # two-hook case this single Steer cannot serve). The orchestrator's 

229 # own context (`note`, and `seeded` when there are findings) is 

230 # appended, never substituted for a human's steer. 

231 repair_steer = Steer(f"{steer.take()}\n\n{context}", source=steer.source) 

232 else: 

233 repair_steer = Steer(context, source="seeded") 

234 # `steps`, not just `tasks`, must name the repair hooks -- `measure_node` 

235 # reads `steps` first, and a stale `steps` here would re-run the node's 

236 # own (still-failing) tasks instead of the repair. 

237 verdict, r_failed, r_excs = await dispatch.measure_node( 

238 db, 

239 run_dirs, 

240 work_item_id, 

241 {**node, "tasks": hooks, "steps": [hooks]}, 

242 row, 

243 registry, 

244 worktree, 

245 round=round, 

246 steer=repair_steer, 

247 launch=launch, 

248 budget=budget, 

249 loop_severities=loop_severities, 

250 ) 

251 if verdict != "ok": 

252 return verdict, r_failed, r_excs 

253 return await dispatch.measure_node( 

254 db, 

255 run_dirs, 

256 work_item_id, 

257 node, 

258 row, 

259 registry, 

260 worktree, 

261 round=round, 

262 steer=steer, 

263 launch=launch, 

264 budget=budget, 

265 loop_severities=loop_severities, 

266 )

551 # should now see the failure for what it actually is.

553 # The repair's own re-measure just wrote its findings/session rows 

554 # under `_REPAIR_ROUND`, not the `round` this loop was on when it 

555 # entered this block (0, on the only entry this ever fires for) -- 

556 # the fall-through below reads `collect_findings`/ 

557 # `needs_context_question` at `round`, so `round` has to move to 

558 # match, or those calls read the stale pre-repair rows instead of 

559 # the fresher re-measure this repair just produced.

563 db, work_item_id, node["id"] 

564 ) 

565 head_sha = _config.git_read(Path(worktree), "rev-parse", "HEAD")

567 # the last measurement. `last_measurement` returns whole findings now 

568 # (the reviewer is handed their messages, and a repeat's severity is 

569 # floored against theirs), so the narrowing that used to live in the 

570 # event payload's `fingerprints` key happens here instead -- identically, 

571 # so `prints == previous_prints` and `repeats` below are unchanged. 

572 previous_prints = ( 

573 sorted({f.fingerprint for f in previous_found if f.severity in policy.loop_severities}) 

574 if previous_found is not None 

575 else None 

576 ) 

577 # `dispatch.previous_fix_session` is history-wide, not scoped to this 

578 # `walk_node` entry -- a `/retry` or an auto-escalation retry clears 

579 # the loop counter (`retry_counters` row deleted) but leaves the old 

580 # `worker_sessions` rows in place, so it alone can't tell "round 1 of 

581 # this entry" from "round 5 of the item's whole history". Round 1 of 

582 # *this* entry always fixes freely, no judge call (spec decision 2): 

583 # gated on the loop counter itself having no row yet (the same signal 

584 # `bump_counter` below uses to tell a fresh fire from a repeat one), 

585 # not on whether a fix session ever ran for this node before. A 

586 # carried-in steer gets the same free pass regardless of the counter: 

587 # it is the human's answer to exactly the trend the judge might stop 

588 # on, and a `stop_needs_human` here would discard it before the 

589 # steered cycle it was meant for ever dispatches, re-stranding the 

590 # item on the trend `/retry` was supposed to escape. 

591 previous_fix = dispatch.previous_fix_session(db, work_item_id, node["id"]) 

592 counter_row = db.read(lambda c: store.read_counter(c, work_item_id, key)) 

593 # Which steers get the free pass the comment above describes is 

594 # `Steer.exempts_judge`'s call, not a `human` test. A *seeded* steer 

595 # (Kraft's own recap of findings the last review already left, 

596 # Kraft-7sec second half) is not a person's deliberate answer to the 

597 # trend this judge exists to brake and must not stand in for one. A gate 

598 # reviewer's verdict is not a person's either, but it is the entire 

599 # content of a `fixed`/`reject` decision and exists nowhere else, so 

600 # discarding it before delivery is the data loss this comment warns 

601 # about rather than the misattribution (Kraft-s7c04.6). 

602 exempt = steer is not None and bool(steer) and steer.exempts_judge 

603 judge_due = previous_fix is not None and counter_row is not None and not exempt 

604 # `reported_hooks`, not `reported`: it is the set of hook points that 

605 # wrote a parseable result file, and `blind_failures` below is computed 

606 # from it. Named for what it holds so nothing else in this function can 

607 # quietly shadow it -- a draft of Kraft-s7c04.3 did, which turned every 

608 # failing hook into a blind failure and put the loop beyond the reach of 

609 # the stuck detector.

611 # A `same_as` is only believable for a tag this round's reviewer was 

612 # actually shown, and `carried_findings_note` is what showed it. An 

613 # invented or stale tag would collapse two distinct defects onto one 

614 # identity and fire the stuck detector on a fiction (Kraft-s7c04.2). 

615 found = _findings.resolve_identity( 

616 found, known={f.fingerprint for f in previous_found or []} 

617 ) 

618 # Parallel to `found`, never a dict keyed on fingerprint: two findings in 

619 # one round can share a tag if the reviewer pointed both at the same 

620 # `same_as`, and a dict would silently drop one of them. 

621 as_reported = [f.severity for f in found] 

622 found = _carry_severity( 

623 found, 

624 previous_found or [], 

625 # Both known and equal, never "both None": a pair of unreadable heads 

626 # is not evidence that nothing moved. 

627 unchanged_tree=head_sha is not None and head_sha == previous_head, 

628 )

632 # nothing, contributes no findings, and is therefore indistinguishable 

633 # in this event from a review that ran and found nothing (Kraft-yenu). 

634 # Naming them costs a dict lookup and is the only signal a reader gets 

635 # that a tier of review is not actually running.

637 t 

638 for t in node["tasks"] 

639 if registry.hooks.get(t, {}).get("kind") == "builtin" 

640 and registry.hooks.get(t, {}).get("handler") == "noop" 

641 ] 

642 # Built here rather than inside the lambda, the same way `judge_payload` 

643 # below is: every name a deferred lambda reads out of this loop has to be 

644 # bound at definition time or it sees the next iteration's value. 

645 # 

646 # `reported_severity` rides along only where the floor overrode the 

647 # reviewer (Kraft-s7c04.3). Refusing a downgrade silently would make a 

648 # reviewer inconsistency indistinguishable from a reviewer agreeing. 

649 # `findings.from_payload` reads key-by-key and ignores the extra key -- 

650 # exactly the case its docstring exists for.

655 # re-rating of an untouched tree from a real change. 

656 "head_sha": head_sha,

659 for f, s in zip(found, as_reported, strict=True)

661 "fingerprints": prints,

671 # findings list refines *why* a task failed, it does not define failure.

696 # failed in a worker-fixable way or reported a finding of its own 

697 # (Kraft-s7c04.24). Refusing it here, ahead of the budget check, the 

698 # judge and `bump_counter`, is the same posture as the `CONFIG_ERROR` 

699 # stop just below: a cycle that cannot be won must not spend an 

700 # attempt or buy a judge session to tell us what we already know. 

701 # 

702 # After `needs_context_question`, not before (ordering correction, 

703 # found reviewing the plan): a question a fix agent has already asked

705 # "reinstall the daemon" -- reachable on the shipped default chain

707 # later, by a re-measure where the in-process hook blind-fails alone.

709 t for t in blind_failures if registry.hooks.get(t, {}).get("kind") in _IN_PROCESS_KINDS 

710 } 

711 if ( 

712 in_process_blind 

713 and set(blind_failures) == in_process_blind

717 prompts.named_with_kind(t, registry) for t in sorted(in_process_blind)

720 f"{named} failed in node {node['id']}, and those tasks are executed by " 

721 "the running Kraft daemon — a worker commit cannot change them. " 

722 "Reinstall and restart, or skip the node." 

723 ) 

724 await db.write( 

725 lambda c, reason=reason: store.mark_needs_human(c, work_item_id, node["id"], reason) 

726 ) 

727 return "needs_human"

729 # Also ahead of the counter bump and the `fix_cycle_started` event: a 

730 # refused launch must not spend a fix-loop attempt or leave a cycle in 

731 # the log for an agent that never ran. The check inside `dispatch_node` 

732 # stays as the backstop for any future agent task reached by another 

733 # path. 

734 # 

735 # After needs_context, not before: a question a task actually asked is 

736 # the more useful thing to hand a human, and raising the cap would not 

737 # answer it. Either order prevents the phantom bump, which is the part 

738 # that matters; only the reason on the card differs.

740 return await stops.stop_for_budget(db, work_item_id, node, budget)

743 # top of the existing cap, never a second way to get stuck. Anything 

744 # the judge itself cannot be trusted on already fell open to 

745 # "continue" inside `dispatch.judge_verdict`; the cap/stuck checks 

746 # below run exactly as they do today regardless of what runs here.

770 return "paused"

788 )

790 return "needs_human"

794 # its eligible findings are downgraded, so gate on `failed` 

795 # itself rather than on the blind subset. Only a task with no

799 return "ok"

807 ) 

808 if _policy.check(count=count, started_at=started_at, cap=cap, now=_now()) == "breached":

812 c, work_item_id, node["id"], list(node["tasks"]) 

813 )

816 await db.write( 

817 lambda c, reason=reason, capped=capped: store.mark_needs_human( 

818 c, work_item_id, node["id"], reason, capped

820 )

825 # a single fingerprint recurring for `_FINGERPRINT_STREAK_LIMIT`

827 # changing shape and the set as a whole never repeats exactly.

835 _FINGERPRINT_STREAK_LIMIT,

837 )

846 # Deliberately NOT mark_sessions_capped_out: these sessions did 

847 # not cap out, and only a real cap breach may claim they did. 

848 await db.write( 

849 lambda c, reason=reason, bundle=bundle: store.mark_needs_human( 

850 c, work_item_id, node["id"], reason, bundle=bundle 

851 ) 

852 ) 

853 return "needs_human"

856 await db.write( 

857 lambda c, payload=payload: events.append(c, work_item_id, "fix_cycle_started", payload) 

858 )

860 prompts.FIX_PROMPT.format(node_id=node["id"], failed=", ".join(failed)) 

861 if failed 

862 else prompts.FIX_PROMPT_FINDINGS_ONLY.format(node_id=node["id"]) 

863 )

865 repeats = set(previous_prints or []) 

866 instruction += prompts.FIX_FINDINGS.format( 

867 findings=prompts.format_findings(eligible, repeats) 

868 )

870 # measurement" is true either way -- but the note claims an earlier 

871 # attempt was made, which is false on the crash/resume path where 

872 # the measurement was recorded and the process died before any 

873 # fix_cycle_started. 

874 if fix_ran and repeats & set(prints): 

875 instruction += prompts.FIX_REPEAT_NOTE 

876 # Only a `continue` reaches here at all -- both stop verdicts return 

877 # above -- but `verdict` is reassigned by the judge and shadows the 

878 # measure verdict from the top of the loop, so this is gated on the 

879 # judge having actually run rather than on the name's current value. 

880 if judge_ran: 

881 instruction += prompts.judge_note(reasoning) 

882 # Every round, not just the last one (Kraft-s7c04.7). `last_measurement` 

883 # gives the fixer one round of memory, which is why round N+2 reverted 

884 # the security property round N established on e983d85c -- the loop had 

885 # the information and could not see two rounds back. Read here rather 

886 # than hoisted: this is the uncommon path (a fix is about to be 

887 # dispatched), and the stuck check above is deliberately lazy about the 

888 # same history so the common path pays nothing. 

889 history = dispatch.judge_history(db, work_item_id, node["id"], policy.loop_severities) 

890 instruction += prompts.round_history_note(history, dispatch.regressed_fingerprints(history)) 

891 instruction += prompts.previous_attempt_note(previous_fix) 

892 fix = await dispatch.dispatch_node( 

893 db, 

894 run_dirs, 

895 "on.implementation.start", 

896 node, 

897 row, 

898 registry, 

899 worktree, 

900 instruction_override=instruction, 

901 round=count, 

902 steer=steer, 

903 launch=launch,

906 # above returns first, so a loop that is stuck never buys a more 

907 # expensive model for the same blind approach (spec 6). 

908 escalate=cap.escalate_after is not None and count > cap.escalate_after, 

909 ) 

910 if fix == "paused": 

911 return "paused"

913 return await stops.stop_for_budget(db, work_item_id, node, budget)

920#: (Kraft-s7c04.23). Distinct from `_REPAIR_ROUND` because one node can 

921#: reach both -- `mr_checks` declares `on_failure` (which writes at

943 """Dispatch `on.implementation.start` to resolve a rebase conflict and

945 supposed to do (design §4.1 -- the second job is the point, not a 

946 formality). 

947

964 reason = f"{key} exhausted after {count - 1} attempt(s): {conflict}" 

965 await db.write( 

966 lambda c, reason=reason: store.mark_needs_human(c, work_item_id, node["id"], reason) 

967 )

969

974 worktree,

977 new_base or "(unknown)", 

978 conflict, 

979 entry.attachments_of(row),

982 db, 

983 run_dirs, 

984 "on.implementation.start", 

985 node, 

986 row, 

987 registry, 

988 worktree, 

989 instruction_override=instruction, 

990 round=_RESOLVE_ROUND, 

991 launch=launch, 

992 ) 

993 if status in ("paused", BUDGET): 

994 return status, None, None 

995 session = db.read( 

996 lambda c: c.execute( 

997 "SELECT result_path FROM worker_sessions WHERE work_item_id = ? AND node_id = ? " 

998 "AND hook_point = 'on.implementation.start' AND round = ? ORDER BY created_at DESC " 

999 "LIMIT 1", 

1000 (work_item_id, node["id"], _RESOLVE_ROUND), 

1001 ).fetchone() 

1002 )

1010 _config.git_read(

1012 "merge-base",

1016 expected_failure=True, 

1017 ) 

1018 is not None 

1019 ) 

1020 if not moved: 

1021 reason = (

1023 f"the rebase: {conflict}" 

1024 ) 

1025 await db.write( 

1026 lambda c, reason=reason: store.mark_needs_human(c, work_item_id, node["id"], reason) 

1027 ) 

1028 return "needs_human", None, None

1044 reason = f"rebase-conflict resolver could not resolve the conflict: {conflict}" 

1045 await db.write( 

1046 lambda c, reason=reason: store.mark_needs_human(c, work_item_id, node["id"], reason) 

1047 ) 

1048 return "needs_human", None, None

1051async def bounce(db, work_item_id: str, node: dict, target: str, policy) -> str: 

1052 """Bump the `rebase_bounce` cap and decide whether another bounce to 

1053 `target` is still allowed (Kraft-4bgg). Same `Cap`/`retry_counters` 

1054 machinery a fix_loop's own cap check uses in `walk_node`, just keyed on 

1055 the whole item's rebase bounces rather than one node's fix cycles. 

1056 """ 

1057 if policy is None: 

1058 raise RuntimeError(f"node {node['id']!r} has rebase_bounce_to but no policy was provided") 

1059 key = "rebase_bounce" 

1060 cap = _policy.resolve_cap(policy, key) 

1061 count, started_at, cap = await db.write(lambda c: store.bump_counter(c, work_item_id, key, cap)) 

1062 if _policy.check(count=count, started_at=started_at, cap=cap, now=_now()) == "breached": 

1063 reason = f"{key} exhausted after {count - 1} bounce(s) back to {target!r}" 

1064 capped = {"cycles": count - 1, "attempts": cap.attempts} 

1065 await db.write( 

1066 lambda c: store.mark_needs_human(c, work_item_id, node["id"], reason, capped) 

1067 ) 

1068 return "needs_human" 

1069 return "ok"

1072async def _report_if_undelivered(db, work_item_id: str, carried: Steer) -> None: 

1073 """Kraft-s7c04.50: `carried` is good for one agent launch, whichever 

1074 dispatch gets there first (Steer's own docstring) -- but a walk that 

1075 stops, or a rebase-bounce that replaces `carried` with its own seeded 

1076 note, before any dispatch ever calls `.take()` loses the original text 

1077 with nothing to show for it. The steer mechanism itself works (verified 

1078 live against acf59aafa6bb4512bd68049705414fc7's actual session logs); 

1079 this only makes the *failure to deliver* case visible instead of silent. 

1080 `.take()` both reads the leftover text and empties it, defensive against 

1081 a future call path invoking this twice on the same `carried`.""" 

1082 if carried: 

1083 await db.write( 

1084 lambda c: events.append(c, work_item_id, "steer_undelivered", {"steer": carried.take()}) 

1085 )

1088async def run_once( 

1089 db, 

1090 run_dirs, 

1091 *, 

1092 work_item_id: str, 

1093 registry: Registry, 

1094 bd_cwd: str | None = None, 

1095 start_index: int = 0, 

1096 policy: _policy.Policy | None = None, 

1097 steer: str | None = None, 

1098 steer_source: str = "human", 

1099 launch: LaunchContext | None = None, 

1100) -> str: 

1101 # the note is good for one agent launch, whichever task gets there first 

1102 carried = Steer(steer, source=steer_source) 

1103 row = db.read( 

1104 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (work_item_id,)).fetchone() 

1105 ) 

1106 if row is None: 

1107 raise LookupError(f"unknown work_item {work_item_id!r}") 

1108 chain = json.loads(row["chain_definition"]) 

1109 nodes = chain["nodes"] 

1110 # Record the chain as loaded before touching the filesystem: this is 

1111 # bookkeeping about the item, not about the worktree, and a work item 

1112 # whose worktree can never be created (bad repo, git failure) must still 

1113 # end up with a `chain_loaded` event and a non-NULL `current_node_id` -- 

1114 # otherwise it looks indistinguishable from a crash between 

1115 # create_work_item and the first load_chain (see the `cur is None` 

1116 # branch in `kraft.executor.resuming.resume`), and the WS gets fewer 

1117 # events than a client waiting on this chain to progress at all is 

1118 # entitled to expect. 

1119 if start_index == 0: 

1120 await db.write(lambda c: store.load_chain(c, work_item_id, nodes[0]["id"])) 

1121 

1122 # Kraft-tsfpk: a bd dependency the tracker already knows about (`bd dep 

1123 # add ... --type blocks`) must stop this walk before it spends a paid 

1124 # agent session re-discovering the blocker from prose. Checked once per 

1125 # dispatch attempt -- intake, resume, retry, and every poller re-entry all 

1126 # pass through here -- not once per node inside the loop below: a merge 

1127 # landing mid-walk is the rare case, and a `bd blocked` call on every node 

1128 # buys correctness nobody has hit yet at the cost of one more subprocess 

1129 # launch per node, every walk, forever. 

1130 # 

1131 # `row["bead_id"]` alone is a near-no-op for the case that motivated this 

1132 # bead (plan-review finding 1): `entry.intake` files a *fresh* tracking 

1133 # bead for every manually created item, and a brand-new bead has no `bd 

1134 # dep` edges of its own -- `bead_id`'s own `blocked_by` is always `[]`. 

1135 # The real edges live on the source beads named in the description, 

1136 # extracted at intake into `implements_beads` (`entry._extract_beads`). 

1137 # Both go into the one `bd blocked --json` call. 

1138 bead_ids = [i for i in (row["bead_id"], *entry._implements_beads(row)) if i] 

1139 if bead_ids: 

1140 # `row["bead_cwd"] or bd_cwd`, the same precedence 

1141 # `entry.close_beads` (`entry.py:221`) already uses -- not `bd_cwd or 

1142 # row["repo"]`. With `KRAFT_BD_CWD` set instance-wide, a bead filed 

1143 # into a per-repo workspace must still be looked up there, or it 

1144 # silently never reads as blocked (plan-review finding 4). 

1145 blocked_by = await beads.blocked_by(bead_ids, cwd=row["bead_cwd"] or bd_cwd) 

1146 # Blockers that are themselves part of this item's own bead set 

1147 # (e.g. two bundled beads with a `blocks` edge between them) aren't 

1148 # external dependencies -- this walk is how they get implemented. 

1149 blocked_by = [b for b in blocked_by if b not in set(bead_ids)] 

1150 if blocked_by: 

1151 await db.write( 

1152 lambda c: store.mark_blocked_by_dependency( 

1153 c, work_item_id, nodes[start_index]["id"], blocked_by 

1154 ) 

1155 ) 

1156 return "paused" 

1157 

1158 # Before the first dispatch, not inside the `env_setup` node: `default.yaml` 

1159 # runs `spec` and `plan` first, and both need a checkout — and, for `plan`'s 

1160 # attached-spec fallback to have anything to find, attachments already 

1161 # copied in — to write into. 

1162 # 

1163 # A git failure here (bad repo, no permission, ...) is attributed to the 

1164 # node that was about to dispatch rather than left to `kraft.api.deps.guard`'s bare 

1165 # crash handler: this call moved out of `env_setup`'s own session so spec 

1166 # and plan can use the checkout too, but the observable shape of "a node 

1167 # failed to start" -- node_started, then needs_human naming it -- should 

1168 # not disappear just because the failure now happens a moment earlier. 

1169 try: 

1170 worktree = await _builtins.ensure_worktree( 

1171 db, 

1172 run_dirs, 

1173 repo=row["repo"], 

1174 work_item_id=work_item_id, 

1175 attachments=entry.attachments_of(row), 

1176 repo_entry=launch.repo_entry if launch else None, 

1177 ) 

1178 except (RuntimeError, _config.ConfigError) as exc: 

1179 failing_node = nodes[start_index]["id"] 

1180 reason = str(exc) 

1181 await db.write(lambda c: store.enter_node(c, work_item_id, failing_node)) 

1182 await db.write(lambda c: store.mark_needs_human(c, work_item_id, failing_node, reason)) 

1183 return "needs_human" 

1184 

1185 i = start_index 

1186 while i < len(nodes): 

1187 # Kraft-e7pm: a session verdict of "paused" is not the only way a walk 

1188 # has to stop. Pausing between two nodes' dispatches leaves no live 

1189 # session to signal at all -- the SIGTERM path in `pause_work_item` 

1190 # never fires -- so without this read the loop would march straight 

1191 # into the next node reading a row that already says paused. One read 

1192 # per node; every non-active status (paused, needs_human by any other 

1193 # door, abandoned) gets the same "paused" verdict every caller of 

1194 # `run_once`/`run` already knows how to handle -- there is nothing a 

1195 # second string would let a caller do that this doesn't. 

1196 if gates.status_of(db, work_item_id) != "active": 

1197 return "paused" 

1198 node = nodes[i] 

1199 bounce_to = node.get("rebase_bounce_to") 

1200 pre_base = _current_base_ref(db, work_item_id) if bounce_to else None 

1201 result = await walk_node( 

1202 db, 

1203 run_dirs, 

1204 work_item_id, 

1205 node, 

1206 row, 

1207 registry, 

1208 worktree, 

1209 policy=policy, 

1210 # `carried` empties itself on the first agent launch, so the note reaches 

1211 # the next agent to run and no later one 

1212 steer=carried, 

1213 launch=launch, 

1214 ) 

1215 if result in ("paused", "needs_human", RATE_LIMITED, WAITING): 

1216 # This call is ending without giving `node` -- or any later node 

1217 # in this same run_once, since none of these statuses continue 

1218 # the loop -- a chance to consume `carried` beyond what it just 

1219 # got. Report it here, once, rather than duplicating this check 

1220 # at each of the four returns below (Kraft-s7c04.50). 

1221 await _report_if_undelivered(db, work_item_id, carried) 

1222 if result == "paused": 

1223 return "paused" 

1224 if result == "needs_human": 

1225 return "needs_human" 

1226 if result == RATE_LIMITED: 

1227 return RATE_LIMITED 

1228 if result == WAITING: 

1229 return WAITING 

1230 if bounce_to: 

1231 new_base = _current_base_ref(db, work_item_id) 

1232 if new_base != pre_base: 

1233 bounced = await bounce(db, work_item_id, node, bounce_to, policy) 

1234 if bounced == "needs_human": 

1235 return "needs_human" 

1236 target = next(j for j, n in enumerate(nodes) if n["id"] == bounce_to) 

1237 # A bounce is a fresh pass over every node in the span it 

1238 # re-enters, not just the target -- an intervening node's own 

1239 # fix-loop counter (e.g. `mr_checks`' `ci_fix_loop` on a bounce 

1240 # from `merge`) is exactly as stale as the target's 

1241 # (Kraft-s7c04.25 spec §2). Inclusive of the bouncing node 

1242 # itself: bouncing from `merge` re-runs `merge` too. 

1243 cleared = [n["id"] for n in nodes[target : i + 1]] 

1244 for n in nodes[target : i + 1]: 

1245 await db.write( 

1246 lambda c, n=n: store.clear_loop_counters( 

1247 c, work_item_id, n["id"], n.get("fix_loop") 

1248 ) 

1249 ) 

1250 await db.write( 

1251 lambda c, cleared=cleared, bounce_to=bounce_to: events.append( 

1252 c, 

1253 work_item_id, 

1254 "loop_counters_reset", 

1255 {"nodes": cleared, "reason": "rebase_bounce", "bounce_to": bounce_to}, 

1256 ) 

1257 ) 

1258 # `carried` is about to be replaced by Kraft's own rebase note 

1259 # -- if the original steer (human or seeded) survived this 

1260 # far untaken, report it before it's overwritten, or it is 

1261 # lost with no trace (Kraft-s7c04.50, found in spec review). 

1262 await _report_if_undelivered(db, work_item_id, carried) 

1263 carried = Steer( 

1264 prompts.rebase_drift_note(worktree, store.branch_for(row), pre_base, new_base), 

1265 # Kraft wrote this one, not a person (the `_REBASE_PROMPT` 

1266 # template already says so); it must not claim the judge 

1267 # exemption a human's own answer gets. 

1268 source="seeded", 

1269 ) 

1270 i = target 

1271 continue 

1272 if await gates.maybe_gate(db, work_item_id, node): 

1273 await _report_if_undelivered(db, work_item_id, carried) 

1274 return "awaiting_gate" 

1275 i += 1 

1276 

1277 # The whole chain ran to completion without any node's dispatch ever 

1278 # taking `carried` -- e.g. every remaining node was builtin/subprocess 

1279 # only. Report it the same as every other exit (Kraft-s7c04.50). 

1280 await _report_if_undelivered(db, work_item_id, carried) 

1281 await db.write(lambda c: store.mark_completed(c, work_item_id)) 

1282 await entry.close_beads(db, row, bd_cwd) 

1283 return "completed"
