13#: A unix socket path must fit `sockaddr_un.sun_path`: 104 bytes on macOS, 108 on Linux. 

14_SOCKET_PATH_MAX = 100

17def short_socket_path(path: Path) -> Path: 

18 """`path`, or where a path that is too long for a unix socket is kept instead: a 

19 private directory under /tmp named by a hash of the path, holding a file of the same name.""" 

20 if len(os.fsencode(path)) < _SOCKET_PATH_MAX: 

21 return path 

22 digest = hashlib.sha1(os.fsencode(path)).hexdigest()[:12] 

23 return Path("/tmp") / f"kraft-{os.getuid()}-{digest}" / path.name

26def private_dir(path: Path) -> None: 

27 """`path`, made if missing, as a directory only this user can enter.""" 

28 path.mkdir(mode=0o700, parents=True, exist_ok=True) 

29 info = path.stat() 

30 if info.st_uid != os.getuid(): 

31 raise PermissionError(f"{path} belongs to another user") 

32 if stat.S_IMODE(info.st_mode) != 0o700: 

33 path.chmod(0o700)

36def take_lock(lock_file: Path) -> int | None: 

37 """The exclusive lock on `lock_file`, as an open fd (closing it releases the lock), or 

38 None when another process holds it. The holder alone binds and removes the socket.""" 

39 fd = os.open(lock_file, os.O_RDWR | os.O_CREAT, 0o600) 

40 try: 

41 fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB) 

42 except OSError: 

43 os.close(fd) 

44 return None 

45 return fd
