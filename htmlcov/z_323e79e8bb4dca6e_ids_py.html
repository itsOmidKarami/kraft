1"""The ids Kraft mints: a work item's 8 characters, a session's 6, both from 

2lowercase Crockford base32 (no i, l, o or u, which read as 1, 1, 0 and v). 

3 

4Drawn with `secrets`, not `random`: processes forked from one parent share 

5`random`'s state and would draw the same ids. 

6 

7A session is unique only within its work item, so outside it a session has 

8two spellings: `session_ref` (`<wid>/<sid>`) for what a person or a header 

9reads, and `qualified` for a name that cannot hold a `/`.""" 

10 

11from __future__ import annotations 

12 

13import secrets 

14from collections.abc import Callable 

15 

16ALPHABET = "0123456789abcdefghjkmnpqrstvwxyz" 

17WORK_ITEM_LENGTH = 8 

18SESSION_LENGTH = 6 

19#: Every id minted before these: `uuid.uuid4().hex`. 

20LEGACY_LENGTH = 32 

21 

22 

23class Exhausted(RuntimeError): 

24 """Every candidate was taken: not bad luck at these lengths, a broken 

25 `taken` or generator, so said loudly rather than looped on.""" 

26 

27 

28def mint(length: int, taken: Callable[[str], bool], *, tries: int = 10) -> str: 

29 """The first of up to `tries` drawn candidates that `taken` does not 

30 reject. `taken` is asked when the id is drawn; the row that holds it is 

31 written later, and its primary key refuses a draw that raced another.""" 

32 for _ in range(tries): 

33 candidate = "".join(secrets.choice(ALPHABET) for _ in range(length)) 

34 if not taken(candidate): 

35 return candidate 

36 raise Exhausted(f"{tries} candidate ids in a row were taken: refusing to guess further") 

37 

38 

39def session_ref(work_item_id: str, session_id: str) -> str: 

40 """A session named outside its item: in `KRAFT_SESSION_ID`, the 

41 `X-Kraft-Session-Id` header, messages and logs. Never stored, and never 

42 one URL path segment: split it first.""" 

43 return f"{work_item_id}/{session_id}" 

44 

45 

46def parse_session_ref(value: str) -> tuple[str | None, str]: 

47 """`(work item id, session id)`. A value with no `/` is a session from 

48 before per-item ids, so its item is None; it is accepted only at that 

49 session's length, `LEGACY_LENGTH`, which is what makes it unique on its 

50 own. A bare per-item id names a session of whichever items have one. 

51 

52 `ValueError` for anything else: an empty value, an empty half, a second 

53 `/`, or a bare value of any other length. Such a value names no session, 

54 and a caller refuses it rather than guess which one it meant.""" 

55 wid, sep, sid = value.partition("/") 

56 if sep: 

57 if not wid or not sid or "/" in sid: 

58 raise ValueError(f"not a session reference: {value!r}") 

59 return wid, sid 

60 if len(value) != LEGACY_LENGTH: 

61 raise ValueError( 

62 f"not a session reference: {value!r} (a bare id is a session's from before " 

63 f"per-item ids, {LEGACY_LENGTH} characters; any other is <work item id>/<session id>)" 

64 ) 

65 return None, value

68def qualified(work_item_id: str, session_id: str) -> str: 

69 """What a machine-wide name for a session is built from: a container, a 

70 volume, a certificate's CN and folder, a socket folder, an in-memory key. 

71 A session minted before per-item ids keeps its bare id, so a session that 

72 ran across the upgrade is found by the names it started with.""" 

73 if len(session_id) == LEGACY_LENGTH: 

74 return session_id 

75 return f"{work_item_id}-{session_id}"
