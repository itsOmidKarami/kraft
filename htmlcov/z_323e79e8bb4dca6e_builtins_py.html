28def _commit_paths(worktree: Path, paths: list[str], message: str) -> None: 

29 """Commit exactly `paths` in `worktree`, or commit nothing. 

30 

31 Kraft's own commit primitive. A document Kraft put in the worktree is not a 

32 change any agent made, so no agent is told to commit it, and it sits 

33 untracked until `forge._assert_clean` refuses to open the merge request over 

34 it (Kraft-8iw6). Kraft-xwen needs the same primitive so a document node can 

35 be given an allowlist with no Bash, which is why the pathspec is explicit 

36 and an unrelated dirty file is left exactly as it was. 

37 

38 `--no-verify`: this runs before `ensure_worktree`'s `uv sync`, so the repo's 

39 `pre-commit` hook has no environment to spawn from yet (Kraft-i047 is the 

40 same failure seen from the other side), and the content is a file Kraft 

41 copied unmodified — there is nothing here for the hook to catch. No `-c 

42 user.email` override either: `ensure_worktree` pins identity into the 

43 worktree (and its submodules) via `_pin_identity` at creation (Kraft-cppp), 

44 which is where every other commit on this branch gets its author. 

45 

46 Best effort. A failure logs a warning and returns: a document that did not 

47 commit becomes a dirty-tree failure at `open_mr` with git's own message 

48 already in the log, which is strictly better than failing worktree creation. 

49 

50 `main_ignore_args`, the same override `forge._work_product_pathspec` 

51 relies on, rides along on the `add`: a spec/plan attachment copied to its 

52 original relative path can land under `docs/superpowers/` or 

53 `.engineering/`, and if `main` ignores that root but this worktree's own 

54 checked-out `.gitignore` predates the rule (Kraft-vu26), a plain `git add` 

55 stages it anyway. Staging nothing here is correct, not a failure -- the 

56 diff-cached check below already treats "nothing staged" as a no-op, and 

57 the attachment stays on disk for the agent to read either way. 

58 """ 

59 if not paths: 

60 return 

61 

62 with main_ignore_args(worktree) as ignore_args: 

63 

64 def git(*args: str) -> subprocess.CompletedProcess: 

65 return subprocess.run( 

66 ["git", *ignore_args, *args], cwd=str(worktree), capture_output=True, text=True 

67 ) 

68 

69 added = git("add", "--", *paths) 

70 if added.returncode != 0: 

71 logger.warning("could not stage %s in %s: %s", paths, worktree, added.stderr.strip()) 

72 return 

73 # Exit 0 means no staged difference for these paths — an ignored or an 

74 # unchanged path stages nothing, and `git commit` on an empty commit exits 

75 # non-zero. Skip it rather than log a failure that is really a no-op. 

76 if git("diff", "--cached", "--quiet", "--", *paths).returncode == 0: 

77 return 

78 done = git("commit", "--no-verify", "-m", message, "--", *paths) 

79 if done.returncode != 0: 

80 logger.warning( 

81 "could not commit %s in %s: %s", 

82 paths, 

83 worktree, 

84 done.stderr.strip() or done.stdout.strip(), 

85 )

88def restore_branch(worktree: Path, branch: str) -> None: 

89 """Force the worktree back onto `branch` if an agent task left it 

90 somewhere else, aborting any in-progress merge first (Kraft-v5qd). 

91 

92 An agent has a real shell and can check out whatever it likes to 

93 investigate something -- a diagnostic test-merge against `main` to look 

94 at a conflict, say. Nothing enforces that it checks back out afterward, 

95 and a merge left mid-conflict when the agent's turn just ends (budget, 

96 a crash, the one-pass skills that only run once) strands the worktree: 

97 every task after it resolves the merge request from *this* branch, and a 

98 diagnostic branch has none. `commit_stragglers` runs right after this in 

99 the agent-kind dispatch, precisely so any straggler ends up on the 

100 branch the item actually owns rather than committed onto whatever the 

101 agent happened to leave checked out. 

102 

103 Best effort throughout, like `commit_stragglers`: a `restore_branch` 

104 that could not recover is a worktree already too broken for a log line 

105 to fix, and the failure it hides here surfaces the same way it always 

106 did -- the next node's own git command refuses on the same tree. 

107 """ 

108 with main_ignore_args(worktree) as ignore_args: 

109 

110 def git(*args: str) -> subprocess.CompletedProcess: 

111 return subprocess.run( 

112 ["git", *ignore_args, *args], cwd=str(worktree), capture_output=True, text=True 

113 ) 

114 

115 current = git("rev-parse", "--abbrev-ref", "HEAD") 

116 if current.returncode != 0 or current.stdout.strip() == branch: 

117 return 

118 logger.warning( 

119 "worktree %s left on %r instead of %r after an agent task; restoring", 

120 worktree, 

121 current.stdout.strip(), 

122 branch, 

123 ) 

124 git("merge", "--abort") # no-op, exit nonzero, if no merge is in progress 

125 checked_out = git("checkout", "-f", branch) 

126 if checked_out.returncode != 0: 

127 logger.warning( 

128 "could not restore %s to %r: %s", worktree, branch, checked_out.stderr.strip() 

129 )

132def _copy_attachments( 

133 repo: Path, worktree: Path, attachments: list[dict], work_item_id: str 

134) -> None: 

135 """Intake attachments that are not committed do not exist in a fresh 

136 worktree (`git worktree add` branches from HEAD), so copy them in. A 

137 committed one arrived through git and is left exactly as git wrote it. 

138 

139 The path was validated against the *repo's* working tree, not this 

140 worktree (checked out from HEAD, possibly a different tree). If HEAD has 

141 a symlink here, `dest.exists()` follows it, which for a dangling symlink 

142 reports False and `shutil.copyfile` would then write through it to 

143 wherever it points. So a symlinked destination, dangling or not, is 

144 always skipped, and the resolved destination must stay inside the 

145 worktree. 

146 

147 `source`, when the validator set it, is the absolute path the document was 

148 actually found at — a working tree of the repo that is not `repo` itself 

149 (Kraft-85wk). Without it the copy would look under `repo`, find nothing, 

150 and skip. 

151 

152 Whatever actually gets copied is committed once, here, so an uncommitted 

153 attachment does not sit untracked and trip `open_mr`'s dirty-worktree guard 

154 (Kraft-8iw6) — no agent is told to commit a file it never wrote.""" 

155 worktree_root = worktree.resolve() 

156 written: list[dict] = [] 

157 for attachment in attachments: 

158 dest = worktree / attachment["path"] 

159 src = Path(attachment["source"]) if attachment.get("source") else repo / attachment["path"] 

160 if dest.is_symlink(): 

161 continue 

162 resolved = dest.resolve() 

163 if resolved != worktree_root and worktree_root not in resolved.parents: 

164 continue 

165 if dest.exists(): 

166 continue 

167 if not src.is_file(): 

168 # Not a skip. `source` is Kraft's own copy since Kraft-eqgn, so a 

169 # file missing here means Kraft lost it — and the gate this 

170 # document justified was trimmed at intake and cannot be put back. 

171 # Continuing would run the item without a document it promised, and 

172 # without a gate to notice. Should be unreachable; loud if not. 

173 raise FileNotFoundError( 

174 f"the {attachment['kind']} attachment for {work_item_id} is missing " 

175 f"from Kraft's storage at {src}; this work item's " 

176 f"{attachment['kind']} gate was trimmed at intake and cannot be " 

177 "restored — re-file the work item" 

178 ) 

179 dest.parent.mkdir(parents=True, exist_ok=True) 

180 shutil.copyfile(src, dest) 

181 written.append(attachment) 

182 

183 if written: 

184 # One commit for the whole intake, not one per document, and no commit 

185 # at all when every copy was skipped. 

186 _commit_paths( 

187 worktree, 

188 [a["path"] for a in written], 

189 f"chore: attach {'+'.join(a['kind'] for a in written)} for {work_item_id}", 

190 )

193def _carry_local_files(repo: Path, worktree: Path, rels: list[str]) -> tuple[list[str], list[str]]: 

194 """Copy `rels` from `repo` into `worktree`. Returns `(carried, refused)`. 

195 

196 `git worktree add` checks out tracked content at HEAD, so a machine-local 

197 file the developer never committed does not exist in the worktree. A 

198 `.python-version` left behind this way is not a missing convenience: uv 

199 falls through to `requires-python`, and `~=3.11` means `>=3.11, <4`, so it 

200 picks whatever interpreter PATH offers first and the chain runs on it 

201 silently -- 3.14 on work item 1e2e6b45898e42298d16232c9cbfb768, ten retries 

202 deep (Kraft-gxcmy). 

203 

204 A file the worktree would not ignore is refused rather than carried. Kraft 

205 has no way to hold an unignored file out of a commit: a per-worktree 

206 `info/exclude` is read from the *common* dir, so an entry there would leak 

207 to every other worktree of the same repo, and a `core.excludesFile` set in 

208 the worktree's own config is outranked by `main_ignore_args`' `-c` -- which 

209 is live during `_commit_paths`, exactly where it would have mattered. 

210 Refusing keeps a carried file clear of `open_mr`'s dirty-worktree guard by 

211 construction, and an unignored machine-local file is one `git add -A` from 

212 being committed with or without Kraft. 

213 

214 Never raises: a refusal or a missing source is reported to the caller, not 

215 turned into a failed worktree. The destination guards are 

216 `_copy_attachments`' (Kraft-85wk) -- a symlinked destination is skipped 

217 whether or not it dangles, and the resolved destination must stay inside 

218 the worktree. 

219 """ 

220 worktree_root = worktree.resolve() 

221 carried: list[str] = [] 

222 refused: list[str] = [] 

223 for rel in rels: 

224 src = repo / rel 

225 dest = worktree / rel 

226 if not src.is_file(): 

227 # config.py only rejects a directory entry that ends in `/` 

228 # (`.venv/`), so a bare `.venv` passes validation and lands here. 

229 # Silently skipping it -- indistinguishable from a source that was 

230 # simply never created -- hides exactly the typo an operator most 

231 # needs to see; refusing it, like an unignored file, at least says 

232 # something went wrong. A genuinely absent source (no file, no 

233 # directory) stays a silent no-op, per the spec. 

234 if src.exists(): 

235 refused.append(rel) 

236 continue 

237 if dest.is_symlink() or dest.exists(): 

238 continue 

239 resolved = dest.resolve() 

240 if resolved != worktree_root and worktree_root not in resolved.parents: 

241 continue 

242 # The worktree's own rules, not `main`'s: `main_ignore_args` widens what 

243 # counts as ignored for Kraft's own commits, but a plain `git add -A` 

244 # from an agent sees only what is checked out here. 

245 ignored = subprocess.run( 

246 ["git", "check-ignore", "-q", "--", rel], 

247 cwd=str(worktree), 

248 capture_output=True, 

249 text=True, 

250 ) 

251 if ignored.returncode != 0: 

252 refused.append(rel) 

253 continue 

254 dest.parent.mkdir(parents=True, exist_ok=True) 

255 shutil.copyfile(src, dest) 

256 carried.append(rel) 

257 return carried, refused

260def _uncarried_local_files(repo: Path, worktree: Path) -> list[str]: 

261 """Root-level files present in `repo` but absent from `worktree`. 

262 

263 `local_files` is opt-in, which means an unconfigured repo behaves exactly 

264 as it did before -- including the part where it is silently wrong. Nobody 

265 knew to configure the repo on work item 1e2e6b45898e42298d16232c9cbfb768 

266 until after it had burned ten retries (Kraft-gxcmy), so the gap says its own 

267 name here instead of waiting to be discovered. 

268 

269 No `--exclude-standard`: an *ignored* root file is the likeliest carry 

270 candidate of all, so the listing has to include it. `--directory` collapses 

271 an untracked directory to a single entry with a trailing slash, which the 

272 filter then drops -- `.venv/` and `node_modules/` are artifacts to rebuild, 

273 never files to carry. 

274 

275 Stateless by design: a carried file exists in the worktree, so nothing needs 

276 to be told what Task 3 copied, and this reads correctly on 

277 `ensure_worktree`'s early-return path where no copy happened at all. 

278 """ 

279 listed = git_read(repo, "ls-files", "--others", "--directory", expected_failure=True) or "" 

280 names = [n for n in listed.splitlines() if n and "/" not in n] 

281 return sorted(n for n in names if not (worktree / n).exists())

284def _pin_identity(repo: Path, worktree: Path, work_item_id: str) -> None: 

285 """Resolve `user.name`/`user.email` from `repo` and write them into 

286 `worktree`'s own git config, plus every submodule's separate gitdir. 

287 

288 `ensure_worktree` otherwise never establishes a commit identity, relying 

289 on the worktree inheriting the repo's config -- true until it isn't: a 

290 repo with no `[user]` block anywhere in its config chain resolves 

291 nothing, and the agent told to commit before it exits invents an identity 

292 to get unblocked (`kraft@local` was seen in the wild), which later nodes 

293 then read back off the branch and treat as sanctioned (Kraft-mxdx). 

294 

295 Raised, not logged: a missing identity means every commit on this branch 

296 is about to either fail or fabricate one, so failing worktree creation 

297 with the missing key named beats a push rejection eight nodes later. 

298 

299 A submodule's gitdir lives separately (under 

300 `.git/worktrees/<id>/modules/...`) and does not inherit config the way 

301 the main worktree does, so it needs the same pin repeated into it. 

302 """ 

303 name = git_read(repo, "config", "--get", "user.name", expected_failure=True) 

304 email = git_read(repo, "config", "--get", "user.email", expected_failure=True) 

305 if not name or not email: 

306 missing = "user.name" if not name else "user.email" 

307 raise RuntimeError(f"no {missing} configured in {repo}; set it before Kraft can commit") 

308 

309 def pin(target: Path) -> None: 

310 subprocess.run( 

311 ["git", "-C", str(target), "config", "user.name", name], 

312 capture_output=True, 

313 text=True, 

314 ) 

315 subprocess.run( 

316 ["git", "-C", str(target), "config", "user.email", email], 

317 capture_output=True, 

318 text=True, 

319 ) 

320 

321 pin(worktree) 

322 submodules = git_read( 

323 worktree, 

324 "submodule", 

325 "foreach", 

326 "--quiet", 

327 "--recursive", 

328 "echo $sm_path", 

329 expected_failure=True, 

330 ) 

331 for line in (submodules or "").splitlines(): 

332 line = line.strip() 

333 if line: 

334 pin(worktree / line)

337async def _setup_submodules( 

338 db, repo: Path, worktree: Path, branch: str, work_item_id: str, paths: list[str] 

339) -> None: 

340 """`git submodule update --init` only the declared paths (design 3 step 2 

341 -- never blanket), check the item's branch out inside each one, and write 

342 one `work_item_repos` row per repo -- deepest submodule first, root last 

343 (3a) -- so the forge nodes later know what to open a merge request 

344 against and in what order. 

345 

346 A submodule is a regular working copy once initialized, not a bare repo, 

347 so getting the item's branch into it is a plain `checkout`/`checkout -b` 

348 -- design 06's "git worktree add inside each submodule" is shorthand for 

349 "this submodule ends up on the item's branch", not a second linked 

350 worktree, which a submodule path does not support the way the root does. 

351 """ 

352 ordered = store.merge_rank_order(paths) 

353 init = await asyncio.to_thread( 

354 subprocess.run, 

355 # `-c protocol.file.allow=always`: git 2.38+ refuses a `file://` 

356 # submodule URL by default (a supply-chain hardening default, not 

357 # something specific to this repo's own submodules). A real remote is 

358 # https/ssh and is unaffected; the fixtures this plan's own tests 

359 # build (`make_repo_with_submodule`) use plain filesystem paths. 

360 [ 

361 "git", 

362 "-c", 

363 "protocol.file.allow=always", 

364 "submodule", 

365 "update", 

366 "--init", 

367 "--", 

368 *ordered, 

369 ], 

370 cwd=str(worktree), 

371 capture_output=True, 

372 text=True, 

373 ) 

374 if init.returncode != 0: 

375 detail = init.stderr.strip() or init.stdout.strip() 

376 raise RuntimeError(f"git submodule update --init failed for {work_item_id}: {detail}") 

377 for rank, rel in enumerate(ordered, start=1): 

378 sub = worktree / rel 

379 exists = git_read( 

380 sub, "rev-parse", "--verify", "--quiet", f"refs/heads/{branch}", expected_failure=True 

381 ) 

382 checkout = await asyncio.to_thread( 

383 subprocess.run, 

384 ["git", "checkout", branch] if exists else ["git", "checkout", "-b", branch], 

385 cwd=str(sub), 

386 capture_output=True, 

387 text=True, 

388 ) 

389 if checkout.returncode != 0: 

390 detail = checkout.stderr.strip() or checkout.stdout.strip() 

391 raise RuntimeError( 

392 f"checkout of {branch!r} failed in submodule {rel} for {work_item_id}: {detail}" 

393 ) 

394 await asyncio.to_thread(_pin_identity, repo, sub, work_item_id) 

395 await db.write( 

396 lambda c, p=str(sub), r=rel, rk=rank: store.add_repo( 

397 c, 

398 work_item_id=work_item_id, 

399 repo_path=p, 

400 role="submodule", 

401 submodule_path=r, 

402 merge_rank=rk, 

403 ) 

404 ) 

405 root_rank = len(ordered) + 1 

406 await db.write( 

407 lambda c, p=str(worktree), rk=root_rank: store.add_repo( 

408 c, work_item_id=work_item_id, repo_path=p, role="root", merge_rank=rk 

409 ) 

410 )

432async def ensure_worktree( 

433 db, 

434 run_dirs, 

435 *, 

436 repo: str, 

437 work_item_id: str, 

438 repo_entry: dict | None, 

439 attachments: list[dict] | None = None, 

440) -> Path: 

441 """The item's worktree, created if it is not there yet, with intake 

442 attachments copied in. 

443 

444 Called by the executor before the first node dispatches, not only by the 

445 `env_setup` builtin: `default.yaml` runs `spec` and `plan` ahead of 

446 `env_setup`, and an agent asked to write a file into a directory that does 

447 not exist fails in a way no chain can recover from (Kraft-bmp). The 

448 attachment copy has to move with it: `plan/SKILL.md` tells a headless 

449 session to fall back to an attached spec when the chain skipped the spec 

450 node, and that document was not there yet if the copy waited for 

451 `env_setup` (node 4) to run. `env_setup` still exists — it is this call, 

452 now unconditionally carrying attachments, plus its own session bookkeeping. 

453 

454 Idempotent in both directions: an existing worktree is returned untouched 

455 *and uncopied-into* (attachments were copied whenever this worktree was 

456 created; that also covers a rejected gate re-entering with the same 

457 attachments — the worktree is gone but the copy already landed on the 

458 branch's prior commits, and `_copy_attachments`'s own `dest.exists()` guard 

459 makes a second copy onto a survived worktree a no-op rather than a 

460 clobber), and an existing branch — the one on the row (a rejected gate 

461 removes the worktree but keeps the branch) is checked out rather than 

462 re-created. 

463 """ 

464 worktree = run_dirs.worktrees / work_item_id 

465 if worktree.is_dir(): 

466 return worktree 

467 # Read from repo_entry before `git worktree add` runs, not after: a 

468 # poisoned entry (malformed repos.yaml) raises ConfigError on `.get`, and 

469 # reading it only after the worktree exists would leave that worktree 

470 # behind for a later `kraft item retry` to find via the early return 

471 # above -- skipping setup_command entirely and dispatching into an 

472 # unprepared worktree. 

473 entry = repo_entry or {} 

474 local_files = entry.get("local_files") or [] 

475 cmd = entry.get("setup_command") 

476 # Pin the base before the worktree exists, so the early return above 

477 # guarantees a crashed-and-retried run never re-pins to a moved HEAD. 

478 row = db.read( 

479 lambda c: c.execute( 

480 "SELECT base_ref, branch, id FROM work_items WHERE id = ?", (work_item_id,) 

481 ).fetchone() 

482 ) 

483 head = None 

484 if row is not None and row["base_ref"] is None: 

485 head = await upstream_head(Path(repo)) 

486 if head: 

487 await db.write(lambda c: store.set_base_ref(c, work_item_id, head)) 

488 else: 

489 # Without a base_ref the diff endpoint answers "no diff available" 

490 # for the rest of the item's life; the reason belongs in the log 

491 # rather than in a reviewer's guesswork. 

492 logger.warning("no base_ref for %s: rev-parse HEAD failed in %s", work_item_id, repo) 

493 # One stored value, not a third derivation of it (Kraft-nhps). The row is 

494 # the truth here; it is None only when the caller asked for a worktree 

495 # before intake committed the row — tests do, and `env_setup` is reachable 

496 # that way — and then the id is all there is to name a branch with. 

497 branch = store.branch_for(row) if row is not None else f"kraft/{work_item_id}" 

498 exists = git_read( 

499 Path(repo), 

500 "rev-parse", 

501 "--verify", 

502 "--quiet", 

503 f"refs/heads/{branch}", 

504 expected_failure=True, 

505 ) 

506 # A worktree directory deleted out from under git leaves a stale 

507 # administrative entry that makes `worktree add` refuse the same path. 

508 await asyncio.to_thread( 

509 subprocess.run, ["git", "worktree", "prune"], cwd=repo, capture_output=True, text=True 

510 ) 

511 args = ["git", "worktree", "add"] 

512 # A sha, not `origin/<default>`: a remote-tracking start point would make 

513 # git set it as the new branch's upstream. 

514 if exists: 

515 args += [str(worktree), branch] 

516 else: 

517 args += [str(worktree), "-b", branch] + ([head] if head else []) 

518 done = await asyncio.to_thread(subprocess.run, args, cwd=repo, capture_output=True, text=True) 

519 if done.returncode != 0: 

520 # Raised, not returned: this runs outside a session, so there is no 

521 # session status to carry the failure. `kraft.api.deps.guard` turns it into 

522 # needs_human with the git stderr in the reason. 

523 detail = done.stderr.strip() or done.stdout.strip() 

524 raise RuntimeError(f"git worktree add failed for {work_item_id}: {detail}") 

525 await asyncio.to_thread(_pin_identity, Path(repo), worktree, work_item_id) 

526 await asyncio.to_thread( 

527 _copy_attachments, Path(repo), worktree, attachments or [], work_item_id 

528 ) 

529 # Before the sync, not after: uv chooses an interpreter when it runs, so a 

530 # pin that lands later is a pin that changed nothing (Kraft-gxcmy). 

531 _, refused = await asyncio.to_thread(_carry_local_files, Path(repo), worktree, local_files) 

532 if refused: 

533 logger.warning( 

534 "not carried into %s (the worktree would not ignore them, so Kraft " 

535 "cannot keep them out of a commit): %s", 

536 work_item_id, 

537 ", ".join(refused), 

538 ) 

539 # Every node from `spec` on can commit, and the shared pre-commit hook 

540 # needs its tooling on PATH, so the worktree must be a working environment 

541 # before the first node dispatches (Kraft-i047). There is no default and no 

542 # marker sniffing: the repo declares how it is prepared, or the chain stops 

543 # (Kraft-kji8w). A failure here used to be a log warning, which dispatched 

544 # a node into a known-broken environment and let the verify node retry a 

545 # deterministic failure ten times over (Kraft-s0w2l). 

546 if cmd is None: 

547 await _discard_worktree(Path(repo), worktree) 

548 raise RuntimeError( 

549 f"no setup_command declared for {repo} in repos.yaml, so {work_item_id}'s " 

550 'worktree cannot be prepared. Declare one (use "" for a repo that ' 

551 "deliberately needs no preparation)." 

552 ) 

553 if cmd: 

554 done = await asyncio.to_thread( 

555 subprocess.run, 

556 cmd, 

557 shell=True, 

558 cwd=worktree, 

559 env=worker_env(repo_entry), 

560 capture_output=True, 

561 text=True, 

562 ) 

563 if done.returncode != 0: 

564 detail = done.stderr.strip() or done.stdout.strip() 

565 left = await _discard_worktree(Path(repo), worktree) 

566 suffix = f" (and its worktree could not be removed: {left})" if left else "" 

567 raise RuntimeError( 

568 f"setup command failed for {work_item_id}: {cmd!r}: {detail}{suffix}" 

569 ) 

570 decl = db.read( 

571 lambda c: c.execute( 

572 "SELECT submodules FROM work_items WHERE id = ?", (work_item_id,) 

573 ).fetchone() 

574 ) 

575 submodules = json.loads(decl["submodules"]) if decl and decl["submodules"] else [] 

576 if submodules: 

577 await _setup_submodules(db, Path(repo), worktree, branch, work_item_id, submodules) 

578 return worktree

581async def upstream_head(repo: Path) -> str | None: 

582 """The tip of origin's default branch, fetched now; the last fetched tip 

583 when the fetch fails; `repo`'s own HEAD only when there is no `origin` 

584 or it was never fetched. 

585 

586 An item's MR targets origin's branch, and the connected checkout is only 

587 as fresh as its owner's last pull -- Kraft merges on the forge, so nothing 

588 here ever moves it (Kraft-k647). Forking and rebasing onto that checkout 

589 started items behind and made every `pre_mr_rebase` answer "nothing to 

590 rebase". A failed fetch (offline, credentials the server process cannot 

591 reach, a concurrent fetch holding the ref lock) still prefers the stale 

592 remote-tracking ref over the checkout, which may be on another branch or 

593 carry unpushed commits that would then ride into the MR unseen. 

594 

595 The refspec is explicit because a `--single-branch` clone's configured 

596 one may not cover the default branch, leaving its ref unmoved. No prompt 

597 may reach a foreground `kraft`'s terminal: `GIT_TERMINAL_PROMPT=0` for 

598 git's own, ssh `BatchMode` for passphrases and unknown hosts -- unless the 

599 user already chose an ssh command, which an env var would override. 

600 """ 

601 if git_read(repo, "remote", "get-url", "origin", expected_failure=True): 

602 default = await git.default_branch(repo) 

603 ref = f"refs/remotes/origin/{default}" 

604 env = {**os.environ, "GIT_TERMINAL_PROMPT": "0"} 

605 if "GIT_SSH_COMMAND" not in env and not git_read( 

606 repo, "config", "core.sshCommand", expected_failure=True 

607 ): 

608 env["GIT_SSH_COMMAND"] = "ssh -o BatchMode=yes" 

609 try: 

610 done = await asyncio.to_thread( 

611 subprocess.run, 

612 ["git", "fetch", "-q", "origin", f"+refs/heads/{default}:{ref}"], 

613 cwd=repo, 

614 stdin=subprocess.DEVNULL, 

615 capture_output=True, 

616 text=True, 

617 timeout=60, 

618 env=env, 

619 ) 

620 detail = (done.stderr.strip() or "failed") if done.returncode else None 

621 except subprocess.TimeoutExpired: 

622 detail = "timed out" 

623 if detail is not None: 

624 logger.warning("could not fetch origin/%s in %s: %s", default, repo, detail) 

625 tip = git_read(repo, "rev-parse", "--verify", "--quiet", ref, expected_failure=True) 

626 if tip: 

627 return tip 

628 return git_read(repo, "rev-parse", "HEAD")

631async def refresh_worktree_base( 

632 worktree: Path, repo: Path, branch: str, *, force: bool = False 

633) -> str | None: 

634 """Rebase `worktree`'s branch onto origin's default branch (`upstream_head`), 

635 so a paused or retried item's next commit lands on top of whatever landed 

636 while the item sat stopped, not the commit it forked from. 

637 

638 Returns the new HEAD sha when the rebase moved the branch (the caller 

639 stores it as the item's `base_ref`), or None when there was nothing to 

640 do: no worktree yet, `repo`'s HEAD unreadable, the branch already 

641 contains that HEAD, the branch already has an `origin` remote-tracking 

642 ref (unless `force`), or the worktree has uncommitted changes -- all are 

643 best-effort skips (logged), same posture as `ensure_worktree`'s other git 

644 steps. The dirty-tree case comes from a pause via SIGTERM 

645 (`kraft.api.routes.lifecycle._terminate`) catching an agent mid-edit with 

646 nothing committed yet, and `git rebase` itself refuses a dirty tree; the 

647 pushed-branch case is covered below. 

648 

649 Raises `RebaseConflict` (a `RuntimeError` subclass), with the rebase 

650 already aborted (`git rebase --abort`), on a conflict -- the worktree is 

651 left clean and consistent either way; what happens next is the caller's 

652 call (Kraft-s7c04.23 reverses the "not to dispatch an agent into" stance 

653 this docstring used to take here, by the human's own call at the design 

654 gate). 

655 

656 `force=True` (only `mr_checks`'s conflict-triggered rebase passes this, 

657 via `mr_rebase_forced` below) skips the "already pushed, don't touch it" 

658 guard: the caller already read the merge request back and confirmed it 

659 cannot land as-is, so rewriting a branch a reviewer is looking at is 

660 exactly what was asked for, not an accident. 

661 """ 

662 if not worktree.is_dir(): 

663 return None 

664 # Cheaper, purely local check first -- ahead of `upstream_head`'s fetch: a 

665 # branch already pushed past a prior open_mr gate must not be silently 

666 # rewritten here -- a reviewer or a pipeline may already be looking at 

667 # those commits, and this is a quiet auto-refresh on resume/retry, not a 

668 # deliberate rebase someone asked for. `forge._push` can publish a 

669 # rewritten branch now (Kraft-z6i8), so this skip is a policy choice about 

670 # *when* to rewrite, not a workaround for push being unable to. 

671 if not force and git_read( 

672 worktree, 

673 "rev-parse", 

674 "--verify", 

675 "--quiet", 

676 f"refs/remotes/origin/{branch}", 

677 expected_failure=True, 

678 ): 

679 logger.info("refresh_worktree_base: %s already pushed to origin, skipping", branch) 

680 return None 

681 head = await upstream_head(repo) 

682 if not head: 

683 logger.warning("refresh_worktree_base: rev-parse HEAD failed in %s", repo) 

684 return None 

685 # Already up to date: exit 0 means head is already an ancestor of the tip. 

686 if ( 

687 git_read(worktree, "merge-base", "--is-ancestor", head, "HEAD", expected_failure=True) 

688 is not None 

689 ): 

690 return None 

691 if git_read(worktree, "status", "--porcelain"): 

692 logger.warning("refresh_worktree_base: %s has uncommitted changes, skipping", worktree) 

693 return None 

694 done = await asyncio.to_thread( 

695 subprocess.run, 

696 ["git", "-C", str(worktree), "rebase", head], 

697 capture_output=True, 

698 text=True, 

699 ) 

700 if done.returncode != 0: 

701 await asyncio.to_thread( 

702 subprocess.run, 

703 ["git", "-C", str(worktree), "rebase", "--abort"], 

704 capture_output=True, 

705 text=True, 

706 ) 

707 # git prints the "CONFLICT (content): Merge conflict in <file>" line to 

708 # stdout; stderr only ever carries the generic "could not apply"/"hint: 

709 # Resolve all conflicts" text. Join both rather than preferring one, so 

710 # the conflicting file's name survives into the raised message and the 

711 # needs_human reason a human reads. 

712 detail = "\n".join(filter(None, [done.stdout.strip(), done.stderr.strip()])) 

713 raise RebaseConflict(f"git rebase failed for {worktree}: {detail}") 

714 return head

717async def mr_rebase( 

718 db, 

719 run_dirs, 

720 *, 

721 session_id: str, 

722 work_item_id: str, 

723 node_id: str, 

724 hook_point: str, 

725 round: int, 

726 repo: str, 

727 worktree: str, 

728 branch: str, 

729 head_sha: str | None = None, 

730) -> str: 

731 """Rebase onto origin's default branch right before `open_mr`, so an item 

732 that ran straight through the chain -- no pause, no `/retry` -- doesn't 

733 open its MR however many commits behind (Kraft-4bgg). A thin wrapper: 

734 `refresh_worktree_base` is already the whole implementation, shared with

748 else:

751 db,

754 work_item_id=work_item_id,

771async def scan_submodules( 

772 db, 

773 run_dirs, 

774 *, 

775 session_id: str, 

776 work_item_id: str, 

777 node_id: str, 

778 hook_point: str, 

779 round: int, 

780 repo: str, 

781 worktree: str, 

782 head_sha: str | None = None, 

783) -> str: 

784 """Design 3a: catch a submodule the agent touched but the item never 

785 declared, give it a `work_item_repos` row and a pinned identity, before 

786 `open_mr` would otherwise have to refuse over it. 

787 

788 Runs in its own node right after `implementation` (see `default.yaml`), 

789 so a plain green `verify` never masks a change `open_mr` 

790 would reject three nodes later -- this is what would have rescued work 

791 item 9d0ab38ff3c9439b90506df0f6966660, which declared nothing. 

792 """ 

793 _, log_path, result_path = await start_session( 

794 db, 

795 run_dirs, 

796 session_id=session_id, 

797 work_item_id=work_item_id, 

798 node_id=node_id, 

799 hook_point=hook_point, 

800 round=round, 

801 head_sha=head_sha, 

802 ) 

803 wt = Path(worktree) 

804 known = { 

805 r["submodule_path"] 

806 for r in db.read( 

807 lambda c: c.execute( 

808 "SELECT submodule_path FROM work_item_repos " 

809 "WHERE work_item_id = ? AND role = 'submodule'", 

810 (work_item_id,), 

811 ).fetchall() 

812 ) 

813 if r["submodule_path"] 

814 } 

815 # '+' means the submodule's checked-out commit no longer matches what the 

816 # superproject's index records -- new commits sitting in the worktree, 

817 # exactly the shape that went unreported before this plan. 

818 raw = git_read(wt, "submodule", "status", expected_failure=True) or "" 

819 touched = set() 

820 for line in raw.splitlines(): 

821 if not line or line[0] != "+": 

822 continue 

823 parts = line[1:].split() 

824 if len(parts) >= 2: 

825 touched.add(parts[1]) 

826 

827 undeclared = sorted(touched - known) 

828 if not undeclared: 

829 return await finish_session( 

830 db, 

831 log_path, 

832 result_path, 

833 session_id=session_id, 

834 status="done", 

835 log="no undeclared submodule changes\n", 

836 ) 

837 

838 existing = db.read( 

839 lambda c: c.execute( 

840 "SELECT COUNT(*) AS n FROM work_item_repos WHERE work_item_id = ?", (work_item_id,) 

841 ).fetchone() 

842 ) 

843 next_rank = existing["n"] + 1 

844 log = "" 

845 for rel in undeclared: 

846 sub = wt / rel 

847 await asyncio.to_thread(_pin_identity, Path(repo), sub, work_item_id) 

848 rank = next_rank 

849 await db.write( 

850 lambda c, p=str(sub), r=rel, rk=rank: store.add_repo( 

851 c, 

852 work_item_id=work_item_id, 

853 repo_path=p, 

854 role="submodule", 

855 submodule_path=r, 

856 merge_rank=rk, 

857 ) 

858 ) 

859 log += f"found undeclared submodule change: {rel}\n" 

860 next_rank += 1 

861 # The root row (written by ensure_worktree, if this item declared any 

862 # submodule at all) now has to merge after these too. 

863 final_rank = next_rank 

864 await db.write( 

865 lambda c, rk=final_rank: c.execute( 

866 "UPDATE work_item_repos SET merge_rank = ? WHERE work_item_id = ? AND role = 'root'", 

867 (rk, work_item_id), 

868 ) 

869 ) 

870 return await finish_session( 

871 db, log_path, result_path, session_id=session_id, status="done", log=log 

872 )

875async def start_session( 

876 db, 

877 run_dirs, 

878 *, 

879 session_id: str, 

880 work_item_id: str, 

881 node_id: str, 

882 hook_point: str, 

883 round: int, 

884 head_sha: str | None = None, 

885 reuse_if_waiting: bool = False, 

886) -> tuple[str, Path, Path]: 

887 """Create the session row before the in-process work starts, not after. 

888 

889 `forge.run_task` can now sit in `_poll_ci` for up to `poll_timeout` 

890 (default 1800s); recording nothing until it returns left pause, abandon 

891 and reattach with no row to find for the whole wait -- pause silently 

892 no-op'd and the chain walked on into merge (Kraft-41b), and a restart 

893 mid-poll left nothing to reattach (Kraft-7xt). Splitting the row's 

894 creation from its exit is what `adapters.subprocess.run_task` already 

895 does for a spawned child; this gives an in-process task the same shape. 

896 

897 Returns `(id, log_path, result_path)`. Normally `id == session_id`; with 

898 `reuse_if_waiting=True` (Kraft-ivh1) a still-`'waiting'` row for this 

899 exact (item, node, hook point, round) is reused instead, and the 

900 returned id/paths are that row's, not `session_id`'s -- the caller must 

901 use the returned id, not `session_id`, for the rest of its work. 

902 """ 

903 log_path = run_dirs.logs / f"{session_id}.log" 

904 result_path = run_dirs.results / f"{session_id}.json" 

905 actual_id, log_str, result_str = await db.write( 

906 lambda c: store.create_session( 

907 c, 

908 id=session_id, 

909 work_item_id=work_item_id, 

910 node_id=node_id, 

911 hook_point=hook_point, 

912 log_path=str(log_path), 

913 result_path=str(result_path), 

914 round=round, 

915 head_sha=head_sha, 

916 reuse_if_waiting=reuse_if_waiting, 

917 ) 

918 ) 

919 return actual_id, Path(log_str), Path(result_str)

922async def finish_session( 

923 db, 

924 log_path: Path, 

925 result_path: Path, 

926 *, 

927 session_id: str, 

928 status: str, 

929 log: str, 

930 findings: list[dict] | None = None, 

931 reused: bool = False, 

932) -> str: 

933 """Write the log and close out a session `start_session` already created. 

934 

935 `findings` is None for every caller that predates this (an agent/ 

936 subprocess task writes its own result file, and every existing builtin 

937 has nothing to report) -- writing `result_path` only when it is given 

938 keeps `_findings.parse`'s "missing file -> no findings" behaviour for 

939 every one of them. 

940 

941 `reused` (Kraft-ivh1): a session `create_session(reuse_if_waiting=True)` 

942 handed back instead of inserting fresh already has a log on disk from an 

943 earlier poll of the same wait episode -- appended to here, not 

944 overwritten, so on.ci.poll's repeated "pipeline pending" reads land in 

945 one running log instead of replacing each other. The `.times` sidecar 

946 is extended the same way: only the newly-added lines get a fresh 

947 timestamp; ones already on disk keep theirs. 

948 """ 

949 previous = log_path.read_text() if reused and log_path.exists() else "" 

950 log_path.write_text(previous + log if previous else log) 

951 if findings is not None: 

952 result_path.write_text(json.dumps({"findings": findings})) 

953 # A subprocess session gets its per-line timestamps from the adapter's drain 

954 # thread (`adapters/subprocess._watch_log`), which stamps each line as it 

955 # appears. A builtin writes its whole log in one call and has no drain 

956 # thread, so without this its lines read back with a blank time column -- 

957 # which `env_setup` used to have, before it stopped going through the 

958 # subprocess adapter. 

959 now = datetime.now(UTC).isoformat() 

960 try: 

961 times_path = logs.times_path(log_path) 

962 previous_line_count = len(logs.split_lines(previous)) if previous else 0 

963 new_entries = [ 

964 json.dumps({"n": previous_line_count + n, "t": now}) + "\n" 

965 for n in range(len(logs.split_lines(log))) 

966 ] 

967 if reused and times_path.exists(): 

968 with times_path.open("a") as fh: 

969 fh.writelines(new_entries) 

970 else: 

971 times_path.write_text("".join(new_entries)) 

972 except OSError: 

973 # Best-effort, the same call `_watch_log` makes about its own sidecar: a 

974 # missing time column is cosmetic, not a reason to fail the node. 

975 logger.warning("no log time sidecar for session %s", session_id) 

976 await db.write(lambda c: store.session_exited(c, session_id, status)) 

977 return status
