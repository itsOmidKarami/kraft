1"""Auto-relaunch a work item once its rate-limit wait is over. 

2 

3Always on, unlike `intake.py`'s poller: waiting out a rate limit is not 

4optional behaviour a human opts into, it is what the rest of this feature 

5promises. Ticks on a fixed interval, records the retry through 

6`store.retry_after_cap` (which resets no cap counter for a relaunch nobody 

7asked for), and relaunches through `executor.run` with no 

8position: the walk resumes at the item's own cursor, like every door that 

9resumes an item (Kraft-c3dab). 

10""" 

11 

12from __future__ import annotations 

13 

14import asyncio 

15import logging 

16 

17from kraft import policy as policy_mod 

18from kraft import store 

19from kraft.store import _now as _now # test seam for wall-clock checks 

20 

21logger = logging.getLogger(__name__) 

22 

23RESUME_PROMPT = ( 

24 "You were working on this task when the agent hit an API rate limit and " 

25 "stopped. That limit has now reset — continue the work from where you " 

26 "left off." 

27) 

28 

29#: How often the poller checks for a due item. Fixed, unlike auto-intake's 

30#: `interval_s`: there is no per-operator tuning knob for this, only a fixed 

31#: cadence cheap enough to run unconditionally (spec "Non-goals"). 

32_INTERVAL_S = 30 

33 

34#: `bump_counter`'s `cap_wall_s` column is NOT NULL, but a rate-limit wait can 

35#: legitimately run for hours -- ponytail: a wall-clock cap this large is 

36#: effectively "no wall-clock limit"; only `count` is ever checked below. 

37_NO_WALL_CAP = 10**9 

38 

39#: `Policy.rate_limit_retries` defaults to 5; mirrored here for the one caller 

40#: whose `app.state.policy` is `None` outright (an unset or invalid policy.yaml). 

41_DEFAULT_RETRIES = 5 

42 

43 

44def _key(node_id: str) -> str: 

45 """Per-node, not per-item: a node that gets rate-limited, recovers, and 

46 later gets rate-limited again on a *different* node must not spend the 

47 same budget the first node already used up.""" 

48 return f"rate_limit:{node_id}" 

49 

50 

51async def tick(app) -> list[str]: 

52 """One poll. Returns the work item ids relaunched, which is usually none.""" 

53 st = app.state 

54 due = st.db.read( 

55 lambda c: c.execute( 

56 # `materialized_chain` too, for `store.node_index` -- see waits.py. 

57 "SELECT id, repo, current_node_id, chain_definition, materialized_chain " 

58 "FROM work_items " 

59 "WHERE status = 'rate_limited' AND retry_at <= ?", 

60 (_now(),), 

61 ).fetchall() 

62 ) 

63 relaunched: list[str] = [] 

64 for row in due: 

65 if await _retry_one(app, row): 

66 relaunched.append(row["id"]) 

67 return relaunched 

68

70async def _retry_one(app, row) -> bool: 

71 from kraft.api import deps 

72 from kraft.executor import stops # deferred, same cycle as `executor` below 

73 

74 st = app.state 

75 wid = row["id"] 

76 node_id = row["current_node_id"] 

77 retries = st.policy.rate_limit_retries if st.policy is not None else _DEFAULT_RETRIES 

78 cap = policy_mod.Cap(attempts=retries, wall_clock_s=_NO_WALL_CAP) 

79 count, _started_at, cap = await st.db.write( 

80 lambda c: store.bump_counter(c, wid, _key(node_id), cap) 

81 ) 

82 if count > cap.attempts: 

83 await st.db.write( 

84 lambda c: store.mark_needs_human( 

85 c, wid, node_id, f"rate_limit retries exhausted after {count - 1} attempt(s)" 

86 ) 

87 ) 

88 return False 

89 

90 # Bracketed from *before* the claim to the hand-off. A claim makes this item 

91 # read `active`, and this poller's own `tick` SELECT filters 

92 # `status = 'rate_limited'`, so an exit from here that neither spawns a walk nor 

93 # moves the status again leaves the item claimed and unowned, and *no later tick 

94 # will ever select it again*. The failed-claim `return False` is inside it 

95 # deliberately, and is nearly inert rather than inert: the claim failed because 

96 # the status left `rate_limited`, and if what it left for was `active` with no 

97 # walk behind it, the bracket writes `needs_human` on the way out. Right answer 

98 # for an orphan, and not nothing -- `handed_off` covers the case where a walk 

99 # really does own it. 

100 async with stops.claimed_or_stopped( 

101 st.db, 

102 wid, 

103 node_id, 

104 reason="the rate-limit poller claimed this item but could not start a walk", 

105 handed_off=lambda: deps.task_is_live(app, wid), 

106 ): 

107 claimed = await st.db.write( 

108 lambda c: store.claim_for_run(c, wid, from_statuses=["rate_limited"]) 

109 ) 

110 if not claimed: 

111 # Something else (a human abandoning it, most plausibly) already moved 

112 # this item off `rate_limited` between the `due` SELECT and here. 

113 # `retry_after_cap` no longer flips status itself -- without this 

114 # check the poller would blindly claw an abandoned item back to 

115 # 'active' and spawn a walk into a worktree that may already be gone. 

116 logger.info("rate-limit retry: %s is no longer rate_limited, skipping", wid) 

117 return False 

118 

119 # Kraft's own relaunch, so no cap counter resets (Kraft-s7c04.22). 

120 await st.db.write( 

121 lambda c: store.retry_after_cap(c, wid, node_id, None, RESUME_PROMPT, by_person=False) 

122 ) 

123 # Same as `waits`': over `store.node_index` so a V1 row's `"{}"` 

124 # `chain_definition` cannot raise, and a node that is not in this item's 

125 # chain stops rather than silently relaunching it at node zero. 

126 if store.node_index(row, node_id) is None: 

127 # `return False`, not a bare `return`: this function's contract is 

128 # "did I relaunch it". The bracket above turns the claim into a stop. 

129 logger.warning( 

130 "rate_limit_retry: %s has no node %r in its chain, not relaunching", wid, node_id 

131 ) 

132 return False 

133 from kraft import executor # deferred: avoids a kraft.api <-> kraft.executor import cycle 

134 

135 try: 

136 deps.spawn( 

137 app, 

138 wid, 

139 deps.guard( 

140 st.db, 

141 wid, 

142 executor.run( 

143 st.db, 

144 st.run_dirs, 

145 work_item_id=wid, 

146 bd_cwd=deps.bd_cwd(), 

147 # No position: the walk resumes at the item's own 

148 # cursor, the step the rate limit stopped (Kraft-c3dab). 

149 policy=st.policy, 

150 steer=RESUME_PROMPT, 

151 launch=deps.launch(st, row["repo"]), 

152 ), 

153 ), 

154 ) 

155 except deps.AlreadyRunning: 

156 logger.warning("rate-limit retry: %s already has a live walk, skipping", wid) 

157 return False 

158 return True

161async def poller(app) -> None: 

162 """`tick` on a fixed interval until cancelled.""" 

163 while True: 

164 await asyncio.sleep(_INTERVAL_S) 

165 try: 

166 await tick(app) 

167 except asyncio.CancelledError: 

168 raise 

169 except Exception: # noqa: BLE001 -- a bad tick must not end the poller 

170 logger.exception("rate-limit retry tick failed")
