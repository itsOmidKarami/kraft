1"""Autonomous pickup from `bd ready` (sub-project E §4-§5). 

2 

3Off by default, and the one thing in this sub-project that starts work rather 

4than refusing to. The boundary it must not cross is in §5 and is worth stating 

5where the code is: **an auto-started work item passes no gate automatically.** 

6It is created with the same `executor.intake` and run with the same 

7`executor.run` as a typed-in one, so it reaches its first gate and stops for a 

8person. Auto-intake removes the typing, not the judgement. An auto-start that 

9also auto-approved would be a different product, and would get its own design 

10document before any of it was written. 

11""" 

12

26def _known_beads(db) -> set[str]: 

27 return { 

28 r[0] 

29 for r in db.read( 

30 lambda c: c.execute( 

31 "SELECT bead_id FROM work_items WHERE bead_id IS NOT NULL" 

32 ).fetchall() 

33 ) 

34 }

37def _daily_breached(db, budget: policy_mod.Budget) -> bool: 

38 """Only the daily cap is evaluable here: a candidate has no work item yet, so 

39 it has no per-item spend. The per-item cap does its work at first dispatch.""" 

40 if budget.daily_usd is None: 

41 return False 

42 since = store.local_midnight_utc() 

43 # The work-item argument is irrelevant to the daily figure; pass a value that 

44 # matches nothing rather than inventing an overload. 

45 _item, daily = db.read(lambda c: store.budget_spend(c, "", since=since)) 

46 return daily >= budget.daily_usd

49async def tick(app) -> list[str]: 

50 """One poll. Returns the work item ids started, which is usually none.""" 

51 from kraft.api import deps 

52 

53 st = app.state 

54 cfg = st.intake 

55 if not cfg.get("enabled"): 

56 return [] 

57 if st.invalid_policy: 

58 return [] 

59 budget = st.policy.budget if st.policy else policy_mod.NO_BUDGET 

60 if _daily_breached(st.db, budget): 

61 logger.info("auto-intake: daily budget reached, starting nothing") 

62 return [] 

63 # Every active item counts, not only auto-started ones: a person working on 

64 # three things must not find the poller adding a fourth. 

65 slots = int(st.policy.max_concurrent if st.policy else 1) - st.db.read(store.active_count) 

66 if slots <= 0: 

67 return [] 

68 

69 try: 

70 repos = config_mod.load_repos(deps.repos_path(st), validate_steering=False) 

71 except config_mod.ConfigError as exc: 

72 logger.warning("auto-intake: repo config invalid, skipping this tick: %s", exc) 

73 return [] 

74 wanted = set(cfg.get("repos") or []) 

75 # `enabled` defaults to True here because `load_repos` does not default it and 

76 # the API's own `RepoBody` does — an entry hand-written without it is on. 

77 repos = [r for r in repos if r.get("enabled", True) and (not wanted or r["path"] in wanted)] 

78 # `repos.yaml` paths are stored as written, so a `~` or a trailing slash in 

79 # the filter matches nothing and the poller ticks forever picking nothing up. 

80 if wanted and not repos: 

81 logger.warning("auto-intake: repos filter %r matched no configured repo", sorted(wanted)) 

82 

83 known = _known_beads(st.db) 

84 ceiling = int(cfg.get("priority_ceiling", 2)) 

85 started: list[str] = [] 

86 for repo in repos: 

87 if slots <= 0: 

88 break 

89 for row in await beads.ready(cwd=repo["path"]): 

90 if slots <= 0: 

91 break 

92 if row["id"] in known or not isinstance(row.get("priority"), int): 

93 continue 

94 # An epic is a container for work, not work: auto-starting one files a 

95 # chain against a title that describes a quarter. 

96 if row.get("issue_type") == "epic": 

97 continue 

98 # P0 is the *highest* priority, so "P2 and below" is `>= ceiling`. 

99 # The highest-priority work is what a human should be looking at; 

100 # unattended pickup is for the backlog. 

101 if row["priority"] < ceiling: 

102 continue 

103 wid = await _start(app, repo, row) 

104 if wid is not None: 

105 started.append(wid) 

106 known.add(row["id"]) 

107 slots -= 1 

108 return started

111async def _start(app, repo: dict, row: dict) -> str | None: 

112 from kraft.api import deps 

113 

114 st = app.state 

115 template = st.templates.valid.get(repo.get("default_chain_template") or "default") 

116 if template is None: 

117 logger.warning("auto-intake: %s has no valid chain template, skipping", repo["path"]) 

118 return None 

119 # Spec §5 is "an auto-started item passes no gate automatically". A template 

120 # with no gate at all satisfies that by having nothing to pass, which is the 

121 # letter of the rule and the opposite of its point: it would run to merge 

122 # unattended. A human can still start such a chain by hand — they are the 

123 # judgement the gates exist to invoke. 

124 if not any(n.get("gate_after") for n in template.nodes): 

125 logger.warning( 

126 "auto-intake: %s uses %r, which has no gate — refusing to start it " 

127 "unattended; a person can start it from the board", 

128 repo["path"], 

129 template.id, 

130 ) 

131 return None 

132 try: 

133 wid = await executor.intake( 

134 st.db, 

135 st.run_dirs, 

136 title=row["title"], 

137 description=row.get("description"), 

138 repo=repo["path"], 

139 template=template, 

140 bd_cwd=deps.bd_cwd(), 

141 bead_id=row["id"], 

142 # the bead was never filed in KRAFT_BD_CWD — it is adopted from the 

143 # repo's own .beads and can only be closed there. 

144 bead_cwd=repo["path"], 

145 source="auto_intake", 

146 bead_priority=row.get("priority"), 

147 ) 

148 except Exception: # noqa: BLE001 -- one bad bead must not stop the poller 

149 logger.exception("auto-intake: could not file %s", row["id"]) 

150 return None 

151 logger.info("auto-intake: started %s from %s", wid, row["id"]) 

152 try: 

153 deps.spawn( 

154 app, 

155 wid, 

156 deps.guard( 

157 st.db, 

158 wid, 

159 executor.run( 

160 st.db, 

161 st.run_dirs, 

162 work_item_id=wid, 

163 registry=st.registry, 

164 bd_cwd=deps.bd_cwd(), 

165 policy=st.policy, 

166 launch=deps.launch(st, repo["path"]), 

167 ), 

168 ), 

169 ) 

170 except deps.AlreadyRunning: 

171 # Structurally unreachable today (every `_start` call is a freshly 

172 # intaken id), kept for the same reason ci_wait/rate_limit_retry are: 

173 # a poller finding its own target already running must log and move 

174 # on, never crash the tick. 

175 logger.warning("auto-intake: %s already has a live walk, not double-starting", wid) 

176 return wid

179async def poller(app) -> None: 

180 """`tick` on a fixed interval until cancelled.""" 

181 raw = app.state.intake.get("interval_s", config_mod.INTAKE_DEFAULT["interval_s"]) 

182 try: 

183 interval = max(30, int(raw)) 

184 except TypeError, ValueError: 

185 # Nothing validates a hand-edited intake.yaml, and raising here would 

186 # kill the poller task at birth — the operator would see a poller they 

187 # just enabled quietly never pick anything up. 

188 interval = config_mod.INTAKE_DEFAULT["interval_s"] 

189 logger.warning("auto-intake: interval_s %r is not a number, using %ss", raw, interval) 

190 while True: 

191 await asyncio.sleep(interval) 

192 try: 

193 await tick(app) 

194 except asyncio.CancelledError: 

195 raise 

196 except Exception: # noqa: BLE001 -- a bad tick must not end the poller 

197 logger.exception("auto-intake tick failed")
