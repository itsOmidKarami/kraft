47async def fail_abandoned_jobs(db, session_id: str, log_path: Path, status: str, reader) -> str: 

48 """`status`, or `failed` when the agent's turn ended with a background job 

49 still running (Kraft-xvugd): the session ends with its turn, so nothing 

50 would ever read that job's result. The prose in `agent._CTX` alone did not 

51 hold (Kraft-nxqft); this is the check behind it. The jobs are named in the 

52 log's last line and a `JOBS_ABANDONED` event rather than left to a generic 

53 missing-result failure. 

54 

55 A `needs_context` stop keeps its status -- it is a person's stop already, 

56 and failing it would lose the question -- but the jobs are still named. 

57 The one door both `run_task` and `reattach` exit an agent session through. 

58 """ 

59 if reader is None or status not in _AGENT_STATUSES: 

60 return status

63 return status

65 reason = ( 

66 f"the turn ended with {len(jobs)} background job(s) still running: {named}. " 

67 "The session ends with its turn, so nothing reads a job's result: run it in " 

68 "the foreground." 

69 )

71 fh.write(f"\nkraft: failed: {reason}\n")

92def _resolve_result_file(path: Path) -> str | None: 

93 """Status from a result file alone, or None if it's missing/empty.""" 

94 if not path.exists(): 

95 return None 

96 try: 

97 raw = path.read_text().strip() 

98 except OSError, UnicodeDecodeError: 

99 # UnicodeDecodeError is a ValueError, not an OSError: a plugin that died 

100 # mid-write leaves bytes that are not valid UTF-8, and reading them must 

101 # fail the session rather than escape as a raw traceback. 

102 return "failed" 

103 if not raw: 

104 return None 

105 try: 

106 data = json.loads(raw) 

107 except json.JSONDecodeError: 

108 return "failed" 

109 status = data.get("status") if isinstance(data, dict) else None 

110 # An unrecognized status still resolves to failed: an agent inventing a status 

111 # is a broken agent, and guessing at its intent is worse than failing the session. 

112 return status if status in _AGENT_STATUSES else "failed"

202def _watch_log( 

203 log_path: Path, times_path: Path, stop: threading.Event, poll_s: float = 0.2 

204) -> None: 

205 """Record when each new line appeared in `log_path`, into a JSONL sidecar. 

206 

207 Reads forward from one open handle rather than re-reading the file: under 

208 `--output-format stream-json` the log grows to megabytes while the worker 

209 works, and re-reading it five times a second would cost more than the 

210 worker. Only `b"\\n"` is counted, so a line is stamped when it completes and 

211 a partial trailing line is carried to the next poll -- the numbering 

212 `logs.split_lines` defines, which `logs.jsonl` reads back by. Best-effort 

213 throughout: an I/O error here must never take the session down with it. 

214 """ 

215 seen = 0 

216 pending = b"" 

217 try: 

218 with open(times_path, "w") as times, open(log_path, "rb") as log: 

219 while True: 

220 done = stop.is_set() 

221 chunk = log.read() # b"" at EOF; the next read picks up new bytes 

222 if chunk: 

223 pending += chunk 

224 complete = pending.count(b"\n") 

225 if complete: 

226 now = datetime.now(UTC).isoformat() 

227 for n in range(seen, seen + complete): 

228 times.write(json.dumps({"n": n, "t": now}) + "\n") 

229 times.flush() 

230 seen += complete 

231 pending = pending[pending.rfind(b"\n") + 1 :] 

232 if done: 

233 return 

234 stop.wait(poll_s) 

235 except OSError: 

236 pass

239def _read_new(path: Path, offset: int) -> tuple[str, int]: 

240 """Complete lines written since `offset`, and the offset to resume from. 

241 

242 A partial trailing line is left where it is rather than decoded half-formed 

243 -- the log is a stream and the writer is mid-flush. Best-effort: an 

244 unreadable log is "nothing new", not a failed session. 

245 """ 

246 try: 

247 with open(path, "rb") as fh: 

248 fh.seek(offset) 

249 data = fh.read() 

250 except OSError: 

251 return "", offset 

252 cut = data.rfind(b"\n") 

253 if cut < 0: 

254 return "", offset 

255 return data[: cut + 1].decode("utf-8", "replace"), offset + cut + 1

258def _progress_usage( 

259 log_path: Path, offset: int, seen: dict[str, _usage.Usage], reader: str | None = None 

260) -> tuple[int, _usage.Usage | None]: 

261 """One tick: new lines since `offset` folded into `seen`, the new offset, and 

262 the running total (or None if nothing has been seen at all). 

263 

264 Shared by `run_task`'s own poll loop and `reattach._adopt`'s: an adopted 

265 session's child survives a Kraft restart (`start_new_session=True`) and 

266 keeps writing its log, so without this loop running there too, tokens_in/out

269 Kraft-41f7 (same frozen-tokens symptom, a different loop missing it). 

270

277 if reader is None: 

278 return offset, None 

279 return offset, _usage.READERS[reader].stream(logs.split_lines(chunk), seen)

310def _resolve_exit_file(path: Path) -> str | None: 

311 """Status from a task's own recorded exit code, or None if absent/garbage. 

312 

313 `reattach` adopts a process Kraft never forked (only pid + create_time, 

314 `reattach.py:38-49`), and a process you did not spawn cannot be reaped --

328 return None

332def _missing_executable(cmd0: str, cwd: Path, env: dict[str, str]) -> bool: 

333 """True when `cmd[0]` names nothing runnable -- the check `Popen` used to do. 

334 

335 `_wrap_with_exit_file` puts `/bin/sh` at argv[0], which always exists, so 

336 `Popen` stopped raising `FileNotFoundError` for a missing binary and `sh` 

337 merely exited 127 -- turning a `config_error` into a `failed` and opening 

338 a fix cycle no agent can win by editing source (Kraft-579). Resolved here 

339 instead, the same two ways `execvp` resolves it: a name containing a 

340 separator is a path relative to the child's cwd, anything else is a PATH 

341 lookup against the child's own env. 

342 """ 

343 if os.sep in cmd0: 

344 candidate = Path(cwd) / cmd0 

345 return not (candidate.is_file() and os.access(candidate, os.X_OK)) 

346 return shutil.which(cmd0, path=env.get("PATH")) is None

349def _wrap_with_exit_file(cmd: list[str], exit_path: Path) -> list[str]: 

350 """`cmd`, wrapped so it writes its own exit code to `exit_path`. 

351 

352 `"$@"` replays argv byte-for-byte -- no re-quoting, so an argument with a 

353 space or a glob survives. `$0` carries the exit path. The child's real 

354 code is propagated, so every existing caller of `proc.returncode` is 

355 unaffected. 

356 

357 Not `exec "$@"`: exec would replace the shell and skip the write on the 

358 success path, which is the whole point. Keeping `sh` alive costs nothing 

359 that matters -- `start_new_session=True` makes `sh` the group leader, 

360 `sh -c` runs with job control off so the real child stays in the same 

361 pgid, and `_kill_group`'s `os.killpg` still reaches both. 

362 """ 

363 return [ 

364 "/bin/sh", 

365 "-c", 

366 'RC=0; "$@" || RC=$?; printf %s "$RC" > "$0"; exit $RC', 

367 str(exit_path), 

368 *cmd, 

369 ]

385async def _kill_group(pgid: int, grace: float, reap: Callable[[], object] | None = None) -> None: 

386 """SIGTERM `pgid`, then SIGKILL whatever is still there after `grace` 

387 seconds. `serve.py`'s `_terminate` is the model this copies. 

388 

389 The session's own child is already gone by the time this runs — this is 

390 for what it backgrounded into the same group and never waited on itself 

391 (a fixture server, `just dev`), which `start_new_session=True` put in the 

392 same process group as the child but this coroutine never touched 

393 (Kraft-1nye). `os.killpg` raises `ProcessLookupError` the moment the group 

394 has no members left, so the common case (nothing backgrounded) returns 

395 almost immediately rather than paying `grace`. 

396 

397 `reap` is the leader's own `Popen.poll`, called before every liveness 

398 check. On a pause or skip this runs from `run_task`'s `finally` while the 

399 leader is still alive, so the SIGTERM above kills it but nothing reaps it 

400 -- the poll loop that normally would was cancelled out from under it. On 

401 Linux an unreaped zombie still counts as a group member, so `killpg(pgid, 

402 0)` kept succeeding and every cancel paid the full `grace` (10s by 

403 default) before falling through to SIGKILL. macOS raises 

404 `PermissionError` for a zombie-only group instead, which is why this 

405 passed locally and failed pause/resume, SIGTERM shutdown and the e2e 

406 pause flow in CI (Kraft-rki). 

407 """ 

408 try: 

409 os.killpg(pgid, signal.SIGTERM) 

410 except ProcessLookupError, PermissionError: 

411 return 

412 deadline = time.monotonic() + grace 

413 while time.monotonic() < deadline: 

414 await asyncio.sleep(0.2) 

415 if reap is not None: 

416 reap() 

417 try: 

418 os.killpg(pgid, 0) 

419 except ProcessLookupError, PermissionError: 

420 return 

421 try: 

422 os.killpg(pgid, signal.SIGKILL) 

423 except ProcessLookupError, PermissionError: 

424 pass

428 db,

431 session_id: str, 

432 work_item_id: str, 

433 node_id: str, 

434 hook_point: str, 

435 cmd: list[str], 

436 cwd: str | Path, 

437 env: dict | None = None, 

438 post_resolve: Callable[[str, Path, int], str] | None = None, 

439 poll_s: float = 0.05, 

440 #: How often the poll loop stops to fold new log lines into live usage. The 

441 #: process poll runs 20x a second; parsing the log that often would cost 

442 #: more than the worker does. A test that cannot wait passes its own. 

443 progress_s: float = 5.0, 

444 #: How long a lingering group member (something the session backgrounded 

445 #: and never waited on) gets after SIGTERM before SIGKILL. serve.py's own 

446 #: `_terminate` uses the same 10s.

448 #: How long to let an interrupted agent flush its result envelope before 

449 #: `_kill_group`'s SIGTERM lands on it (Kraft-s7c04.18). Paid only on a 

450 #: pause. ~1s observed against claude 2.1.273; 2s is the margin. 

451 flush_grace: float = 2.0, 

452 round: int = 0, 

453 head_sha: str | None = None, 

454 thread: int = 1, 

455 sandbox: dict | None = None, 

456 #: Names a log schema in `usage.READERS`, or `None` when the harness 

457 #: declared no log-based reading (`source: result_file`, no 

458 #: `rate_limit_signal`). Resolved by the caller from the harness, never 

459 #: guessed here -- this adapter stays harness-agnostic. 

460 reader: str | None = None, 

461 #: Every agent hook is told by `_CTX` to write $KRAFT_RESULT_PATH, 

462 #: regardless of whether it also declares `artifact:` -- this holds it to 

463 #: that half of the contract on its own (Kraft-avpe). Only run_agent_task 

464 #: sets this; a subprocess-kind hook a template binds directly has no such 

465 #: contract. 

466 require_result_file: bool = False, 

467 repo_entry: RepoEntry | None = None, 

468 #: The tightest time cap over this launch (`caps.at_launch`): past its 

469 #: deadline the process group, and a sandbox's container, is killed and 

470 #: the session exits `capped_out` with `caps.REACHED` naming the scope. 

471 time_cap: caps.Deadline | None = None, 

472 #: The name the result file (and its sidecars) goes under, when not the 

473 #: session's own: an escalation thread's, shared by its turns 

474 #: (`escalate.thread_files`, Kraft-s7c04.54). The log stays per session. 

475 files: str | None = None, 

476 #: `{"harness": <harness id>, "model": <resolved model or None>}`, which a 

477 #: `rate_limit_hit` carries so a later launch can skip that pair until its 

478 #: reset (`executor.fallback.known_limited`). Only `run_agent_task` sets it. 

479 rate_limit_key: dict | None = None, 

480 #: The `harnesses.yaml` harness this session runs on, recorded on its row 

481 #: for `worker.reattach` (Kraft-9elw1). Only `run_agent_task` sets it. 

482 harness: str | None = None, 

483) -> str: 

484 log_path = run_dirs.logs / f"{session_id}.log" 

485 result_path = result_path_for(run_dirs, files or session_id) 

486 # The task's own exit code, written by the launch wrapper on the way out. 

487 # A sidecar, never `result_path` itself -- see `_resolve_exit_file`. 

488 exit_path = result_path.with_suffix(".exit") 

489 # docker's own sidecar, written only once it created the container. 

490 cidfile = result_path.with_suffix(".cid") 

491 # Captured before any sandbox wrap reassigns `cmd` below (Kraft-s7c04.35) -- 

492 # this must read as what actually ran, never a docker-wrapped invocation. 

493 command_ran = shlex.join(cmd) 

494 

495 await db.write( 

496 lambda c: store.create_session( 

497 c, 

498 id=session_id, 

499 work_item_id=work_item_id, 

500 node_id=node_id, 

501 hook_point=hook_point, 

502 log_path=str(log_path), 

503 result_path=str(result_path), 

504 round=round, 

505 head_sha=head_sha, 

506 thread=thread, 

507 command=command_ran, 

508 harness=harness, 

509 ) 

510 ) 

511 

512 full_env = worker_env(repo_entry, {**(env or {}), "KRAFT_RESULT_PATH": str(result_path)}) 

513 if sandbox: 

514 # The container writes its result here, and `result_path` is mounted 

515 # read-write into it by name -- docker can only bind-mount a file 

516 # that already exists, so create it now (empty) rather than letting 

517 # docker invent a directory at that path. 

518 result_path.touch(exist_ok=True) 

519 # docker refuses a cidfile that already exists. 

520 cidfile.unlink(missing_ok=True) 

521 cmd = _sandbox.docker_argv( 

522 cmd, 

523 cwd, 

524 sandbox, 

525 run_dirs.results, 

526 env={**(repo_entry.env if repo_entry is not None else {}), **(env or {})}, 

527 name=_sandbox.container_name(session_id), 

528 result_path=result_path, 

529 cidfile=cidfile, 

530 ) 

531 # Kraft-qx1q: `create_session` above inserts this row 'pending' with no 

532 # pid yet. `pause_work_item`, `chain.skip_node`, and 

533 # `stop_escalation_session` can all mark a row stopped in the window 

534 # between that insert and here -- SIGTERM has nothing to signal without a 

535 # pid, so without this check Popen launches an untracked agent into a 

536 # worktree the caller believes is idle. One read, right before the one 

537 # place that can actually refuse the launch, covers every caller that 

538 # marks a row stopped: no log file is opened and no process is started. 

539 current_status = await db.write(lambda c: store.session_status(c, session_id)) 

540 if current_status != "pending": 

541 return current_status 

542 log = open(log_path, "w") 

543 try: 

544 try: 

545 # A missing cwd still raises from `Popen` below and is unaffected; 

546 # only the missing-binary half has to move up here, because the 

547 # wrapper puts `/bin/sh` at argv[0] and that always exists. 

548 # Checked inside the `try` so both halves land in the one handler. 

549 if Path(cwd).is_dir() and _missing_executable(cmd[0], cwd, full_env): 

550 raise FileNotFoundError(errno.ENOENT, os.strerror(errno.ENOENT), cmd[0]) 

551 # stdin is DEVNULL for every task, not just agents: a CLI in 

552 # stream-json mode waits on stdin before it starts (measured: a 3s 

553 # "no stdin data received" stall), and no hook has any business 

554 # reading the server's own stdin. 

555 proc = subprocess.Popen( 

556 _wrap_with_exit_file(cmd, exit_path), 

557 cwd=str(cwd), 

558 stdout=log, 

559 stderr=subprocess.STDOUT, 

560 start_new_session=True, 

561 env=full_env, 

562 stdin=subprocess.DEVNULL, 

563 ) 

564 # `start_new_session=True` makes this the group's pgid for life, 

565 # captured now rather than re-derived after the leader exits. 

566 pgid = proc.pid 

567 except FileNotFoundError as exc: 

568 # Popen raises this for a missing cwd as well as a missing 

569 # executable, and the two are fixed in different places. Say which: 

570 # the alternative is what this branch used to leave behind — a 

571 # zero-byte log, a 5ms "failed", and no way to tell them apart. 

572 # Safe to write: in this path the child never took the fd. 

573 missing = "working directory" if not Path(cwd).is_dir() else f"command {cmd[0]!r}" 

574 log.write(f"could not start {' '.join(cmd)} in {cwd}: no such {missing} ({exc})\n") 

575 # Not "failed": a task that never launched is a configuration 

576 # problem, and reporting it as a task failure opens a fix cycle no 

577 # agent can win by editing source (Kraft-579). The caller 

578 # short-circuits this straight to needs_human. 

579 await db.write(lambda c: store.session_exited(c, session_id, "config_error")) 

580 return "config_error" 

581 finally: 

582 log.close() # the child holds its own dup'd fd 

583 # The child owns the log fd, so it keeps writing across a Kraft restart — which 

584 # is the whole premise of reattach. A watcher thread therefore *tails the file* 

585 # rather than draining a pipe: piping through the parent would give the 

586 # surviving agent a SIGPIPE the moment Kraft exited. It records when each new 

587 # line appeared into a sidecar, so the log modal has a time column (design 6c) 

588 # without the plain-text log — read by the copy button and by the envelope 

589 # parser — gaining a prefix. 

590 stop = threading.Event() 

591 watcher = threading.Thread( 

592 target=_watch_log, 

593 args=(log_path, logs.times_path(log_path), stop), 

594 daemon=True, 

595 ) 

596 watcher.start() 

597 

598 try: 

599 try: 

600 pid_start_time = psutil.Process(proc.pid).create_time() 

601 except psutil.Error: 

602 pid_start_time = None 

603 await db.write(lambda c: store.session_running(c, session_id, proc.pid, pid_start_time)) 

604 

605 # Poll instead of `await asyncio.to_thread(proc.wait)`: a blocked thread is 

606 # uncancellable, so on SIGTERM the executor task's cancel() could not 

607 # interrupt it and uvicorn fell back to a hard exit (same fix as 

608 # reattach._adopt). 

609 # 

610 # The same loop, with a second job: read the log forward from our own byte 

611 # offset every `progress_s` and write the running usage onto the session 

612 # row. Before this the row held NULL tokens and a NULL model for the whole 

613 # run, so `usage_rollup` reported 0 for exactly the node a human was 

614 # watching (Kraft-54dk, Kraft-2r8s). 

615 seen_usage: dict[str, _usage.Usage] = {} 

616 log_offset = 0 

617 next_progress = time.monotonic() + progress_s 

618 capped = False 

619 while proc.poll() is None: 

620 await asyncio.sleep(poll_s) 

621 if time_cap is not None and caps.monotonic() >= time_cap.at: 

622 # Left through the `finally` below, which kills the group and 

623 # tears down a sandbox's container. 

624 capped = True 

625 break 

626 if time.monotonic() < next_progress: 

627 continue 

628 next_progress = time.monotonic() + progress_s 

629 # A bad line or a write hiccup here must not kill this task: the child 

630 # keeps running and writing its own log fd regardless, so losing this 

631 # loop silently freezes tokens_in/out for the rest of the session with 

632 # nothing to show it happened (Kraft-41f7). 

633 try: 

634 log_offset, live = _progress_usage(log_path, log_offset, seen_usage, reader) 

635 if live is not None: 

636 await db.write(lambda c, u=live: store.session_progress(c, session_id, u)) 

637 except Exception: 

638 logger.exception("usage progress tick failed for session %s", session_id) 

639 finally: 

640 # A pause or skip cancels this task from inside the poll loop above 

641 # (deps.cancel -> CancelledError), which used to skip straight past 

642 # the stop/join pair that only ever sat after the loop -- leaking the 

643 # watcher thread, and its two open fds, for the rest of the server's 

644 # life. `finally` runs on that path too, so the watcher stops however 

645 # this function leaves the loop. 

646 stop.set() 

647 watcher.join(timeout=2) 

648 # A paused row means a human interrupted this session and `_terminate` 

649 # SIGINT'd it rather than SIGTERM'ing it, so the agent CLI would flush 

650 # the result envelope that carries its cost (Kraft-s7c04.18). The wait 

651 # has to be here and not in the pause route: `_kill_group` below 

652 # SIGTERMs the whole group the moment this loop leaves, which on a 

653 # cancel is milliseconds after the SIGINT, and a sleep in the route 

654 # would run concurrently and protect nothing. The group leader is the 

655 # `/bin/sh` wrapper, not the agent, so `proc.poll()` reaping the 

656 # wrapper says nothing about whether the agent has finished writing. 

657 # `db.write` for a SELECT is deliberate, not a slip: the pause is 

658 # written by the API route concurrently with this coroutine, and 

659 # `db.write` queues onto the serialized writer so it observes that 

660 # write. `db.read` goes to the separate `_reader` connection, whose 

661 # snapshot may predate the pause -- which would skip the flush wait 

662 # and lose exactly the cost this fix exists to keep. 

663 paused = await db.write(lambda c: store.session_status(c, session_id)) == "paused" 

664 if paused and flush_grace: 

665 await _flush_sleep(flush_grace) 

666 # Same cancellation as above: without this inside `finally`, a pause 

667 # or skip propagates CancelledError past the kill calls entirely, and 

668 # a sandboxed session's container -- parented by the docker daemon, 

669 # not `pgid`'s process group -- keeps running against the bind-mounted 

670 # worktree with nothing left to stop it (Kraft-rki). The container 

671 # kill gets its own nested `finally` so a `_kill_group` failure (a 

672 # dead pgid can still raise -- PermissionError, not just 

673 # ProcessLookupError, once its last member is gone) can't shadow it. 

674 try: 

675 await _kill_group(pgid, group_kill_grace, reap=proc.poll) 

676 finally: 

677 if sandbox: 

678 await _sandbox.teardown(session_id) 

679 if paused: 

680 # The cancelled path never reaches `session_exited`, so Task 6's 

681 # hook inside it never fires. Idempotent with that hook for the 

682 # pause that was not cancelled: both write the same numbers and 

683 # both are guarded on the row still being paused. 

684 await db.write( 

685 lambda c: store.record_pause_usage( 

686 c, session_id, _usage.read(log_path, result_path, reader), reader 

687 ) 

688 ) 

689 if capped: 

690 hit = time_cap.hit 

691 

692 def _capped(c): 

693 store.session_exited( 

694 c, 

695 session_id, 

696 "capped_out", 

697 None, 

698 _usage.read(log_path, result_path, reader), 

699 reader=reader, 

700 ) 

701 events.append( 

702 c, 

703 work_item_id, 

704 caps.REACHED, 

705 hit.payload(node_id=node_id, task=hook_point, session_id=session_id), 

706 ) 

707 

708 with open(log_path, "a") as fh: 

709 fh.write(f"\nkraft: stopped: {hit.reason}\n") 

710 await db.write(_capped) 

711 return caps.TIME_CAPPED 

712 returncode = proc.returncode 

713 status = _resolve(result_path, returncode) 

714 # `docker run` itself failing to launch (daemon down, image pull failed) 

715 # is Kraft's own launcher not reaching the sandboxed command at all -- 

716 # the same "config problem, not a task failure" class as the 

717 # FileNotFoundError branch above, one step later and inside the sandboxed 

718 # path only (Kraft-nc9gm). Checked before every other status adjustment 

719 # below so it can't be shadowed by require_result_file or post_resolve. 

720 if sandbox and status == "failed" and _docker_launch_failed(cidfile): 

721 status = "config_error" 

722 # A session that exits clean with no result file at all never reached the 

723 # end of its own contract -- `_resolve`'s exit-code fallback cannot tell 

724 # "no contract" (a plain subprocess hook) from "broke the contract" (an 

725 # agent hook), so the caller who knows which one this is says so 

726 # explicitly (Kraft-avpe). Before the rate-limit check below: a rejected 

727 # launch also has no result file "by construction", and that branch's own 

728 # unconditional overwrite is what protects it, not an exclusion here. 

729 if require_result_file and status == "done" and _resolve_result_file(result_path) is None: 

730 status = "failed" 

731 if require_result_file: 

732 status = await fail_abandoned_jobs(db, session_id, log_path, status, reader) 

733 rate_limit = _rate_limit_rejection(log_path, reader=reader) 

734 if rate_limit is not None: 

735 # A rejected launch produced no artifact by construction, so this 

736 # skips `post_resolve` (agent.py's artifact-presence check) entirely 

737 # rather than let it downgrade an already-correct status to "failed". 

738 status = "rate_limited" 

739 await db.write( 

740 lambda c, rl=rate_limit: events.append( 

741 c, 

742 work_item_id, 

743 "rate_limit_hit", 

744 {**asdict(rl), **(rate_limit_key or {}), "node_id": node_id}, 

745 ) 

746 ) 

747 elif post_resolve is not None: 

748 status = post_resolve(status, log_path, returncode) 

749 # A human pausing the item SIGTERMs this child, so a non-zero rc here may mean 

750 # "stopped on purpose" rather than "failed". The row is the authority. 

751 if await db.write(lambda c: store.session_status(c, session_id)) == "paused": 

752 return "paused" 

753 fields = read_result_fields(result_path) 

754 seen = _usage.read(log_path, result_path, reader) 

755 await db.write( 

756 lambda c: store.session_exited( 

757 c, 

758 session_id, 

759 status, 

760 fields["summary_ref"], 

761 seen, 

762 concerns=fields["concerns"], 

763 question=fields["question"], 

764 reader=reader, 

765 ) 

766 ) 

767 return status
