74def _git_common_dir(path: Path) -> Path | None: 

75 """The `.git` that every working tree of one repository shares, or None. 

76 

77 Identical for a main checkout and every worktree linked to it; different for 

78 an unrelated repo, and different for a submodule of this one (whose common 

79 dir is `<super>/.git/modules/<path>`). `config_mod.git_read` never raises, 

80 so a directory that is not a repo, or is not readable, is None rather than a 

81 500. 

82 """ 

83 common = config_mod.git_read( 

84 path, "rev-parse", "--path-format=absolute", "--git-common-dir", expected_failure=True 

85 ) 

86 return Path(common).resolve() if common else None

89def _attachment_roots(repo: str, cwd: str | None) -> list[Path]: 

90 """The roots an attachment path may be resolved against, in order. 

91 

92 The registered repo always, and first. The caller's own working tree second, 

93 and only when it proves it is a working tree of that same repository — 

94 same `.git`, or no second root (Kraft-85wk). With no `cwd` the reachable set 

95 is exactly what it has always been. 

96 

97 This check is also what enforces Kraft-vwv's decision: an intake attachment 

98 belongs to the item's root worktree only. A `cwd` inside a submodule 

99 contributes no root, because a submodule's common dir is not the 

100 superproject's — so a submodule-local spec cannot be attached by accident, 

101 and `ensure_worktree` copies into the root worktree only. One change has one 

102 spec; N copies in the MR would be N places for it to drift. 

103 """ 

104 root = Path(repo).resolve() 

105 if not cwd: 

106 return [root] 

107 common = _git_common_dir(Path(cwd)) 

108 if common is None or common != _git_common_dir(root): 

109 return [root] 

110 toplevel = config_mod.git_read(Path(cwd), "rev-parse", "--show-toplevel", expected_failure=True) 

111 if toplevel is None: 

112 return [root] 

113 top = Path(toplevel).resolve() 

114 return [root] if top == root else [root, top]

117def _validated_attachments( 

118 repo: str, attachments: list[Attachment], cwd: str | None = None 

119) -> list[dict]: 

120 """Trust boundary: `path` comes from a browser or a local agent and is used 

121 to read a file and to write into a worktree. Resolve under a candidate root 

122 and reject any escape. 

123 

124 The widening `cwd` buys is small and proven: other working trees of the same 

125 repository, for a caller the perimeter middleware has already established is 

126 local and same-site. A path is taken at the first root it both stays inside 

127 and exists under. 

128 """ 

129 kinds = [a.kind for a in attachments] 

130 if len(set(kinds)) != len(kinds): 

131 raise HTTPException(422, "at most one attachment per kind") 

132 roots = _attachment_roots(repo, cwd) 

133 out = [] 

134 for a in attachments: 

135 inside = False 

136 for root in roots: 

137 target = (root / a.path).resolve() 

138 if not target.is_relative_to(root): 

139 continue 

140 inside = True 

141 # Working tree, not HEAD: a document written minutes ago is legal 

142 # input, and ensure_worktree copies it into the worktree. 

143 if not target.is_file(): 

144 continue 

145 entry = {"kind": a.kind, "path": str(target.relative_to(root))} 

146 if root != roots[0]: 

147 # The copy reads `repo / path` by default and this file is not 

148 # in `repo` at all, so it would silently copy nothing and leave 

149 # a trimmed gate with no document. Absolute, so the copy needs 

150 # to know nothing about roots. 

151 entry["source"] = str(target) 

152 out.append(entry) 

153 break 

154 else: 

155 raise HTTPException( 

156 422, 

157 f"attachment not found: {a.path}" 

158 if inside 

159 else f"attachment path escapes the repo: {a.path}", 

160 ) 

161 return out

164@api_router.post("/work-items", status_code=201) 

165async def create_work_item(body: NewWorkItem, request: Request): 

166 st = request.app.state 

167 if st.invalid_policy: 

168 # Spec §9: a malformed policy.yaml makes the process refuse work, same 

169 # posture as an invalid registry — do not accept a run we cannot bound. 

170 detail = "; ".join(st.invalid_policy) 

171 raise HTTPException(503, f"policy config invalid, refusing work: {detail}") 

172 template = st.templates.valid.get( 

173 body.chain_template if body.chain_template is not None else "default" 

174 ) 

175 if template is None: 

176 raise HTTPException(422, "unknown or invalid template") 

177 if body.root_merge_policy not in store.ROOT_MERGE_POLICIES: 

178 raise HTTPException(422, f"unknown root_merge_policy {body.root_merge_policy!r}") 

179 # Before `executor.intake`, which no longer 502s on a bd failure (Kraft-7gy) 

180 # and would file the item with no bead and a warning nobody reads. An 

181 # explicit check rather than `Field(max_length=...)`: pydantic's 422 body is 

182 # a list of error dicts, and `kraft item create` prints `detail` straight 

183 # through -- one sentence is the contract every other CLI error keeps. 

184 if len(body.title) > beads_mod.MAX_TITLE: 

185 raise HTTPException( 

186 422, 

187 f"title is {len(body.title)} characters; the tracker's limit is {beads_mod.MAX_TITLE}", 

188 ) 

189 if not Path(body.repo).is_dir(): 

190 raise HTTPException(422, f"repo path does not exist: {body.repo}") 

191 attachments = _validated_attachments(body.repo, body.attachments, body.cwd) 

192 node_ids = {n["id"] for n in template.nodes} 

193 unknown_skip = set(body.skip_nodes) - node_ids 

194 if unknown_skip: 

195 raise HTTPException(422, f"unknown node id(s) to skip: {sorted(unknown_skip)}") 

196 # A kept node's rebase_bounce_to naming a skipped node is a dangling bounce 

197 # target: walk.py's `next(j for j, n in ... if n["id"] == bounce_to)` has no 

198 # fallback like `reject_target`'s and raises StopIteration mid-run, crashing 

199 # the executor into needs_human. Reject the skip at intake instead. 

200 dangling_bounce = { 

201 n["id"]: n["rebase_bounce_to"] 

202 for n in template.nodes 

203 if n["id"] not in body.skip_nodes and n.get("rebase_bounce_to") in body.skip_nodes 

204 } 

205 if dangling_bounce: 

206 raise HTTPException( 

207 422, 

208 f"cannot skip node(s) {sorted(set(dangling_bounce.values()))}: named as " 

209 f"rebase_bounce_to by kept node(s) {sorted(dangling_bounce)}", 

210 ) 

211 # skip_nodes alone, or together with an attachment's gate trim, must not 

212 # empty the chain: materialize would hand `intake` nothing to run, and 

213 # `create_work_item`/`executor.run_once` both index `nodes[0]` unguarded 

214 # (code-review). The bead is already filed and the run spawned by the 

215 # time either of those would crash, so this has to be checked first. 

216 satisfied = frozenset(ATTACHMENT_GATES[a["kind"]] for a in attachments) 

217 if not materialize(template, satisfied_gates=satisfied, skip_nodes=body.skip_nodes)["nodes"]: 

218 raise HTTPException(422, "skip_nodes would leave no nodes in the chain") 

219 for node_id, fields in body.node_overrides.items(): 

220 if node_id not in node_ids: 

221 raise HTTPException(422, f"unknown node id {node_id!r}") 

222 field_errs = validate_node_override_fields(fields) 

223 if field_errs: 

224 raise HTTPException(422, f"node {node_id!r}: {field_errs[0]}") 

225 try: 

226 wid = await executor.intake( 

227 st.db, 

228 st.run_dirs, 

229 title=body.title, 

230 description=body.description, 

231 repo=body.repo, 

232 template=template, 

233 # The raw request value, not the resolved template's id (Kraft-cd47): 

234 # None here means no explicit template was chosen, and must stay 

235 # None in the row -- `intake`'s own default would otherwise store 

236 # `template.id`, indistinguishable from an item that named 

237 # `chain_template: "default"` outright. 

238 chain_template=body.chain_template, 

239 bd_cwd=deps.bd_cwd(), 

240 submodules=body.submodules, 

241 root_merge_policy=body.root_merge_policy, 

242 attachments=attachments, 

243 status="active" if body.autostart else "paused", 

244 # Folded into intake's own INSERT transaction, not a separate 

245 # `active_count` read here: two autostart creates racing a few 

246 # milliseconds apart must not both see a free slot and both win 

247 # one (Kraft-m43g, Kraft-nxht). `None` when not autostarting -- 

248 # `status` is already "paused" and needs no capacity decision. 

249 limit=(st.policy.max_concurrent if st.policy else 1) if body.autostart else None, 

250 auto_gate=body.auto_gate, 

251 skip_nodes=frozenset(body.skip_nodes), 

252 budget_set="budget_usd" in body.model_fields_set, 

253 budget_usd=body.budget_usd, 

254 node_overrides=body.node_overrides or None, 

255 ) 

256 except Exception as exc: # noqa: BLE001 -- executor.intake raises several unrelated types 

257 # No longer reachable for a bd failure — `executor.intake` degrades 

258 # instead (Kraft-7gy). A 502 here is now a template or a DB problem. 

259 raise HTTPException(502, f"intake failed: {exc}") from exc 

260 

261 # Present only when there is one: a null field on every successful create is 

262 # noise in `kraft item create`'s kv block and in the API. 

263 warning = deps._bead_warning(st, wid) 

264 extra = {"bead_warning": warning} if warning else {} 

265 

266 if not body.autostart: 

267 # Created, not started. `/resume` begins it at node zero, because a NULL 

268 # current_node_id falls through that handler's `next(..., 0)` default. 

269 return {"id": wid, "status": "paused", **extra} 

270 

271 row = st.db.read( 

272 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (wid,)).fetchone() 

273 ) 

274 if row["status"] != "active": 

275 # Lost the capacity race inside `intake`'s own INSERT (Kraft-m43g, 

276 # Kraft-nxht): landed "paused" same as an explicit `not autostart`, 

277 # not rejected -- there is no walk to spawn. 

278 return {"id": wid, "status": row["status"], **extra} 

279 

280 deps.spawn( 

281 request.app, 

282 wid, 

283 deps.guard( 

284 st.db, 

285 wid, 

286 executor.run( 

287 st.db, 

288 st.run_dirs, 

289 work_item_id=wid, 

290 registry=st.registry, 

291 bd_cwd=deps.bd_cwd(), 

292 policy=st.policy, 

293 launch=deps.launch(st, body.repo), 

294 on_approve=deps._on_approve(st), 

295 ), 

296 ), 

297 ) 

298 chain = json.loads(row["chain_definition"]) 

299 return JSONResponse( 

300 status_code=201, 

301 content={ 

302 "id": wid, 

303 "bead_id": row["bead_id"], 

304 "status": row["status"], 

305 "chain_definition": chain, 

306 # the run task advances this asynchronously; before its first write the 

307 # chain still starts at node 0 by definition. 

308 "current_node_id": row["current_node_id"] or chain["nodes"][0]["id"], 

309 **extra, 

310 }, 

311 )

363class WorkItemPatch(BaseModel): 

364 #: Absent means untouched, in every field. This route is still not a 

365 #: general table editor -- a body carrying anything else is ignored, not 

366 #: applied -- but a screen that edits one field must not blank another, 

367 #: so no field has a default that means "clear it" except where the field 

368 #: itself says otherwise. `description: ""` clears the brief; 

369 #: `description` omitted leaves it alone. 

370 title: str | None = None 

371 description: str | None = None 

372 #: A template name switches a not-yet-started item onto that template's 

373 #: own materialized chain (Kraft-gwn6). 404s on an unknown name; 409s once 

374 #: `current_node_id` is set -- the chain is fixed for the life of a 

375 #: started item. 

376 chain_template: str | None = None 

377 #: `None` (default) leaves the override alone. `{}` clears every field 

378 #: back to the template's own binding; a non-empty object *replaces* the 

379 #: whole stored override -- it does not merge with what is already there 

380 #: (Kraft-4k6l). No `current_node_id` restriction, unlike `chain_template`: 

381 #: a model/effort dial can change mid-chain, including on a paused item -- 

382 #: that is the point, making a stuck item cheaper before its next retry. 

383 agent_overrides: dict | None = None 

384 #: Per-node field overrides (UI v2 · 04 point 1). `None` (default) leaves 

385 #: overrides alone. `{}` resets every node to the template -- refused 

386 #: (409) once the item has started. A non-empty object is per node id: 

387 #: `{node_id: {}}` drops that node's overrides, `{node_id: {field: 

388 #: value}}` sets fields on it -- merged into what's already stored, not a 

389 #: whole-object replace (unlike `agent_overrides`), so toggling one 

390 #: node's switch never wipes another's. Refused (409) for any node id 

391 #: that has already started. 

392 node_overrides: dict[str, dict] | None = None 

393 #: Per-item spend cap (point 4). Presence, not value, is what matters: 

394 #: omitted leaves the cap alone; sent as `null` sets an explicit "no 

395 #: cap"; sent as a number sets that cap. Distinguished via 

396 #: `model_fields_set` below, because `None` is both "untouched" and a 

397 #: legal explicit value here. 

398 budget_usd: float | None = None

401def _validate_node_overrides(st, row, patch: dict[str, dict]) -> None: 

402 """422 on an unknown node id or field, 409 on a node that has started 

403 (UI v2 · 04 point 1). `patch == {}` (reset to template) 409s if the item 

404 has started at all -- point 2, "only allowed on unstarted nodes". 

405 """ 

406 chain = json.loads(row["chain_definition"]) 

407 node_ids = {n["id"] for n in chain["nodes"]} 

408 if not patch: 

409 if row["current_node_id"] is not None: 

410 raise HTTPException(409, "work item has already started; overrides cannot be reset") 

411 return 

412 

413 def check(c): 

414 for node_id, fields in patch.items(): 

415 if node_id not in node_ids: 

416 raise HTTPException(422, f"unknown node id {node_id!r}") 

417 field_errs = validate_node_override_fields(fields) 

418 if field_errs: 

419 raise HTTPException(422, f"node {node_id!r}: {field_errs[0]}") 

420 if store.node_started(c, row["id"], node_id): 

421 raise HTTPException(409, f"node {node_id!r} has started; its config is locked") 

422 

423 st.db.read(lambda c: check(c))

426@api_router.patch("/work-items/{wid}") 

427async def update_work_item(wid: str, body: WorkItemPatch, request: Request): 

428 st = request.app.state 

429 row = deps._work_item_row(st, wid) # 404s on an unknown work item, before any 422 

430 fields_set = body.model_fields_set 

431 if ( 

432 body.title is None 

433 and body.description is None 

434 and body.chain_template is None 

435 and body.agent_overrides is None 

436 and body.node_overrides is None 

437 and "budget_usd" not in fields_set 

438 ): 

439 raise HTTPException( 

440 422, 

441 "nothing to patch: send a title, description, chain_template, agent_overrides, " 

442 "node_overrides, or budget_usd", 

443 ) 

444 if body.title is not None and not body.title.strip(): 

445 raise HTTPException(422, "title cannot be empty") 

446 

447 if body.node_overrides is not None: 

448 _validate_node_overrides(st, row, body.node_overrides) 

449 

450 new_chain_definition = None 

451 if body.chain_template is not None: 

452 template = st.templates.valid.get(body.chain_template) 

453 if template is None: 

454 raise HTTPException(404, f"unknown chain template {body.chain_template!r}") 

455 if row["current_node_id"] is not None: 

456 raise HTTPException( 

457 409, "work item has already started; template is fixed for its life" 

458 ) 

459 # The exact expression `executor.intake` uses today, against the 

460 # item's existing attachments -- switching template must not force a 

461 # re-attach to get back a trim the item already earned. 

462 satisfied = frozenset(ATTACHMENT_GATES[a["kind"]] for a in entry.attachments_of(row)) 

463 new_chain_definition = json.dumps(materialize(template, satisfied_gates=satisfied)) 

464 

465 if body.agent_overrides is not None: 

466 errs = validate_agent_overrides(body.agent_overrides) 

467 if errs: 

468 raise HTTPException(422, errs[0]) 

469 

470 def apply(c): 

471 # One write, so a multi-field patch is one transaction and cannot land half. 

472 if body.title is not None: 

473 store.set_title(c, wid, body.title) 

474 if body.description is not None: 

475 store.set_description(c, wid, body.description) 

476 if body.chain_template is not None: 

477 store.set_chain_template(c, wid, body.chain_template, new_chain_definition) 

478 if body.agent_overrides is not None: 

479 store.set_agent_overrides( 

480 c, wid, json.dumps(body.agent_overrides) if body.agent_overrides else None 

481 ) 

482 if body.node_overrides is not None: 

483 store.set_node_overrides(c, wid, body.node_overrides) 

484 if "budget_usd" in fields_set: 

485 store.set_budget(c, wid, body.budget_usd) 

486 

487 await st.db.write(apply) 

488 # `model_dump(exclude_none=True)` would drop an explicit `budget_usd: 

489 # null` along with every untouched field, so build the echo from 

490 # `fields_set` (what the caller actually sent) instead. 

491 return {"id": wid, **{f: getattr(body, f) for f in fields_set}}
