93def _git_common_dir(path: Path) -> Path | None: 

94 """The `.git` that every working tree of one repository shares, or None. 

95 

96 Identical for a main checkout and every worktree linked to it; different for 

97 an unrelated repo, and different for a submodule of this one (whose common 

98 dir is `<super>/.git/modules/<path>`). `config_mod.git_read` never raises, 

99 so a directory that is not a repo, or is not readable, is None rather than a 

100 500. 

101 """ 

102 common = config_mod.git_read( 

103 path, "rev-parse", "--path-format=absolute", "--git-common-dir", expected_failure=True 

104 ) 

105 return Path(common).resolve() if common else None

108def _attachment_roots(repo: str, cwd: str | None) -> list[Path]: 

109 """The roots an attachment path may be resolved against, in order. 

110 

111 The registered repo always, and first. The caller's own working tree second, 

112 and only when it proves it is a working tree of that same repository — 

113 same `.git`, or no second root (Kraft-85wk). With no `cwd` the reachable set 

114 is exactly what it has always been. 

115 

116 This check is also what enforces Kraft-vwv's decision: an intake attachment 

117 belongs to the item's root worktree only. A `cwd` inside a submodule 

118 contributes no root, because a submodule's common dir is not the 

119 superproject's — so a submodule-local spec cannot be attached by accident, 

120 and `ensure_worktree` copies into the root worktree only. One change has one 

121 spec; N copies in the MR would be N places for it to drift. 

122 """ 

123 root = Path(repo).resolve() 

124 if not cwd: 

125 return [root] 

126 common = _git_common_dir(Path(cwd)) 

127 if common is None or common != _git_common_dir(root): 

128 return [root] 

129 toplevel = config_mod.git_read(Path(cwd), "rev-parse", "--show-toplevel", expected_failure=True) 

130 if toplevel is None: 

131 return [root] 

132 top = Path(toplevel).resolve() 

133 return [root] if top == root else [root, top]

136def _validated_attachments( 

137 repo: str, attachments: list[Attachment], cwd: str | None = None 

138) -> list[dict]: 

139 """Trust boundary: `path` comes from a browser or a local agent and is used 

140 to read a file and to write into a worktree. Resolve under a candidate root 

141 and reject any escape. 

142 

143 The widening `cwd` buys is small and proven: other working trees of the same 

144 repository, for a caller the perimeter middleware has already established is 

145 local and same-site. A path is taken at the first root it both stays inside 

146 and exists under. 

147 """ 

148 kinds = [a.kind for a in attachments] 

149 if len(set(kinds)) != len(kinds): 

150 raise HTTPException(422, "at most one attachment per kind") 

151 roots = _attachment_roots(repo, cwd) 

152 out = [] 

153 for a in attachments: 

154 inside = False 

155 for root in roots: 

156 target = (root / a.path).resolve() 

157 if not target.is_relative_to(root): 

158 continue 

159 inside = True 

160 # Working tree, not HEAD: a document written minutes ago is legal 

161 # input, and ensure_worktree copies it into the worktree. 

162 if not target.is_file(): 

163 continue 

164 entry = {"kind": a.kind, "path": str(target.relative_to(root))} 

165 if root != roots[0]: 

166 # The copy reads `repo / path` by default and this file is not 

167 # in `repo` at all, so it would silently copy nothing and leave 

168 # a trimmed gate with no document. Absolute, so the copy needs 

169 # to know nothing about roots. 

170 entry["source"] = str(target) 

171 out.append(entry) 

172 break 

173 else: 

174 raise HTTPException( 

175 422, 

176 f"attachment not found: {a.path}" 

177 if inside 

178 else f"attachment path escapes the repo: {a.path}", 

179 ) 

180 return out

428class WorkItemPatch(BaseModel): 

429 #: Absent means untouched, in every field. This route is still not a 

430 #: general table editor -- a body carrying anything else is ignored, not 

431 #: applied -- but a screen that edits one field must not blank another, 

432 #: so no field has a default that means "clear it" except where the field 

433 #: itself says otherwise. `description: ""` clears the brief; 

434 #: `description` omitted leaves it alone. 

435 title: str | None = None 

436 description: str | None = None 

437 #: A template name switches a not-yet-started item onto that template's 

438 #: own materialized chain (Kraft-gwn6). 404s on an unknown name; 409s once 

439 #: `current_node_id` is set -- the chain is fixed for the life of a 

440 #: started item. 

441 chain_template: str | None = None 

442 #: `None` (default) leaves the override alone. `{}` clears every field 

443 #: back to the template's own binding; a non-empty object *replaces* the 

444 #: whole stored override -- it does not merge with what is already there 

445 #: (Kraft-4k6l). No `current_node_id` restriction, unlike `chain_template`: 

446 #: a model/effort dial can change mid-chain, including on a paused item -- 

447 #: that is the point, making a stuck item cheaper before its next retry. 

448 agent_overrides: dict | None = None 

449 #: Per-node field overrides (UI v2 · 04 point 1). `None` (default) leaves 

450 #: overrides alone. `{}` resets every node to the template -- refused 

451 #: (409) once the item has started. A non-empty object is per node id: 

452 #: `{node_id: {}}` drops that node's overrides, `{node_id: {field: 

453 #: value}}` sets fields on it -- merged into what's already stored, not a 

454 #: whole-object replace (unlike `agent_overrides`), so toggling one 

455 #: node's switch never wipes another's. Refused (409) for any node id 

456 #: that has already started. 

457 node_overrides: dict[str, dict] | None = None 

458 #: Per-item spend cap (point 4). Presence, not value, is what matters: 

459 #: omitted leaves the cap alone; sent as `null` sets an explicit "no 

460 #: cap"; sent as a number sets that cap. Distinguished via 

461 #: `model_fields_set` below, because `None` is both "untouched" and a 

462 #: legal explicit value here. 

463 budget_usd: float | None = None 

464 #: The item's own policy override, as at intake (`NewWorkItem.policy`). 

465 #: `None` (default) leaves it alone, `{}` clears it, and a non-empty 

466 #: object *replaces* the whole stored override. Accepted on any item that 

467 #: has not ended, running or waiting included. A change takes effect at 

468 #: the next node the item enters and at the next observation of a wait it 

469 #: is parked on: every node entry re-reads the row (`walk.walk_node`), and 

470 #: every observation is a fresh entry. 

471 policy: dict | None = None 

472 #: Per kind, like `node_overrides` (Kraft-s7c04.28): `{"spec": PATH}` 

473 #: re-snapshots that kind from PATH, `{"spec": null}` drops it and so 

474 #: restores the gate it trimmed (.29), and a kind not named keeps the copy 

475 #: it has. The chain is re-trimmed from the item's own snapshot, never 

476 #: the live template, so the trim follows the documents both ways and 

477 #: nodes skipped at intake stay skipped. 409 once `current_node_id` is 

478 #: set (a started item's worktree already holds its documents, committed 

479 #: on its branch, and its chain is fixed), or when another write changed 

480 #: the attachments or chain since this one read them. 

481 attachments: dict[Literal["spec", "plan"], str | None] | None = None 

482 #: The caller's working directory, as at intake (`NewWorkItem.cwd`). 

483 cwd: str | None = None

486def _check_node_overrides( 

487 overrides: dict[str, dict], 

488 node_ids: set[str], 

489 nodes: dict | None, 

490 maxima: PolicyMaximaInput | None = None, 

491) -> None: 

492 """422 on an override naming no node of the chain, an unknown field, a 

493 model/effort a node's harness refuses (`overrides.harness_refusal`), or an 

494 `auto_escalate: true` on a gate declaring no `auto_review`. One check for 

495 intake and `PATCH`, so the two doors cannot answer differently (review E 

496 #3). `nodes` maps node id to its `ResolvedNode`; `None` (a legacy row, with 

497 no typed chain) skips the last two checks. 

498 

499 Said in the error, because that is what the caller reads. The override 

500 *permits or suppresses* a reviewing task the chain declares; it does not 

501 name one (`store.effective_nodes`). Accepting `auto_escalate: true` on a 

502 gate with no `auto_review` would persist a switch, answer 2xx, and change 

503 nothing -- the shape a human reads as "I turned it on". 

504 """ 

505 for node_id, fields in overrides.items(): 

506 if node_id not in node_ids: 

507 raise HTTPException(422, f"unknown node id {node_id!r}") 

508 field_errs = validate_node_override_fields(fields) 

509 if field_errs: 

510 raise HTTPException(422, f"node {node_id!r}: {field_errs[0]}") 

511 if nodes is not None and (why := harness_refusal(nodes[node_id], fields)): 

512 raise HTTPException(422, f"node {node_id!r}: {why}") 

513 # A per-item fix-loop bound is an operational value, held to the 

514 # administrator maximum like the item's own policy override is 

515 # (Kraft-3br6j): the two doors must not bound it differently. 

516 for key, name, seconds in ( 

517 ("attempts", "max_attempts", 1), 

518 ("wall_clock_s", "timeout_minutes", 60), 

519 ): 

520 ceiling = getattr(maxima, name, None) 

521 if ( 

522 fields.get(key) is not None 

523 and ceiling is not None 

524 and fields[key] > ceiling * seconds 

525 ): 

526 raise HTTPException( 

527 422, 

528 f"node {node_id!r}: {key} {fields[key]} cannot exceed the administrator " 

529 f"maximum {name} {ceiling}", 

530 ) 

531 if ( 

532 nodes is not None 

533 and fields.get("auto_escalate") is True 

534 and nodes[node_id].auto_review is None 

535 ): 

536 raise HTTPException( 

537 422, 

538 f"node {node_id!r} declares no 'auto_review' task, so agent gate review " 

539 f"cannot be switched on for it: an override permits or suppresses the " 

540 f"reviewer its chain declares, it cannot supply one", 

541 )

544def _validate_node_overrides(st, row, patch: dict[str, dict]) -> None: 

545 """422 on an unknown node id or field, 409 on a node that has started 

546 (UI v2 · 04 point 1). `patch == {}` (reset to template) 409s if the item 

547 has started at all -- point 2, "only allowed on unstarted nodes". 

548 """ 

549 # Node ids come from the shared reader, which answers for either chain shape 

550 # and never raises; only `nodes` needs the typed snapshot, because 

551 # `auto_review` and a task's harness have no legacy equivalent. 

552 node_ids = set(store.chain_node_ids(row)) 

553 v1 = store.materialized_chain_of(row) 

554 nodes = {n.id: n for n in v1.chain.nodes} if v1 is not None else None 

555 if not patch: 

556 if row["current_node_id"] is not None: 

557 raise HTTPException(409, "work item has already started; overrides cannot be reset") 

558 return

565 raise HTTPException(409, f"node {node_id!r} has started; its config is locked") 

566 

567 st.db.read(lambda c: check(c))

570def _item_policy(row, patch: dict | None, new_materialized: str | None): 

571 """The item's own policy override after this PATCH, checked against the 

572 chain it will run: the new template's on a switch, else its own. A switch 

573 re-checks the override already stored, since a path it names may not 

574 exist on the new chain. 422 naming the field refused.""" 

575 if patch is None and new_materialized is None: 

576 return None 

577 chain = ( 

578 MaterializedChain.from_json(new_materialized) 

579 if new_materialized is not None 

580 else store.materialized_chain_of(row) 

581 ) 

582 if chain is None: 

583 raise HTTPException(409, "this work item has no V1 chain to hold a policy override") 

584 try: 

585 # `{}` clears: nothing to check. 

586 wanted = store.policy_override_of(row) if patch is None else (patch or None) 

587 return chain.with_item_policy(wanted).item_policy 

588 except PolicyError as exc: 

589 raise HTTPException(422, str(exc)) from exc

592def _retrimmed(row, filed_kinds: frozenset[str], kinds: frozenset[str]) -> str: 

593 """The item's own snapshot, re-trimmed for `kinds` (Kraft-2fyjt): never the 

594 live template, whose nodes, policy and steering may have changed since 

595 intake. Trimming needs only the snapshot; putting a trimmed node back needs 

596 the chain as it was before the trim, which a snapshot keeps as `untrimmed` 

597 from the moment anything is attached. 409 naming it when that is missing.""" 

598 previous = store.materialized_chain_of(row) 

599 if previous is None: 

600 raise HTTPException(409, "this work item has no V1 chain to re-trim") 

601 untrimmed = previous.untrimmed if filed_kinds else previous.chain.chain 

602 if untrimmed is None and filed_kinds - kinds: 

603 raise HTTPException( 

604 409, 

605 f"this item's snapshot does not carry the nodes its " 

606 f"{', '.join(sorted(filed_kinds))} attachment trimmed at intake (it was filed " 

607 "before snapshots kept them), so dropping one cannot put them back; replace " 

608 "the document instead", 

609 ) 

610 base = ( 

611 ResolvedChain.from_chain(untrimmed, steering=previous.chain.steering) 

612 if untrimmed is not None 

613 else previous.chain 

614 ) 

615 try: 

616 chain = base.trim_for_attachments(kinds) 

617 except ValueError as exc: 

618 raise HTTPException(422, str(exc)) from exc 

619 return dataclasses.replace( 

620 previous, chain=chain, untrimmed=untrimmed if kinds else None 

621 ).to_json()

624@api_router.patch("/work-items/{wid}") 

625async def update_work_item(wid: str, body: WorkItemPatch, request: Request): 

626 st = request.app.state 

627 # 404s on an unknown item and 409s on an ended one (Kraft-6vni1), before any 422 

628 row = deps._live_work_item_row(st, wid) 

629 fields_set = body.model_fields_set 

630 if ( 

631 body.title is None 

632 and body.description is None 

633 and body.chain_template is None 

634 and body.agent_overrides is None 

635 and body.node_overrides is None 

636 and body.policy is None 

637 and body.attachments is None 

638 and "budget_usd" not in fields_set 

639 ): 

640 raise HTTPException( 

641 422, 

642 "nothing to patch: send a title, description, chain_template, agent_overrides, " 

643 "node_overrides, policy, attachments, or budget_usd", 

644 ) 

645 if body.title is not None and not body.title.strip(): 

646 raise HTTPException(422, "title cannot be empty") 

647 

648 if body.node_overrides is not None: 

649 _validate_node_overrides(st, row, body.node_overrides) 

650 

651 filed_kinds = frozenset(a["kind"] for a in entry.attachments_of(row)) 

652 kinds, added = filed_kinds, [] 

653 if body.chain_template is not None: 

654 # `st.library is None` first, through the shared door: a 404 "unknown 

655 # chain template 'default'" for a library that did not parse tells the 

656 # operator their chain id is wrong, which is the one misleading answer 

657 # the 503 was written to replace. The membership check answers 404 only 

658 # once there *is* a library to be absent from. 

659 deps.library_or_503(st) 

660 if body.chain_template not in st.library.chain_ids: 

661 raise HTTPException(404, f"unknown chain template {body.chain_template!r}") 

662 if row["current_node_id"] is not None: 

663 raise HTTPException( 

664 409, "work item has already started; template is fixed for its life" 

665 ) 

666 if body.attachments is not None: 

667 if row["current_node_id"] is not None: 

668 raise HTTPException( 

669 409, 

670 "work item has already started; its worktree already holds the documents " 

671 "it was filed with, so its attachments are fixed for its life", 

672 ) 

673 wanted = [Attachment(kind=k, path=p) for k, p in body.attachments.items() if p] 

674 added = _validated_attachments(row["repo"], wanted, body.cwd) 

675 kinds = (kinds - body.attachments.keys()) | {a["kind"] for a in added} 

676 

677 new_materialized = None 

678 if body.chain_template is not None: 

679 # Re-materialized the way intake would have, attachments included: a 

680 # switch that re-authored a document the item already carries would 

681 # undo the trim it was filed with. The one door that reads the live 

682 # library: the caller asked for a different template. 

683 previous = store.materialized_chain_of(row) 

684 target = previous.target if previous is not None else None 

685 try: 

686 new_materialized = ( 

687 deps.resolve_chain_or_422(st, body.chain_template) 

688 .materialize( 

689 target=target or entry.single_repo_target(row["repo"]), 

690 # The instance and repository layers, never 

691 # `previous.policy`: that one already carries the old 

692 # chain's own override, and layering the new chain's on 

693 # top of it stacks the two (Kraft-yaq99). 

694 effective_policy=deps.item_policy(st, row["repo"], target), 

695 repository_policies=deps.repository_policies(st, target), 

696 # Not the chain's: the repositories' steering the item was 

697 # filed with stays frozen across a template switch. 

698 repository_steering=previous.repository_steering if previous else None, 

699 attachment_kinds=kinds, 

700 ) 

701 .to_json() 

702 ) 

703 except ValueError as exc: 

704 raise HTTPException(422, str(exc)) from exc 

705 elif body.attachments is not None: 

706 new_materialized = _retrimmed(row, filed_kinds, kinds) 

707 

708 item_policy = _item_policy(row, body.policy, new_materialized) 

709 

710 if body.agent_overrides is not None: 

711 errs = validate_agent_overrides(body.agent_overrides) 

712 if errs: 

713 raise HTTPException(422, errs[0]) 

714 v1 = store.materialized_chain_of(row) 

715 if v1 is not None: 

716 nodes = {n.id: n for n in v1.chain.nodes} 

717 if (why := item_harness_refusal(nodes, body.agent_overrides)) is not None: 

718 raise HTTPException(422, why) 

719 

720 def apply(c): 

721 # One write, so a multi-field patch is one transaction and cannot land half. 

722 if body.title is not None: 

723 store.set_title(c, wid, body.title) 

724 if body.description is not None: 

725 store.set_description(c, wid, body.description) 

726 # Attachments first: its compare-and-set is against the row as read. 

727 if body.attachments is not None: 

728 store.set_attachments( 

729 c, 

730 wid, 

731 stored, 

732 new_materialized, 

733 seen=(row["attachments"], row["materialized_chain"]), 

734 ) 

735 if body.chain_template is not None: 

736 store.set_chain_template(c, wid, body.chain_template, new_materialized) 

737 if body.agent_overrides is not None: 

738 store.set_agent_overrides( 

739 c, wid, json.dumps(body.agent_overrides) if body.agent_overrides else None 

740 ) 

741 if body.node_overrides is not None: 

742 store.set_node_overrides(c, wid, body.node_overrides) 

743 if "budget_usd" in fields_set: 

744 store.set_budget(c, wid, body.budget_usd) 

745 if body.policy is not None: 

746 store.set_policy_override(c, wid, item_policy) 

747 

748 filed = stored = entry.attachments_of(row) 

749 won = False 

750 try: 

751 if body.attachments is not None: 

752 try: 

753 stored = entry.replace_attachments( 

754 st.run_dirs, wid, filed, body.attachments, added, repo=row["repo"] 

755 ) 

756 except ValueError as exc: 

757 raise HTTPException(422, str(exc)) from exc 

758 await st.db.write(apply) 

759 won = True 

760 except ValueError as exc: 

761 if body.attachments is None: 

762 raise 

763 # `store.set_attachments`' compare-and-set lost: the item started, or 

764 # another PATCH changed its attachments or chain since `row` was read. 

765 raise HTTPException(409, f"{exc}; nothing was changed") from exc 

766 finally: 

767 # Only this request's own files: the superseded copies if its write 

768 # won, its fresh ones if not. Never "whatever the row does not name", 

769 # which would delete a concurrent PATCH's copies before it writes. 

770 if body.attachments is not None: 

771 keep, drop = (stored, filed) if won else (filed, stored) 

772 entry.discard_attachments(st.run_dirs, wid, drop, keep) 

773 # `model_dump(exclude_none=True)` would drop an explicit `budget_usd: 

774 # null` along with every untouched field, so build the echo from 

775 # `fields_set` (what the caller actually sent) instead. 

776 return {"id": wid, **{f: getattr(body, f) for f in fields_set}}
