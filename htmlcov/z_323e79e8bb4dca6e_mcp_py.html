1"""`kraft admin mcp` — the MCP front door onto `client.py`. 

2 

3A dispatch table and nothing more. Every tool here is a docstring plus one call 

4into `client`, because `kraft <verb>` is the same functions behind a different 

5door and the two must not drift. 

6 

7Docstrings are the tool descriptions an agent reads to decide whether to call 

8something, so they are written for that reader, not for a maintainer. 

9""" 

10

20def build() -> MCPServer: 

21 """The server, tools registered. Split from `serve_stdio` so a test can list 

22 the tools without owning a transport.""" 

23 server = MCPServer("kraft") 

24 

25 @server.tool() 

26 async def list_work_items(status: str | None = None) -> list[dict]: 

27 """List Kraft work items — the board. Optionally filter by status 

28 ("active", "paused", "completed", "failed"). Returns id, title, repo, 

29 status, current node, and any gate waiting on a human.""" 

30 return await client.list_work_items(status) 

31 

32 @server.tool() 

33 async def get_work_item(work_item_id: str | None = None) -> dict: 

34 """Get one Kraft work item. With no id, resolves the work item this 

35 session is standing in — correct when the cwd is a Kraft worktree or the 

36 session was started by Kraft itself.""" 

37 return await client.get_work_item(work_item_id) 

38 

39 @server.tool() 

40 async def get_gate_artifact(work_item_id: str | None = None) -> dict: 

41 """The spec or plan the work item's pending gate is a decision about.""" 

42 return await client.artifact(work_item_id)

44 @server.tool() 

45 async def search(q: str, limit: int = 20) -> dict: 

46 """Search Kraft's cross-repo index of specs, plans, and session 

47 summaries. Use this before writing a spec, to find whether the decision 

48 was already made somewhere else.""" 

49 return await client.search(q, limit)

51 @server.tool() 

52 async def create_work_item( 

53 title: str, 

54 repo: str | None = None, 

55 chain_template: str = "default", 

56 description: str | None = None, 

57 attachments: list[dict] | None = None, 

58 auto_gate: bool = True, 

59 ) -> dict: 

60 """File a new Kraft work item. It is created **paused** and does not run: 

61 a human starts it from the board. Use this to hand finished work off to 

62 Kraft rather than doing it in this session. `repo` defaults to the repo 

63 of the work item this session is standing in. 

64 

65 `description` is the brief — what the work actually is, in prose. The 

66 title is only a label; the spec node writes its design from the 

67 description, so put the intent there rather than packing it into the 

68 title. 

69 

70 `attachments` hands over documents that already exist: 

71 `[{"kind": "spec", "path": ".engineering/specs/x.md"}]`, kind `spec` or 

72 `plan`, at most one of each. An attached document trims the node whose 

73 gate it satisfies, so nobody re-approves what you already agreed, and 

74 the implementing agent is told to follow it rather than guess. A path 

75 is resolved against the repo and against the working tree you are 

76 standing in, so a spec you just wrote in a worktree can be attached as 

77 it is — relative to that tree, or absolute.""" 

78 return await client.create_work_item(title, repo, chain_template, description, attachments)

80 @server.tool() 

81 async def ensure_repo(path: str | None = None) -> dict: 

82 """Connect a repository to Kraft if it is not already connected, so work 

83 items can be created against it. Idempotent — safe to call every time. 

84 `path` defaults to the current working directory.""" 

85 return await client.ensure_repo(path)

87 @server.tool() 

88 async def approve_gate(gate: str | None = None, work_item_id: str | None = None) -> dict: 

89 """Approve the human gate a Kraft work item is waiting on, letting the 

90 chain continue. With no gate name, approves whichever gate is pending. 

91 Gates are spec_approval, plan_approval, chain_finalized, and 

92 human_review_approval. Only a human should decide this — ask first.""" 

93 return await client.approve_gate(gate, work_item_id) 

94 

95 @server.tool() 

96 async def reject_gate( 

97 note: str, 

98 gate: str | None = None, 

99 work_item_id: str | None = None, 

100 node: str | None = None, 

101 ) -> dict: 

102 """Reject the human gate a Kraft work item is waiting on, sending the 

103 chain back to the node that can address the note. `note` says what is 

104 wrong and is required. `node` overrides where the chain re-enters and 

105 must name a node at or before the gate's own; the default is the one 

106 the chain declares. Only a human should decide this — ask first.""" 

107 return await client.reject_gate(note, gate, work_item_id, node) 

108 

109 @server.tool() 

110 async def pause_work_item(work_item_id: str | None = None) -> dict: 

111 """Stop a running Kraft work item's current attempt. Pair with 

112 resume_work_item to redirect work that is going wrong: there is no way to 

113 talk to a running agent, so steering means pausing and resuming.""" 

114 return await client.pause(work_item_id) 

115 

116 @server.tool() 

117 async def report_progress(task: int, work_item_id: str | None = None) -> dict: 

118 """Say which task of the plan you are starting while implementing a 

119 Kraft work item. `task` is the N of the plan's `## Task N` heading. 

120 Defaults to the work item this session is running in.""" 

121 return await client.report_progress(task, work_item_id) 

122 

123 @server.tool() 

124 async def resume_work_item(steer: str | None = None, work_item_id: str | None = None) -> dict: 

125 """Start or restart a paused Kraft work item. `steer` is carried into the 

126 next attempt's prompt. This is also how a work item created by 

127 create_work_item is started for the first time.""" 

128 return await client.resume(steer, work_item_id) 

129 

130 @server.tool() 

131 async def retry_work_item(steer: str | None = None, work_item_id: str | None = None) -> dict: 

132 """Re-run the node a stopped Kraft work item stopped on, with `steer` 

133 carried into the retry's prompt. This is the only way back onto an item 

134 that stopped for a human: resume only takes a paused item.""" 

135 return await client.retry(steer, work_item_id) 

136 

137 @server.tool() 

138 async def skip_work_item(note: str | None = None, work_item_id: str | None = None) -> dict: 

139 """Advance a Kraft work item past its current node or pending gate, 

140 without running or approving it. Works while active (kills the 

141 running session first), paused, or stopped for a human. Only a human 

142 should decide this — ask first.""" 

143 return await client.skip(note, work_item_id) 

144 

145 @server.tool() 

146 async def escalate_work_item(message: str, work_item_id: str | None = None) -> dict: 

147 """Send a message into a Kraft work item's escalation thread — the 

148 door onto a needs_human stop that wants back-and-forth with an agent 

149 rather than a one-shot retry. Resumes the same thread on every later 

150 call for the same item, so whatever was already tried carries 

151 forward.""" 

152 return await client.escalate(message, work_item_id) 

153 

154 @server.tool() 

155 async def set_mr_labels(labels: list[str], work_item_id: str | None = None) -> dict: 

156 """Label this Kraft work item's merge request and re-create its 

157 pipeline. For an `on_failure` repair task that has read a red 

158 `on.ci.poll` and worked out which labels the pipeline wants — this 

159 applies that decision, it does not make it.""" 

160 return await client.mr_labels(labels, work_item_id) 

161 

162 @server.tool() 

163 async def set_chain_template(template: str, work_item_id: str | None = None) -> dict: 

164 """Switch a not-yet-started Kraft work item onto a different chain 

165 template. Only works before the chain has started (no current node 

166 set yet) -- 404s on an unknown template name, 409s once the item is 

167 running.""" 

168 return await client.set_chain_template(template, work_item_id) 

169 

170 @server.tool() 

171 async def set_agent_overrides( 

172 model: str | None = None, 

173 escalate_model: str | None = None, 

174 effort: str | None = None, 

175 clear: bool = False, 

176 work_item_id: str | None = None, 

177 ) -> dict: 

178 """Set or clear a Kraft work item's own model/effort override, applied 

179 to every agent node in its chain without changing the chain itself. 

180 `clear` resets every field back to the template's own binding; naming 

181 a field replaces the whole stored override rather than merging with 

182 it.""" 

183 return await client.set_agent_overrides( 

184 model, escalate_model, effort, clear=clear, work_item_id=work_item_id 

185 ) 

186 

187 @server.tool() 

188 async def set_node_overrides( 

189 node_id: str, 

190 auto_escalate: bool | None = None, 

191 auto_escalate_stuck: bool | None = None, 

192 auto_escalate_delay_s: int | None = None, 

193 clear: bool = False, 

194 work_item_id: str | None = None, 

195 ) -> dict: 

196 """Set or clear one node's per-item auto-escalate override on a Kraft 

197 work item, without touching the Policy screen's system defaults or the 

198 chain template everyone else uses. `clear` resets this node back to 

199 the template's own binding; naming a field replaces the whole stored 

200 override for that node rather than merging with it. 409s once the 

201 node has started.""" 

202 return await client.set_node_overrides( 

203 node_id, 

204 auto_escalate, 

205 auto_escalate_stuck, 

206 auto_escalate_delay_s, 

207 clear=clear, 

208 work_item_id=work_item_id, 

209 ) 

210 

211 @server.tool() 

212 async def permission_request( 

213 tool_name: str, input: dict, tool_use_id: str | None = None 

214 ) -> dict: 

215 """Not for you to call directly. Kraft passes this tool to the agent CLI 

216 as `--permission-prompt-tool`, and the CLI calls it when it wants to ask 

217 whether a tool use is allowed. It answers from the permission grant on 

218 the node this session is running, and records the decision on the work 

219 item.""" 

220 return await client.permission_request(tool_name, input, tool_use_id) 

221 

222 return server

225def serve_stdio() -> None: 

226 # Tag every call this process makes through `kraft.client` as MCP's 

227 # (Kraft-s7c04.43). `setdefault`, so an embedder that has already named 

228 # itself keeps its own answer. 

229 os.environ.setdefault("KRAFT_CLIENT", "mcp") 

230 build().run(transport="stdio")
