27def _refusals_reach_the_agent(fn: Callable[..., Awaitable[Any]]) -> Callable[..., Awaitable[Any]]: 

28 """`fn`, with Kraft's refusals raised as `ToolError`. 

29 

30 The MCP SDK treats any other exception as a crash: the agent reads only 

31 "Error executing tool <name>" and the reason stays in the server log. The 

32 client raises `ValueError` for every API refusal (a 404, a 409, no work 

33 item to act on) and `PermissionError` for the worker self-action guard, 

34 and an agent needs that sentence to recover. So both go out as the line 

35 `kraft <verb>` prints. Anything else is still a crash, logged with its 

36 traceback. 

37 """ 

38 

39 @functools.wraps(fn) 

40 async def tool(*args: Any, **kwargs: Any) -> Any: 

41 try: 

42 return await fn(*args, **kwargs) 

43 except (ValueError, PermissionError) as exc: 

44 raise ToolError(client.refusal(exc)) from exc 

45 

46 return tool

49class _Server(MCPServer): 

50 """`MCPServer`, with every tool it registers wrapped by 

51 `_refusals_reach_the_agent`, so no tool can be added without it.""" 

52 

53 def tool(self, *args: Any, **kwargs: Any) -> Callable[[Callable], Callable]: 

54 register = super().tool(*args, **kwargs) 

55 

56 def wrap(fn: Callable) -> Callable: 

57 # The wrapper awaits `fn`, so a sync tool would fail on every call 

58 # rather than here, where its author sees it. 

59 if not inspect.iscoroutinefunction(fn): 

60 raise TypeError(f"MCP tool {fn.__name__} must be `async def`") 

61 return register(_refusals_reach_the_agent(fn)) 

62 

63 return wrap

106 @server.tool() 

107 async def create_work_item( 

108 title: str, 

109 repo: str | None = None, 

110 chain: str | None = None, 

111 description: str | None = None, 

112 attachments: list[dict] | None = None, 

113 auto_gate: bool = True, 

114 implements_beads: list[str] | None = None, 

115 depends_on: list[str] | None = None, 

116 policy: dict | None = None, 

117 base_branch: str | None = None, 

118 skip_nodes: list[str] | None = None, 

119 budget_usd: float | None = None, 

120 node_overrides: dict | None = None, 

121 chain_template: str | None = None, 

122 ) -> dict: 

123 """File a new Kraft work item. It is created **paused** and does not run: 

124 a human starts it from the board. Use this to hand finished work off to 

125 Kraft rather than doing it in this session. `repo` defaults to the repo 

126 of the work item this session is standing in. `chain` defaults to the 

127 repo's `default_chain`, else `default`. 

128 

129 `description` is the brief — what the work actually is, in prose. The 

130 title is only a label; the spec node writes its design from the 

131 description, so put the intent there rather than packing it into the 

132 title. 

133 

134 `attachments` hands over documents that already exist: 

135 `[{"kind": "spec", "path": ".engineering/specs/x.md"}]`, kind `spec` or 

136 `plan`, at most one of each. An attached document trims the node whose 

137 gate it satisfies, so nobody re-approves what you already agreed, and 

138 the implementing agent is told to follow it rather than guess. A path 

139 is resolved against the repo and against the working tree you are 

140 standing in, so a spec you just wrote in a worktree can be attached as 

141 it is — relative to that tree, or absolute. 

142 

143 `implements_beads` are bead ids this item implements; they are closed 

144 when it completes. Ids mentioned in the description are not parsed — 

145 naming a bead in prose promises nothing. 

146 

147 `depends_on` are work item ids this item comes after. A human still 

148 starts it; started while one of them is unfinished it is blocked, and 

149 Kraft starts it when they complete. They cannot be added later. 

150 

151 `policy` is the item's own policy override, as `set_work_item_policy` 

152 takes it. Leave it out unless a human asked for one. 

153 

154 `base_branch` is the branch the work starts from and its merge request 

155 targets -- a release branch, say. Unset, it is the repo's default 

156 branch. It must already exist on the repo's origin. 

157 

158 `skip_nodes` drops named nodes from the item's chain; `budget_usd` caps 

159 its spend in dollars (unset, the policy's cap applies -- this door can 

160 set a cap but not lift one); `node_overrides` is `{node_id: {field: 

161 value}}`, the fields `set_node_overrides` takes. Leave all three out 

162 unless a human asked for them. `chain_template` is `chain`'s 1.x 

163 name, still read; pass `chain`.""" 

164 return await client.create_work_item( 

165 title, 

166 repo=repo, 

167 chain=chain or chain_template, 

168 description=description, 

169 attachments=attachments, 

170 auto_gate=auto_gate, 

171 implements_beads=implements_beads, 

172 depends_on=depends_on, 

173 policy=policy, 

174 base_branch=base_branch, 

175 skip_nodes=skip_nodes, 

176 node_overrides=node_overrides, 

177 **({"budget_usd": budget_usd} if budget_usd is not None else {}), 

178 )
