1"""Fire a delayed `auto_escalate`/`auto_escalate_stuck` once its configured 

2wait has elapsed (docs/superpowers/specs/2026-09-12-auto-escalate-delay-and-preemption-design.md). 

3 

4Both mechanisms already carry their own "has the delay elapsed" check 

5(`executor.gates.review_gates`/`auto_escalate_stuck`, Kraft-vyk8) and already 

6run inline, synchronously, right after every step in 

7`walk.run`/`resuming.resume` -- so at `auto_escalate_delay_s == 0` (the 

8default) that inline call always fires immediately, exactly as before this 

9module existed. This poller is the backstop for a delay greater than 0: the 

10item that inline call left alone because the delay hadn't elapsed yet has 

11nothing else watching it once that call returns, so something has to 

12re-check it once the delay passes. Same fixed-interval shape as 

13`rate_limit_retry.poller`/`waits.poller`. 

14 

15Always on, like those two: a delay is a per-node config an operator opts 

16into, but once they have, nothing except this poller ever revisits an item 

17sitting past it -- there is no coroutine left waiting on it the way the 

18pre-poller inline call briefly was. 

19""" 

20 

21from __future__ import annotations 

22 

23import asyncio 

24import logging 

25 

26from kraft import events, store 

27from kraft.executor import gates 

28 

29logger = logging.getLogger(__name__) 

30 

31#: Same fixed cadence `rate_limit_retry`/`waits` use -- no per-operator 

32#: tuning knob for this either. 

33_INTERVAL_S = 30 

34 

35 

36async def tick(app) -> list[str]: 

37 """One poll over every `needs_human` row -- the one DB status value both 

38 a pending gate and a genuine stuck stop share (`db.py`'s work_items CHECK 

39 constraint has no separate `awaiting_gate` value; `"awaiting_gate"` only 

40 ever exists as the in-memory status string `walk`/`resuming` hand 

41 `review_gates`, never as a column value). `gates.pending_gate` is what 

42 tells the two apart, the same door `review_gates`/`auto_escalate_stuck` 

43 already use for it: a row with a pending gate goes through 

44 `gates.review_gates` (seeded with the `"awaiting_gate"` status string, 

45 same as the inline caller), everything else through 

46 `gates.auto_escalate_stuck`. Both are safe to call unconditionally -- 

47 each re-checks its own preconditions (pending gate, reason, cap, budget, 

48 and now elapsed delay) and returns its `status` unchanged if nothing is 

49 due, the same "a delayed re-check that finds the status already changed 

50 simply doesn't match its own scan" guarantee a human's own action 

51 already relies on. 

52 

53 Skips a row with a live task already in `app.state.tasks`: that is 

54 either the original inline call from `run()`/`resume()` still in flight, 

55 or a previous tick's own dispatch that hasn't finished yet, and a second 

56 concurrent `review_gates`/`auto_escalate_stuck` call for the same item is 

57 exactly the failure both functions' own docstrings warn against. The row 

58 settles once that task's `deps.spawn` done-callback pops it, and the next 

59 tick (or the item's own resolution) picks it up from there. 

60 

61 Returns the work item ids this tick spawned a check against, which is 

62 usually none -- most calls land on an item whose delay has not elapsed, 

63 or that a human already cleared, and that check returns its status 

64 unchanged. 

65 """ 

66 from kraft.api import deps 

67 

68 st = app.state 

69 attempted: list[str] = [] 

70 # Same bound `intake.tick`/manual resume already enforce: a dispatch here 

71 # launches a real agent session, same as a resume does, so it must count 

72 # against the same `max_concurrent` slots rather than opening one per 

73 # `needs_human` row regardless of how many are already running (code 

74 # review finding) -- otherwise a backlog parked before this feature 

75 # existed launches all at once the first tick after a server start. 

76 # 

77 # `active_count` alone only bounds one tick: `escalate.dispatch` and 

78 # `gate_review.review` both leave the row `needs_human`/`awaiting_gate` 

79 # for the whole agent turn, so a session this poller spawned is invisible 

80 # to it -- the next tick's `active_count` read would not see it and would 

81 # hand out a fresh batch of slots on top of the ones still running (code 

82 # review finding). `app.state.tasks` is where every dispatch, from this 

83 # poller or any other, actually lives for as long as it runs (`deps.spawn` 

84 # populates it, its done-callback empties it), so union it with the 

85 # `active`-status rows rather than trust either alone: a row can be 

86 # `active` before its task lands in `st.tasks` (claim, then spawn) or 

87 # have a live task while its row is still `needs_human` (this poller's 

88 # own case), and `OR` over both catches whichever is true without 

89 # double-counting a row that happens to satisfy both. 

90 live = tuple(st.tasks) 

91 placeholders = ",".join("?" * len(live)) 

92 busy = st.db.read( 

93 lambda c: c.execute( 

94 "SELECT COUNT(*) FROM work_items WHERE status = 'active'" 

95 + (f" OR id IN ({placeholders})" if live else ""), 

96 live, 

97 ).fetchone()[0] 

98 ) 

99 slots = int(st.policy.max_concurrent if st.policy else 1) - busy 

100 if slots <= 0: 

101 return attempted 

102 # ORDER BY random() -- a row this poller can never resolve on its own (a 

103 # human-only gate, a needs_context stop, an already-capped run) stays 

104 # `needs_human` forever. Without a rotating scan order it would sit at a 

105 # stable rowid position at the head of every tick's results and eat every 

106 # slot below, starving rows behind it for good (code review finding). 

107 # Shuffling means a blocked row wins some ticks and loses others, so the 

108 # rest of the backlog still gets its turn. 

109 due = st.db.read( 

110 lambda c: c.execute( 

111 "SELECT * FROM work_items WHERE status = 'needs_human' ORDER BY RANDOM()" 

112 ).fetchall() 

113 ) 

114 default_delay = int(st.policy.auto_escalate_delay_s if st.policy else 0) 

115 for row in due: 

116 if slots <= 0: 

117 break 

118 wid = row["id"] 

119 if wid in st.tasks: 

120 continue 

121 # Only rows that actually opted into a delay (code review finding). At 

122 # the default 0 the inline call from `run()`/`resume()` already fired 

123 # immediately and there is nothing left for a backstop to back-stop -- 

124 # re-checking those here would hand a stuck item a *fresh* 

125 # `auto_escalate_stuck` turn every `_INTERVAL_S` (up to the cap) where 

126 # before this module it got one per `run()`/`resume()`. That is the 

127 # opposite of this feature's "zero regression at the default". 

128 if store.effective_auto_escalate_delay_s(row, default_delay) <= 0: 

129 continue 

130 evts = st.db.read(lambda c, wid=wid: events.read_after(c, 0, wid)) 

131 gate = gates.pending_gate(st.db, wid, evts=evts) 

132 # Armed and actually due, checked here rather than left to the 

133 # spawned call (code review finding): an unarmed or already-handled 

134 # row can only return its status unchanged, and spawning it anyway 

135 # spends a `max_concurrent` slot the rows that *are* due needed, and 

136 # briefly installs a task under this wid that 409s a human's retry. 

137 try: 

138 if not gates.auto_check_due(row, gate, evts, st.policy): 

139 continue 

140 except LookupError: 

141 # A legacy row with a pending gate and a delay > 0: `auto_check_due` 

142 # walks the V1 chain and `walk.chain_of` raises for a row the legacy 

143 # intake path wrote. Per row, not per tick -- uncaught it aborted the 

144 # whole scan every `_INTERVAL_S` and starved every other due row 

145 # behind it. Not a compatibility path: nothing is retried against the 

146 # legacy shape, the row is simply skipped, which is what it already 

147 # gets from every other V1-converted door. 

148 continue 

149 status, fn = ( 

150 ("awaiting_gate", gates.review_gates) 

151 if gate is not None 

152 else ("needs_human", gates.auto_escalate_stuck) 

153 ) 

154 deps.spawn( 

155 app, 

156 wid, 

157 deps.guard( 

158 st.db, 

159 wid, 

160 fn( 

161 status, 

162 st.db, 

163 st.run_dirs, 

164 work_item_id=wid, 

165 policy=st.policy, 

166 launch=deps.launch(st, row["repo"]), 

167 bd_cwd=deps.bd_cwd(), 

168 on_approve=deps._on_approve(st), 

169 ), 

170 ), 

171 ) 

172 attempted.append(wid) 

173 slots -= 1 

174 return attempted 

175 

176 

177async def poller(app) -> None: 

178 """`tick` on a fixed interval until cancelled.""" 

179 while True: 

180 await asyncio.sleep(_INTERVAL_S) 

181 try: 

182 await tick(app) 

183 except asyncio.CancelledError: 

184 raise 

185 except Exception: # noqa: BLE001 -- a bad tick must not end the poller 

186 logger.exception("auto-escalate delay tick failed")
