17#: Lines kept: a chatty command must not grow the panel without bound. 

18MAX_LINES = 5000

20 

21class OutputPanel(ModalScreen[None]): 

22 """Modal, so it holds the focus while open: its `x` never reaches the 

23 view underneath. Installed once (`KraftConsole.SCREENS`), so closing it 

24 keeps what it shows for the next Ctrl+O.""" 

25 

26 BINDINGS: ClassVar[list[BindingType]] = [ 

27 Binding("escape,q", "dismiss", "close"), 

28 Binding("x", "stop", "stop"), 

29 ] 

30 

31 def __init__(self) -> None: 

32 super().__init__() 

33 self._pending: list[Text] = [] 

34 

35 def compose(self) -> ComposeResult: 

36 with Vertical(id="output"): 

37 yield RichLog(id="output-log", wrap=True, max_lines=MAX_LINES) 

38 with Horizontal(id="output-bar"): 

39 yield Button("Stop", id="stop", disabled=True) 

40 yield Static("x stops the command · Esc closes", id="output-keys") 

41 

42 def on_mount(self) -> None: 

43 self.query_one(RichLog).focus() 

44 

45 def begin(self, argv: list[str]) -> None: 

46 self._flush() 

47 log = self.query_one(RichLog) 

48 if log.lines: 

49 log.write("") 

50 log.write(Text(f"kraft {shlex.join(argv)}", style="bold")) 

51 self.query_one("#stop", Button).disabled = False 

52 

53 def write_line(self, line: str) -> None: 

54 # One write per burst, not per line: 100k lines would stall the loop. 

55 # from_ansi, not markup: a child's own colours draw, and `[/]` is text. 

56 # The `\r` of CRLF output would blank the line: from_ansi keeps what follows the last one. 

57 if not self._pending: 

58 self.app.call_later(self._flush) 

59 self._pending.append(Text.from_ansi(line.removesuffix("\r"))) 

60 

61 def _flush(self) -> None: 

62 lines, self._pending = self._pending, [] 

63 if lines: 

64 self.query_one(RichLog).write(Text("\n").join(lines)) 

65 

66 def end(self, code: int) -> None: 

67 self._flush() 

68 theme = self.app.current_theme 

69 colour = theme.error if code else self.app.theme_variables["kraft-muted"] 

70 self.query_one(RichLog).write(Text(f"exit {code}", style=colour)) 

71 self.query_one("#stop", Button).disabled = True 

72 

73 def note(self, text: str) -> None: 

74 """A muted line of the console's own, among the command's.""" 

75 self._flush() 

76 self.query_one(RichLog).write(Text(text, style=self.app.theme_variables["kraft-muted"])) 

77 

78 def action_stop(self) -> None: 

79 self.app.stop_command() 

80 

81 @on(Button.Pressed, "#stop") 

82 def _stop(self) -> None: 

83 self.app.stop_command()
