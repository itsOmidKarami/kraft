1"""A person takes over a stopped item's agent session (`kraft item takeover`). 

2 

3A takeover is a lock on the item's row, held by the CLI's own pid. The server 

4checks and takes it in one write (`prepare`), the CLI runs the agent in the 

5foreground on the person's terminal, and the lock is released when the CLI 

6exits (`release`) -- or, when the CLI was killed hard, by the next check that 

7finds its pid gone (`live`). No `worker_sessions` row is written: a row would 

8be adopted by `reattach`, counted by the local worker, stopped by every stop 

9loop and shown by the board, none of which is a person's terminal. 

10 

11The item's status never changes while a person has it. 

12"""

60def cwd_of(db, run_dirs, row) -> Path: 

61 """The folder the agent runs in: a workspace's root, else the item's own 

62 folder (its worktree, or the checkout when it runs in place).""" 

63 from kraft import inplace 

64 

65 root = db.read( 

66 lambda c: c.execute( 

67 "SELECT repo_path FROM work_item_repos WHERE work_item_id = ? AND role = 'root'", 

68 (row["id"],), 

69 ).fetchone() 

70 ) 

71 return Path(root["repo_path"]) if root else inplace.item_folder(run_dirs, row)

75 for node in nodes:

83def _latest_agent_session(db, row, node): 

84 """The node's most recent agent session row, which picks the task and the 

85 harness, or None when no agent task of it has run.""" 

86 paths = [t.path for t in node.tasks() if isinstance(t.task, AgentTask)] 

87 if not paths: 

88 return None, {} 

89 session = db.read( 

90 lambda c: c.execute( 

91 "SELECT * FROM worker_sessions WHERE work_item_id = ? AND node_id = ? " 

92 f"AND hook_point IN ({marks(paths)}) ORDER BY created_at DESC, rowid DESC LIMIT 1", 

93 (row["id"], node.id, *paths), 

94 ).fetchone() 

95 ) 

96 return session, {t.path: t for t in node.tasks()}

104def _live_worker(conn: sqlite3.Connection, wid: str) -> bool: 

105 rows = conn.execute( 

106 "SELECT pid, pid_start_time FROM worker_sessions WHERE work_item_id = ? " 

107 f"AND status IN ({in_list(LIVE)}) AND pid IS NOT NULL", 

108 (wid,),

110 return any(proc_identity.identity_ok(r["pid"], r["pid_start_time"]) for r in rows)

119def _tip_of(cwd: Path, branch: str) -> str: 

120 done = _git(cwd, "rev-parse", "--verify", "--quiet", f"refs/heads/{branch}")

124async def prepare( 

125 db, 

126 run_dirs, 

127 row, 

128 harnesses: _harness.HarnessSet, 

129 *, 

130 launch, 

131 pid: int, 

132 pid_start: float, 

133 starting: set[str], 

134 walk_running: bool, 

135) -> Prepared: 

136 """Refuse a takeover that can't go ahead, or take the lock and say how to 

137 start the agent. Raises `TakeoverRefused`. 

138 

139 What can change under the check -- the item's status, another takeover, a 

140 worker still dying -- is checked again inside the write that takes the lock, 

141 so two takeovers at once leave exactly one of them holding it.""" 

142 wid = row["id"] 

143 status = row["status"] 

144 if status not in ("paused", "needs_human"): 

145 raise TakeoverRefused(f"work item is {status}; takeover needs paused or needs_human") 

146 try: 

147 policy = dispatch.item_sandbox(row, launch) 

148 except dispatch.InPlaceSandboxed: 

149 policy = None # an in-place item runs unsandboxed whatever is configured 

150 except dispatch.SandboxUnresolved as exc: 

151 raise TakeoverRefused(str(exc)) from exc 

152 if policy is not None: 

153 raise TakeoverRefused(f"takeover doesn't support sandboxed items yet ({policy['kind']})") 

154 

155 chain = store.materialized_chain_of(row) 

156 node = ( 

157 _find_node(chain.chain.nodes, row["current_node_id"]) 

158 if chain and row["current_node_id"] 

159 else None 

160 ) 

161 session, tasks = _latest_agent_session(db, row, node) if node is not None else (None, {}) 

162 if session is None: 

163 raise TakeoverRefused("nothing has run on this step yet; resume it instead") 

164 task = tasks[session["hook_point"]] 

165 

166 machine = session["runner"] or placement.runner_for(row, task.runner, node.id) 

167 if machine not in (placement.LOCAL, placement.ANY): 

168 raise TakeoverRefused(f"this step ran on {machine}; takeover needs a local worktree") 

169 

170 h = harnesses.valid.get(session["harness"] or _DEFAULT_HARNESS) 

171 if h is None: 

172 raise TakeoverRefused(f"harness {session['harness']!r} is not available") 

173 if h.interactive is None: 

174 # The same words `build_interactive_argv` refuses with. 

175 raise TakeoverRefused(f"takeover isn't supported for {h.id} yet") 

176 

177 step = next((s for s in node.steps_in() if any(t.path == task.path for t in s.tasks)), None) 

178 agents = [t.path for t in step.tasks if isinstance(t.task, AgentTask)] if step else [] 

179 if len(agents) > 1: 

180 raise TakeoverRefused( 

181 f"this step has several tasks ({', '.join(agents)}); takeover needs one" 

182 ) 

183 

184 if (lock := await live(db, run_dirs, wid, harnesses)) is not None: 

185 raise TakeoverRefused(f"a person has this item's session open (since {since(lock)})") 

186 still_stopping = "the item's last session is still stopping; try again in a moment" 

187 if walk_running or wid in starting: 

188 raise TakeoverRefused(still_stopping) 

189 if escalate.escalation_running(db, wid): 

190 raise TakeoverRefused("an escalation is running on this item") 

191 

192 # The session to fork: the task's own paused one, found as a resume finds it. 

193 resumed = None 

194 if h.interactive.resume is not None: 

195 resumed = dispatch._resumable_session(db, wid, node, task, h) 

196 cwd = cwd_of(db, run_dirs, row) 

197 if not cwd.is_dir(): 

198 raise TakeoverRefused(f"this work item's worktree is not on disk ({cwd})") 

199 branch = store.branch_for(row) 

200 tip = await asyncio.to_thread(_tip_of, cwd, branch) 

201 minted = str(uuid.uuid4()) 

202 argv = _harness.build_interactive_argv( 

203 h, session_id=minted, resume=resumed[1] if resumed else None 

204 ) 

205 resumed_from = resumed[0] if resumed else None 

206 

207 def take(conn: sqlite3.Connection) -> None: 

208 current = conn.execute("SELECT status FROM work_items WHERE id = ?", (wid,)).fetchone() 

209 if current is None or current["status"] != status: 

210 raise TakeoverRefused( 

211 f"work item is {current['status'] if current else 'gone'}; " 

212 "takeover needs paused or needs_human" 

213 ) 

214 if (held := store.takeover_lock(conn, wid)) is not None: 

215 raise TakeoverRefused(f"a person has this item's session open (since {since(held)})") 

216 if _live_worker(conn, wid): 

217 raise TakeoverRefused(still_stopping) 

218 store.take_takeover_lock( 

219 conn, 

220 wid, 

221 pid=pid, 

222 pid_start=pid_start, 

223 session_id=minted, 

224 resumed_from=resumed_from, 

225 branch=branch, 

226 tip=tip, 

227 ) 

228 events.append( 

229 conn, 

230 wid, 

231 WorkItemEvent.TAKEOVER_STARTED, 

232 {"session_id": minted, "resumed_from": resumed_from}, 

233 ) 

234 

235 await db.write(take) 

236 return Prepared(argv=argv, cwd=str(cwd), session_id=minted, resumed_from=resumed_from)

239async def live(db, run_dirs, work_item_id: str, harnesses: _harness.HarnessSet): 

240 """The item's takeover lock when the CLI that took it is still running. 

241 A lock whose pid is gone (a SIGKILLed CLI, a closed laptop) is released the 

242 way the CLI would have, so it never wedges the item; None then.""" 

243 lock = db.read(lambda c: store.takeover_lock(c, work_item_id)) 

244 if lock is None: 

245 return None 

246 if proc_identity.identity_ok(lock["takeover_pid"], lock["takeover_pid_start"]): 

247 return lock 

248 await release(db, run_dirs, work_item_id, harnesses) 

249 return None

252async def held(db, run_dirs, work_item_id: str): 

253 """`live`, for a caller with no harnesses at hand: the lock a running CLI 

254 holds, or None. Reads the harness files only when a lock is there.""" 

255 if db.read(lambda c: store.takeover_lock(c, work_item_id)) is None: 

256 return None 

257 return await live(db, run_dirs, work_item_id, _harness.load(None))

260def _warnings(cwd: Path, lock) -> list[str]: 

261 """What a person left behind that the next run would trip over.""" 

262 if not cwd.is_dir(): 

263 return [] 

264 out = [] 

265 branch, tip = lock["takeover_branch"], lock["takeover_tip"] 

266 if _git(cwd, "status", "--porcelain").stdout.strip(): 

267 out.append("uncommitted changes in the worktree") 

268 if _git(cwd, "symbolic-ref", "--quiet", "HEAD").stdout.strip() != f"refs/heads/{branch}": 

269 out.append(f"HEAD is not on {branch}") 

270 if ( 

271 tip 

272 and _git(cwd, "merge-base", "--is-ancestor", tip, f"refs/heads/{branch}").returncode == 1 

273 ): 

274 out.append("history was rewritten since the takeover started") 

275 return out

278def _session_recorded(h: _harness.Harness | None, session_id: str) -> bool: 

279 """The takeover's session was written: a person who quit before any turn 

280 left no file, and there is nothing to continue.""" 

281 pattern = h.interactive.session_file if h and h.interactive else None 

282 if not pattern: 

283 return False 

284 return bool(glob.glob(os.path.expanduser(pattern.replace("{value}", session_id))))

287async def release(db, run_dirs, work_item_id: str, harnesses: _harness.HarnessSet) -> Released: 

288 """End the item's takeover: clear the lock, leave the session on the worker 

289 row it forked from when it was written, and report what to check. A 

290 release with no lock held is a no-op, so the CLI's own and a stale-lock 

291 sweep can race.""" 

292 row = db.read( 

293 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (work_item_id,)).fetchone() 

294 ) 

295 lock = db.read(lambda c: store.takeover_lock(c, work_item_id)) 

296 if row is None or lock is None: 

297 return Released(status=row["status"] if row else "", recorded=False, warnings=[]) 

298 session_id = lock["takeover_session_id"] 

299 forked = lock["takeover_resumed_from"] 

300 warnings = await asyncio.to_thread(_warnings, cwd_of(db, run_dirs, row), lock) 

301 recorded = False 

302 if forked is not None: 

303 source = db.read( 

304 lambda c: c.execute( 

305 "SELECT harness FROM worker_sessions WHERE work_item_id = ? AND id = ?", 

306 (work_item_id, forked), 

307 ).fetchone() 

308 ) 

309 h = harnesses.valid.get((source["harness"] if source else None) or _DEFAULT_HARNESS) 

310 recorded = await asyncio.to_thread(_session_recorded, h, session_id) 

311 

312 def end(conn: sqlite3.Connection) -> bool: 

313 held = store.takeover_lock(conn, work_item_id) 

314 if held is None or held["takeover_session_id"] != session_id: 

315 return False # released, or taken again, by someone else meanwhile 

316 store.clear_takeover_lock(conn, work_item_id) 

317 if recorded and forked is not None: 

318 store.set_cli_session_id(conn, work_item_id, forked, session_id) 

319 events.append( 

320 conn, 

321 work_item_id, 

322 WorkItemEvent.TAKEOVER_ENDED, 

323 {"recorded": recorded, "warnings": warnings}, 

324 ) 

325 return True 

326 

327 await db.write(end) 

328 return Released(status=row["status"], recorded=recorded, warnings=warnings)
