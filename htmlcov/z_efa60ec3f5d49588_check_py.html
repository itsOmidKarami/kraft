36def _git_lfs() -> dict: 

37 """Without git-lfs a machine is not given a step of an LFS repo; that is not a failure. The 

38 worker looks for git-lfs once, when it starts, so one installed later is listed only after the 

39 worker restarts.""" 

40 path = shutil.which("git-lfs") 

41 if path is not None: 

42 return _row("git-lfs", True, path)

47 "(it looks for git-lfs only when it starts)", 

48 skipped=True,

52def _agents() -> list[dict]: 

53 """One row per agent CLI Kraft knows, and a failure when none is installed. 

54 A credential this check cannot see is only a warning: a CLI may keep its 

55 login in a store of its own.""" 

56 rows, installed = [], 0 

57 for hid, h in sorted(harness.load(None).valid.items()): 

58 path = shutil.which(h.command[0]) 

59 if path is None: 

60 rows.append(_row(f"agent: {hid}", True, "not installed", skipped=True)) 

61 continue 

62 installed += 1 

63 rows.append(_row(f"agent: {hid}", True, path)) 

64 wanted = sorted({c.env for c in h.credentials}) 

65 if wanted and not any(os.environ.get(name) for name in wanted): 

66 rows.append( 

67 _row( 

68 f"credential: {hid}", 

69 True, 

70 f"{' or '.join(wanted)} is not set here. A login the CLI keeps " 

71 "itself is not visible to this check", 

72 warn=True, 

73 ) 

74 ) 

75 if not installed: 

76 rows.append(_row("agent CLIs", False, "none of the agent CLIs Kraft knows is on PATH")) 

77 return rows

80async def _sandboxes() -> list[dict]: 

81 """One row per sandbox backend, from its own health check: a sandboxed item's step needs a 

82 backend that works here, or it stops `config_error` rather than run unsandboxed. None usable 

83 is only a warning: a machine may be for unsandboxed items.""" 

84 rows, usable = [], 0 

85 for backend in backends.every(): 

86 try: 

87 ok, detail = await backend.health({"kind": backend.kind}, refresh=True) 

88 except Exception as exc: # noqa: BLE001 -- a check reports, it does not crash 

89 ok, detail = False, f"could not check: {exc}" 

90 usable += ok 

91 rows.append(_row(f"sandbox: {backend.kind}", True, detail, skipped=not ok)) 

92 if not usable: 

93 rows.append( 

94 _row("sandbox runtime", True, "none usable: sandboxed items cannot run here", warn=True) 

95 ) 

96 return rows

99async def _server(server: str, token: str, kraft_version: str) -> dict: 

100 """One handshake with the server, then hang up.""" 

101 from websockets.asyncio.client import connect 

102 from websockets.exceptions import WebSocketException 

103 

104 try: 

105 url = machine.dial_url(server) 

106 except ValueError as exc: 

107 return _row("server", False, str(exc)) 

108 try: 

109 async with connect(url, max_size=FRAME_LIMIT, ping_interval=None, open_timeout=10) as ws: 

110 t = WebSocketTransport(ws.send, ws.recv, ws.close) 

111 welcome = await protocol.machine_handshake( 

112 t, kraft_version=kraft_version, token=token, host=socket.gethostname() 

113 ) 

114 except protocol.Refused as exc: 

115 return _row("server", False, f"refused: {exc}") 

116 except (OSError, TimeoutError, WebSocketException, protocol.Closed) as exc: 

117 return _row("server", False, f"could not connect to {url}: {exc or type(exc).__name__}") 

118 return _row("server", True, f"welcomed as {welcome['name']} (kraft {welcome['kraft']})")
