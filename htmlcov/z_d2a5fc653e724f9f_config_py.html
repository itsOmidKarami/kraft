28#: A collection name, plugin name or alias: an `Identifier` short enough to be 

29#: a directory name under `run/plugins/`. 

30Name = Annotated[Identifier, Field(max_length=64)] 

31#: `<plugin>@<collection>`. 

32PluginId = Annotated[StrictStr, Field(pattern=r"^[a-z][a-z0-9_-]{0,63}@[a-z][a-z0-9_-]{0,63}$")] 

33#: A URL git clones: https without credentials, ssh, file, or scp-like 

34#: `user@host:path`. Never the GitHub shorthand: `collection add owner/repo` 

35#: writes https://github.com/owner/repo.git. No form may start with `-`. 

36_HOST = r"[A-Za-z0-9][A-Za-z0-9.-]*(:[0-9]+)?" 

37_USER = r"[A-Za-z0-9_][A-Za-z0-9._-]*@" 

38_GIT_URL = re.compile( 

39 rf"^(https://{_HOST}(/[^\s\x00-\x1f]*)?" 

40 rf"|ssh://({_USER})?{_HOST}/[^\s\x00-\x1f]*" 

41 r"|file:///[^\s\x00-\x1f]+" 

42 rf"|{_USER}[A-Za-z0-9][A-Za-z0-9.-]*:[^\s\x00-\x1f-][^\s\x00-\x1f]*)$" 

43) 

44#: A plugin's directory in its collection, as `collection.json` writes it. 

45_SOURCE = re.compile(r"^\./[^/\x00-\x1f\\]+(/[^/\x00-\x1f\\]+)*$") 

46_SEMVER = ( 

47 r"^(0|[1-9][0-9]*)\.(0|[1-9][0-9]*)\.(0|[1-9][0-9]*)" 

48 r"(-(0|[1-9][0-9]*|[0-9]*[A-Za-z-][0-9A-Za-z-]*)(\.(0|[1-9][0-9]*|[0-9]*[A-Za-z-][0-9A-Za-z-]*))*)?$" 

49) 

50_OID = r"^[0-9a-f]{40}([0-9a-f]{24})?$" # a SHA-1 or SHA-256 object name 

51#: Reserved for Kraft's own components: no plugin name or alias may be it. 

52RESERVED_NAMESPACE = "kraft"

59def source_problem(source: str) -> str | None: 

60 """Why `source` is not a plugin directory inside its collection, or None. 

61 Segment checks in Python, not a negative regex, so `.KRAFT` is caught 

62 however a filesystem folds case.""" 

63 if not _SOURCE.match(source) or source.endswith("\n"): 

64 return "must be ./ followed by a relative path, with no empty segment or trailing /" 

65 parts = source[2:].split("/") 

66 if any(p in (".", "..") for p in parts): 

67 return "may not have a . or .. segment" 

68 if parts[0].casefold() == ".kraft": 

69 return "may not be inside .kraft/" 

70 return None

73class CollectionConfig(_Model): 

74 """One `collections:` entry, keyed by the name its `collection.json` 

75 declares. A git collection is fetched at `ref`; a directory collection is 

76 read in place, so it has no ref and nothing to auto-update.""" 

77 

78 model_config = ConfigDict( 

79 frozen=True, 

80 strict=True, 

81 extra="forbid", 

82 # The validator below, for an editor. 

83 json_schema_extra={ 

84 "oneOf": [ 

85 { 

86 "required": ["git"], 

87 "not": {"required": ["path"], "properties": {"path": {"type": "string"}}}, 

88 }, 

89 { 

90 "required": ["path"], 

91 "properties": { 

92 "path": {"type": "string", "pattern": "^/."}, 

93 "auto_update": {"const": False}, 

94 }, 

95 "not": {"required": ["ref"], "properties": {"ref": {"type": "string"}}}, 

96 }, 

97 ] 

98 }, 

99 )

101 git: StrictStr | None = Field( 

102 default=None, 

103 description="A URL git clones: https://, ssh://, file:/// or user@host:path.", 

104 json_schema_extra={"pattern": _GIT_URL.pattern}, 

105 ) 

106 path: StrictStr | None = Field( 

107 default=None, description="An absolute directory, in normal form." 

108 ) 

109 ref: StrictStr | None = Field( 

110 default=None, 

111 description="A branch, tag or commit of a git collection. " 

112 "None: the remote's default branch.", 

113 ) 

114 auto_update: StrictBool = Field( 

115 default=False, 

116 description="Update every installed plugin of this collection on server start. " 

117 "A plugin cannot opt out.", 

118 )

120 @field_validator("git") 

121 @classmethod 

122 def _git(cls, git: str | None) -> str | None: 

123 if git is not None and not _GIT_URL.match(git): 

124 raise _refuse( 

125 "git must be a full URL: https://host/path (no credentials), ssh://..., " 

126 "file:///... or user@host:path; for GitHub's owner/repo write " 

127 "https://github.com/owner/repo.git" 

128 ) 

129 return git

131 @field_validator("ref", mode="before") 

132 @classmethod 

133 def _ref_is_text(cls, ref: object) -> object: 

134 """YAML reads `ref: 1.10` as the number 1.1.""" 

135 if ref is not None and not isinstance(ref, str): 

136 raise _refuse(f"ref must be a string; quote it: ref: '{ref}'") 

137 return ref

139 @field_validator("ref") 

140 @classmethod 

141 def _ref(cls, ref: str | None) -> str | None: 

142 if ref is not None and (problem := branch_name_problem(ref)): 

143 raise _refuse(f"ref {problem}") 

144 return ref

146 @field_validator("path") 

147 @classmethod 

148 def _path(cls, path: str | None) -> str | None: 

149 if path is None: 

150 return None 

151 if path.startswith("~"): 

152 raise _refuse("path must be absolute; ~ is not expanded") 

153 if not path.startswith("/") or path == "/" or posixpath.normpath(path) != path: 

154 raise _refuse( 

155 "path must be an absolute directory in normal form (no trailing /, . or ..)" 

156 ) 

157 return path

159 @model_validator(mode="after") 

160 def _source(self) -> CollectionConfig: 

161 if (self.git is None) == (self.path is None): 

162 raise _refuse("give exactly one of git or path") 

163 if self.path is not None and self.ref is not None: 

164 raise _refuse("ref is for a git collection; a directory collection is read as it is") 

165 if self.path is not None and self.auto_update: 

166 raise _refuse("a directory collection never auto-updates") 

167 return self

170class PluginEntry(_Model): 

171 """One installed plugin, in its long form; `true` and `false` are short 

172 for `{enabled: true}` and `{enabled: false}`.""" 

173 

174 # By alias only: `as_:` in the file is a typo, refused like any other. 

175 model_config = ConfigDict( 

176 frozen=True, 

177 strict=True, 

178 extra="forbid", 

179 validate_by_name=False, 

180 validate_by_alias=True, 

181 serialize_by_alias=True, 

182 ) 

183 

184 as_: Name | None = Field( 

185 default=None, 

186 alias="as", 

187 description="The namespace its components are referenced under, in place of its name.", 

188 ) 

189 enabled: StrictBool = True 

190 auto_update: StrictBool | None = Field( 

191 default=None, 

192 description="Only when its collection does not auto-update, and only for a git collection.", 

193 )

197#: form the operator wrote ("plugins.x@y.map.enabeld") rather than both. 

198Entry = Annotated[

204class PluginsConfig(_Model): 

205 """`config/plugins.yaml`.""" 

206 

207 FILE: ClassVar[str] = "plugins.yaml" 

208 model_config = ConfigDict(frozen=True, strict=True, extra="forbid")

210 # pydantic exports a key pattern as patternProperties, which lets any 

211 # other key through; closing the object makes the schema refuse it too. 

212 collections: dict[Name, CollectionConfig] = Field( 

213 default={}, json_schema_extra={"additionalProperties": False, "type": ["object", "null"]} 

214 ) 

215 plugins: dict[PluginId, Entry] = Field( 

216 default={}, json_schema_extra={"additionalProperties": False, "type": ["object", "null"]} 

217 )

219 @field_validator("collections", "plugins", mode="before") 

220 @classmethod 

221 def _empty(cls, value: object) -> object: 

222 """`plugins:` with every entry deleted reads as YAML null.""" 

223 return {} if value is None else value

225 def entry(self, plugin_id: str) -> PluginEntry: 

226 value = self.plugins[plugin_id] 

227 return PluginEntry(enabled=value) if isinstance(value, bool) else value 

228 

229 def namespace(self, plugin_id: str) -> str: 

230 return self.entry(plugin_id).as_ or plugin_id.split("@", 1)[0] 

231 

232 @model_validator(mode="after") 

233 def _plugins(self) -> PluginsConfig: 

234 seen: dict[str, str] = {} 

235 for plugin_id in self.plugins: 

236 name, collection_name = plugin_id.split("@", 1) 

237 collection = self.collections.get(collection_name) 

238 if collection is None: 

239 raise _refuse(f"{plugin_id}: no collection {collection_name!r} under collections") 

240 entry = self.entry(plugin_id) 

241 if entry.auto_update is not None: 

242 if collection.path is not None: 

243 raise _refuse(f"{plugin_id}: a directory collection never auto-updates") 

244 if collection.auto_update: 

245 raise _refuse( 

246 f"{plugin_id}: {collection_name} auto-updates every plugin; " 

247 "drop auto_update here" 

248 ) 

249 if RESERVED_NAMESPACE in (name, entry.as_): 

250 raise _refuse(f"{plugin_id}: {RESERVED_NAMESPACE!r} is Kraft's own namespace") 

251 namespace = self.namespace(plugin_id) 

252 if namespace in seen: 

253 raise _refuse( 

254 f"{plugin_id}: namespace {namespace!r} is taken by {seen[namespace]}; " 

255 "install one under another name with --as" 

256 ) 

257 seen[namespace] = plugin_id 

258 return self

261class LockEntry(_Model): 

262 """What one installed plugin resolved to at its last install or update.""" 

263 

264 model_config = ConfigDict(frozen=True, strict=True, extra="forbid") 

265 

266 #: The namespace and the collection `ref` it was reviewed under. A 

267 #: `plugins.yaml` that says otherwise waits for `update` to apply it. 

268 namespace: Name 

269 ref: StrictStr | None # None: a directory collection 

270 #: The collection's `git:` URL it was fetched from: another repository 

271 #: under the same name is a change a person reviews. None for a directory. 

272 git: StrictStr | None = None 

273 #: The collection's commit, and the tree at `source` in it. Both None for 

274 #: a directory collection. 

275 commit: Annotated[StrictStr, Field(pattern=_OID)] | None 

276 tree: Annotated[StrictStr, Field(pattern=_OID)] | None 

277 #: The entry's `source` in `collection.json`, as written there. 

278 source: StrictStr 

279 digest: Annotated[StrictStr, Field(pattern=r"^sha256:[0-9a-f]{64}$")] 

280 version: Annotated[StrictStr, Field(pattern=_SEMVER)] 

281 # Not strict: written back by `model_dump(mode="json")`, it reads as a string. 

282 updated_at: Annotated[AwareDatetime, Field(strict=False)] 

283 

284 @field_validator("source") 

285 @classmethod 

286 def _source(cls, source: str) -> str: 

287 if problem := source_problem(source): 

288 raise _refuse(f"source {problem}") 

289 return source 

290 

291 @model_validator(mode="after") 

292 def _pinned(self) -> LockEntry: 

293 if (self.commit is None) != (self.tree is None): 

294 raise _refuse("commit and tree are both set (git) or both null (directory)") 

295 return self
