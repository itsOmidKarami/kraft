1"""What an install or update of one add-on would change, said before anything 

2is written. 

3 

4A review compares the candidate's files with the locked store's (or with 

5nothing, on install). It leads with **reach**: what can change how far a run 

6goes, which is also what `auto_update` holds for a person. The rest is 

7**content**. Each resolved chain is flattened to `path -> value` facts keyed 

8by component ids, so a diff is a diff of facts and a new field in the chain 

9schema shows up as content without a line of code here. 

10"""

55def precedence(version: str) -> tuple: 

56 """A sort key for a SemVer 2.0.0 version without build metadata: a 

57 pre-release sorts below its own release, numeric identifiers below 

58 alphanumeric ones, and a longer pre-release above its own prefix.""" 

59 core, _, pre = version.partition("-") 

60 numbers = tuple(int(part) for part in core.split(".")) 

61 if not pre: 

62 return (numbers, (1,)) 

63 identifiers = tuple( 

64 (0, int(part), "") if part.isdigit() else (1, 0, part) for part in pre.split(".") 

65 ) 

66 return (numbers, (0, identifiers))

69def escape(text: str) -> str: 

70 """`text` with every character a reader cannot see written out. Newline 

71 and tab are kept.""" 

72 return "".join( 

73 f"\\u{{{ord(ch):04X}}}" 

74 if ch not in "\n\t" and unicodedata.category(ch) in ("Cc", "Cf") 

75 else ch 

76 for ch in text

80def render(review: Review) -> str: 

81 """The review as the terminal, `--check` and `--json` all print it.""" 

82 was = f"{review.old_version} -> " if review.old_version else "new, " 

83 lines = [f"{review.addon_id} {was}{review.new_version}"] 

84 for title, entries in (("Reach", review.reach), ("Content", review.content)): 

85 if entries: 

86 lines.append(f"{title}:") 

87 lines += [f" - {entry}" for entry in entries] 

88 if not review.changed: 

89 lines.append(" up to date") 

90 return escape("\n".join(lines))

115def _flatten(value: object, at: str, out: dict[str, object], lists: dict[str, list[str]]) -> None: 

116 """Facts go to `out`; the order of each id-keyed list, which a fact keyed 

117 by id cannot show, goes to `lists`.""" 

118 if isinstance(value, Mapping): 

119 for key, item in value.items(): 

120 if key != "id": 

121 _flatten(item, f"{at}.{key}", out, lists) 

122 elif ( 

123 isinstance(value, list) 

124 and value 

125 and all(isinstance(item, Mapping) and "id" in item for item in value) 

126 ): 

127 lists[at] = [str(item["id"]) for item in value] 

128 for item in value: 

129 _flatten(item, f"{at}[{item['id']}]", out, lists) 

130 elif ( 

131 value is not None 

132 and value != {} 

133 and (value != [] or at.endswith((".fallback", ".reviewers"))) 

134 ): 

135 # `fallback: []` says "none"; absent, the task takes its profile's. 

136 # `reviewers: []` likewise expects none, where absent defers to the repo. 

137 out[at] = value

199def _facts(extracted: Extracted | None, ancestors: Sequence[InstalledAddon] = ()) -> _Facts: 

200 """What `extracted` declares and its chains resolve to. A child's chains 

201 resolve only with its `ancestors` (parent first) in place; only its own 

202 chains are read, not theirs.""" 

203 if extracted is None: 

204 return _Facts("", {}, {}, {}, {}, {}, {}, {}, {}) 

205 declared = json.loads(extracted.files[manifest.ADDON_JSON][1]) 

206 name = declared.get("name", "addon") 

207 library = _yaml(extracted, "library.yaml") 

208 chains: dict[str, object] = {} 

209 lists: dict[str, list[str]] = {} 

210 order: dict[str, list[tuple[str, str]]] = {} 

211 stale: frozenset[str] = frozenset() 

212 with tempfile.TemporaryDirectory() as tmp: 

213 root = write_scratch(Path(tmp), extracted) 

214 addon = _candidate(declared, name, root) 

215 try: 

216 loaded = TemplateLibrary.from_mappings( 

217 {}, (), library_path=root / "local.yaml", addons=[*ancestors, addon] 

218 ) 

219 except TemplateLibraryError: 

220 loaded = None # lint refuses it; there is nothing to compare 

221 if loaded is not None: 

222 own = [c for c in loaded.chain_ids if c.startswith(f"{name}:")] 

223 _chain_facts(loaded, own, chains, lists, order) 

224 stale = frozenset(e for e in loaded.stale_removals if e.startswith(f"{name}: ")) 

225 return _Facts( 

226 version=str(declared.get("version", "")), 

227 requires=declared.get("requires") if isinstance(declared.get("requires"), dict) else {}, 

228 chains=chains, 

229 lists=lists, 

230 order=order, 

231 components={ 

232 f"{section}.{key}": body 

233 for section in ("tasks", "steps", "nodes") 

234 for key, body in _section(library, section).items() 

235 }, 

236 steering=_section(library, "steering"), 

237 skills={ 

238 rel.split("/")[1]: data.decode() 

239 for rel, (_mode, data) in extracted.files.items() 

240 if rel.startswith("skills/") 

241 }, 

242 profiles=_section(_yaml(extracted, "profiles.yaml"), "profiles"), 

243 stale=stale, 

244 )

262 """The names `extracted`'s own files declare, for each removable section.""" 

263 library = _yaml(extracted, "library.yaml") 

264 files = extracted.files

282def _inherited(ancestors: Sequence[InstalledAddon]) -> dict[str, dict[str, str]]: 

283 """For each section, the names the parent's merged library has, each with 

284 the id of the ancestor that defines it: from the root down, a level drops 

285 what it removes and then adds what it declares.""" 

286 if not ancestors: 

287 return {} 

288 declared: list[dict[str, frozenset[str]]] = [] 

289 removed: list[dict[str, tuple[str, ...]]] = [] 

290 for ancestor in ancestors: 

291 try: 

292 files = fetch.verify_store(ancestor.root) 

293 level = ( 

294 _declared(files), 

295 inherit.removals(ancestor.root / "library.yaml", _yaml(files, "library.yaml")), 

296 ) 

297 except (fetch.AddonRefused, fetch.FetchError, TemplateLibraryError): 

298 level = ({}, {}) # the store does not read; loading says why 

299 declared.append(level[0]) 

300 removed.append(level[1]) 

301 seen = inherit.Visible.build(ancestors, declared, removed).levels[0] 

302 ids = {a.namespace: a.id for a in ancestors} 

303 return { 

304 section: {name: ids[owner.partition(":")[0]] for name, owner in names.items()} 

305 for section, names in seen.items() 

306 }

309def _parent_lines(was: Mapping[str, str], now: Mapping[str, str]) -> list[str]: 

310 """What a change of the `requires.addons` entry (parent id to range) says.""" 

311 if was == now: 

312 return [] 

313 (old_id, old_range), (new_id, new_range) = ( 

314 next(iter(side.items()), (None, None)) for side in (was, now) 

315 ) 

316 if old_id is None: 

317 return [f"now builds on {new_id} {new_range}"] 

318 if new_id is None: 

319 return [f"no longer builds on {old_id}"] 

320 if old_id == new_id: 

321 return [f"parent range: {old_id} {old_range} -> {new_range}"] 

322 return [f"parent changed: {old_id} {old_range} -> {new_id} {new_range}"]

344def _chain_change( 

345 path: str, old: object, new: object, was: _Facts, now: _Facts, names: Collection[str] 

346) -> tuple[str, str]: 

347 """`(section, line)` for one changed fact of a resolved chain. `names`: 

348 the namespaces under review, whose skills are their own.""" 

349 where, _, last = path.rpartition(".") 

350 if ".auto_review" in path: 

351 return "reach", f"{where}: a gate's own agent review changed: {last} {_how(old, new)}" 

352 if last in _GATE_FIELDS: 

353 return "reach", f"{where}: {last} {_how(old, new)}" 

354 if last == "kind": 

355 if old == "gate": # the fact changed, so `new` is not a gate 

356 return "reach", f"{where}: gate removed{'' if new is None else f', now {new}'}" 

357 return "content", f"{where}: {'removed' if new is None else f'added ({new})'}" 

358 if last == "target": 

359 chain_id = path.split(".nodes[", 1)[0] 

360 used = { 

361 v 

362 for p, v in was.chains.items() 

363 if p.startswith(f"{chain_id}.") and p.endswith(".target") 

364 } 

365 if new in _MERGES: 

366 node = _NODE.search(path) 

367 before = [] 

368 for node_id, kind in now.order.get(chain_id, ()): 

369 if node is not None and node_id == node.group(1): 

370 break 

371 before.append(kind) 

372 gated = "a gate comes before it" if "gate" in before else "NO gate comes before it" 

373 return "reach", f"{where}: {new} step {_how(old, new)}; {gated}" 

374 if new is not None and new not in used: 

375 return "reach", f"{where}: forge target {new!r}, which this chain did not use before" 

376 if new is None: 

377 # A wait for CI or for a person's approval is a forge step too. 

378 return "reach", f"{where}: forge step {old!r} removed" 

379 return "content", f"{where}: forge target {_how(old, new)}" 

380 if last == "read_only" and old and not new: 

381 return "reach", f"{where}: no longer read-only" 

382 if last == "scope" and old is not None: 

383 # `each_repository` runs the task once more per repository. 

384 return "reach", f"{where}: scope {_how(old, new)}" 

385 if (".policy." in path or ".fix_loop." in path) and last in LIMITS: 

386 if last == "deny_tools": 

387 dropped = sorted(set(old or ()) - set(new or ())) 

388 if dropped: 

389 return "reach", f"{where}: deny_tools no longer denies {dropped}" 

390 return "content", f"{where}: deny_tools {_how(old, new)}" 

391 lowered = isinstance(old, int | float) and isinstance(new, int | float) and new < old 

392 if lowered: 

393 return "content", f"{where}: {last} lowered, {_how(old, new)}" 

394 word = "removed" if new is None else "raised" if old is not None else "set" 

395 return "reach", f"{where}: limit {last} {word}, {_how(old, new)}" 

396 if last in _ROUTE: 

397 return "reach", f"{where}: {last} {_how(old, new)}" 

398 if last == "prompt": 

399 return "content", f"{where}: prompt {'new' if old is None else 'changed'}:\n{new or ''}" 

400 if last == "skill": 

401 qualifier = new.split(":", 1)[0] if isinstance(new, str) and ":" in new else None 

402 if qualifier not in (None, "kraft", *names): 

403 return "content", f"{where}: new plugin skill reference {new!r}, handed to the agent" 

404 return "content", f"{where}: skill {_how(old, new)}" 

405 if last == "reviewers": 

406 # Who the review wait reads decides whose feedback reaches the fix loop. 

407 return "reach", f"{where}: reviewers {_how(_named(old), _named(new))}" 

408 if last in _INPUTS or last == "steering": 

409 return "content", f"{where}: {last} {_how(old, new)}" 

410 return "content", f"{path}: {_how(old, new)}"

413def _reordered(path: str, before: list[str], after: list[str]) -> tuple[str, str]: 

414 """`(section, line)` for an id-keyed list whose common entries changed

416 and a moved step on the other side of a wait or a check in its node; the

422def _chain_diff( 

423 was: _Facts, now: _Facts, names: Collection[str], reach: list[str], content: list[str] 

424) -> None: 

425 """Every changed fact and every reordered list of the resolved chains.""" 

426 for path in sorted(was.chains.keys() | now.chains.keys()): 

427 before, after = was.chains.get(path), now.chains.get(path) 

428 if before != after: 

429 section, line = _chain_change(path, before, after, was, now, names) 

430 (reach if section == "reach" else content).append(line) 

431 for path in sorted(was.lists.keys() & now.lists.keys()): 

432 common = set(was.lists[path]) & set(now.lists[path]) 

433 before = [i for i in was.lists[path] if i in common] 

434 after = [i for i in now.lists[path] if i in common] 

435 if before != after: 

436 section, line = _reordered(path, before, after) 

437 (reach if section == "reach" else content).append(line)

440def local_changes( 

441 before: TemplateLibrary | None, 

442 after: TemplateLibrary, 

443 namespace: str | Collection[str], 

444 *, 

445 extra: Collection[str] = (), 

446) -> tuple[list[str], list[str]]: 

447 """`(reach, content)` for the instance's own chains: what each one that 

448 resolves through the add-on under `namespace` (or the add-ons of a tree, 

449 under each of a collection) would run differently once `after` replaces 

450 `before`. A chain the add-on does not reach has no line. `extra`: other 

451 chains to diff the same way, a repo library's 

452 (`the-review-covers-repo-library-chains`).""" 

453 names = (namespace,) if isinstance(namespace, str) else tuple(namespace) 

454 sides = [] 

455 for loaded in (before, after): 

456 facts = _Facts("", {}, {}, {}, {}, {}, {}, {}, {}) 

457 if loaded is not None: 

458 local = [ 

459 chain_id 

460 for chain_id in loaded.chain_ids 

461 if ":" not in chain_id or chain_id in extra 

462 ] 

463 _chain_facts(loaded, local, facts.chains, facts.lists, facts.order) 

464 sides.append(facts) 

465 reach: list[str] = [] 

466 content: list[str] = [] 

467 _chain_diff(sides[0], sides[1], names, reach, content) 

468 return reach, content

471def review( 

472 addon_id: str, 

473 old: Extracted | None, 

474 new: Extracted, 

475 *, 

476 old_ancestors: Sequence[InstalledAddon] = (), 

477 new_ancestors: Sequence[InstalledAddon] = (), 

478) -> Review: 

479 """What replacing `old` (None on install) with `new` would change. A child 

480 add-on's chains resolve only through its parent: each side is read with 

481 that side's ancestors (parent first) in place, the installed tree for 

482 `old` and the one it would load in for `new`.""" 

483 was, now = _facts(old, old_ancestors), _facts(new, new_ancestors) 

484 name = addon_id.split("@", 1)[0] 

485 reach: list[str] = [] 

486 content: list[str] = [] 

487 if old is not None: 

488 if precedence(now.version) < precedence(was.version): 

489 reach.append(f"downgrade: {was.version} -> {now.version}") 

490 if "-" in now.version and "-" not in was.version: 

491 reach.append(f"moves onto a pre-release: {now.version}") 

492 own = [{k: v for k, v in f.requires.items() if k != "addons"} for f in (was, now)] 

493 if own[0] != own[1]: 

494 reach.append(f"requires changed: {own[0]} -> {own[1]}") 

495 reach += _parent_lines(*(f.requires.get("addons") or {} for f in (was, now))) 

496 before, after = _inherited(old_ancestors), _inherited(new_ancestors) 

497 for section, names in _declared(new).items(): 

498 offered, offered_before = after.get(section, {}), before.get(section, {}) 

499 kind = _KINDS[section] 

500 reach += [ 

501 f"its {kind} {declared} now overrides {offered[declared]}'s new {kind} {declared}" 

502 for declared in sorted(names) 

503 if declared in offered and declared not in offered_before 

504 ] 

505 _chain_diff(was, now, (name,), reach, content) 

506 for label, before_map, after_map in ( 

507 ("component", was.components, now.components), 

508 ("steering", was.steering, now.steering), 

509 ("agent profile", was.profiles, now.profiles), 

510 ): 

511 for key in sorted(before_map.keys() | after_map.keys()): 

512 if before_map.get(key) != after_map.get(key): 

513 state = ( 

514 "added" 

515 if key not in before_map 

516 else "removed" 

517 if key not in after_map 

518 else "changed" 

519 ) 

520 detail = f": {_how(before_map.get(key), after_map.get(key))}" 

521 # A task keeps `profile: fast` while what `fast` runs on changes. 

522 rerouted = label == "agent profile" and state == "changed" 

523 (reach if rerouted else content).append( 

524 f"{label} {key} {state}{detail if label != 'component' else ''}" 

525 ) 

526 content += [f"warning: {entry}; it does nothing" for entry in sorted(now.stale - was.stale)] 

527 for key in sorted(was.skills.keys() | now.skills.keys()): 

528 before_text, after_text = was.skills.get(key, ""), now.skills.get(key, "") 

529 if before_text != after_text: 

530 diff = "\n".join( 

531 difflib.unified_diff( 

532 before_text.splitlines(), 

533 after_text.splitlines(), 

534 f"skills/{key}/SKILL.md (locked)", 

535 f"skills/{key}/SKILL.md", 

536 lineterm="", 

537 ) 

538 ) 

539 content.append(f"skill {key} changed:\n{diff}") 

540 return Review(
