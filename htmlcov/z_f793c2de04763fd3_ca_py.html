1"""The Kraft CA (sandbox part 2, P4b; spec §4 "VM-backed runtimes" and §6): 

2one EC P-256 root under `run/ca/`, generated once, whose key never leaves 

3there (mode 0600, directory 0700). 

4 

5It signs, for now, the daemon's loopback TLS listener (`server_cert`) and 

6one client certificate per egress session on a VM-backed runtime 

7(`mint_session_cert`), whose subject CN is the session id -- the only 

8identity the listener trusts, since every relay-B connection arrives from 

9127.0.0.1. And one server certificate per injected host (`mint_host_leaf`), 

10which the egress proxy shows a worker's client when it terminates TLS to 

11inject a credential (P6). 

12 

13Every certificate carries what Python 3.13+'s `VERIFY_X509_STRICT` (on in 

14`ssl.create_default_context`) demands: critical basicConstraints and 

15keyUsage, SKI, and AKI on a leaf. An ad-hoc `openssl req -x509` CA is 

16refused under it (spike 4.5a); the flag is never relaxed instead. 

17 

18Call it from the daemon only: generation is not locked against a second 

19process doing the same at once. 

20"""

39def strict(context: ssl.SSLContext) -> ssl.SSLContext: 

40 """`context` with the two verify flags `ssl.create_default_context` sets only 

41 from Python 3.13, set by hand so 3.12 behaves the same as 3.13 and 3.14: 

42 VERIFY_X509_STRICT (nothing looser than the RFC 5280 checks) and 

43 VERIFY_X509_PARTIAL_CHAIN (a trusted intermediate is enough of an anchor, 

44 which a sandbox `ca_bundle` that is not a self-signed root relies on).""" 

45 context.verify_flags |= ssl.VERIFY_X509_STRICT | ssl.VERIFY_X509_PARTIAL_CHAIN 

46 return context

49_CA_DAYS = 3650 

50#: The listener's certificate lives as long as the root it is cached beside. 

51_SERVER_DAYS = _CA_DAYS 

52# ponytail: no revocation; the registry forgetting the session is the whole 

53# mechanism -- add a CRL if a leaked leaf outliving its session ever matters. 

54# A session that outlives this fails closed: relay B's handshake is refused. 

55_SESSION_DAYS = 7 

56#: Backdating, for a runtime VM's clock a little behind the host's. 

57_SKEW = datetime.timedelta(minutes=5) 

58#: The names relay B dials the daemon by from a VM-backed runtime (Docker 

59#: Desktop's, podman machine's); socat verifies the listener's certificate 

60#: against the one it dialled. 

61GATEWAY_HOSTS = ("host.docker.internal", "host.containers.internal")

71def write_whole(path: Path, data: bytes, mode: int) -> None: 

72 """`data` at `path` whole or not at all: written to a file of its own, 

73 created `mode` (a key's 0600 never readable by anyone else even for an 

74 instant), then renamed over `path`. A reader never sees half a file, 

75 and a crash never leaves one where the next mint would reuse it.""" 

76 tmp = path.with_name(f".{path.name}.{os.getpid()}.{threading.get_ident()}") 

77 tmp.unlink(missing_ok=True) # one a crash left, perhaps not `mode` 

78 fd = os.open(tmp, os.O_WRONLY | os.O_CREAT | os.O_EXCL, mode) 

79 try: 

80 with os.fdopen(fd, "wb") as out: 

81 out.write(data) 

82 os.replace(tmp, path) 

83 except BaseException: 

84 tmp.unlink(missing_ok=True) 

85 raise

103def ensure_ca(run_dirs) -> tuple[Path, Path]: 

104 """`run/ca/ca.pem` and `ca.key`: generated when either is missing, 

105 reused forever after (no rotation: delete both to rotate by hand).""" 

106 directory = _dir(run_dirs) 

107 cert_path, key_path = directory / "ca.pem", directory / "ca.key" 

108 if cert_path.is_file() and key_path.is_file(): 

109 return cert_path, key_path 

110 key = ec.generate_private_key(ec.SECP256R1()) 

111 now = datetime.datetime.now(datetime.UTC) 

112 ski = x509.SubjectKeyIdentifier.from_public_key(key.public_key()) 

113 cert = ( 

114 x509.CertificateBuilder() 

115 .subject_name(_name("Kraft CA")) 

116 .issuer_name(_name("Kraft CA")) 

117 .public_key(key.public_key()) 

118 .serial_number(x509.random_serial_number()) 

119 .not_valid_before(now - _SKEW) 

120 .not_valid_after(now + datetime.timedelta(days=_CA_DAYS)) 

121 .add_extension(x509.BasicConstraints(ca=True, path_length=0), critical=True) 

122 .add_extension( 

123 x509.KeyUsage( 

124 digital_signature=False, 

125 content_commitment=False, 

126 key_encipherment=False, 

127 data_encipherment=False, 

128 key_agreement=False, 

129 key_cert_sign=True, 

130 crl_sign=True, 

131 encipher_only=False, 

132 decipher_only=False, 

133 ), 

134 critical=True, 

135 ) 

136 .add_extension(ski, critical=False) 

137 .add_extension( 

138 x509.AuthorityKeyIdentifier.from_issuer_subject_key_identifier(ski), critical=False 

139 ) 

140 .sign(key, hashes.SHA256()) 

141 ) 

142 _write_key(key_path, key) 

143 _write_cert(cert_path, cert) 

144 return cert_path, key_path

147def _leaf( 

148 run_dirs, 

149 cert_path: Path, 

150 key_path: Path, 

151 cn: str, 

152 *, 

153 usage: x509.ObjectIdentifier, 

154 days: int, 

155 san: list[x509.GeneralName] | None = None, 

156) -> tuple[Path, Path]: 

157 """One end-entity certificate for `cn`, signed by the CA, written to 

158 `cert_path`/`key_path`.""" 

159 ca_cert_path, ca_key_path = ensure_ca(run_dirs) 

160 ca_cert = x509.load_pem_x509_certificate(ca_cert_path.read_bytes()) 

161 ca_key = serialization.load_pem_private_key(ca_key_path.read_bytes(), None) 

162 key = ec.generate_private_key(ec.SECP256R1()) 

163 now = datetime.datetime.now(datetime.UTC) 

164 builder = ( 

165 x509.CertificateBuilder() 

166 .subject_name(_name(cn)) 

167 .issuer_name(ca_cert.subject) 

168 .public_key(key.public_key()) 

169 .serial_number(x509.random_serial_number()) 

170 .not_valid_before(now - _SKEW) 

171 .not_valid_after(now + datetime.timedelta(days=days)) 

172 .add_extension(x509.BasicConstraints(ca=False, path_length=None), critical=True) 

173 .add_extension( 

174 x509.KeyUsage( 

175 digital_signature=True, 

176 content_commitment=False, 

177 key_encipherment=False, 

178 data_encipherment=False, 

179 key_agreement=False, 

180 key_cert_sign=False, 

181 crl_sign=False, 

182 encipher_only=False, 

183 decipher_only=False, 

184 ), 

185 critical=True, 

186 ) 

187 .add_extension(x509.ExtendedKeyUsage([usage]), critical=False) 

188 .add_extension(x509.SubjectKeyIdentifier.from_public_key(key.public_key()), critical=False) 

189 .add_extension( 

190 x509.AuthorityKeyIdentifier.from_issuer_subject_key_identifier( 

191 ca_cert.extensions.get_extension_for_class(x509.SubjectKeyIdentifier).value 

192 ), 

193 critical=False, 

194 ) 

195 ) 

196 if san: 

197 builder = builder.add_extension(x509.SubjectAlternativeName(san), critical=False) 

198 _write_key(key_path, key) 

199 _write_cert(cert_path, builder.sign(ca_key, hashes.SHA256())) 

200 return cert_path, key_path

203def server_cert(run_dirs) -> tuple[Path, Path]: 

204 """`run/ca/server.pem` and `server.key`, the daemon listener's: CN 

205 `kraft-daemon`, SAN `IP:127.0.0.1` and every `GATEWAY_HOSTS` name. 

206 Minted once, again only when missing or older than the CA (rotated by 

207 hand).""" 

208 ca_cert_path, _ = ensure_ca(run_dirs) 

209 directory = _dir(run_dirs) 

210 cert_path, key_path = directory / "server.pem", directory / "server.key" 

211 if ( 

212 cert_path.is_file() 

213 and key_path.is_file() 

214 and cert_path.stat().st_mtime >= ca_cert_path.stat().st_mtime 

215 ): 

216 return cert_path, key_path 

217 return _leaf( 

218 run_dirs, 

219 cert_path, 

220 key_path, 

221 "kraft-daemon", 

222 usage=ExtendedKeyUsageOID.SERVER_AUTH, 

223 days=_SERVER_DAYS, 

224 san=[ 

225 x509.IPAddress(ipaddress.ip_address("127.0.0.1")), 

226 *(x509.DNSName(name) for name in GATEWAY_HOSTS), 

227 ], 

228 )

231#: One exact DNS name, lowercase: no wildcard, no port, nothing a path 

232#: could make more of. 

233_EXACT_HOST = re.compile(r"[a-z0-9](?:[a-z0-9-]*[a-z0-9])?(?:\.[a-z0-9](?:[a-z0-9-]*[a-z0-9])?)*")

236def mint_host_leaf(run_dirs, host: str) -> tuple[Path, Path]: 

237 """`run/ca/hosts/<host>/leaf.pem` and `leaf.key`: a server certificate 

238 for exactly `host` (SAN `DNSName(host)`), for the egress proxy to show a 

239 worker's client when it terminates TLS to inject a credential (spec §6). 

240 Keyed by the host as DNS compares it (case, a trailing dot), cached 

241 like `server_cert`: minted again only when missing or older than the 

242 CA. `ValueError` for anything but an exact host -- the name is a 

243 directory. One at a time: the proxy mints on each CONNECT's thread, and 

244 a CLI opens several connections at once; hold `HOST_LEAF_LOCK` around 

245 this and the load too, or another re-mint can land between the files.""" 

246 name = host.lower().rstrip(".") 

247 if not _EXACT_HOST.fullmatch(name): 

248 raise ValueError(f"{host!r} is not an exact host name to mint a certificate for") 

249 # ponytail: one lock for every host; per-host locks if minting ever shows up. 

250 with HOST_LEAF_LOCK: 

251 return _host_leaf(run_dirs, name) 

252 

253 

254#: Held while a host leaf is checked and minted (`mint_host_leaf`), and by 

255#: the proxy while it loads the pair: a re-mint rewrites both files. 

256HOST_LEAF_LOCK = threading.RLock()

259def _host_leaf(run_dirs, name: str) -> tuple[Path, Path]: 

260 ca_cert_path, _ = ensure_ca(run_dirs) 

261 directory = _dir(run_dirs) / "hosts" / name 

262 for d in (directory.parent, directory): 

263 d.mkdir(parents=True, exist_ok=True, mode=0o700) 

264 d.chmod(0o700) 

265 cert_path, key_path = directory / "leaf.pem", directory / "leaf.key" 

266 if ( 

267 cert_path.is_file() 

268 and key_path.is_file() 

269 and cert_path.stat().st_mtime >= ca_cert_path.stat().st_mtime 

270 ): 

271 return cert_path, key_path 

272 return _leaf( 

273 run_dirs, 

274 cert_path, 

275 key_path, 

276 name, 

277 usage=ExtendedKeyUsageOID.SERVER_AUTH, 

278 days=_SERVER_DAYS, 

279 san=[x509.DNSName(name)], 

280 )

289def mint_session_cert(run_dirs, session_id: str) -> tuple[Path, Path]: 

290 """The session's client certificate and key, CN = the whole session id, 

291 beside a copy of the CA's certificate (never its key). Idempotent: an 

292 existing pair is returned as it is.""" 

293 ca_cert_path, _ = ensure_ca(run_dirs) 

294 directory = session_dir(run_dirs, session_id) 

295 for d in (directory.parent, directory): 

296 d.mkdir(parents=True, exist_ok=True, mode=0o700) 

297 d.chmod(0o700) 

298 cert_path, key_path = directory / "client.pem", directory / "key.pem" 

299 shutil.copyfile(ca_cert_path, directory / "ca.pem") 

300 if cert_path.is_file() and key_path.is_file(): 

301 return cert_path, key_path 

302 return _leaf( 

303 run_dirs, 

304 cert_path, 

305 key_path, 

306 session_id, 

307 usage=ExtendedKeyUsageOID.CLIENT_AUTH, 

308 days=_SESSION_DAYS, 

309 )

312def discard_session_cert(run_dirs, session_id: str) -> None: 

313 """Remove the session's directory, key and all. Best-effort; a no-op for
