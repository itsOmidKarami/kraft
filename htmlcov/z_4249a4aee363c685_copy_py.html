1"""A throwaway copy of an in-place item's branch, for an agent that only reads. 

2 

3The checks-first gate reviewer and the review reply agent work in one instead 

4of the person's checkout, so nothing they do there reaches the person's files. 

5It is a shared clone (`git clone --shared --no-checkout`), not a linked 

6worktree: a linked worktree shares the person's refs, stash and config, so a 

7`git stash` or `git branch -f` in it would change theirs. A shared clone reads 

8their objects through `alternates` and writes only into its own `.git`; its 

9`origin` is removed, so a plain `git push` has nowhere to go. 

10 

11It lives in the session's own folder (`copy_path`), never under 

12`run_dirs.scratch`, whose startup sweep runs before reattach and would take a 

13re-adopted session's copy from under it. 

14""" 

15

34class CopyFailed(Exception): 

35 """The copy could not be made. Never falls back to the person's checkout."""

41def copy_path(run_dirs, wid: str, sid: str) -> Path: 

42 """Where session `sid` of item `wid` keeps its copy.""" 

43 return item_paths.session_files(run_dirs, wid, sid).folder / "copy"

50def remove_copy(path: Path) -> None: 

51 # A global `core.fsmonitor` may have started a daemon in it: best effort. 

52 shutil.rmtree(path, ignore_errors=True) 

53 (path.parent / MCP_SERVERS).unlink(missing_ok=True)

56def watch_copy( 

57 db, row, launch, copy: Path, folder: Path 

58) -> Callable[[], tuple[list[str], list[str]]] | None: 

59 """Snapshot `copy` whole, and the item's branch tip in the person's 

60 `folder`, which a commit by any route moves (a push from the copy, or the 

61 person). Returns what lists the files changed since in the copy, and on 

62 the tip, or None when either cannot be read.""" 

63 from kraft.executor import read_only # it imports this package 

64 

65 was = ( 

66 read_only.snapshot(db, row, launch, copy, full=True), 

67 read_only.snapshot(db, row, launch, folder), 

68 ) 

69 if None in was: 

70 return None 

71 return lambda: ( 

72 read_only.changed(db, row, launch, copy, was[0], full=True), 

73 read_only.changed(db, row, launch, folder, was[1]), 

74 )

82async def make_copy(row, dest: Path, *, timeout_s: float | None) -> None: 

83 """Fill the empty folder `dest` with `row`'s branch tip, detached. Raises 

84 `CopyFailed` (with `dest` empty again) when the branch is gone, git fails, 

85 `timeout_s` runs out (None means no limit of its own) or anything else 

86 goes wrong. A cancel stops git at once, empties `dest` and is re-raised. 

87 A `dest` that is not empty is refused untouched: the failure path empties 

88 it.""" 

89 from kraft import builtins # builtins imports this package 

90 

91 branch = store.branch_for(row) 

92 if dest.exists() and any(dest.iterdir()): 

93 raise CopyFailed(branch, f"{dest} is not empty") 

94 repo = str(row["repo"]) 

95 # Hooks off, and LFS pointers left as they are: nothing is fetched into the 

96 # person's object store, and no hook of theirs runs. 

97 env = _sandbox.hardened_git_env() | {"GIT_LFS_SKIP_SMUDGE": "1"} 

98 deadline = None if timeout_s is None else time.monotonic() + timeout_s 

99 

100 async def git(*args: str, silent: str = "git failed") -> str: 

101 left = None if deadline is None else max(deadline - time.monotonic(), 0.0) 

102 done = await builtins._git_group(["git", *args], timeout=left, env=env) 

103 if done.returncode != 0: 

104 lines = (done.stderr or done.stdout).strip().splitlines() 

105 raise CopyFailed(branch, lines[-1] if lines else silent) 

106 return done.stdout.strip() 

107 

108 try: 

109 # Read once, in the person's repository; nothing ever creates it. 

110 ref = f"refs/heads/{branch}" 

111 sha = await git( 

112 "-C", repo, "rev-parse", "-q", "--verify", ref, silent=f"{repo} has no {ref}" 

113 ) 

114 await git("clone", "-q", "--shared", "--no-checkout", repo, str(dest)) 

115 await git("-C", str(dest), "remote", "remove", "origin") 

116 # The checkout's branches and origin/* refs, for `git diff main...HEAD`: 

117 # read from the person's repository, objects through alternates, no 

118 # remote left behind. 

119 refs = [f"+refs/{kind}/*:refs/{kind}/*" for kind in ("heads", "remotes", "tags")] 

120 await git("-C", str(dest), "fetch", "-q", "--no-tags", "--update-head-ok", repo, *refs) 

121 await git("-C", str(dest), "checkout", "-q", "--detach", sha) 

122 except asyncio.CancelledError: 

123 await _subprocess.uncancellable(asyncio.to_thread(_empty, dest)) 

124 raise 

125 except Exception as exc: # anything at all: a failed copy never stops the item 

126 await asyncio.to_thread(_empty, dest) 

127 if isinstance(exc, CopyFailed): 

128 raise 

129 why = "git took too long" if isinstance(exc, subprocess.TimeoutExpired) else str(exc) 

130 raise CopyFailed(branch, why) from exc

133async def drop_dead_copies(db, run_dirs, *, before: float) -> list[Path]: 

134 """Remove every copy whose session is gone or no longer live, made before 

135 `before` (the boot time: a session started since has its folder before its 

136 row). What it removed.""" 

137 dead = [] 

138 for copy in sorted((run_dirs.base / item_paths.ITEMS).glob("*/sessions/*/copy")): 

139 sid, wid = copy.parent.name, copy.parent.parent.parent.name 

140 status = db.read(lambda c, sid=sid, wid=wid: store.session_status(c, sid, work_item_id=wid)) 

141 if status not in LIVE and copy.lstat().st_mtime < before: 

142 dead.append(copy) 

143 for copy in dead: 

144 await asyncio.to_thread(remove_copy, copy) 

145 logger.info("removed %s: its session is no longer running", copy) 

146 return dead
