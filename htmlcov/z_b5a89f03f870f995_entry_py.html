29def single_repo_target(repo: str, *, base_branch: str | None = None) -> WorkItemTarget: 

30 """The immutable target of an item filed against one repository. 

31 

32 The id is the constant `"target"`, not the repository's own configured id: 

33 a single-repository item has exactly one target, nothing reads the name, 

34 and a `repos.yaml` id is not available at every intake door. A *workspace* 

35 target carries real member ids and is Phase 6's. 

36 

37 Shared with the `/work-items` route, which dry-runs the same 

38 materialization before it files a bead -- two copies of this would let the 

39 dry run and the real one disagree about what gets trimmed. 

40 """ 

41 return WorkItemTarget.for_repository("target", base_branch=base_branch)

219def _store_attachments( 

220 run_dirs, work_item_id: str, attachments: list[dict] | None, *, repo: str, tag: str = "" 

221) -> list[dict]: 

222 """Copy each attachment into Kraft's own storage; return the rewritten records. 

223 

224 `path` is left alone — it is the destination inside the worktree, and 

225 `ensure_worktree` still needs it. Only `source` changes, from "where the 

226 caller had it" to "where Kraft keeps it", which is why `ensure_worktree` 

227 needs no change at all: it already prefers `source`. 

228 

229 Raises rather than skipping. Every other reader of an attachment is 

230 best-effort, and that is right for them; this one backs an irreversible 

231 decision. 

232 

233 `tag` names a re-snapshot apart from the copy it supersedes 

234 (`replace_attachments`), so the old one stays intact until the row stops 

235 naming it. 

236 """ 

237 if not attachments: 

238 return attachments or [] 

239 dest_dir = run_dirs.attachments / work_item_id 

240 dest_dir.mkdir(parents=True, exist_ok=True) 

241 stored = [] 

242 for a in attachments: 

243 src = Path(a["source"]) if a.get("source") else Path(repo) / a["path"] 

244 dest = dest_dir / f"{a['kind']}{tag}{Path(a['path']).suffix or '.md'}" 

245 # Not shutil.copyfile's own error message: it names two absolute paths 

246 # under $KRAFT_HOME and says nothing about which attachment this was. 

247 try: 

248 shutil.copyfile(src, dest) 

249 except OSError as exc: 

250 raise ValueError(f"cannot read the {a['kind']} attachment at {src}: {exc}") from exc 

251 stored.append({**a, "source": str(dest)}) 

252 return stored

255def replace_attachments( 

256 run_dirs, work_item_id: str, current: list[dict], changes: dict, added: list[dict], *, repo: str 

257) -> list[dict]: 

258 """A not-yet-started item's attachments after a PATCH (Kraft-s7c04.28): 

259 each kind named in `changes` is dropped, `added` (validated, one per kind) 

260 is snapshotted under a fresh name, and every other kind keeps the copy it 

261 has. Raises like intake does; `discard_attachments` clears whichever set 

262 the write did not keep.""" 

263 kept = [a for a in current if a["kind"] not in changes] 

264 tag = f"-{uuid.uuid4().hex[:8]}" 

265 fresh = _store_attachments(run_dirs, work_item_id, added, repo=repo, tag=tag) 

266 return sorted(kept + fresh, key=lambda a: a["kind"] != "spec")

269def discard_attachments(run_dirs, work_item_id: str, drop: list[dict], keep: list[dict]) -> None: 

270 """Delete the stored copies `drop` names and `keep` does not: only this 

271 item's own, and only the ones the caller made or superseded.""" 

272 stored = run_dirs.attachments / work_item_id 

273 kept = {a.get("source") for a in keep} 

274 for a in drop: 

275 source = a.get("source") 

276 if source and source not in kept and Path(source).parent == stored: 

277 Path(source).unlink(missing_ok=True)

288#: `Fixes Kraft-abc12` / `Closes: Kraft-abc12` in a commit message. Unlike a 

289#: bare id anywhere in prose, this is a promise the author made deliberately, 

290#: and it is the convention every forge already reads. It is also the signal 

291#: that tracks implementation: a bead fixed by a merged commit stayed open for 

292#: a day because nothing here read it. A line of its own, not the word 

293#: anywhere: a body retelling the brief ("the old path closes Kraft-x early") 

294#: promises nothing (Kraft-iaou3). 

295_TRAILER_RE = re.compile( 

296 r"^[ \t]*(?:Fixes|Closes):?[ \t]+(Kraft-[a-z0-9]+(?:\.[0-9]+)*)", re.I | re.M 

297)

300def _trailer_beads(messages: list[str]) -> list[str]: 

301 """Bead ids promised by `Fixes`/`Closes` trailers, deduped, order preserved.""" 

302 seen: list[str] = [] 

303 for message in messages: 

304 for match in _TRAILER_RE.findall(message): 

305 if match not in seen: 

306 seen.append(match) 

307 return seen

318async def close_beads(db, row, bd_cwd: str | None, run_dirs, *, by_hand: bool = False) -> None: 

319 """Close `row['bead_id']`, every id in `row['implements_beads']`, and every id 

320 a `Fixes`/`Closes` trailer on the item's own commits names -- but only when 

321 the item's change carries them (Kraft-iaou3). 

322 

323 A chain running to its end is not evidence on its own: two P1s once read 

324 "closed" with no code behind them, and the next work was planned on top. 

325 The evidence is the item's own branch changing something besides its own 

326 attachments (`base_ref..HEAD`, read from the row as it is *now* -- a 

327 rebase mid-walk moves `base_ref`, and the stale one would count the base's 

328 commits as the item's), or a member's merge request having merged. Without 

329 it the beads stay open and a `beads_left_open` event says why: a bead left 

330 open is found, a bead wrongly closed is not. `by_hand` is a person 

331 completing the item and asking for the close (Ruling 167) -- they have said 

332 where the work landed, so nothing here second-guesses it. 

333 

334 An item filed while bd was down has no `bead_id` (Kraft-7gy); this backfills 

335 one via a late `beads.intake` first, rather than leaving it untracked 

336 forever (Kraft-dr3n) -- and persists it to the row, same as if intake had 

337 filed it originally. Each id's failure is logged, not raised -- one bad bead 

338 must not stop the others from closing, same as the single-bead path before. 

339 """ 

340 row = db.read( 

341 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (row["id"],)).fetchone() 

342 ) 

343 cwd = row["bead_cwd"] or bd_cwd 

344 bead_id = row["bead_id"] 

345 if bead_id is None: 

346 try: 

347 bead_id = await beads.intake(row["title"], description=row["description"], cwd=cwd) 

348 except Exception as exc: # noqa: BLE001 -- any bd failure degrades, same as intake 

349 logger.warning("late bead intake failed for %r: %r", row["title"], exc) 

350 else: 

351 await db.write(lambda c: store.set_bead_id(c, row["id"], bead_id)) 

352 # `git_read` never raises, so a worktree already cleaned up, or a row with no 

353 # `base_ref`, reads as no change at all. 

354 worktree = run_dirs.worktrees / row["id"] 

355 base = row["base_ref"] 

356 log = git_read(worktree, "log", "--format=%B%x00", f"{base}..HEAD") if base else None 

357 changed = git_read(worktree, "diff", "--name-only", base, "HEAD") if base else None 

358 stated = _implements_beads(row) 

359 trailers = _trailer_beads(log.split("\0") if log else []) 

360 ids = [b for b in (bead_id, *stated) if b] 

361 ids += [x for x in trailers if x not in ids] 

362 if not by_hand and not _carries_a_change(db, row, changed): 

363 if ids: 

364 reason = ( 

365 "the item completed without changing anything but its own attachments, " 

366 "and no member merge request merged; close them by hand once the work lands" 

367 ) 

368 logger.warning("beads left open for %s: %s", row["id"], reason) 

369 await db.write( 

370 lambda c: events.append( 

371 c, row["id"], "beads_left_open", {"beads": ids, "reason": reason} 

372 ) 

373 ) 

374 return 

375 for one in ids: 

376 try: 

377 await beads.complete(one, cwd=cwd) 

378 except Exception as exc: # noqa: BLE001 

379 logger.warning("bead close failed for %s: %r", one, exc)

382def _carries_a_change(db, row, changed: str | None) -> bool: 

383 """Whether the item's branch changed a path that is not one of its own 

384 attachments, or one of its member repos' merge requests merged.""" 

385 attached = {a["path"] for a in attachments_of(row)} 

386 if set((changed or "").splitlines()) - attached: 

387 return True 

388 return any(r["state"] == "merged" for r in db.read(lambda c: store.repos_for(c, row["id"])))
