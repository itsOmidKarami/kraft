33def one_line_title(title: str, description: str | None) -> tuple[str, str | None]: 

34 """`title` folded to its first line, the rest put at the top of `description`. 

35 

36 A title is one line: a line break broke `kraft view list`'s table. The 

37 doors a person types at refuse one (`api/routes/work_items._check_title`), 

38 but a `policy.yaml` cron trigger has no one to refuse, and a YAML block 

39 scalar (`title: >`, unless `>-`) ends in a line break. Duplicate copies a title 1.4 

40 stored as it was. Those are folded here instead, so the item still files 

41 with nothing of its text lost. Control characters are dropped and a tab 

42 becomes a space. A title with no text takes the description's first line 

43 that has some; with none there either, it comes back empty, for the 

44 caller to refuse. 

45 """ 

46 lines = [plain_text(line) for line in title.splitlines()] 

47 first = next((i for i, line in enumerate(lines) if line.strip()), None) 

48 if first is None: 

49 fallback = next((ln for ln in (description or "").splitlines() if ln.strip()), "") 

50 return (one_line_title(fallback, None)[0] if fallback else ""), description 

51 if len(lines) == 1 and title.splitlines() == [title]: 

52 return lines[0], description 

53 head = lines[first].strip() 

54 rest = "\n".join(lines[first + 1 :]).strip() 

55 if not rest: 

56 return head, description 

57 return head, f"{rest}\n\n{description}" if description else rest

388def _store_attachments( 

389 run_dirs, work_item_id: str, attachments: list[dict] | None, *, repo: str, tag: str = "" 

390) -> list[dict]: 

391 """Copy each attachment into Kraft's own storage; return the rewritten records. 

392 

393 `path` is left alone — it is the destination inside the worktree, and 

394 `ensure_worktree` still needs it. Only `source` changes, from "where the 

395 caller had it" to "where Kraft keeps it", which is why `ensure_worktree` 

396 needs no change at all: it already prefers `source`. 

397 

398 Raises rather than skipping. Every other reader of an attachment is 

399 best-effort, and that is right for them; this one backs an irreversible 

400 decision. 

401 

402 `tag` names a re-snapshot apart from the copy it supersedes 

403 (`replace_attachments`), so the old one stays intact until the row stops 

404 naming it. 

405 

406 The copy goes to the item's `filed/` (`item_paths.filed_dir`), which no 

407 sandbox mounts and no harness is granted, so a worker cannot rewrite 

408 what a person filed. 

409 """ 

410 if not attachments: 

411 return attachments or [] 

412 dest_dir = item_paths.filed_dir(run_dirs, work_item_id) 

413 dest_dir.mkdir(parents=True, exist_ok=True) 

414 stored = [] 

415 for a in attachments: 

416 src = Path(a["source"]) if a.get("source") else Path(repo) / a["path"] 

417 dest = dest_dir / f"{a['kind']}{tag}{Path(a['path']).suffix or '.md'}" 

418 # Not shutil.copyfile's own error message: it names two absolute paths 

419 # under $KRAFT_HOME and says nothing about which attachment this was. 

420 try: 

421 shutil.copyfile(src, dest) 

422 except OSError as exc: 

423 raise ValueError(f"cannot read the {a['kind']} attachment at {src}: {exc}") from exc 

424 stored.append({**a, "source": str(dest)}) 

425 return stored

428def replace_attachments( 

429 run_dirs, work_item_id: str, current: list[dict], changes: dict, added: list[dict], *, repo: str 

430) -> list[dict]: 

431 """A not-yet-started item's attachments after a PATCH (Kraft-s7c04.28): 

432 each kind named in `changes` is dropped, `added` (validated, one per kind) 

433 is snapshotted under a fresh name, and every other kind keeps the copy it 

434 has. Raises like intake does; `discard_attachments` clears whichever set 

435 the write did not keep.""" 

436 kept = [a for a in current if a["kind"] not in changes] 

437 tag = f"-{uuid.uuid4().hex[:8]}" 

438 fresh = _store_attachments(run_dirs, work_item_id, added, repo=repo, tag=tag) 

439 return sorted(kept + fresh, key=lambda a: a["kind"] != "spec")

465#: `Fixes Kraft-abc12` / `Closes: Kraft-abc12` in a commit message. Unlike a 

466#: bare id anywhere in prose, this is a promise the author made deliberately, 

467#: and it is the convention every forge already reads. It is also the signal 

468#: that tracks implementation: a bead fixed by a merged commit stayed open for 

469#: a day because nothing here read it. A line of its own, not the word 

470#: anywhere: a body retelling the brief ("the old path closes Kraft-x early") 

471#: promises nothing (Kraft-iaou3). 

472_TRAILER_RE = re.compile( 

473 r"^[ \t]*(?:Fixes|Closes):?[ \t]+(Kraft-[a-z0-9]+(?:\.[0-9]+)*)", re.I | re.M 

474)

478 """Bead ids promised by `Fixes`/`Closes` trailers, deduped, order preserved."""

497 which a late intake may have just filed, else the row's), those it 

498 implements, then those a `Fixes`/`Closes` trailer on its own commits names. 

499 The cancel preview reads it too, so Mark complete names every bead its box 

500 would close.""" 

501 worktree = inplace.item_folder(run_dirs, row) 

502 base = row["base_ref"] 

503 head = inplace.head_ref(row) 

504 log = git_read(worktree, "log", "--format=%B%x00", f"{base}..{head}") if base else None 

505 own = bead_id or row["bead_id"] 

506 ids = [b for b in (own, *_implements_beads(row)) if b] 

507 return list(dict.fromkeys([*ids, *_trailer_beads(log.split("\0") if log else [])]))

510async def close_beads(db, row, bd_cwd: str | None, run_dirs, *, by_hand: bool = False) -> None: 

511 """Close `row['bead_id']`, every id in `row['implements_beads']`, and every id 

512 a `Fixes`/`Closes` trailer on the item's own commits names -- but only when 

513 the item's change carries them (Kraft-iaou3). 

514 

515 A chain running to its end is not evidence on its own: two P1s once read 

516 "closed" with no code behind them, and the next work was planned on top. 

517 The evidence is the item's own branch changing something besides its own 

518 attachments (`base_ref..HEAD`, read from the row as it is *now* -- a 

519 rebase mid-walk moves `base_ref`, and the stale one would count the base's 

520 commits as the item's), or a member's merge request having merged. Without 

521 it the beads stay open and a `beads_left_open` event says why: a bead left 

522 open is found, a bead wrongly closed is not. `by_hand` is a person 

523 completing the item and asking for the close (Ruling 167) -- they have said 

524 where the work landed, so nothing here second-guesses it. 

525 

526 An item filed while bd was down has no `bead_id` (Kraft-7gy); this backfills 

527 one via a late `beads.intake` first, rather than leaving it untracked 

528 forever (Kraft-dr3n) -- and persists it to the row, same as if intake had 

529 filed it originally. Each id's failure is logged, not raised -- one bad bead 

530 must not stop the others from closing, same as the single-bead path before. 

531 """ 

532 row = db.read( 

533 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (row["id"],)).fetchone() 

534 ) 

535 cwd = row["bead_cwd"] or bd_cwd 

536 bead_id = row["bead_id"] 

537 if bead_id is None: 

538 try: 

539 bead_id = await beads.intake(row["title"], description=row["description"], cwd=cwd) 

540 except Exception as exc: # noqa: BLE001 -- any bd failure degrades, same as intake 

541 logger.warning("late bead intake failed for %r: %r", row["title"], exc) 

542 else: 

543 await db.write(lambda c: store.set_bead_id(c, row["id"], bead_id)) 

544 # `git_read` never raises, so a worktree already cleaned up, or a row with no 

545 # `base_ref`, reads as no change at all. 

546 base = row["base_ref"] 

547 worktree = inplace.item_folder(run_dirs, row) 

548 changed = ( 

549 git_read(worktree, "diff", "--name-only", base, inplace.head_ref(row)) if base else None 

550 ) 

551 ids = named_beads(row, run_dirs, bead_id) 

552 if not by_hand and not _carries_a_change(db, row, changed): 

553 if ids: 

554 reason = ( 

555 "the item completed without changing anything but its own attachments, " 

556 "and no member merge request merged; close them by hand once the work lands" 

557 ) 

558 logger.warning("beads left open for %s: %s", row["id"], reason) 

559 await db.write( 

560 lambda c: events.append( 

561 c, row["id"], ChainEvent.BEADS_LEFT_OPEN, {"beads": ids, "reason": reason} 

562 ) 

563 ) 

564 return 

565 for one in ids: 

566 try: 

567 await beads.complete(one, cwd=cwd) 

568 except Exception as exc: # noqa: BLE001 

569 logger.warning("bead close failed for %s: %r", one, exc)

572async def settle_beads_on_abandon( 

573 row, bd_cwd: str | None, reason: str | None, *, ended: str = "abandoned" 

574) -> None: 

575 """Abandoning or cancelling (`ended` says which) is not finishing: the bead 

576 intake filed for the item is closed with the reason, and the beads it 

577 implements stay open with a note saying why the item was dropped. A failure is 

578 logged, never raised -- one bad bead must not fail the abandon, nor stop the 

579 others.""" 

580 cwd = row["bead_cwd"] or bd_cwd 

581 why = f"{ended}: {reason}" if reason else ended 

582 if row["bead_id"]: 

583 try: 

584 await beads.complete(row["bead_id"], cwd=cwd, reason=why) 

585 except Exception as exc: # noqa: BLE001 

586 logger.warning("bead close failed for %s: %r", row["bead_id"], exc) 

587 for one in _implements_beads(row): 

588 try: 

589 await beads.annotate(one, f"Kraft work item {row['id']} was {why}", cwd=cwd) 

590 except Exception as exc: # noqa: BLE001 

591 logger.warning("bead note failed for %s: %r", one, exc)

594def _carries_a_change(db, row, changed: str | None) -> bool: 

595 """Whether the item's branch changed a path that is not one of its own 

596 attachments, or one of its member repos' merge requests merged.""" 

597 attached = {a["path"] for a in attachments_of(row)} 

598 if set((changed or "").splitlines()) - attached: 

599 return True 

600 return any(r["state"] == "merged" for r in db.read(lambda c: store.repos_for(c, row["id"])))
