191def rebase_drift_note(worktree, base: str, old_base: str, new_base: str) -> str: 

192 """A short pointer at what changed upstream during a rebase bounce back 

193 to `verify` (Kraft-4bgg): commit subjects and touched files, not a full 

194 diff, so the next review agent knows what to go looking for instead of 

195 blindly re-reviewing everything from scratch. 

196 """ 

197 log = git_read(worktree, "log", "--oneline", f"{old_base}..{new_base}") or "(no log)"

200 if len(body) > _REBASE_NOTE_MAX: 

201 body = body[:_REBASE_NOTE_MAX] + "\n... (truncated)" 

202 return _REBASE_PROMPT.format(base=base, body=body)

205#: What the rebase-conflict resolver (`walk.resolve_rebase_conflict`, 

206#: Kraft-s7c04.23) is dispatched with. Two jobs, not one -- resolving the 

207#: conflict is mechanical; judging whether the upstream commits it rebased 

208#: onto change what this work item is supposed to do is the reason an agent 

209#: is doing this rather than a script (design §4.1). A real diff, not a 

210#: `--stat`: that relevance judgement cannot be made from file names alone. 

211_REBASE_RESOLVE_PROMPT = ( 

212 "A rebase conflict stopped this work item. Branch {branch} could not be " 

213 "replayed onto {new_base} (it was based on {old_base}); the rebase has " 

214 "already been aborted{folder} git reported:\n\n" 

215 "{conflict}\n\n" 

216 "What landed upstream, {old_base}..{new_base}:\n\n{diff}\n\n" 

217 "{attachments}" 

218 "You have two jobs:\n" 

219 "1. {rerun}\n" 

220 "2. Judge whether the commits that landed upstream change what this work " 

221 "item is supposed to do -- not just whether the text merges cleanly. A " 

222 "clean rebase can still invalidate the plan; a conflict can be " 

223 "irrelevant to it.\n\n" 

224 "Report through the normal result file:\n" 

225 "- status: done -- resolved, and upstream does not affect what this " 

226 "item does.\n" 

227 "- status: done_with_concerns -- resolved, but upstream touches what " 

228 "this item does; say what in `concerns`.\n" 

229 "- status: needs_context -- upstream invalidates the agreed spec or " 

230 "plan; ask in `question`.\n" 

231 "- status: failed -- could not resolve the conflict.\n" 

232) 

233 

234#: Where the resolver works and how it re-runs the rebase there: a clean 

235#: worktree; or in place a person's checkout, whose uncommitted edits git 

236#: refuses a plain rebase over and the agent may not stash (`agent._IN_PLACE`), 

237#: so `--autostash` sets them aside, as Kraft's own rebase does 

238#: (`inplace.rebase_in_checkout`). A put-back that fails stops the item after 

239#: the task (`inplace.check_task`). 

240_RESOLVE_IN = { 

241 False: ( 

242 " and the worktree is clean.", 

243 "{run} the rebase (`git rebase {new_base}`) and resolve the conflict " 

244 "yourself, then finish it (`git rebase --continue`).", 

245 ), 

246 True: ( 

247 ". This is a person's own checkout: it may hold their uncommitted " 

248 "edits, which you must neither commit nor discard.", 

249 "{run} the rebase with `git -c rebase.updateRefs=false rebase " 

250 "--autostash {new_base}`: git sets their edits aside for the rebase and " 

251 "puts them back when it ends. That is the one exception to the rule " 

252 "against `git stash`; never stash, pop or drop anything yourself. " 

253 "Resolve the conflict by editing the conflicted files and `git add`-ing " 

254 "them by literal path, then finish it (`git rebase --continue`). If you " 

255 "cannot resolve it, `git rebase --abort` puts their edits back too. When " 

256 "the rebase ends and git says applying the autostash resulted in " 

257 "conflicts, the conflict markers in those files are the person's to " 

258 "resolve: leave those files and the stash alone, and report " 

259 "`done_with_concerns` naming them. If git refuses to start because an " 

260 "untracked file of theirs would be overwritten, do not move or remove " 

261 "it: report `failed`.", 

262 ), 

263} 

264 

265#: Same posture as `_REBASE_NOTE_MAX`, but larger: the resolver's whole job 

266#: is judging this diff's relevance, unlike the drift note's reviewer, who 

267#: only needs a pointer at what to go re-look at. 

268_REBASE_RESOLVE_DIFF_MAX = 6000

494 "\n\nImplementing this work item is a different node's job. Your own task is " 

495 "the `## Method` section of your system prompt; the brief above is the " 

496 "context that task runs in." 

497)

499# A repo's own tracking-issue guidance (e.g. CLAUDE.md's beads workflow) tells 

500# any agent to close a bead once it judges the work done. That is right for a

504# closes the work item's own tracking bead itself, once the chain actually 

505# completes (see `kraft.executor.entry.close_beads` below) -- a worker closing 

506# any bead, including its own, only duplicates or races that (Kraft-a03). 

507BEAD_NOTE = ( 

508 "\n\nDo not run `bd close` on any bead, including this work item's own " 

509 "tracking bead. Kraft closes it automatically once the whole chain " 

510 "completes; closing it here would mark work done before verify, review "

515def brief(work_item_row) -> str: 

516 """What the work item is, as an agent is told it. 

517 

518 The title is a label; the description is the actual brief, and the spec node 

519 is expected to write a design from it. A work item with no description is 

520 the title alone — exactly the string this returned before descriptions 

521 existed. 

522 """ 

523 description = work_item_row["description"] 

524 if not description: 

525 return work_item_row["title"] 

526 return f"{work_item_row['title']}\n\n{description}"

529def attachment_note(attachments: list[dict], *, method_is_own: bool = False) -> str: 

530 """Documents attached at intake, as an agent is told about them. 

531 

532 The implementer (`method_is_own=False`, the default) gets the imperative: 

533 follow the documents, do not re-plan. A hook with its own skill 

534 (`method_is_own=True`) gets the same paths but is told to judge the change 

535 against them, not to build from them. 

536 """ 

537 if not attachments: 

538 return "" 

539 lines = "\n".join(f"{a['kind'].capitalize()}: {a['path']}" for a in attachments) 

540 template = _ATTACHMENT_REFERENCE if method_is_own else _ATTACHMENT_PROMPT 

541 return template.format(lines=lines)

553_PROGRESS_NOTE_NO_CHANNEL = (

558def progress_note( 

559 task: AgentTask, work_item_row, worktree, db=None, sandbox=None, *, run_dirs 

560) -> str: 

561 """Only for the task doing the work from the brief, and only for a plan with 

562 `## Task N` headings to count. 

563 

564 "The task doing the work" is an agent task with no `skill:` of its own: a 

565 task that selects a skill states its own method (write a spec, judge a 

566 change), and only the one working from the brief walks a plan. V1 has no 

567 hook name to key this on, and keying it on a node id was already wrong 

568 (Kraft-s7c04.45). 

569 """ 

570 if task.skill is not None: 

571 return "" 

572 if db is not None and _progress.rework_run(db, work_item_row): 

573 # Kraft-hj2q9: a run a rejection bounced back follows the note, not the plan. 

574 return "" 

575 # In place, the plan on the item's branch: an escalation's or a reply's 

576 # prompt is built before the take, while the checkout may be the person's. 

577 in_place = _store.is_in_place(work_item_row) 

578 tasks = _progress.tasks_for(work_item_row, Path(worktree), run_dirs, in_place=in_place) 

579 note = _PROGRESS_NOTE if _callback.reachable(sandbox) else _PROGRESS_NOTE_NO_CHANNEL 

580 return note.format(total=len(tasks)) if tasks else ""

584#: (`task.skill is None`) is told to address and reply to each; a reviewer is 

585#: told to judge the change against them and never reply itself -- that is 

586#: the implementing agent's job, and a reviewer replying would blur who is 

587#: answering whom (review threads anywhere §1). 

588_THREADS_FOR_WORKER = ( 

589 "\n\nA person reviewing this work left these threads. Address each one, then " 

590 'answer it: `kraft item reply <thread-id> --claim fixed --body "..."` when you ' 

591 "changed the code, `--claim answered` when you are replying without a change.\n\n" 

592) 

593_THREADS_FOR_WORKER_NO_CHANNEL = ( 

594 "\n\nA person reviewing this work left these threads. Address each one. " 

595 "This sandbox has no route to Kraft, so you cannot reply to them yourself: " 

596 "say in your result's summary what you did about each, by thread id.\n\n" 

597) 

598_THREADS_FOR_REVIEWER = ( 

599 "\n\nA person reviewing this work left these threads. Check whether the change " 

600 "addresses each one. Report an unaddressed `must_fix` thread as a finding. Do not " 

601 "reply to the threads; that is the implementing agent's job.\n\n" 

602)

605def review_threads_note(task: AgentTask, work_item_row, db, note: str | None, sandbox=None) -> str: 

606 """The item's unanswered review threads, for every agent launch, framed by 

607 role (review threads anywhere §1). A thread the note already lists -- the 

608 rejection note on a rework run -- is left out, so none appears twice.""" 

609 threads = db.read(lambda c: _store.unanswered(c, work_item_row["id"])) 

610 threads = [t for t in threads if f"[{t['id']}]" not in (note or "")] 

611 body = _store.render_threads(threads) 

612 if not body: 

613 return "" 

614 if task.skill is not None: 

615 return _THREADS_FOR_REVIEWER + body 

616 return ( 

617 _THREADS_FOR_WORKER if _callback.reachable(sandbox) else _THREADS_FOR_WORKER_NO_CHANNEL 

618 ) + body
