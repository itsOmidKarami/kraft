1"""Where a sandboxed task runs: one backend per `sandbox: {kind: ...}`. 

2 

3What a `sandbox:` value looks like is `policy.SandboxPolicy`'s, and which one 

4an item runs in is `dispatch.item_sandbox`'s. A backend is how a command 

5actually runs inside one, from the image check before a session exists to 

6cleaning up after the item is gone. `kraft.worker.sandbox` holds what every 

7backend shares: host git hardening and the guards on a sandboxed worktree. 

8 

9A backend owns how code crosses its boundary. Docker bind-mounts the worktree, 

10so its `collect` has nothing to fetch; a backend with no shared filesystem 

11(a VM, a cloud sandbox) would copy code in and results out instead. So a 

12caller never reads what a sandboxed session left in the worktree before the 

13session's backend has had its `code_out` and `collect`: that ordering is what 

14lets such a backend exist without reworking every caller. 

15"""

49 async def prepare(self, sandbox: dict, *, kraft_ca: Path | None = None) -> Path | None: 

50 """Ready what a launch in `sandbox` needs on the host side, after a 

51 successful `probe`: the CA bundle to hand `wrap` as `ca_bundle`, or 

52 None for none -- with `kraft_ca` (the Kraft CA's certificate, when 

53 the launch has a proxy-managed credential) trusted too. Raises 

54 `worker.sandbox.SandboxNotReady` saying why the launch cannot go 

55 ahead (`config_error`)."""

57 def code_in(self, run_base: Path, cwd: Path, branch: str | None, **kw) -> Sequence[RefStore]: 

58 """Put the worktree's code where the session will see it, and return 

59 what `wrap` mounts as `refs=` and `code_out` publishes. `branch` 

60 None is a setup command: it sees the code and publishes nothing. 

61 `members=` is each workspace member's trusted `(common gitdir, admin 

62 dir)` by mount path (`sandbox.Checkout`), code the session sees too."""

64 def wrap(self, cmd: list[str], cwd, sandbox: dict, results, env=None, **kw) -> list[str]: 

65 """The argv that runs `cmd` inside the sandbox; `results` is the files of 

66 the results folder it may read (`store.result_files`), None for none; 

67 `ca_bundle=` is 

68 what `prepare` returned, `sentinels=` each proxy-managed 

69 credential's variable and what it holds in the value's place, 

70 `workdir=` where under `cwd` the command runs when not at its top."""

72 def oneshot(self, sandbox: dict, cwd, home): 

73 """Short commands in the session's image, asked as its session would

95 async def collect(self, session_id: str, result_path: Path) -> None: 

96 """Bring the session's result file back to `result_path`. A no-op

99 async def oom_killed(self, session_id: str) -> OomKill | None: 

100 """Whether the session's sandbox was killed under its memory limit, 

101 and whether the runtime confirmed the limit did it; None when it was 

102 not, or this backend cannot tell (a session it never ran). Asked 

103 after the command exits and before `close`, which removes the 

104 evidence. An exit code alone never counts: an unconfirmed kill needs 

105 a memory limit, and `record_oom_kill` drops one Kraft stopped itself.""" 

106 

107 async def close(self, session_id: str) -> None: 

108 """Stop and remove the session's sandbox, best-effort and bounded. A 

109 no-op for a session this backend never ran.""" 

110 

111 async def egress_transport(self) -> str: 

112 """How this backend's sessions reach the egress channel, asked before 

113 it opens: "unix", a socket per session, or "tls", the daemon's one 

114 mTLS listener (a runtime in a VM cannot connect to a host socket). 

115 Raises `worker.sandbox.SandboxNotReady` when it cannot tell."""

117 async def open_session(self, session_id: str, sandbox: dict, sock_path: Path | None) -> dict: 

118 """Ready the session's egress route under `sandbox['network']` 

119 before it starts, to the channel listening at `sock_path` 

120 (`worker.channel`), None under the "tls" transport: docker starts a 

121 relay the worker joins. Returns 

122 the environment the launch must add (its proxy, and a PATH with the 

123 `kraft` shim first); `{}` for a sandbox 

124 with no `network`. Raises `worker.sandbox.SandboxNotReady` when the 

125 route cannot be made (`config_error`), never opens egress instead."""

127 async def close_session(self, session_id: str) -> None: 

128 """Undo `open_session`, best-effort and bounded like `close`, and a 

129 no-op for a session with no route. The channel itself is closed by 

130 whoever opened it (`run_task`, reattach), not by the backend."""

149def for_sandbox(sandbox: Mapping) -> SandboxBackend: 

150 """The backend that runs `sandbox` (a `SandboxPolicy` dumped to a dict, 

151 the form `dispatch.item_sandbox` hands out).""" 

152 return _BACKENDS[sandbox["kind"]]

155def every() -> tuple[SandboxBackend, ...]: 

156 return tuple(_BACKENDS.values())

159def for_session(kind: str | None) -> tuple[SandboxBackend, ...]: 

160 """The backends a session row may have run in: the one it recorded, or 

161 every one for a row that recorded none -- unsandboxed, or written before 

162 the column existed. Each backend's session calls are no-ops for a 

163 session it never ran, which is what makes asking all of them safe; so is 

164 a kind no backend runs any more.""" 

165 backend = _BACKENDS.get(kind) if kind is not None else None 

166 return (backend,) if backend is not None else every()
