1"""What of the daemon's network setup crosses into a docker container: its 

2proxy, and the extra root certificates it trusts. 

3 

4Nothing did before (Kraft sandbox part 2, P3): `docker_argv` forwarded only 

5Kraft's own variables and the repository's, so a host behind a proxy or a 

6TLS-intercepting CA had sandboxed workers that could reach nothing. The 

7daemon's values cannot cross as they are, either: 

8 

9- `SSL_CERT_FILE` names a host file the container does not have, which breaks 

10 TLS rather than extending it. So when there is an extra CA, Kraft builds one 

11 bundle -- the image's own roots plus the extra CA -- and mounts it at one 

12 fixed container path (`CONTAINER_CA`), the only path any CA variable names. 

13 A combined bundle, because `SSL_CERT_FILE` and most of its siblings replace 

14 the store rather than add to it. 

15- A proxy on the host's loopback means the container's own loopback inside 

16 it, where nothing listens. Such a proxy is not forwarded (`forwarded_proxies`) 

17 and doctor says so (`loopback_proxies`); `network:` (a later part) is the fix. 

18 

19The proxy variables are forwarded whenever the daemon has them. The CA side 

20only acts when there is an extra CA, or a launch whose credentials the 

21egress proxy manages (P6), whose bundle also holds the Kraft CA the proxy 

22terminates their TLS with: otherwise there is no probe, no mount and no CA 

23variable, as before. The bundle `prepare` builds is handed to the 

24launch explicitly (`DockerBackend.wrap(..., ca_bundle=)`), never remembered. 

25"""

85def _is_loopback(value: str) -> bool: 

86 """Does a proxy URL point at this machine's loopback, where a container 

87 cannot follow it? `127.0.0.0/8` (IPv4-mapped too), `::1`, `localhost` 

88 (and the unspecified address, which a client connects to as loopback).""" 

89 target = value if "://" in value else f"http://{value}" 

90 try: 

91 host = urlsplit(target).hostname 

92 except ValueError: 

93 return False 

94 if not host: 

95 return False 

96 if host == "localhost" or host.endswith(".localhost"): 

97 return True 

98 try: 

99 address = ipaddress.ip_address(host) 

100 except ValueError: 

101 return False 

102 # An IPv4-mapped `::ffff:127.0.0.1` is unwrapped by hand: `is_loopback` and 

103 # `is_unspecified` read through `ipv4_mapped` themselves only from Python 

104 # 3.13, and from a late 3.12.x patch release. 

105 if getattr(address, "ipv4_mapped", None) is not None: 

106 address = address.ipv4_mapped 

107 return address.is_loopback or address.is_unspecified

110def loopback_proxies(environ: Mapping[str, str] | None = None) -> dict[str, str]: 

111 """The daemon's proxy variables that point at its loopback: `{name: value}`.""" 

112 environ = os.environ if environ is None else environ 

113 return { 

114 name: environ[name] 

115 for name in PROXY_VARS 

116 if name.lower() != "no_proxy" and environ.get(name) and _is_loopback(environ[name]) 

117 }

120def forwarded_proxies(environ: Mapping[str, str] | None = None) -> list[str]: 

121 """The proxy variables to forward bare into a container: every one the 

122 daemon sets, except a proxy on its loopback. Forwarded, that one fails 

123 every request (nothing listens on the container's own loopback); 

124 left out, the container at least goes direct where the host can.""" 

125 environ = os.environ if environ is None else environ 

126 skip = loopback_proxies(environ) 

127 return [name for name in PROXY_VARS if environ.get(name) and name not in skip]

133def certificates(text: str) -> list[str]: 

134 """The PEM certificates in `text`, in order; anything else dropped.""" 

135 return _PEM.findall(text)

138def _read_pem(path: Path) -> list[str]: 

139 """`path`'s certificates. Raises `ValueError` saying why there are none 

140 to use: unreadable (missing, a directory, no permission) or no PEM.""" 

141 try: 

142 text = path.read_text(errors="replace") 

143 except OSError as exc: 

144 raise ValueError(f"{path} cannot be read ({exc.strerror or exc})") from exc 

145 found = certificates(text) 

146 if not found: 

147 raise ValueError(f"{path} holds no PEM certificate") 

148 return found

158def extra_ca(environ: Mapping[str, str] | None = None) -> tuple[str, list[str]] | None: 

159 """`(where it came from, its certificates)`: `sandbox.yaml`'s 

160 `ca_bundle`, else the daemon's `SSL_CERT_FILE`, else None. 

161 

162 A `ca_bundle` that cannot be used raises `ConfigError`: someone named it 

163 for sandboxes, and skipping it would launch a container that fails TLS 

164 later with nothing saying why. An unusable `SSL_CERT_FILE` is only 

165 ignored (`ignored_ssl_cert_file` says why, for doctor): it was never set 

166 for sandboxes, and refusing on it would stop launches that ran before.""" 

167 environ = os.environ if environ is None else environ 

168 host = _sandbox_host(environ) 

169 if host.ca_bundle is not None: 

170 try: 

171 return f"sandbox.yaml `ca_bundle` ({host.ca_bundle})", _read_pem(host.ca_bundle) 

172 except ValueError as exc: 

173 raise ConfigError( 

174 f"sandbox.yaml `ca_bundle`: {exc}, so a sandboxed task could not trust its " 

175 "CA; fix the path or remove it (see the sandbox.yaml reference)" 

176 ) from exc 

177 if not environ.get("SSL_CERT_FILE"): 

178 return None 

179 path = Path(environ["SSL_CERT_FILE"]) 

180 try: 

181 return f"the daemon's SSL_CERT_FILE ({path})", _read_pem(path) 

182 except ValueError: 

183 return None

186def ignored_ssl_cert_file(environ: Mapping[str, str] | None = None) -> str | None: 

187 """Why the daemon's `SSL_CERT_FILE` is not used as the extra CA although 

188 it is set, or None when it is used or does not apply. Raises 

189 `ConfigError` for a `sandbox.yaml` that does not parse.""" 

190 environ = os.environ if environ is None else environ 

191 if not environ.get("SSL_CERT_FILE") or _sandbox_host(environ).ca_bundle is not None: 

192 return None 

193 try: 

194 _read_pem(Path(environ["SSL_CERT_FILE"])) 

195 except ValueError as exc: 

196 return str(exc) 

197 return None

200def _write(path: Path, text: str) -> None: 

201 """`text` at `path`, atomically, and only when it changed: a running 

202 container keeps the file it mounted, the next launch gets the new one.""" 

203 try: 

204 if path.read_text() == text: 

205 return 

206 except OSError: 

207 pass 

208 fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=f".{path.name}.") 

209 try: 

210 with os.fdopen(fd, "w") as fh: 

211 fh.write(text) 

212 os.chmod(tmp, 0o644) 

213 os.replace(tmp, path) 

214 except BaseException: 

215 Path(tmp).unlink(missing_ok=True) 

216 raise

228async def _read_image_roots(image: str) -> list[str] | None: 

229 """The image's own roots, read by a container with no network and its 

230 entrypoint bypassed (whatever one prints is not a certificate). `[]` 

231 when the image holds none of `IMAGE_ROOTS`; None when the run itself 

232 failed (a pull that timed out, no `sh`, the daemon).""" 

233 from kraft.worker.backends.docker import docker_call, home_label 

234 

235 ran = await docker_call( 

236 "run", 

237 "--rm", 

238 "--label", 

239 home_label(), 

240 "--network=none", 

241 "--cap-drop=ALL", 

242 "--security-opt=no-new-privileges", 

243 "--entrypoint=sh", 

244 image, 

245 "-c", 

246 'for f in "$@"; do if [ -f "$f" ]; then cat "$f"; exit 0; fi; done', 

247 "sh", 

248 *IMAGE_ROOTS, 

249 timeout=300, 

250 ) 

251 if ran is None or ran[0] != 0: 

252 return None 

253 return certificates(ran[1])

261async def prepare(image: str, *, kraft_ca: Path | None = None) -> Path | None: 

262 """`image`'s combined CA bundle on the host, built now, or None when there 

263 is neither an extra CA nor a `kraft_ca`: the Kraft CA's certificate, 

264 given when the launch has a proxy-managed credential (spec §7), whose 

265 bundle is a file of its own -- a launch without one never trusts it, 

266 and never has its bundle rewritten under it. Raises `ConfigError` for a 

267 `ca_bundle` that cannot be used (`extra_ca`) and for an image whose own 

268 roots could not be read: a bundle of the extra CA alone would replace 

269 the store and fail TLS to every public host. 

270 

271 One bundle per image id under `run/sandbox-ca/`: `<id>.roots.pem`, the 

272 image's own roots, read once per image id (an id is immutable); `<id>.pem`, 

273 those plus the extra CA, rewritten when the extra CA changes; 

274 `<id>.kraft.pem`, those plus the Kraft CA.""" 

275 extra = extra_ca() 

276 if extra is None and kraft_ca is None: 

277 return None 

278 extra_certs = extra[1] if extra is not None else [] 

279 if kraft_ca is not None: 

280 extra_certs = [*extra_certs, *certificates(kraft_ca.read_text())] 

281 base = ca_dir() 

282 base.mkdir(parents=True, exist_ok=True) 

283 image_id = await _image_id(image) 

284 roots_file = base / f"{image_id}.roots.pem" if image_id else None 

285 if roots_file is not None and roots_file.is_file(): 

286 roots = certificates(roots_file.read_text()) 

287 else: 

288 roots = await _read_image_roots(image) 

289 if roots is None: 

290 raise ConfigError( 

291 f"could not read image {image!r}'s own root certificates (the image did not " 

292 "run: not pulled, no `sh`, or the runtime did not answer), and trusting the " 

293 "extra CA alone would fail TLS to every other host; pull the image, or check " 

294 "that it runs `sh`" 

295 ) 

296 # Run first, then ask its id: the run is what pulled an image that 

297 # was not here yet. 

298 image_id = image_id or await _image_id(image) 

299 if image_id: 

300 _write(base / f"{image_id}.roots.pem", "".join(f"{c}\n" for c in roots)) 

301 name = image_id or "image-" + hashlib.sha256(image.encode()).hexdigest()[:32] 

302 bundle = base / f"{name}{'.kraft' if kraft_ca is not None else ''}.pem" 

303 _write(bundle, "".join(f"{c}\n" for c in dict.fromkeys([*roots, *extra_certs]))) 

304 return bundle

307def ca_args(bundle: str | Path | None) -> tuple[list[str], dict[str, str]]: 

308 """`docker run` mount arguments and variables for a prepared `bundle`;
