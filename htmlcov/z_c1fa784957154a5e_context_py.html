14def absolute_path(path: str) -> str: 

15 """`path` made absolute against the caller's cwd, `~` expanded. 

16 

17 The server resolves a relative path against its OWN cwd, so `kraft repo 

18 connect .` connected whatever directory the daemon started in (Kraft-9efnk.32). 

19 Every client call that sends a filesystem path goes through this first. An 

20 absolute path comes back unchanged (no symlink resolution): the server 

21 already matches either spelling (`deps._connected`). 

22 """ 

23 return os.path.abspath(os.path.expanduser(path))

26def attachment_paths(spec: str | None, plan: str | None) -> list[dict] | None: 

27 """A spec and a plan to attach, as `create_work_item` takes them, each 

28 made absolute against the caller's cwd. The server joins a relative path 

29 onto a candidate root (the repo, or the worktree the call is made from), 

30 never onto the cwd, so `specs/x.md` from a subdirectory of either would 

31 miss both; absolute, it is still accepted only if it lands inside one of 

32 them. None for neither.""" 

33 found = [ 

34 {"kind": kind, "path": str(Path(value).expanduser().resolve())} 

35 for kind, value in (("spec", spec), ("plan", plan)) 

36 if value 

37 ] 

38 return found or None

41def resolve_context(cwd: Path | None = None) -> tuple[str | None, str]: 

42 """`(work_item_id, origin)` for the session calling us. 

43 

44 `origin` is `"worker"` only when `$KRAFT_WORK_ITEM_ID` is set, which the 

45 executor injects into sessions it starts. The self-action guard keys on it, 

46 so a worker that lost the variable would read as a human — which is why the 

47 guard and the injection have to ship together (design §4). 

48 

49 A human-started session is identified by its directory instead: the executor 

50 lays worktrees out as `run_dirs.worktrees / work_item_id`. 

51 """ 

52 wid = os.environ.get("KRAFT_WORK_ITEM_ID") 

53 if wid: 

54 return wid, "worker" 

55 worktrees = RunDirs(Path(os.environ.get("KRAFT_RUN_DIR") or default_run_dir())).worktrees 

56 try: 

57 worktrees = worktrees.resolve() 

58 here = (cwd or Path.cwd()).resolve() 

59 except OSError: 

60 return None, "user" 

61 for candidate in (here, *here.parents): 

62 if candidate.parent == worktrees: 

63 return candidate.name, "user" 

64 return None, "user"

67async def in_place_item(cwd: Path | None = None) -> str | None: 

68 """The in-place item of the cwd's connected repo whose branch is checked 

69 out there, or None. Kraft names every item's branch `kraft/...`, so on any 

70 other branch the server is not asked.""" 

71 branch = config.git_read( 

72 cwd or Path.cwd(), "symbolic-ref", "--short", "-q", "HEAD", expected_failure=True 

73 ) 

74 if not branch or not branch.startswith("kraft/"): 

75 return None 

76 repo = await resolve_repo(cwd) 

77 if repo is None: 

78 return None 

79 # shortcut: reads the whole board to find one row; a `?branch=` filter on 

80 # the list route if boards grow large enough for that to matter. 

81 items = (await transport._get("/work-items"))["items"] 

82 # `.get`: an older server's rows carry no `in_place`. 

83 return next( 

84 ( 

85 i["id"] 

86 for i in items 

87 if i.get("in_place") and i.get("branch") == branch and i["repo"] == repo 

88 ), 

89 None, 

90 )

93async def standing_item(cwd: Path | None = None) -> tuple[str | None, str]: 

94 """`resolve_context`, else the in-place item checked out where the 

95 session stands (`in_place_item`).""" 

96 wid, origin = resolve_context(cwd) 

97 return (wid, origin) if wid else (await in_place_item(cwd), origin)

100async def resolve_repo(cwd: Path | None = None) -> str | None: 

101 """The connected repo the cwd is inside, or None. 

102 

103 Reads `GET /repos` rather than parsing `repos.yaml`: a local YAML read would 

104 work with the server down, but it puts a second reader on config the API 

105 already owns and shapes, and every verb that uses this answer needs the 

106 server anyway. 

107 

108 Parents are walked so a submodule checkout resolves to the connected 

109 superproject. `config.normalized_repo_root`, not a bare `--show-toplevel`, 

110 so a linked worktree ($KRAFT_HOME worktree or any other) resolves to its 

111 main checkout instead of having no connected ancestor at all (Kraft-tc33). 

112 

113 Auto-connected child repos (`managed: false`, from `_auto_connect_children`) 

114 are skipped rather than returned: Phase 1 registers submodules as disabled 

115 entries for the Settings screen, but does not change item routing, so 

116 walking outward past a nearer disabled child still lands on the nearest 

117 *managed* ancestor, not the outermost one (Kraft, review of repos.py:173, 

118 context.py:60). 

119 """ 

120 here = cwd or Path.cwd() 

121 root = config.normalized_repo_root(here) 

122 if root is None: 

123 return None 

124 payload = await transport._get("/repos") 

125 connected = {entry["path"]: entry for entry in payload["repos"]} 

126 for candidate in (root, *root.parents): 

127 entry = connected.get(str(candidate)) 

128 if entry is not None and entry.get("managed", True): 

129 return str(candidate) 

130 return None

139async def _forbid_self_action(work_item_id: str | None) -> str: 

140 """Resolve the target of an act call, refusing a worker's own item. 

141 

142 Design §6 rule 2. A worker session approving its own gate would collapse the 

143 human-gate model, so the refusal sits at the boundary the agent cannot route 

144 around rather than in a skill's prose. Client-side because `origin` is known 

145 only here: `kraft.api.perimeter` sees an authenticated local caller either way. 

146 

147 A value the worker's own id starts with, `_MIN_PREFIX` characters or more, 

148 is its own item too: the server reads it as that id whenever no other id 

149 starts the same way, and no other id is the whole of it, since intake 

150 refuses a new id that is the start of an existing one. 

151 """ 

152 resolved, origin = resolve_context() if work_item_id else await standing_item() 

153 target = work_item_id or resolved 

154 if target is None: 

155 raise transport.ClientError( 

156 "no work item: pass an id, or run from a Kraft worktree", kind=transport.Failure.REFUSED 

157 ) 

158 own = target == resolved or ( 

159 resolved is not None and len(target) >= _MIN_PREFIX and resolved.startswith(target) 

160 ) 

161 if origin == "worker" and own: 

162 raise PermissionError( 

163 f"a worker session cannot act on its own work item ({resolved}). " 

164 "Gates are where a human decides; report what you found instead." 

165 ) 

166 return target

169async def resolve_work_item(work_item_id: str | None) -> str: 

170 """An explicit id, or the item this session is standing in. 

171 

172 Not `_forbid_self_action`: reading your own logs is exactly what a worker 

173 session should be able to do. The guard is about acting, not looking. 

174 

175 Public because `cli.py` must not become a second definition of what a 

176 work item is (Kraft-t5s9). 

177 """ 

178 if work_item_id: 

179 return work_item_id 

180 resolved, _origin = await standing_item() 

181 if resolved is None: 

182 raise transport.ClientError( 

183 "no work item: pass an id, or run from a Kraft worktree", kind=transport.Failure.REFUSED 

184 ) 

185 return resolved
