1"""The scopes below policy that set a cap, for the Policy page: its "set below 

2policy" column (`below_for`, in the policy draft's resolve) and a chain's 

3effective caps under the draft (`chain_scopes`, the "Preview on a chain" read). 

4 

5Both are derived from the published library's chains; the policy draft changes 

6only the instance policy they are held against.""" 

7 

8from __future__ import annotations

10import copy 

11 

12from kraft.cap_levels import LEVEL_OF, SCOPE_CAP_FIELDS 

13from kraft.drafts import authored, resolve 

14from kraft.policy import InstancePolicy, PolicyError 

15from kraft.templates.library import TemplateLibraryError 

16from kraft.templates.models import _scoped

18#: A source that is a layer below policy, not the policy's own value. 

19_BELOW = ("chain", "library:") 

20 

21 

22def _own(source: str) -> bool: 

23 return source.startswith(_BELOW) 

24

26def _cap(value, source: str, bound: tuple[str, float] | None) -> dict: 

27 return { 

28 "value": value, 

29 "source": source, 

30 "maximum": {"value": bound[1], "level": bound[0]} if bound else None, 

31 "exceeds": value is not None and bound is not None and value > bound[1], 

32 }

35def chain_scopes(st, chain_id: str, instance: InstancePolicy) -> list[dict] | None: 

36 """Per scope of the published chain (the chain itself, then every node, 

37 step and task; a gate waits for a person and has none), each cap with the 

38 value it runs under, the layer that set it (`chain`, `library:<section>.<name>`, 

39 `policy`, `default`) and the maximum bounding it at its level. `None` for a 

40 chain that does not resolve.""" 

41 library = getattr(st, "library", None) 

42 if library is None: 

43 return None 

44 try: 

45 resolved = library.resolve_chain(chain_id) 

46 data = authored.normalise( 

47 copy.deepcopy(authored.parse(library.chain_file(chain_id).read_text())) 

48 ) 

49 except (TemplateLibraryError, OSError): 

50 return None 

51 sources = resolve._sources(data, resolve._library_model(st, {}), resolved, instance) 

52 

53 base = instance 

54 own = resolved.chain.policy 

55 try: 

56 if own is not None: 

57 base = instance.apply_template_override(own) 

58 except PolicyError: 

59 base = instance 

60 out = [ 

61 { 

62 "path": chain_id, 

63 "kind": "chain", 

64 "level": "work_item", 

65 "caps": { 

66 cap: _cap( 

67 getattr(own, cap, None), 

68 "chain" if getattr(own, cap, None) is not None else "default", 

69 instance.maxima.nearest("work_item", cap), 

70 ) 

71 for cap in SCOPE_CAP_FIELDS 

72 }, 

73 } 

74 ] 

75 for path, kind, scope in resolved.cap_scopes(): 

76 if kind == "gate": 

77 continue 

78 level = LEVEL_OF[kind] 

79 try: 

80 effective = _scoped(scope, base) 

81 except PolicyError: 

82 effective = None # a scope the draft's maximum refuses: its own values still show 

83 fields = sources.get(path, {}) 

84 caps = {} 

85 for cap in SCOPE_CAP_FIELDS: 

86 set_here = fields.get(f"policy.{cap}") 

87 if set_here is not None and _own(set_here["source"]): 

88 value, source = set_here["value"], set_here["source"] 

89 else: 

90 value = getattr(effective, cap, None) 

91 source = "policy" if value is not None else "default" 

92 caps[cap] = _cap(value, source, instance.maxima.nearest(level, cap)) 

93 out.append({"path": path, "kind": kind, "level": level, "caps": caps}) 

94 return out

97def below_for(st, instance: InstancePolicy) -> dict[tuple[str, str], list[dict]]: 

98 """`{(level, cap): [{layer, chain, path, via, value, exceeds}]}`: every 

99 chain scope, in the published library, that sets `cap` itself, `layer` being 

100 `chain` (the chain file) or `library` (a component it extends, named in 

101 `via`), `exceeds` against the draft's maximum at that level.""" 

102 library = getattr(st, "library", None) 

103 out: dict[tuple[str, str], list[dict]] = {} 

104 for chain_id in library.chain_ids if library is not None else (): 

105 for scope in chain_scopes(st, chain_id, instance) or (): 

106 for cap, c in scope["caps"].items(): 

107 if not _own(c["source"]): 

108 continue 

109 library_layer = c["source"].startswith("library:") 

110 out.setdefault((scope["level"], cap), []).append( 

111 { 

112 "layer": "library" if library_layer else "chain", 

113 "chain": chain_id, 

114 "path": scope["path"], 

115 "via": c["source"] if library_layer else None, 

116 "value": c["value"], 

117 "exceeds": c["exceeds"], 

118 } 

119 ) 

120 return out
