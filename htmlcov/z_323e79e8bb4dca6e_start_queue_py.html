1"""The start queue: one scheduler that starts `queued` work items as slots free. 

2 

3A door that would start a walk and finds every `max_concurrent` slot busy 

4queues the item instead of refusing (`routes.lifecycle.queued_answer`). The 

5item keeps the request it was asked: the verb, its body and who asked. 

6 

7An item started while something it comes after is unfinished is `blocked` 

8(`routes.lifecycle.blocked_answer`) and holds the same saved request. Each 

9pass first moves every blocked item whose dependencies have all completed 

10into the queue (`store.release_blocked`), so there is one path to `active`. 

11 

12`tick` takes the earliest queued item while a slot is free, puts it back in 

13the status it came from, and makes that request again through the same route 

14function, with a `Request` rebuilt from the saved caller. So a queued start 

15runs every check, the rebase, the counter clearing and the steer delivery a 

16fresh request would, against the item as it is now. Four ends: 

17 

18* it starts, and the door writes its own event; 

19* the door now refuses it, or fails: the item stays where it came from, and 

20 `work_item_dequeued` carries the refusal or the error; 

21* another request took the slot first: the door queues it again, and it is 

22 given back the place it had; 

23* the server shuts down before the door claims it: it is queued again, in its 

24 place, and the next server starts it. 

25"""

44def _request(app, headers: dict[str, str]) -> Request: 

45 """The saved caller as the request a door reads: the app, and the headers 

46 that say who asked (`deps.CALLER_HEADERS`).""" 

47 return Request( 

48 { 

49 "type": "http",

51 "path": "/", 

52 "query_string": b"", 

53 "headers": [(k.encode(), v.encode()) for k, v in headers.items()], 

54 "app": app, 

55 }

59def starting(app) -> set[str]: 

60 """The ids `_start_one` is starting right now. Between its take and the 

61 door's claim the row reads `paused` or `needs_human`, so this is the only 

62 place that says a start is under way; `pause` reads it. In memory: the 

63 server is one process, and a restart ends every start in flight.""" 

64 ids = getattr(app.state, "_starting", None) 

65 if ids is None: 

66 ids = app.state._starting = set() 

67 return ids

70async def tick(app) -> list[str]: 

71 """One pass. Returns the work item ids started, which is usually none.""" 

72 st = app.state 

73 limit = st.policy.max_concurrent if st.policy else 1 

74 # Before the queue: an item whose dependencies have all completed joins 

75 # it, and is started in this same pass when a slot is free. 

76 await st.db.write(store.release_blocked) 

77 started: list[str] = [] 

78 for wid in st.db.read(store.queued_ids): 

79 if st.db.read(store.active_count) >= limit: 

80 break 

81 if storage.holds(st, wid): 

82 # Keeps its place, and is not asked again each tick (its door would 

83 # write another `work_item_storage_held`). A start behind it that 

84 # has a worktree still goes. 

85 continue 

86 if await _start_one(app, wid): 

87 started.append(wid) 

88 return started

91async def _start_one(app, wid: str) -> bool: 

92 starting(app).add(wid) # before the take, so no pause can read the row in between 

93 try: 

94 return await _take_and_start(app, wid) 

95 finally: 

96 starting(app).discard(wid)

99async def _take_and_start(app, wid: str) -> bool: 

100 from kraft.api.routes import lifecycle # deferred: kraft.api starts this module's poller 

101 

102 st = app.state 

103 saved = await st.db.write(lambda c: store.take_queued(c, wid)) 

104 if saved is None: 

105 return False # paused, ended or started since `tick` selected it 

106 request = _request(app, saved["headers"]) 

107 try: 

108 if saved["verb"] == "retry": 

109 await lifecycle._retry(wid, lifecycle.Retry(**saved["body"]), request) 

110 else: 

111 await lifecycle.resume_work_item(wid, lifecycle.Resume(**saved["body"]), request) 

112 except asyncio.CancelledError: 

113 # The server is shutting down mid-start. An item the door has not 

114 # claimed goes back in the queue, in its place, for the next server to 

115 # start: left where it came from, its request would be lost unsaid. 

116 await st.db.write(lambda c: _requeue(c, wid, saved)) 

117 raise 

118 except HTTPException as exc: 

119 await _dequeued(st, wid, "refused", str(exc.detail)) 

120 return False 

121 except Exception as exc: # noqa: BLE001 -- it has left the queue: say so, and let the pass go on 

122 logger.exception("start queue: starting %s failed", wid) 

123 await _dequeued(st, wid, "failed", f"{type(exc).__name__}: {exc}") 

124 return False 

125 status = st.db.read( 

126 lambda c: c.execute("SELECT status FROM work_items WHERE id = ?", (wid,)).fetchone()[ 

127 "status" 

128 ] 

129 ) 

130 if status == WorkItemStatus.QUEUED: 

131 # Another request took the slot between `tick`'s look and the door's 

132 # claim, and the door queued this one again: not to the back. 

133 await st.db.write(lambda c: store.keep_place(c, wid, saved["at"])) 

134 return False 

135 return True

138def _requeue(conn, wid: str, saved: dict) -> None: 

139 """Queue `saved` again, as the door does when it finds the slot gone. Only 

140 while the item is still where the take left it: a claimed item is running.""" 

141 if store.queue_work_item( 

142 conn, 

143 wid, 

144 verb=saved["verb"], 

145 body=saved["body"], 

146 headers=saved["headers"], 

147 from_statuses=[saved["from"]], 

148 ): 

149 store.keep_place(conn, wid, saved["at"])

152async def _dequeued(st, wid: str, why: str, detail: str) -> None: 

153 """The item was put back and did not start: its timeline says why. `to` is 

154 the status it is in now, which an open board shows: where it came from, or 

155 the stop of a door that failed after its claim (`stops.claimed_or_stopped`).""" 

156 

157 def write(c): 

158 to = c.execute("SELECT status FROM work_items WHERE id = ?", (wid,)).fetchone()["status"] 

159 events.append(c, wid, WorkItemEvent.DEQUEUED, {"why": why, "detail": detail, "to": to}) 

160 

161 await st.db.write(write)

164async def poller(app) -> None: 

165 """`tick` on a fixed interval until cancelled.""" 

166 while True: 

167 await asyncio.sleep(_INTERVAL_S) 

168 try: 

169 await tick(app) 

170 except asyncio.CancelledError: 

171 raise 

172 except Exception: # noqa: BLE001 -- one bad pass must not end the queue 

173 logger.exception("start queue: tick failed")
