9def kraft_home() -> Path: 

10 """The directory an installed Kraft keeps everything in. 

11 

12 One home, not a path guessed from `__file__`: the source tree is the install 

13 only while Kraft runs from a checkout, and under site-packages counting 

14 parent directories lands somewhere meaningless. `KRAFT_HOME` moves it, which 

15 is how `just dev` gets an instance that cannot touch the real one. 

16 """ 

17 return Path(os.environ.get("KRAFT_HOME") or Path.home() / ".kraft").expanduser()

30def default_config_dir() -> Path: 

31 """Where an install keeps its configuration: the policy, the connected 

32 repositories, the harness profiles, access, and the templates (the library 

33 and the chains) an item is filed from. `config/` since 2.0; a home still 

34 holding the 1.x `templates/` is renamed once at start, leaving `templates` 

35 as a link to it (`cli.admin.adopt_pre_2_home`)."""

45def config_dir(environ: Mapping[str, str] | None = None) -> Path: 

46 """The config directory this process runs against: `KRAFT_CONFIG_DIR`, else 

47 the 1.x `KRAFT_TEMPLATES_DIR`, else `default_config_dir()`. Read at call 

48 time, never at import, so a test or a reload that sets the variable after 

49 import still takes effect.""" 

50 env = os.environ if environ is None else environ 

51 named = env.get("KRAFT_CONFIG_DIR") or env.get(LEGACY_CONFIG_DIR_VAR) 

52 default = default_config_dir() 

53 if named and not names_default_config_dir(named): 

54 return Path(named) 

55 legacy = pre_2_config_dir() 

56 renamed = legacy is None or legacy.is_symlink() or (default / "library.yaml").exists() 

57 if not renamed and legacy != default: 

58 # Until the first 2.0 start renames it (`cli.admin.adopt_pre_2_home`), 

59 # a 1.x home's files are where they were: `kraft admin doctor`, 

60 # `kraft view list` and the start's own bind address, read before 

61 # that rename, must read them, not an empty home beside them. That 

62 # includes a `config/` made by hand before it (the harness guide's 

63 # `config/harnesses`), which holds no `library.yaml` until the 

64 # start merges the home into it -- the same test the merge uses. 

65 return legacy 

66 return default

69def names_default_config_dir(named: str | os.PathLike) -> bool: 

70 """Whether a variable names `default_config_dir()` itself: read as if it 

71 were unset, so a unit edited to the new default before the first 2.0 

72 start still has its 1.x home adopted, not an empty one seeded beside it 

73 (R12c-04). The last part is compared as spelled: `templates`, a link to 

74 `config/` once renamed, still names `templates`.""" 

75 

76 def spelled(path: Path) -> tuple[str, str]: 

77 path = Path(os.path.abspath(path.expanduser())) 

78 return os.path.realpath(path.parent), path.name 

79 

80 return spelled(Path(named)) == spelled(default_config_dir())

83def pre_2_config_dir() -> Path | None: 

84 """`$KRAFT_HOME/templates`, when it holds a 1.x home (a `library.yaml`) 

85 or a 0.x one (a `registry.yaml`): what `default_config_dir()` was called 

86 before 2.0. None when there is no such directory.""" 

87 old = kraft_home() / "templates" 

88 if (old / "library.yaml").is_file() or (old / "registry.yaml").is_file(): 

89 return old 

90 return None

93def linked_pre_2_config_dir() -> Path | None: 

94 """The directory `$KRAFT_HOME/templates` links to, when it is a 1.x home 

95 (see `pre_2_config_dir`) that this process does not read: a dotfiles 

96 repo linked in as `templates`. 2.0 reads `config/`, and the link is not 

97 renamed, so what it holds is ignored unless `KRAFT_CONFIG_DIR` names it. 

98 None for the link the 2.0 rename leaves (`templates -> config`), for none 

99 at all, and once the directory in use is the target (R14c-01).""" 

100 old = kraft_home() / "templates" 

101 if not old.is_symlink() or pre_2_config_dir() is None: 

102 return None 

103 target = Path(os.path.realpath(old)) 

104 if target in ( 

105 Path(os.path.realpath(default_config_dir())), 

106 Path(os.path.realpath(config_dir())), 

107 ): 

108 return None 

109 return target

131def default_harnesses_dir() -> Path: 

132 """Where an operator may override or add a harness definition. 

133 

134 Same shape and same reason as `default_skills_dir`: the shipped harnesses 

135 live in the package (`kraft.harness.BUNDLED`), and this directory exists 

136 only when someone has deliberately added or overridden one. Not seeded by 

137 `cli.seed_home` -- a seeded copy would freeze at whichever version the 

138 operator first installed, which is the drift measured live on 2026-09-13 

139 (Kraft-717xy). 

140 

141 1.x kept it at `$KRAFT_HOME/templates/harnesses` whatever 

142 `KRAFT_TEMPLATES_DIR` named, so a home whose config lived elsewhere has 

143 it there, in a `templates/` the 2.0 rename never adopts: read until a 

144 `config/harnesses` exists. 

145 """ 

146 current = default_config_dir() / "harnesses" 

147 legacy = kraft_home() / "templates" / "harnesses" 

148 if not current.is_dir() and legacy.is_dir(): 

149 return legacy 

150 return current

232 def ensure(self) -> RunDirs: 

233 """Also makes the run dir private (Kraft-9efnk.18): it holds whole 

234 agent sessions. The chmod tightens an install made before this, and 

235 only the directory itself, so nothing under it is walked. Worktrees 

236 stay reachable: every sandbox runs as the operator's own uid 

237 (`docker.Host.user_args`).""" 

238 self.base.mkdir(parents=True, exist_ok=True, mode=0o700) 

239 self.base.chmod(0o700) 

240 for d in (self.logs, self.results, self.worktrees, self.attachments): 

241 d.mkdir(parents=True, exist_ok=True) 

242 return self
