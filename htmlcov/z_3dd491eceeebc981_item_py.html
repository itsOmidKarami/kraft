1"""Item chain drafts (B15): a person's unapplied edits to one work item's own 

2chain (`store.materialized_chain_of`: the run fork's copy once a retry made 

3one), applied all at once or not at all. 

4 

5Four ops, each bounded the way the door it stands in for is: 

6 

7* `override {path, task_config?, policy?}` -- `validate_retry_override`, so a 

8 draft sets exactly what a retry override may. 

9* `add_node {after, node}` and `remove_node {node}` -- `revision.revise`, the 

10 chain-revision gate's own code: library components only, and never a gate, 

11 an unskippable node or one that lands the merge request removed. 

12* `skip {path}` -- a step or task path, recorded at apply as `scope_skipped`. 

13 

14An op is *passed* once the item has moved past where it acts (`passed`); a 

15passed op is never applied, and applying a draft that holds one is refused. 

16"""

67def _node(op: Op) -> str: 

68 """The node `op` acts on; for `add_node`, the node it follows.""" 

69 if isinstance(op, AddNodeOp): 

70 return op.after 

71 if isinstance(op, RemoveNodeOp): 

72 return op.node 

73 return op.path.split(PATH_SEPARATOR)[0]

76def passed(op: Op, ids: list[str], current: str | None) -> bool: 

77 """Whether the item has moved past where `op` acts: its node is at or 

78 before `current` (an `add_node`: its `after` is before it). Never before 

79 the item starts (`current` None); always once its chain has run to the 

80 end (`current` no longer one of `ids`). A node an earlier op adds is never 

81 passed: it follows one that is not.""" 

82 if current is None: 

83 return False 

84 if current not in ids: 

85 return True 

86 node, here = _node(op), ids.index(current) 

87 if node not in ids: 

88 return False 

89 return ids.index(node) < here if isinstance(op, AddNodeOp) else ids.index(node) <= here

92def _override(chain: MaterializedChain, op: OverrideOp) -> MaterializedChain: 

93 def one(c: MaterializedChain) -> ResolvedChain: 

94 return validate_retry_override( 

95 c, op.path, task_config=op.task_config, policy=op.policy 

96 ).chain.chain 

97 

98 revised = replace(chain, chain=one(chain)) 

99 if chain.untrimmed is not None: 

100 # The copy before the attachment trim gets the override too, as 

101 # `revise` keeps it for a skip or an add. 

102 whole = replace( 

103 chain, 

104 chain=ResolvedChain.from_chain(chain.untrimmed, steering=chain.chain.steering), 

105 untrimmed=None, 

106 ) 

107 revised = replace(revised, untrimmed=one(whole).chain) 

108 return revised

111def _skip(chain: MaterializedChain, op: SkipOp) -> None: 

112 target = ChainPath.parse(chain, op.path) 

113 if target.scope is ControlScope.NODE: 

114 raise PathError(f"{op.path!r} is a node; skip names a step or a task (remove_node a node)") 

115 if not target.skippable: 

116 raise PathError(f"{op.path!r} does not allow skipping")

119@dataclass(frozen=True) 

120class Evaluated: 

121 #: Each op as stored, with `passed`. 

122 ops: list[dict] 

123 #: `[{op: index, message}]`, one per op that cannot apply. 

124 problems: list[dict] 

125 #: The chain with every op applied that is neither passed nor a problem. 

126 chain: MaterializedChain 

127 #: The paths of the `skip` ops applied. 

128 skips: list[str] 

129 

130 @property 

131 def passed(self) -> list[int]: 

132 return [i for i, op in enumerate(self.ops) if op["passed"]] 

133 

134 

135def evaluate( 

136 chain: MaterializedChain, 

137 current: str | None, 

138 ops: list[dict], 

139 library: TemplateLibrary | None, 

140) -> Evaluated: 

141 """`ops` applied in order to `chain`, the item standing on `current` (None: 

142 not started). Each op is its own change set, so a problem names its op; 

143 the next op applies to the chain as the ones before it left it.""" 

144 parsed = OPS.validate_python(ops) 

145 ids = [n.id for n in chain.chain.nodes] 

146 at = current if current in ids else None 

147 marked, problems, skips = [], [], [] 

148 for i, (raw, op) in enumerate(zip(ops, parsed, strict=True)): 

149 marked.append({**raw, "passed": passed(op, ids, current)}) 

150 if marked[-1]["passed"]: 

151 continue 

152 try: 

153 if isinstance(op, OverrideOp): 

154 chain = _override(chain, op) 

155 elif isinstance(op, SkipOp): 

156 _skip(chain, op) 

157 skips.append(op.path) 

158 else: 

159 change = ( 

160 {"add": [{"after": op.after, "node": op.node, "evidence": EVIDENCE}]} 

161 if isinstance(op, AddNodeOp) 

162 else {"skip": [{"node": op.node, "evidence": EVIDENCE}]} 

163 ) 

164 changes = revision.ChangeSet.model_validate({"rationale": EVIDENCE, **change}) 

165 chain = revision.revise(chain, changes, at=at, library=library) 

166 except (revision.RevisionError, RetryOverrideError, PathError) as exc: 

167 problems.append({"op": i, "message": str(exc)}) 

168 return Evaluated(ops=marked, problems=problems, chain=chain, skips=skips)
