14def running_agents_warning(doing: str, items: list[dict], *, now: bool = False) -> list[str]: 

15 """The lines naming the active `items` whose agents a server stop ends, 

16 `doing` being what ends them ("restarting"); none when none ends. 

17 

18 A clean stop leaves the local steps of worktree items running: only an item running in 

19 place (`in_place`) ends with it. `now` is a stop that ends every active item's agent 

20 (`kraft admin stop --now`).""" 

21 ending = items if now else [item for item in items if item.get("in_place")] 

22 if not ending: 

23 return [] 

24 which = "any active item" if now else "any item running in place" 

25 return [ 

26 f"{doing} the server ends the agent of {which} ({len(ending)}):", 

27 *( 

28 f" {item['id']} {item.get('current_node_id') or '-'} {item['title']}" 

29 for item in ending 

30 ), 

31 " each one stops when Kraft starts again; retry it then. To avoid that, " 

32 "pause them first (kraft item pause ID) and resume them after.", 

33 ]

45def is_item_row(result: object) -> bool: 

46 """A whole `work_items` row, which the gate and lifecycle routes hand back 

47 (`deps.work_item_answer`): some 30 KB with its frozen chain, which a person 

48 who typed `kraft item approve` never asked to read (R7a-05).""" 

49 return isinstance(result, dict) and "chain_definition" in result and "status" in result

52#: What a row's `status` reads as, after "the item is now". 

53_NOW = total( 

54 WorkItemStatus, 

55 { 

56 WorkItemStatus.ACTIVE: "running", 

57 WorkItemStatus.PAUSED: "paused", 

58 WorkItemStatus.COMPLETED: "complete", 

59 # Cancel and abandon both store `abandoned`: only the verb that ended it can tell. 

60 WorkItemStatus.ABANDONED: "ended", 

61 WorkItemStatus.WAITING: "waiting on something outside Kraft", 

62 WorkItemStatus.RATE_LIMITED: "waiting for the agent's or the forge's rate limit to reset", 

63 WorkItemStatus.QUEUED: "queued for a free slot", 

64 WorkItemStatus.BLOCKED: "blocked until the items it comes after complete", 

65 WorkItemStatus.NEEDS_HUMAN: None, # item_now words it, with the id 

66 }, 

67 name="_NOW", 

68)

71def item_now(row: dict) -> str: 

72 """Where the item stands, as the end of a sentence.""" 

73 if row["status"] == WorkItemStatus.NEEDS_HUMAN: 

74 return f"stopped for a person: kraft view show {row['id']} says why" 

75 return _NOW.get(row["status"]) or row["status"]

78def item_action( 

79 done: str | Callable[[dict], str], 

80 *, 

81 status: bool = True, 

82 small: Callable[[dict], str] | None = None, 

83) -> Callable[[dict], str]: 

84 """The sentence for a verb whose answer may be the item's whole row: what 

85 was done (`done`, a format string taking `{id}`, or a function of the 

86 answer) and, unless `status` is off, what the item is now. Any other 

87 answer is small: `small` says it when given, else `done` alone does.""" 

88 

89 def say(result: dict) -> str: 

90 row = is_item_row(result) 

91 if small is not None and not row: 

92 return small(result) 

93 head = done(result) if callable(done) else done.format(id=result["id"]) 

94 return f"{head}; the item is now {item_now(result)}" if status and row else head 

95 

96 return say

103def hold_reason(held: dict, storage: dict | None) -> str | None: 

104 """Why a queued start waits when it is not a busy slot, in the web UI's 

105 words: another in-place item is using its checkout (`held`'s `why`, 

106 `held_by_title`), or the worktree storage limit holds it (`storage`, the 

107 figures `slots.storage` and `queued.storage` carry). None for a slot.""" 

108 if held.get("why") == "checkout": 

109 title = held.get("held_by_title") 

110 who = f"“{render.plain_text(str(title))}”" if title is not None else "another item" 

111 return f"waiting for the checkout: {who} is using it" 

112 if storage: 

113 used = render.human_size(storage["used_bytes"]) 

114 limit = render.human_size(storage["limit_bytes"]) 

115 return f"held by the worktree storage limit: {used} counted against the {limit} limit" 

116 return None

119def queued(result: dict) -> str | None: 

120 """A start the board had no slot for, or None for any other answer.""" 

121 if result.get("status") != WorkItemStatus.QUEUED: 

122 return None 

123 slots = result.get("slots") or {} 

124 storage = slots.get("storage") 

125 why = hold_reason(result, storage) 

126 pause = f"kraft item pause {result['id']} takes it out of the queue" 

127 if result.get("why") == "checkout": 

128 return f"queued {result['id']}: {why}. It starts when the checkout frees; {pause}" 

129 if why: 

130 return ( 

131 f"queued {result['id']}: {why}. It starts when space frees; " 

132 f"kraft item archive --reclaimable archives finished items to make room, {pause}" 

133 ) 

134 return ( 

135 f"queued {result['id']}: {slots.get('busy')} of {slots.get('limit')} slots are busy. " 

136 f"It starts when one frees; {pause}" 

137 )

140def blocked(result: dict) -> str | None: 

141 """A start held behind unfinished items, or None for any other answer.""" 

142 if result.get("status") != WorkItemStatus.BLOCKED: 

143 return None 

144 on = ", ".join(f"{d['id']} ({d['status']})" for d in result.get("waiting_on") or []) 

145 return ( 

146 f"blocked {result['id']}: it comes after {on}. It starts when they complete; " 

147 f"kraft item unblock {result['id']} drops them" 

148 )

151def held(result: dict) -> str | None: 

152 """A start that did not start: queued for a slot, or blocked.""" 

153 return queued(result) or blocked(result)

156def filed(result: dict) -> str | None: 

157 """What a create that asked to start says when every slot was busy: 

158 queued for one, or filed paused until one frees. None for any other 

159 create. `kraft item create` prints it under the item; the console's New 

160 item says it in its toast.""" 

161 slots = result.get("slots") 

162 if not slots: 

163 return None 

164 return held(result) or ( 

165 f"filed paused: {slots['busy']} of {slots['limit']} slots are busy. " 

166 f"Start it when one frees: kraft item resume {result['id']}" 

167 )

170def resumed(result: dict) -> str: 

171 """resume's usual answer: the node it went on from, and the agent tasks 

172 a steer reached. No node is an item that had never started.""" 

173 if waiting := held(result): 

174 return waiting 

175 node = result.get("node_id") 

176 line = f"resumed {result['id']} at {node}" if node else f"started {result['id']}" 

177 if steered := result.get("steered"): 

178 line += f"; steered {', '.join(steered)}" 

179 return line

201def unblocked(result: dict) -> str: 

202 """unblock's answer: what the item no longer comes after, and what it 

203 still waits on.""" 

204 dropped = result.get("dropped") or [] 

205 if not dropped: 

206 return f"{result['id']} comes after nothing unfinished: nothing was dropped" 

207 line = f"{result['id']} no longer comes after {', '.join(dropped)}" 

208 if left := result.get("waiting_on"): 

209 line += f"; it still waits on {', '.join(d['id'] for d in left)}" 

210 return line

213def dollars(raw: str) -> float | None: 

214 """A cap as typed: dollars, or `none` for an explicit no-cap. ValueError 

215 names what it could not read.""" 

216 if raw.lower() == "none": 

217 return None 

218 try: 

219 value = float(raw) 

220 except ValueError: 

221 value = math.nan 

222 if not math.isfinite(value): # inf and nan cannot go in the JSON body 

223 raise ValueError(f"dollars or `none`, not {raw!r}") 

224 return value

261def reload_summary(result: dict) -> str: 

262 """What a reload took and what it refused: `kraft admin reload` prints it, 

263 the console's `l` on the Server says it.""" 

264 refused = result.get("refused_policy") 

265 refused_intake = result.get("refused_intake") 

266 refused_workers = result.get("refused_workers") 

267 refused_forges = result.get("refused_forges") 

268 files = [ 

269 n 

270 for n, bad in ( 

271 ("policy.yaml", refused), 

272 ("intake.yaml", refused_intake), 

273 ("workers.yaml", refused_workers), 

274 ("forges.yaml", refused_forges), 

275 ) 

276 if not bad 

277 ] 

278 also = f" and {' and '.join(files)}" if files else "" 

279 lines = [f"reloaded {len(result.get('valid', []))} template(s){also}"] 

280 for name, reason in (result.get("invalid_templates") or {}).items(): 

281 lines.append(f" invalid: {name}: {reason}") 

282 if refused: 

283 lines.append(f" refused: policy.yaml: {refused} (the running policy is kept)") 

284 if refused_intake: 

285 # `ConfigError` names the file itself. 

286 lines.append( 

287 f" refused: {refused_intake} (the running auto-intake and schedules are kept)" 

288 ) 

289 if refused_workers: 

290 lines.append(f" refused: {refused_workers} (the running machines are kept)") 

291 if refused_forges: 

292 lines.append(f" refused: {refused_forges} (the running accounts are kept)") 

293 return "\n".join(lines)
