12def usage_rollup(conn: sqlite3.Connection, work_item_id: str) -> dict: 

13 """Per-node and per-item usage for one work item (handoff spec §8). 

14 

15 Rounds count *distinct* fix cycles seen on the node, not sessions: a node 

16 that ran three tasks in one pass has run one round, not three. 

17 

18 `cost_complete` is false when some session spent tokens but reported no cost 

19 — cost comes only from the agent, so a sum over those is a floor, not a 

20 total. Treating a missing cost as zero would quietly under-report the bill, 

21 which is the one thing a cost figure must not do. 

22 

23 `wall_ms` is derived when the column is NULL (`_common.session_wall_ms`) -- 

24 unlike cost, time is knowable for a session that never reported, because 

25 the row carries the same two stamps `session_exited` would have used. 

26 """ 

27 rows = conn.execute( 

28 "SELECT node_id, round, tokens_in, tokens_out, cost_usd, wall_ms, status, " 

29 "started_at, created_at, exited_at " 

30 "FROM worker_sessions WHERE work_item_id = ?", 

31 (work_item_id,), 

32 ).fetchall() 

33 by_node: dict[str, dict] = {} 

34 rounds: dict[str, set[int]] = {} 

35 for r in rows: 

36 node = by_node.setdefault( 

37 r["node_id"], 

38 { 

39 "node": r["node_id"], 

40 "tokens_in": 0, 

41 "tokens_out": 0, 

42 "cost_usd": 0.0, 

43 "wall_ms": 0, 

44 "sessions": 0, 

45 "rounds": 0, 

46 "capped_out": 0, 

47 "cost_complete": True, 

48 }, 

49 ) 

50 node["tokens_in"] += r["tokens_in"] or 0 

51 node["tokens_out"] += r["tokens_out"] or 0 

52 node["cost_usd"] += r["cost_usd"] or 0.0 

53 # a session that spent tokens but reported no cost makes the sum a floor 

54 if r["cost_usd"] is None and (r["tokens_in"] or r["tokens_out"]): 

55 node["cost_complete"] = False 

56 # Not `r["wall_ms"] or 0`: only `session_exited` writes that column, so 

57 # a paused, skipped, capped or still-running session contributed 

58 # nothing and a node that had been running 75 minutes summed to 0 

59 # (Kraft-s7c04.18, .47). The stamps are on the row either way. 

60 node["wall_ms"] += session_wall_ms(r) or 0 

61 node["sessions"] += 1 

62 node["capped_out"] += 1 if r["status"] == "capped_out" else 0 

63 rounds.setdefault(r["node_id"], set()).add(r["round"] or 0) 

64 for node_id, seen in rounds.items(): 

65 by_node[node_id]["rounds"] = len(seen) 

66 

67 nodes = list(by_node.values()) 

68 total = { 

69 k: sum(n[k] for n in nodes) 

70 for k in ("tokens_in", "tokens_out", "cost_usd", "wall_ms", "sessions", "capped_out") 

71 } 

72 # An item's rounds is the deepest a single node had to loop, not the sum: 

73 # summing would read as "this item retried nine times" for nine clean nodes. 

74 total["rounds"] = max((n["rounds"] for n in nodes), default=0) 

75 total["cost_complete"] = all(n["cost_complete"] for n in nodes) 

76 return {"total": total, "by_node": sorted(nodes, key=lambda n: n["node"])}

79def local_midnight_utc(now: datetime | None = None) -> str: 

80 """Midnight of `now`'s own day, in `now`'s own zone, as a UTC ISO string. 

81 

82 "Daily" means the operator's day, not UTC's — a cap that rolls over at 5pm 

83 local is a cap nobody can reason about. Normalized to UTC on the way out so 

84 it compares as a string against the `created_at` values `_now()` writes. 

85 

86 ponytail: the offset is the one in force *now*, not the one in force at 

87 midnight, so on a DST-transition day the window starts an hour early or 

88 late. A spend cap does not care; if something here ever does, resolve the 

89 offset at the midnight instant instead. 

90 """ 

91 now = now or datetime.now().astimezone() 

92 return now.replace(hour=0, minute=0, second=0, microsecond=0).astimezone(UTC).isoformat()

95def budget_spend( 

96 conn: sqlite3.Connection, work_item_id: str, *, since: str | None = None 

97) -> tuple[float, float]: 

98 """`(spent on this work item, spent instance-wide since `since`)`, in dollars. 

99 

100 A query, not a counter: `worker_sessions.cost_usd` is already the source of 

101 truth and a parallel counter is a second thing to get wrong. A NULL 

102 `cost_usd` — a session still running, or a subprocess or builtin task that 

103 has no cost — contributes zero, so this is a floor on in-flight spend by 

104 construction. 

105 """ 

106 item = conn.execute( 

107 "SELECT COALESCE(SUM(cost_usd), 0.0) FROM worker_sessions WHERE work_item_id = ?", 

108 (work_item_id,), 

109 ).fetchone()[0] 

110 if since is None: 

111 daily = conn.execute("SELECT COALESCE(SUM(cost_usd), 0.0) FROM worker_sessions").fetchone()[ 

112 0 

113 ] 

114 else: 

115 daily = conn.execute( 

116 "SELECT COALESCE(SUM(cost_usd), 0.0) FROM worker_sessions WHERE created_at >= ?", 

117 (since,), 

118 ).fetchone()[0] 

119 return float(item), float(daily)

122def effective_work_item_cap(row, policy_budget) -> tuple[float | None, str]: 

123 """`(cap_usd, source)` for one work item -- `source` is `"item"` when the 

124 item has its own cap (set or explicitly cleared to "no cap"), else 

125 `"policy"` for `policy_budget.work_item_usd` (UI v2 · 04, point 4). 

126 

127 `row["budget_set"]` is what makes an explicit "no cap" (`budget_usd` 

128 NULL, `budget_set` 1) distinguishable from "never customized" (`budget_usd` 

129 NULL, `budget_set` 0, defer to policy) -- a bare nullable column alone 

130 cannot tell those apart. 

131 """ 

132 if row["budget_set"]: 

133 return row["budget_usd"], "item" 

134 return policy_budget.work_item_usd, "policy"

137def effective_budget(row, policy_budget): 

138 """`policy_budget` with `work_item_usd` replaced by this item's effective 

139 cap (`effective_work_item_cap`). `daily_usd` is always policy-wide -- 

140 there is no per-item daily cap to override. 

141 """ 

142 cap, _source = effective_work_item_cap(row, policy_budget) 

143 return replace(policy_budget, work_item_usd=cap)

146def set_budget(conn: sqlite3.Connection, work_item_id: str, budget_usd: float | None) -> None: 

147 """Set this item's own spend cap: a number, or `None` for an explicit "no 

148 cap" (point 4). Always marks `budget_set`, so the item's choice -- even 

149 "no cap" -- is never confused with "not customized, use the policy 

150 default". 

151 """ 

152 conn.execute( 

153 "UPDATE work_items SET budget_set = 1, budget_usd = ?, updated_at = ? WHERE id = ?", 

154 (budget_usd, _now(), work_item_id), 

155 ) 

156 events.append(conn, work_item_id, "budget_changed", {"budget_usd": budget_usd})
