62class AlreadyRunning(RuntimeError): 

63 """A live executor task already holds this key. Not `HTTPException`: 

64 three of `spawn`'s callers are pollers, not routes, and making `waits` 

65 import FastAPI to catch its own race is backwards. Routes translate this 

66 into a 409; pollers catch it and log-and-skip, because a poller finding 

67 an item already running is a normal race, not a poller crash."""

71 try: 

72 await coro 

73 except asyncio.CancelledError:

75 except AssertionError:

110def task_is_live(app: FastAPI, wid: str) -> bool: 

111 """True if `wid` already has an undone task -- the same test `spawn` makes 

112 internally, exposed so a gate route can check it *before* a mutating store 

113 write. A gate has no status column to claim (§1 doesn't cover it): calling 

114 `store.approve_gate`/`apply_rejection` and only then discovering `spawn` 

115 refuses would leave the gate cleared and the item `active` with no walk 

116 behind it -- the exact stranding `claim_for_run`'s ordering exists to 

117 prevent everywhere else. Checking first, with no `await` before the write 

118 that follows, shrinks that race to the width of the write itself.""" 

119 existing = app.state.tasks.get(wid) 

120 return existing is not None and not existing.done()

123def discard(coro) -> None: 

124 """Close a coroutine that will never run, and any coroutine it was handed 

125 as an argument. One that is never awaited raises "coroutine was never 

126 awaited" at garbage-collection time and leaks whatever it closed over; 

127 closing an unstarted `guard(...)` does not reach the `executor.run(...)` 

128 it wraps, so that one is closed too.""" 

129 for arg in coro.cr_frame.f_locals.values() if coro.cr_frame else (): 

130 if asyncio.iscoroutine(arg): 

131 arg.close() 

132 coro.close()

135def spawn(app: FastAPI, wid: str, coro) -> asyncio.Task: 

136 if task_is_live(app, wid): 

137 discard(coro) 

138 raise AlreadyRunning(wid) 

139 task = asyncio.ensure_future(coro) 

140 app.state.tasks[wid] = task

142 def _done(t, wid=wid): 

143 # Preemption paths (approve/reject/retry) SIGTERM the old task and 

144 # immediately spawn a new one under the same wid before the old 

145 # task's callback fires. Popping unconditionally would let the old 

146 # task's callback evict the new, still-live task -- dropping the 

147 # tick loop's only concurrency guard and orphaning the new run. 

148 # Only pop the slot if it still holds *this* task. 

149 if app.state.tasks.get(wid) is t: 

150 app.state.tasks.pop(wid, None) 

151 

152 task.add_done_callback(_done) 

153 return task

156async def cancel(app: FastAPI, wid: str, timeout: float | None = None) -> None: 

157 """Cancel `wid`'s live walk task, if any, and wait for it to actually 

158 finish before returning. 

159 

160 A bare `.cancel()` only *schedules* a `CancelledError` at the task's next 

161 await point -- `task.done()` stays False until the event loop resumes it. 

162 `skip` cancels the current walk and spawns its replacement in the same 

163 request, with no `await` in between; without waiting here, `spawn`'s 

164 `AlreadyRunning` check would see the old task as still live and refuse the 

165 new one every time. Awaiting the task makes cancellation synchronous from 

166 the caller's point of view: by the time this returns, `spawn`'s own 

167 done-callback has already popped `app.state.tasks[wid]`. 

168 

169 `deps.guard` re-raises `CancelledError` and swallows every other 

170 exception itself, so `CancelledError` is the only thing `await task` can 

171 raise here. 

172 

173 `timeout`, when given, bounds the wait (Kraft-c5ui): a task parked in 

174 `asyncio.to_thread` (a git/forge call) does not see the cancellation 

175 until that call returns, so an unbounded wait here can block the whole 

176 request for as long as that call takes. Past the deadline this returns 

177 anyway, with the task still cancelling in the background -- its own 

178 done-callback still pops `app.state.tasks[wid]` whenever it actually 

179 exits. Only safe for a caller with nothing that depends on the task 

180 being gone by the time this returns: `skip` still needs the unbounded 

181 wait, because it spawns a replacement off exactly that assumption. 

182 `asyncio.shield` keeps the timeout from cancelling anything itself -- 

183 the task already got its `.cancel()` above and is left to finish that on 

184 its own. 

185 """ 

186 task = app.state.tasks.get(wid) 

187 if task is None or task.done(): 

188 return 

189 task.cancel() 

190 try: 

191 if timeout is None: 

192 await task 

193 else: 

194 await asyncio.wait_for(asyncio.shield(task), timeout=timeout) 

195 except asyncio.CancelledError: 

196 # The walk's own, unless this caller was cancelled as well 

197 # (Kraft-1zhnj): a `Task.cancel()` on a caller waiting here reaches 

198 # the walk instead and comes back looking the same, and swallowing it 

199 # leaves a shutdown waiting on a task that never ends. 

200 if asyncio.current_task().cancelling(): 

201 raise 

202 except TimeoutError: 

203 logger.warning( 

204 "cancel: task for %s did not unwind within %.1fs; proceeding anyway", wid, timeout 

205 )

208def skip_lock(app: FastAPI, wid: str) -> asyncio.Lock: 

209 """A per-`wid` lock serializing `/skip` end to end. 

210 

211 `/skip` can start from `paused`/`needs_human`, where there is no live task 

212 yet for `spawn`'s own check to catch, and `claim_for_run`'s `to_status` 

213 (`"active"`) is also one of `/skip`'s own `from_statuses` -- so two 

214 concurrent skips both win that claim too and both write `skip_node` 

215 before either reaches `spawn`. The caller checks `.locked()` before 

216 entering, refusing a second caller outright rather than queuing it behind 

217 the first -- a queued second caller would wake up to a state the first 

218 already advanced, and would then skip the *next* node too instead of 

219 getting the 409 two callers racing the same node are supposed to get. 

220 """ 

221 locks = getattr(app.state, "_skip_locks", None) 

222 if locks is None: 

223 locks = {} 

224 app.state._skip_locks = locks 

225 return locks.setdefault(wid, asyncio.Lock())

273def forbid_self_action(st, request, wid: str, *, escalation_may: bool = True) -> None: 

274 """403 when the caller names a worker session of `wid` itself: design §6 

275 rule 2, a worker does not approve, reject, pause, resume, skip, abandon, 

276 retry, complete, cancel or escalate its own item, nor revise its 

277 attachments, agent or node overrides, policy or chain draft. Review comments and 

278 submit-review need no call here: `routes.review._refuse_agents` refuses 

279 any session header there outright. 

280 

281 The server-side twin of `client.context._forbid_self_action`, for a 

282 client other than `kraft` or the MCP server that honestly sends 

283 `X-Kraft-Session-Id`. Consistency, not a security boundary: a caller 

284 that leaves the header off passes, and an unsandboxed worker can. 

285 

286 An escalation turn is no worker (`identify_as_worker=False`), so it may: 

287 the client lets its `kraft item retry` through (it has no 

288 `KRAFT_WORK_ITEM_ID`), and so does this -- except where `escalation_may` 

289 is False, approve and reject (Kraft-9efnk.16): a gate is a human's call, 

290 and the escalation agent is not the human. The client cannot refuse 

291 those itself, as nothing in an escalation's environment names its item, 

292 so this is the only refusal. A sandboxed session reaches 

293 only `/retry` of these, through its channel (`worker.callback.ROUTES`), 

294 which sets this header on every call: this is what refuses a sandboxed 

295 worker's retry of its own item while its escalation's self-retry passes, 

296 since the channel does not tell the two apart (Kraft-9efnk.14). 

297 """ 

298 sid = request.headers.get("x-kraft-session-id") 

299 if not sid: 

300 return 

301 session = st.db.read( 

302 lambda c: c.execute( 

303 "SELECT work_item_id, hook_point FROM worker_sessions WHERE id = ?", (sid,) 

304 ).fetchone() 

305 ) 

306 if ( 

307 session is not None 

308 and session["work_item_id"] == wid 

309 and (not escalation_may or session["hook_point"] != executor.ESCALATION_HOOK) 

310 ): 

311 raise HTTPException( 

312 403, 

313 f"a worker session cannot act on its own work item ({wid}). " 

314 "Gates are where a human decides; report what you found instead.", 

315 )

392async def restore_plugins(app) -> None: 

393 """Put back every locked store that is gone (a config directory copied to 

394 a new machine, a cleared run directory, a teammate's newer lock), then 

395 load what came back and collect what nothing reads. Off the event loop: 

396 it may fetch. One that cannot be restored stays left out, with the reason 

397 under `invalid_templates`.""" 

398 from kraft import apply 

399 

400 st = app.state 

401 # `verify`, as the library load reads: a store edited in place stays left 

402 # out here too, so it never counts as "loads now". 

403 fresh = await asyncio.to_thread(installed_plugins, st, verify=True) 

404 missing = tuple(p for p in fresh if not p.quiet and not p.root.is_dir()) 

405 loads = {p.id for p in fresh if p.left_out is None} 

406 if missing: 

407 failed = await asyncio.to_thread( 

408 plugins_load.restore_missing, missing, st.templates_dir, st.run_dirs.plugins 

409 ) 

410 _reload_templates(st) 

411 apply.notify(app) 

412 if failed: 

413 logger.warning("plugin stores not restored: %s", "; ".join(failed.values())) 

414 elif any( 

415 # Left out by the running library, loading now. Not a plugin enabled 

416 # by hand (quiet while disabled): that edit waits for its own reload. 

417 p.left_out is not None and not p.quiet and p.id in loads 

418 for p in (st.library.plugins if st.library is not None else ()) 

419 ): 

420 # A launch put the store back (`dispatch.restore_pins`). Load it. 

421 _reload_templates(st) 

422 apply.notify(app) 

423 collect_plugin_stores(st)

470def collect_plugin_stores(st) -> list[str]: 

471 """Delete each plugin store that none of these names: the lock, the 

472 library this server has loaded, or the snapshot of a work item that has 

473 not ended (a paused or stopped one can still be retried onto it). A 

474 directory collection's store is the lock's, so it stays while installed. 

475 Nothing is deleted while the lock cannot be read, while another plugin 

476 change is running, or when the store cannot be written.""" 

477 from kraft.plugins import update as plugin_update 

478 from kraft.plugins.config import PluginsLock 

479 

480 def keep() -> set[str]: 

481 # Asked under the update lock: an install that just finished is in the lock. 

482 locked = PluginsLock.load(st.templates_dir / PluginsLock.FILE).plugins.values() 

483 kept = {entry.digest.removeprefix("sha256:") for entry in locked} 

484 kept |= {p.root.name for p in (st.library.plugins if st.library is not None else ())} 

485 ended = sorted(ENDED) 

486 rows = st.db.read( 

487 lambda c: c.execute( 

488 # A retried item runs on `run_chain`, and a revision adds its pins there. 

489 "SELECT materialized_chain, run_chain FROM work_items " 

490 f"WHERE status NOT IN ({', '.join('?' * len(ended))})", 

491 ended, 

492 ).fetchall() 

493 ) 

494 for stored in (snapshot for row in rows for snapshot in row if snapshot): 

495 for pin in (json.loads(stored).get("plugins") or {}).values(): 

496 kept.add(str(pin["digest"]).removeprefix("sha256:")) 

497 return kept 

498 

499 try: 

500 return plugins_load.gc(keep, st.run_dirs.plugins) 

501 except (config_mod.ConfigError, plugin_update.Busy, OSError): 

502 return []

545 the caller's own, else the repo's `default_chain`, else None -- 

546 the `default` template, un-chosen (Kraft-cd47). Every intake door asks this 

547 one function (Kraft-9efnk.11). A workspace item asks its root repo. 

548 

549 A repo default other than `default` is stored by name, not as None: every 

550 reader of the row (the board, the stage graph, analytics) reads None as 

551 `default`, so a None here would misreport the chain the item runs. 

552 """ 

553 if chain_template is not None: 

554 return chain_template 

555 default = entry.default_chain 

556 return None if default in (None, "default") else default

559def resolve_chain(st, chain_template: str | None): 

560 """The resolved V1 chain `chain_template` names, or `None`. 

561 

562 `None` for `chain_template` means "no explicit template was chosen" 

563 (Kraft-cd47) and resolves `default`, the same way the legacy lookup did -- 

564 the *stored* value stays `None`, which is the distinction that matters. 

565 Resolution failures are `None` too: a chain that does not resolve is not 

566 selectable, and `lint()` is where an author reads why. 

567 """ 

568 if st.library is None: 

569 return None 

570 try: 

571 return st.library.resolve_chain(chain_template if chain_template is not None else "default") 

572 except TemplateLibraryError: 

573 logger.warning("chain %r does not resolve", chain_template, exc_info=True) 

574 return None

577def resolve_chain_or_422(st, chain_template: str | None): 

578 """`resolve_chain`, as the error every intake door owes its caller. One 

579 function so the three doors (`POST /work-items`, `POST /triggers`, and the 

580 auto-intake poller's own check) cannot answer differently. 

581 

582 **503 when the library itself did not load, 422 only when the chain is 

583 genuinely unknown.** `TemplateLibrary.from_yaml_dir` raises on any one bad 

584 file, so a single malformed `chains/*.yaml` leaves `st.library is None` and 

585 *every* chain id unresolvable -- and "unknown or invalid template" then 

586 tells the operator their chain id is wrong when the truth is that one file 

587 does not parse. Same posture and same shape as `invalid_policy`'s 503: name 

588 the file, refuse the work, and leave the Settings screens reachable. 

589 """ 

590 library = library_or_503(st) 

591 name = chain_template if chain_template is not None else "default" 

592 try: 

593 return library.resolve_chain(name) 

594 except TemplateLibraryError as exc: 

595 # The resolver's own message: it names the unknown id, or the path and 

596 # reference that broke the chain (Kraft-n1zp9). 

597 for plugin in library.plugins: 

598 # Not "unknown chain": its plugin is installed and its store is on the way back. 

599 away = not plugin.quiet and not plugin.root.is_dir() 

600 named = re.search(rf"(?<![a-z0-9_-]){re.escape(plugin.namespace)}:", f"{name} {exc}") 

601 if away and named: 

602 raise HTTPException(503, f"plugin {plugin.id} is being restored; retry") from exc 

603 raise HTTPException(422, f"chain {name!r}: {exc}") from exc

606def intake_chain_or_422(st, repo: str, chain_template: str | None): 

607 """`(stored, chain)` for an intake door: `chain_template_for`'s value, to

609 one answers 503 before the repo lookup can answer 422.""" 

610 library_or_503(st) 

611 stored = chain_template_for(connected_or_422(st, repo), chain_template) 

612 return stored, resolve_chain_or_422(st, stored)

624def _connected(repos: list[RepoEntry], path: str) -> RepoEntry | None: 

625 """Find a connected repo by path. 

626 

627 `POST /repos` stores git's `--show-toplevel`, which resolves symlinks (on 

628 macOS /var -> /private/var), so the path a client added with is not always 

629 the path stored. Match either, or a caller cannot patch or delete the repo 

630 it just connected. 

631 

632 A relative path matches nothing: this process would resolve it against its 

633 own cwd, not the caller's, and find the wrong repo (Kraft-9efnk.39). 

634 """ 

635 if not _absolute(path): 

636 return None 

637 entry = next((r for r in repos if r.path == path), None) 

638 if entry is not None: 

639 return entry 

640 resolved = str(Path(path).expanduser().resolve()) 

641 return next((r for r in repos if r.path == resolved), None)

644def connected_or_422(st, repo: str) -> RepoEntry: 

645 """`repo`'s entry, or a 422 saying how to connect it (Kraft-ta8nv). 

646 

647 Both intake doors file only against a connected repo: any directory would 

648 make the check an existence oracle on the host's filesystem, and an item in 

649 an unconnected repo has no entry for dispatch to read its config from.""" 

650 try: 

651 repos = config_mod.load_repos(repos_path(st)) 

652 except config_mod.ConfigError as exc: 

653 raise HTTPException(422, str(exc)) from exc 

654 if not _absolute(repo): 

655 raise HTTPException(422, f"repo must be an absolute path, not {repo!r}") 

656 entry = _connected(repos, repo) 

657 if entry is None: 

658 raise HTTPException( 

659 422, f"{repo} is not a connected repo; connect it first: kraft repo connect {repo}" 

660 ) 

661 return entry

664async def base_branch_or_422(repo: str, branch: str | None) -> str | None: 

665 """`branch`, once origin is known to have it (Kraft-v9gbi) -- an item's 

666 work starts from it, so a branch that is not there would only fail at the 

667 first node, after the bead is filed. None names no branch: the 

668 repository's default, which needs no check.""" 

669 if branch is None: 

670 return None 

671 path = Path(repo) 

672 # The model's own rules, checked here first so the refusal is one 

673 # sentence rather than pydantic's error list, and before any git call. 

674 if (problem := branch_name_problem(branch)) is not None: 

675 raise HTTPException(422, f"base branch {branch!r} is not a valid branch name: it {problem}") 

676 if not config_mod.git_read(path, "remote", "get-url", "origin", expected_failure=True): 

677 raise HTTPException( 

678 422, f"base branch {branch!r} cannot be checked: {repo} has no origin remote" 

679 ) 

680 try: 

681 done = await asyncio.to_thread( 

682 subprocess.run, 

683 ["git", "ls-remote", "--exit-code", "--heads", "origin", f"refs/heads/{branch}"], 

684 cwd=path, 

685 stdin=subprocess.DEVNULL, 

686 capture_output=True, 

687 text=True, 

688 timeout=30, 

689 env=builtins_mod.promptless_git_env(path), 

690 ) 

691 except subprocess.TimeoutExpired: 

692 raise HTTPException( 

693 422, f"could not reach {repo}'s origin to check base branch {branch!r}: timed out" 

694 ) from None 

695 if done.returncode == 2: 

696 raise HTTPException( 

697 422, 

698 f"no branch {branch!r} on origin of {repo}; push it first, or name a branch origin has", 

699 ) 

700 if done.returncode != 0: 

701 detail = done.stderr.strip() or "failed" 

702 raise HTTPException( 

703 422, f"could not reach {repo}'s origin to check base branch {branch!r}: {detail}" 

704 ) 

705 return branch

708def workspace_target( 

709 st, 

710 repo: str, 

711 *, 

712 workspace: str | None, 

713 members: list[str], 

714 root_pointer_policy: RootPointerPolicy | None, 

715 base_branch: str | None = None, 

716) -> WorkItemTarget | None: 

717 """The workspace target an intake selects, or None for a plain repository 

718 item. Raises a 422 for a selection that cannot assemble: an undeclared 

719 workspace, one rooted elsewhere, a member it does not mount. The 

720 pointer policy defaults to the workspace's own 

721 (`workspace-root-pointer-update-is-explicit`).""" 

722 if workspace is None: 

723 if members: 

724 raise HTTPException(422, "members are selected, but the item names no workspace") 

725 return None 

726 try: 

727 declared = config_mod.load_workspaces(repos_path(st)) 

728 repos = config_mod.load_repos(repos_path(st)) 

729 except config_mod.ConfigError as exc: 

730 raise HTTPException(422, str(exc)) from exc 

731 ws = declared.get(workspace) 

732 if ws is None: 

733 raise HTTPException(422, f"no workspace {workspace!r} is declared in repos.yaml") 

734 root = next(r for r in repos if r.id == ws.root) 

735 # A string comparison, never a filesystem lookup on the request's path: 

736 # the stored root is git's own `--show-toplevel`, which is what every 

737 # client (the intake modal, `kraft item create`) sends back. 

738 if root.path != os.path.normpath(os.path.expanduser(repo)): 

739 raise HTTPException(422, f"workspace {workspace!r} is rooted at {root.path}, not at {repo}") 

740 try: 

741 return WorkItemTarget.from_selection( 

742 ws, members=members, root_pointer_policy=root_pointer_policy, base_branch=base_branch 

743 ) 

744 except TemplateEnvironmentError as exc: 

745 raise HTTPException(422, str(exc)) from exc

748def _repository_layers(st, repo: str, target: WorkItemTarget | None) -> list[tuple[str, RepoEntry]]: 

749 """`(repository id, entry)` for every repository whose layer binds an item 

750 filed in `repo`: its own entry, or for a workspace target the root's and 

751 each selected member's. Raises `PolicyError` on an unreadable file.""" 

752 try: 

753 repos = config_mod.load_repos(repos_path(st)) 

754 except config_mod.ConfigError as exc: 

755 raise PolicyError(f"cannot read the repository policy layer: {exc}") from exc 

756 if target is None or target.kind != "workspace": 

757 entry = _connected(repos, repo) 

758 return [(entry.id or "", entry)] if entry is not None else [] 

759 by_id = {r.id: r for r in repos if r.id} 

760 return [(rid, by_id[rid]) for rid in target.repositories() if rid in by_id]

778def item_policy(st, repo: str, target: WorkItemTarget | None = None) -> InstancePolicy: 

779 """The policy a work item filed in `repo` starts from: the instance policy 

780 with that repository's layer on top (`RepoEntry.repository_override`), which 

781 may only tighten it (`repository-policy-cannot-relax-instance-safety`). 

782 Every intake door materializes from this, so the layer is frozen into every 

783 item filed in the repository. 

784 

785 A workspace `target` runs its ordinary tasks in an assembled checkout that 

786 holds every selected repository at once, so its layer is the meet of all 

787 of theirs -- each checked on its own first, so a refusal names the 

788 repository (Kraft-jc39p; `repository_policies` has each one alone). 

789 

790 Unlike `launch`, this raises: a `repos.yaml` that cannot be read has 

791 restrictions nobody can see, and filing an item without them is exactly 

792 the relaxation the layer exists to prevent. A `PolicyError`, so each door 

793 answers it the way it answers a chain policy past a ceiling.""" 

794 base = instance_policy(st) 

795 layers = _repository_layers(st, repo, target) 

796 for _rid, entry in layers: 

797 _layered(base, entry) 

798 overrides = [o for _rid, e in layers if (o := e.repository_override()) is not None] 

799 if not overrides: 

800 return base 

801 return base.apply_template_override(TemplatePolicyOverride.meet(overrides))

805 """A workspace item's starting policy per selected repository id -- the

807 out to it runs under (Kraft-jc39p). Empty for a single-repository item,

847class _PoisonedRepoEntry: 

848 """Stand-in `repo_entry` (and `repositories`) for a `repos.yaml` that 

849 failed to load. 

850 

851 Any read of it -- an attribute (`entry.sandbox`, `entry.env`), a key or 

852 a membership test on the `repositories` mapping -- raises the original 

853 `ConfigError` from *inside* the dispatch this entry was actually needed 

854 for, which is already wrapped in `guard`'s blanket `except Exception` -- 

855 the same needs_human treatment a `SteeringError` raised deep in a 

856 dispatch already gets. Truthy, so `if launch.repo_entry` never skips it 

857 for "nothing configured" before that read runs -- a repo config a caller 

858 can't parse must not quietly resolve to "no sandbox, no deny_tools, no 

859 steering" instead."""

889def repository_steering( 

890 st, repo: str, target: WorkItemTarget | None = None 

891) -> dict[str, dict[str, str]]: 

892 """What an item filed in `repo` freezes as its repository steering 

893 (`MaterializedChain.repository_steering`): each repository it runs in 

894 that names steering in `repos.yaml`, by path, to those profiles' texts 

895 from the library. Raises `SteeringError` (a `ValueError`) naming a name 

896 the library does not define, or an unreadable `repos.yaml`, so an intake 

897 door refuses the item rather than filing it unsteered.""" 

898 try: 

899 repos = config_mod.load_repos(repos_path(st)) 

900 except config_mod.ConfigError as exc: 

901 raise steering_mod.SteeringError(str(exc)) from exc 

902 entries = [_connected(repos, repo)] 

903 if target is not None and target.kind == "workspace": 

904 by_id = {r.id: r for r in repos if r.id} 

905 entries += [by_id.get(rid) for rid in target.repositories()] 

906 profiles = library_steering(st) or {} 

907 return { 

908 e.path: steering_mod.select(e.steering, profiles, where=f"repos.yaml: {e.path}") 

909 for e in entries 

910 if e is not None and e.steering 

911 }

914def launch(st, repo: str) -> executor.LaunchContext: 

915 """The repo config for one agent dispatch. 

916 

917 A malformed `repos.yaml` must not crash `lifespan` on reattach (locking 

918 an operator out of the Settings UI that would let them fix it) or 500 the 

919 approve/reject/resume/retry routes, so this itself never raises. But a 

920 repo entry can carry `sandbox:` now, so silently degrading to 

921 `repo_entry=None` would run that dispatch's worker unsandboxed on the 

922 host with nothing but a log line nobody reads -- the exact bare-metal 

923 fallback the sandbox design rules out. Instead the failure is carried in 

924 a `_PoisonedRepoEntry` that raises the instant the dispatch that actually 

925 needed this repo's config reads it, landing in `guard` as needs_human 

926 the same way a bad `sandbox:` value would if it were only caught then. 

927 

928 The live library's steering profiles ride along for an item filed before 

929 repository steering was frozen into its snapshot 

930 (`steering.for_repository`); every other launch reads its snapshot's. 

931 

932 `chain_revision_digest` rides along too (Kraft-rndd1): `st.library` bound 

933 into `revision.artifact_digest`, so a gate's own agent review can capture 

934 what its artifact resolves to before dispatching, the same way `GET 

935 .../artifact` does for a person.""" 

936 steering = library_steering(st) or {} 

937 digest = functools.partial(revision.artifact_digest, library=getattr(st, "library", None)) 

938 try: 

939 repos = config_mod.load_repos(repos_path(st)) 

940 except config_mod.ConfigError as exc: 

941 logger.warning("repo config invalid, failing dispatch that reads it: %s", exc) 

942 # The members' entries too: a fanned-out task must fail the same 

943 # way, never run with no sandbox because its entry was unreadable. 

944 return executor.LaunchContext( 

945 repo_entry=cast("RepoEntry", _PoisonedRepoEntry(exc)), 

946 skills_dir=st.skills_dir, 

947 repositories=cast("dict[str, RepoEntry]", _PoisonedRepoEntry(exc)), 

948 library_steering=steering, 

949 chain_revision_digest=digest, 

950 ) 

951 return executor.LaunchContext( 

952 repo_entry=_connected(repos, repo), 

953 skills_dir=st.skills_dir, 

954 repositories={r.id: r for r in repos if r.id}, 

955 library_steering=steering, 

956 chain_revision_digest=digest, 

957 )
