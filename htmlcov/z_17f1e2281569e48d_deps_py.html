59class AlreadyRunning(RuntimeError): 

60 """A live executor task already holds this key. Not `HTTPException`: 

61 three of `spawn`'s callers are pollers, not routes, and making `waits` 

62 import FastAPI to catch its own race is backwards. Routes translate this 

63 into a 409; pollers catch it and log-and-skip, because a poller finding 

64 an item already running is a normal race, not a poller crash."""

67async def guard(db, wid: str, coro) -> None: 

68 try: 

69 await coro 

70 except asyncio.CancelledError: 

71 raise 

72 except AssertionError: 

73 # Kraft's own broken invariant (and, under pytest, the real-agent 

74 # guard) -- never evidence of an executor crash, so it must not 

75 # become a needs_human stop. `dispatch.measure_node` carves out the 

76 # same exception for the same reason (Kraft-cpotk); mirror it here. 

77 raise 

78 except Exception as exc: # noqa: BLE001 

79 logger.exception("executor task crashed for %s", wid) 

80 reason = f"executor crashed: {exc!r}" 

81 try: 

82 await db.write(lambda c: store.mark_needs_human(c, wid, None, reason)) 

83 except Exception: # noqa: BLE001 

84 logger.exception("could not mark %s needs_human after crash", wid)

103def task_is_live(app: FastAPI, wid: str) -> bool: 

104 """True if `wid` already has an undone task -- the same test `spawn` makes 

105 internally, exposed so a gate route can check it *before* a mutating store 

106 write. A gate has no status column to claim (§1 doesn't cover it): calling 

107 `store.approve_gate`/`apply_rejection` and only then discovering `spawn` 

108 refuses would leave the gate cleared and the item `active` with no walk 

109 behind it -- the exact stranding `claim_for_run`'s ordering exists to 

110 prevent everywhere else. Checking first, with no `await` before the write 

111 that follows, shrinks that race to the width of the write itself.""" 

112 existing = app.state.tasks.get(wid) 

113 return existing is not None and not existing.done()

116def discard(coro) -> None: 

117 """Close a coroutine that will never run, and any coroutine it was handed 

118 as an argument. One that is never awaited raises "coroutine was never 

119 awaited" at garbage-collection time and leaks whatever it closed over; 

120 closing an unstarted `guard(...)` does not reach the `executor.run(...)` 

121 it wraps, so that one is closed too.""" 

122 for arg in coro.cr_frame.f_locals.values() if coro.cr_frame else (): 

123 if asyncio.iscoroutine(arg): 

124 arg.close() 

125 coro.close()

128def spawn(app: FastAPI, wid: str, coro) -> asyncio.Task: 

129 if task_is_live(app, wid): 

130 discard(coro) 

131 raise AlreadyRunning(wid) 

132 task = asyncio.ensure_future(coro) 

133 app.state.tasks[wid] = task 

134 

135 def _done(t, wid=wid): 

136 # Preemption paths (approve/reject/retry) SIGTERM the old task and 

137 # immediately spawn a new one under the same wid before the old 

138 # task's callback fires. Popping unconditionally would let the old 

139 # task's callback evict the new, still-live task -- dropping the 

140 # tick loop's only concurrency guard and orphaning the new run. 

141 # Only pop the slot if it still holds *this* task. 

142 if app.state.tasks.get(wid) is t: 

143 app.state.tasks.pop(wid, None) 

144 

145 task.add_done_callback(_done) 

146 return task

149async def cancel(app: FastAPI, wid: str, timeout: float | None = None) -> None: 

150 """Cancel `wid`'s live walk task, if any, and wait for it to actually 

151 finish before returning. 

152 

153 A bare `.cancel()` only *schedules* a `CancelledError` at the task's next 

154 await point -- `task.done()` stays False until the event loop resumes it. 

155 `skip` cancels the current walk and spawns its replacement in the same 

156 request, with no `await` in between; without waiting here, `spawn`'s 

157 `AlreadyRunning` check would see the old task as still live and refuse the 

158 new one every time. Awaiting the task makes cancellation synchronous from 

159 the caller's point of view: by the time this returns, `spawn`'s own 

160 done-callback has already popped `app.state.tasks[wid]`. 

161 

162 `deps.guard` re-raises `CancelledError` and swallows every other 

163 exception itself, so `CancelledError` is the only thing `await task` can 

164 raise here. 

165 

166 `timeout`, when given, bounds the wait (Kraft-c5ui): a task parked in 

167 `asyncio.to_thread` (a git/forge call) does not see the cancellation 

168 until that call returns, so an unbounded wait here can block the whole 

169 request for as long as that call takes. Past the deadline this returns 

170 anyway, with the task still cancelling in the background -- its own 

171 done-callback still pops `app.state.tasks[wid]` whenever it actually 

172 exits. Only safe for a caller with nothing that depends on the task 

173 being gone by the time this returns: `skip` still needs the unbounded 

174 wait, because it spawns a replacement off exactly that assumption. 

175 `asyncio.shield` keeps the timeout from cancelling anything itself -- 

176 the task already got its `.cancel()` above and is left to finish that on 

177 its own. 

178 """ 

179 task = app.state.tasks.get(wid) 

180 if task is None or task.done(): 

181 return 

182 task.cancel() 

183 try: 

184 if timeout is None: 

185 await task 

186 else: 

187 await asyncio.wait_for(asyncio.shield(task), timeout=timeout) 

188 except asyncio.CancelledError: 

189 pass 

190 except TimeoutError: 

191 logger.warning( 

192 "cancel: task for %s did not unwind within %.1fs; proceeding anyway", wid, timeout 

193 )

196def skip_lock(app: FastAPI, wid: str) -> asyncio.Lock: 

197 """A per-`wid` lock serializing `/skip` end to end. 

198 

199 `/skip` can start from `paused`/`needs_human`, where there is no live task 

200 yet for `spawn`'s own check to catch, and `claim_for_run`'s `to_status` 

201 (`"active"`) is also one of `/skip`'s own `from_statuses` -- so two 

202 concurrent skips both win that claim too and both write `skip_node` 

203 before either reaches `spawn`. The caller checks `.locked()` before 

204 entering, refusing a second caller outright rather than queuing it behind 

205 the first -- a queued second caller would wake up to a state the first 

206 already advanced, and would then skip the *next* node too instead of 

207 getting the 409 two callers racing the same node are supposed to get. 

208 """ 

209 locks = getattr(app.state, "_skip_locks", None) 

210 if locks is None: 

211 locks = {} 

212 app.state._skip_locks = locks 

213 return locks.setdefault(wid, asyncio.Lock())

396async def base_branch_or_422(repo: str, branch: str | None) -> str | None: 

397 """`branch`, once origin is known to have it (Kraft-v9gbi) -- an item's 

398 work starts from it, so a branch that is not there would only fail at the 

399 first node, after the bead is filed. None names no branch: the 

400 repository's default, which needs no check.""" 

401 if branch is None: 

402 return None 

403 path = Path(repo) 

404 # The model's own rules, checked here first so the refusal is one 

405 # sentence rather than pydantic's error list, and before any git call. 

406 if (problem := branch_name_problem(branch)) is not None: 

407 raise HTTPException(422, f"base branch {branch!r} is not a valid branch name: it {problem}") 

408 if not config_mod.git_read(path, "remote", "get-url", "origin", expected_failure=True): 

409 raise HTTPException( 

410 422, f"base branch {branch!r} cannot be checked: {repo} has no origin remote" 

411 ) 

412 try: 

413 done = await asyncio.to_thread( 

414 subprocess.run, 

415 ["git", "ls-remote", "--exit-code", "--heads", "origin", f"refs/heads/{branch}"], 

416 cwd=path, 

417 stdin=subprocess.DEVNULL, 

418 capture_output=True, 

419 text=True, 

420 timeout=30, 

421 env=builtins_mod.promptless_git_env(path), 

422 ) 

423 except subprocess.TimeoutExpired: 

424 raise HTTPException( 

425 422, f"could not reach {repo}'s origin to check base branch {branch!r}: timed out" 

426 ) from None 

427 if done.returncode == 2: 

428 raise HTTPException( 

429 422, 

430 f"no branch {branch!r} on origin of {repo}; push it first, or name a branch origin has", 

431 ) 

432 if done.returncode != 0: 

433 detail = done.stderr.strip() or "failed" 

434 raise HTTPException( 

435 422, f"could not reach {repo}'s origin to check base branch {branch!r}: {detail}" 

436 ) 

437 return branch

510def item_policy(st, repo: str, target: WorkItemTarget | None = None) -> InstancePolicy: 

511 """The policy a work item filed in `repo` starts from: the instance policy 

512 with that repository's layer on top (`RepoEntry.repository_override`), which 

513 may only tighten it (`repository-policy-cannot-relax-instance-safety`). 

514 Every intake door materializes from this, so the layer is frozen into every 

515 item filed in the repository. 

516 

517 A workspace `target` runs its ordinary tasks in an assembled checkout that 

518 holds every selected repository at once, so its layer is the meet of all 

519 of theirs -- each checked on its own first, so a refusal names the 

520 repository (Kraft-jc39p; `repository_policies` has each one alone). 

521 

522 Unlike `launch`, this raises: a `repos.yaml` that cannot be read has 

523 restrictions nobody can see, and filing an item without them is exactly 

524 the relaxation the layer exists to prevent. A `PolicyError`, so each door 

525 answers it the way it answers a chain policy past a ceiling.""" 

526 base = instance_policy(st) 

527 layers = _repository_layers(st, repo, target) 

528 for _rid, entry in layers: 

529 _layered(base, entry) 

530 overrides = [o for _rid, e in layers if (o := e.repository_override()) is not None] 

531 if not overrides: 

532 return base 

533 return base.apply_template_override(TemplatePolicyOverride.meet(overrides))

579class _PoisonedRepoEntry: 

580 """Stand-in `repo_entry` (and `repositories`) for a `repos.yaml` that 

581 failed to load. 

582 

583 Any read of it -- an attribute (`entry.sandbox`, `entry.env`), a key or 

584 a membership test on the `repositories` mapping -- raises the original 

585 `ConfigError` from *inside* the dispatch this entry was actually needed 

586 for, which is already wrapped in `guard`'s blanket `except Exception` -- 

587 the same needs_human treatment a `SteeringError` raised deep in a 

588 dispatch already gets. Truthy, so `if launch.repo_entry` never skips it 

589 for "nothing configured" before that read runs -- a repo config a caller 

590 can't parse must not quietly resolve to "no sandbox, no deny_tools, no 

591 steering" instead.""" 

592

594 self._exc = exc

596 def __bool__(self) -> bool: 

597 return True

600 raise self._exc

603 raise self._exc

605 def __contains__(self, _key): 

606 raise self._exc

612def library_steering(st) -> dict[str, str] | None: 

613 """The loaded library's steering profiles, name to instructions; `None` 

614 when no library loaded.""" 

615 library = getattr(st, "library", None) 

616 if library is None: 

617 return None 

618 return {n: p.instructions for n, p in library.steering.items()}

621def repository_steering( 

622 st, repo: str, target: WorkItemTarget | None = None 

623) -> dict[str, dict[str, str]]: 

624 """What an item filed in `repo` freezes as its repository steering 

625 (`MaterializedChain.repository_steering`): each repository it runs in 

626 that names steering in `repos.yaml`, by path, to those profiles' texts 

627 from the library. Raises `SteeringError` (a `ValueError`) naming a name 

628 the library does not define, or an unreadable `repos.yaml`, so an intake 

629 door refuses the item rather than filing it unsteered.""" 

630 try: 

631 repos = config_mod.load_repos(repos_path(st)) 

632 except config_mod.ConfigError as exc: 

633 raise steering_mod.SteeringError(str(exc)) from exc 

634 entries = [_connected(repos, repo)] 

635 if target is not None and target.kind == "workspace": 

636 by_id = {r.id: r for r in repos if r.id} 

637 entries += [by_id.get(rid) for rid in target.repositories()] 

638 profiles = library_steering(st) or {} 

639 return { 

640 e.path: steering_mod.select(e.steering, profiles, where=f"repos.yaml: {e.path}") 

641 for e in entries 

642 if e is not None and e.steering 

643 }

646def launch(st, repo: str) -> executor.LaunchContext: 

647 """The repo config for one agent dispatch. 

648 

649 A malformed `repos.yaml` must not crash `lifespan` on reattach (locking 

650 an operator out of the Settings UI that would let them fix it) or 500 the 

651 approve/reject/resume/retry routes, so this itself never raises. But a 

652 repo entry can carry `sandbox:` now, so silently degrading to 

653 `repo_entry=None` would run that dispatch's worker unsandboxed on the 

654 host with nothing but a log line nobody reads -- the exact bare-metal 

655 fallback the sandbox design rules out. Instead the failure is carried in 

656 a `_PoisonedRepoEntry` that raises the instant the dispatch that actually 

657 needed this repo's config reads it, landing in `guard` as needs_human 

658 the same way a bad `sandbox:` value would if it were only caught then. 

659 

660 The live library's steering profiles ride along for an item filed before 

661 repository steering was frozen into its snapshot 

662 (`steering.for_repository`); every other launch reads its snapshot's. 

663 

664 `chain_revision_digest` rides along too (Kraft-rndd1): `st.library` bound 

665 into `revision.artifact_digest`, so a gate's own agent review can capture 

666 what its artifact resolves to before dispatching, the same way `GET 

667 .../artifact` does for a person.""" 

668 steering = library_steering(st) or {} 

669 digest = functools.partial(revision.artifact_digest, library=getattr(st, "library", None)) 

670 try: 

671 repos = config_mod.load_repos(repos_path(st)) 

672 except config_mod.ConfigError as exc: 

673 logger.warning("repo config invalid, failing dispatch that reads it: %s", exc) 

674 # The members' entries too: a fanned-out task must fail the same 

675 # way, never run with no sandbox because its entry was unreadable. 

676 return executor.LaunchContext( 

677 repo_entry=cast("RepoEntry", _PoisonedRepoEntry(exc)), 

678 skills_dir=st.skills_dir, 

679 repositories=cast("dict[str, RepoEntry]", _PoisonedRepoEntry(exc)), 

680 library_steering=steering, 

681 chain_revision_digest=digest, 

682 ) 

683 return executor.LaunchContext( 

684 repo_entry=_connected(repos, repo), 

685 skills_dir=st.skills_dir, 

686 repositories={r.id: r for r in repos if r.id}, 

687 library_steering=steering, 

688 chain_revision_digest=digest, 

689 )
