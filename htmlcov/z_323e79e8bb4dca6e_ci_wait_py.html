1"""Re-enter a node parked on a pipeline that has not settled. 

2 

3`ci_poll` (`adapters/forge/run.py`) no longer blocks in-process for up to 

4`poll_timeout`: it makes one CI check, and a pending pipeline becomes a row 

5the scheduler owns -- `status = 'waiting'` plus `retry_at`, set by 

6`store.mark_waiting` (Kraft-ru98). This is the poller that owns that row: 

7structurally the same job `rate_limit_retry.py` does for a rate limit, ticking 

8on a fixed interval and relaunching through `executor.run(start_index=...)`, 

9the same path a manual retry uses. 

10 

11Always on, for the same reason the rate-limit poller is: a work item parked on 

12a pipeline has to be woken by something, and that something cannot be the 

13coroutine that used to sit in the wait. 

14""" 

15 

16from __future__ import annotations 

17 

18import asyncio 

19import json 

20import logging 

21 

22from kraft import policy as policy_mod 

23from kraft import store 

24from kraft.adapters import forge as _forge 

25from kraft.store import _now as _now # test seam for wall-clock checks 

26 

27logger = logging.getLogger(__name__) 

28 

29#: How often the poller checks for a due item. Same fixed cadence 

30#: `rate_limit_retry` uses -- no per-operator tuning knob for this. 

31_INTERVAL_S = 30 

32 

33#: Fallback when `app.state.policy` is `None` outright (an unset or invalid 

34#: policy.yaml) -- the same shape `poll_timeout`'s old 1800s default carried, 

35#: with attempts high enough not to bind before the wall clock does. 

36_DEFAULT_CAP = policy_mod.Cap(attempts=1000, wall_clock_s=int(_forge.DEFAULT_POLL_TIMEOUT)) 

37 

38 

39def _key(node_id: str) -> str: 

40 """Per-node, not per-item: mirrors `rate_limit_retry._key` for the same 

41 reason -- a node that waits, resolves, and later waits again on a 

42 *different* node must not spend the same budget the first node used. 

43 

44 Duplicated in `store.counters.retry_after_cap`, which clears this same 

45 key on `/retry` (Kraft-cs4s) -- edit both together. 

46 """ 

47 return f"ci_wait:{node_id}" 

48 

49 

50async def tick(app) -> list[str]: 

51 """One poll. Returns the work item ids re-entered, which is usually none.""" 

52 st = app.state 

53 due = st.db.read( 

54 lambda c: c.execute( 

55 "SELECT id, repo, current_node_id, chain_definition FROM work_items " 

56 "WHERE status = 'waiting' AND retry_at <= ?", 

57 (_now(),), 

58 ).fetchall() 

59 ) 

60 reentered: list[str] = [] 

61 for row in due: 

62 if await _re_enter_one(app, row): 

63 reentered.append(row["id"]) 

64 return reentered 

65 

66 

67async def _re_enter_one(app, row) -> bool: 

68 from kraft.api import deps 

69 

70 st = app.state 

71 wid = row["id"] 

72 node_id = row["current_node_id"] 

73 cap = policy_mod.resolve_cap(st.policy, "ci_wait") if st.policy is not None else _DEFAULT_CAP 

74 

75 def _bump_and_reenter(c): 

76 # One transaction: the counter bump and the status flip back to 

77 # 'active' must land together, or a tick between them could re-select 

78 # this same row (Kraft-ppk9). 

79 result = store.bump_counter(c, wid, _key(node_id), cap) 

80 store.mark_reentered(c, wid) 

81 return result 

82 

83 count, started_at, cap = await st.db.write(_bump_and_reenter) 

84 if policy_mod.check(count=count, started_at=started_at, cap=cap, now=_now()) == "breached": 

85 await st.db.write( 

86 lambda c: store.mark_needs_human( 

87 c, wid, node_id, f"CI wait exceeded its cap after {count - 1} re-entry(ies)" 

88 ) 

89 ) 

90 return False 

91 

92 chain = json.loads(row["chain_definition"]) 

93 start = next(i for i, n in enumerate(chain["nodes"]) if n["id"] == node_id) 

94 from kraft import executor # deferred: avoids a kraft.api <-> kraft.executor import cycle 

95 

96 # No steer: there is no agent to address here, and a note handed to a 

97 # forge node would sit unconsumed until the next agent launch -- a 

98 # different node's business. 

99 try: 

100 deps.spawn( 

101 app, 

102 wid, 

103 deps.guard( 

104 st.db, 

105 wid, 

106 executor.run( 

107 st.db, 

108 st.run_dirs, 

109 work_item_id=wid, 

110 registry=st.registry, 

111 bd_cwd=deps.bd_cwd(), 

112 start_index=start, 

113 policy=st.policy, 

114 launch=deps.launch(st, row["repo"]), 

115 ), 

116 ), 

117 ) 

118 except deps.AlreadyRunning: 

119 logger.warning("ci-wait: %s already has a live walk, skipping this tick", wid) 

120 return False 

121 return True

124async def poller(app) -> None: 

125 """`tick` on a fixed interval until cancelled.""" 

126 while True: 

127 await asyncio.sleep(_INTERVAL_S) 

128 try: 

129 await tick(app) 

130 except asyncio.CancelledError: 

131 raise 

132 except Exception: # noqa: BLE001 -- a bad tick must not end the poller 

133 logger.exception("ci-wait retry tick failed")
