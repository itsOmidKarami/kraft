1"""The FastAPI app. Split out of what used to be a single `kraft/api.py` -- 

2see the sibling modules for the actual routes: 

3 

4- `deps` -- helpers shared by several route modules 

5- `startup` -- `lifespan()` and its background tasks 

6- `perimeter` -- SPA navigation shortcut, local-client check, auth middleware 

7- `routes/*` -- one module per resource, each decorating `api_router` below 

8 

9Nothing outside this package imports anything from here except `app` -- 

10`kraft.intake`, `kraft.waits`, and `kraft.rate_limit_retry` import 

11`kraft.api.deps` directly instead (see that module's docstring). 

12"""

16from fastapi import APIRouter, FastAPI, HTTPException, Request 

17from fastapi.responses import FileResponse, JSONResponse 

18 

19from kraft import config as config_mod 

20from kraft.api import perimeter 

21from kraft.api.startup import lifespan 

22 

23app = FastAPI(lifespan=lifespan) 

24 

25#: Every JSON endpoint lives under here, so it can never share a path with an 

26#: SPA client-side route (e.g. GET /work-items/<id> the page vs. the same path 

27#: as a JSON handler) — that collision used to make a browser render raw JSON 

28#: on some deep links instead of the app shell. 

29api_router = APIRouter(prefix="/api") 

30 

31# Registration order matters: Starlette runs the *last*-declared middleware 

32# first, so `_perimeter` (who may talk to this server at all) has to be 

33# declared last, after `_authenticate` (session/bearer), after 

34# `_spa_navigation` (the SPA-shell fast path) -- see `perimeter._perimeter`'s 

35# own docstring. 

36app.middleware("http")(perimeter._spa_navigation) 

37app.middleware("http")(perimeter._authenticate) 

38app.middleware("http")(perimeter._perimeter)

40# Each of these decorates `api_router` (imported above) with its own routes.

58@app.get("/{path:path}") 

59async def spa(path: str, request: Request): 

60 if perimeter._is_api_path(f"/{path}"): 

61 # A real API prefix with no matching route is a bad request, not a 

62 # missing page — this must not fall through to the SPA shell. 

63 raise HTTPException(404, "not found") 

64 dist = request.app.state.frontend_dist 

65 if dist is None: 

66 raise HTTPException(404, "not found") 

67 candidate = (dist / path).resolve() 

68 if dist.resolve() in candidate.parents and candidate.is_file(): 

69 return FileResponse(candidate) 

70 return FileResponse(dist / "index.html")

73@app.exception_handler(config_mod.ConfigError) 

74async def _bad_config_file(request: Request, exc: config_mod.ConfigError) -> JSONResponse: 

75 """A legible 422 for any `load_repos`/`Intake.load`/etc. caller that does 

76 not catch `ConfigError` itself. `templates/` is a plain directory an 

77 operator can hand-edit, and the message already names the file and what is 

78 wrong with it — a global backstop is the fix, not a guard at each call 

79 site, because the next caller of `load_repos` would just inherit the same 

80 trap a per-route `try/except` does nothing to close. 

81 

82 A route that already catches `ConfigError` and raises its own 4xx 

83 (`_validate_repos`, `probe_repo`, `_launch`) never reaches this handler — 

84 only an *uncaught* `ConfigError` does.""" 

85 return JSONResponse({"detail": str(exc)}, status_code=422)

88@app.exception_handler(404) 

89async def _spa_deep_link(request: Request, exc: HTTPException): 

90 # Every non-/api GET is answered directly by the spa() catch-all above — 

91 # it returns a file or index.html unconditionally and never raises — so 

92 # the only 404s that reach here are genuinely /api/ ones (a bad path, or a 

93 # handler's own `raise HTTPException(404, ...)`). Always JSON: nothing 

94 # under /api/ is ever the shell, whatever Accept header asks for it. 

95 return JSONResponse({"detail": exc.detail}, status_code=404)
