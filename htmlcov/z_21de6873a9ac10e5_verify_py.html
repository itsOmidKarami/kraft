1"""`kraft repo connect --verify`: rehearse a repo's declared commands once, in 

2a throwaway worktree, while a person is still watching. 

3 

4A work item's worktree is a fresh checkout of the commit the probe read 

5(`detect.source_ref`: origin's default branch, else HEAD) with nothing 

6untracked in it but the entry's `local_files`. So this runs exactly what a 

7work item would -- the entry's `setup_command` through a shell, then each 

8test scope's command split into argv, from the worktree's root, with the 

9worker's allowlisted environment and nothing more (no `CI=true`: workers do 

10not get one either, and a rehearsal that differs proves less) -- and reports 

11each one's outcome and time. 

12 

13Two things a work item would trip over are checked on the way: 

14 

15- a step that does not finish in `timeout` minutes, the usual cause being a 

16 test runner in watch mode, which a work item would wait on too; 

17- a step that leaves files git would commit (an unlocked `uv sync` writing 

18 uv.lock, a `build/` directory nobody ignored): every work item would commit 

19 them into its merge request. 

20 

21On the host, always. A sandboxed repo's commands run in its sandbox when a 

22work item runs them, so rehearsing them on the host is refused unless the 

23caller says `on_host`. 

24"""

57def _run( 

58 argv: list[str] | str, cwd: Path, env: dict, log: Path, timeout_s: float 

59) -> tuple[str, float]: 

60 """("passed" | "failed" | "timed out", seconds). A step that times out, 

61 or is interrupted (Ctrl-C), is killed with every process it started: 

62 `sh -c` dying alone would leave its npm and node children writing into a 

63 worktree being removed. Its own session keeps the terminal's SIGINT from 

64 reaching it, so the kill is this function's to do.""" 

65 started = time.monotonic() 

66 with log.open("w") as out: 

67 try: 

68 proc = subprocess.Popen( 

69 argv, 

70 cwd=cwd, 

71 env=env, 

72 shell=isinstance(argv, str), 

73 stdout=out, 

74 stderr=subprocess.STDOUT, 

75 stdin=subprocess.DEVNULL, 

76 start_new_session=True, 

77 ) 

78 except OSError as exc: # a test command whose program is not installed 

79 out.write(f"{exc}\n") 

80 return "failed", time.monotonic() - started 

81 try: 

82 code = proc.wait(timeout=timeout_s) 

83 except subprocess.TimeoutExpired: 

84 os.killpg(proc.pid, signal.SIGKILL) 

85 proc.wait() 

86 return "timed out", time.monotonic() - started 

87 except BaseException: 

88 os.killpg(proc.pid, signal.SIGKILL) 

89 proc.wait() 

90 raise 

91 return ("passed" if code == 0 else "failed"), time.monotonic() - started

94def _tail(log: Path) -> str: 

95 lines = log.read_text("utf-8", "replace").splitlines() 

96 return "\n".join(f" {line}" for line in lines[-_TAIL:])

99def _changed(worktree: Path, *, setup: bool = False) -> tuple[set[str], set[str]]: 

100 """(what `git add -A` would stage, the installs left out of it). The 

101 first: untracked files git does not ignore, and tracked files changed, 

102 less Kraft's own roots and an untracked virtualenv or `node_modules`, 

103 which Kraft's sweep leaves out (`forge.git.work_product_pathspec`), and, 

104 after the `setup` step, a lockfile it wrote, which the sweep leaves out 

105 too. The second: those installs, by directory or file. The agent commits 

106 on its own too, and a `git add -A` of its takes one the repo does not 

107 ignore.""" 

108 from kraft.adapters.forge.git import WRITTEN_LOCKFILES, is_environment 

109 

110 out = subprocess.run( 

111 ["git", "status", "--porcelain=v1", "-z", "--untracked-files=all"], 

112 cwd=worktree, 

113 capture_output=True, 

114 text=True, 

115 check=False, 

116 ).stdout 

117 entries = [(entry[:2], entry[3:]) for entry in out.split("\0") if len(entry) > 3] 

118 kraft_roots = tuple(f"{r}/" for r in config_mod.KRAFT_ROOTS) 

119 

120 def install(path: str) -> str | None: 

121 parts = path.split("/") 

122 if setup and parts[-1] in WRITTEN_LOCKFILES: 

123 return path 

124 for n in range(1, len(parts)): 

125 if is_environment(worktree.joinpath(*parts[:n])): 

126 return "/".join(parts[:n]) 

127 return None 

128 

129 changed: set[str] = set() 

130 installs: set[str] = set() 

131 for code, p in entries: 

132 if p.startswith(kraft_roots): 

133 continue 

134 where = install(p) if code == "??" else None 

135 if where is None: 

136 changed.add(p) 

137 else: 

138 installs.add(where) 

139 return changed, installs

142def _scopes(entry: config_mod.RepoEntry) -> list[tuple[str, str | None, str]]: 

143 """(label, setup to run first, command) per test scope, the repository's 

144 then each area's -- `dispatch._select_scopes`' table, with every scope 

145 selected, as a change touching everything would.""" 

146 scopes = [(", ".join(s.paths), None, s.command) for s in entry.test_scopes or ()] 

147 if not scopes and entry.test_command: 

148 scopes = [("**", None, entry.test_command)] 

149 for name, area in entry.areas.items(): 

150 for s in (area.get("verification") or {}).get("test_scopes") or []: 

151 scopes.append( 

152 (f"area {name}: {', '.join(s['paths'])}", area.get("setup"), s["command"]) 

153 ) 

154 return scopes

157def _steps(entry: config_mod.RepoEntry, say) -> tuple[list[tuple[str, list[str] | str]], bool]: 

158 """The commands to run, in order, and whether the entry declares enough 

159 for a work item to start at all.""" 

160 ok = True 

161 steps: list[tuple[str, list[str] | str]] = [] 

162 if entry.setup_command is None: 

163 say(" setup: none declared -- a work item on this repo stops before it starts") 

164 ok = False 

165 elif entry.setup_command: 

166 steps.append(("setup", entry.setup_command)) 

167 else: 

168 say(' setup: "" (nothing to prepare)') 

169 scopes = _scopes(entry) 

170 if not scopes: 

171 if entry.test_command == "": 

172 say(' tests: "" (this repo declares no tests)') 

173 else: 

174 say(" tests: none declared -- verify stops every work item here") 

175 ok = False 

176 ran_setups: set[str] = set() 

177 for label, area_setup, command in scopes: 

178 if area_setup and area_setup not in ran_setups: 

179 ran_setups.add(area_setup) 

180 steps.append((f"area setup ({label})", shlex.split(area_setup))) 

181 steps.append((f"test [{label}]", shlex.split(command))) 

182 return steps, ok

185def verify(stored: dict, *, say=print, timeout_minutes: float = 30, on_host: bool = False) -> bool: 

186 """Run `stored`'s (a repos.yaml entry, as the API returns it) setup and 

187 test commands in a fresh worktree; True when every one passed and left 

188 nothing for a commit to pick up.""" 

189 entry = config_mod.RepoEntry.model_validate( 

190 {k: v for k, v in stored.items() if k not in _TOLD}, 

191 context={"unrecognised_keys_reported": True}, 

192 ) 

193 repo = Path(entry.path) 

194 if entry.effective_sandbox is not None and not on_host: 

195 say( 

196 "verify: this repo runs sandboxed, and --verify would run its commands on this " 

197 "machine instead; pass --on-host to do that anyway" 

198 ) 

199 return False 

200 env = worker_env(entry) 

201 ref = detect.source_ref(repo) or "HEAD" 

202 logs = Path(tempfile.mkdtemp(prefix="kraft-verify-logs-")) 

203 worktree = Path(tempfile.mkdtemp(prefix="kraft-verify-")) / "checkout" 

204 added = subprocess.run( 

205 ["git", "-C", str(repo), "worktree", "add", "--detach", "-q", str(worktree), ref], 

206 capture_output=True, 

207 text=True, 

208 check=False, 

209 ) 

210 if added.returncode != 0: 

211 shutil.rmtree(worktree.parent, ignore_errors=True) 

212 shutil.rmtree(logs, ignore_errors=True) 

213 say(f"verify: could not cut a worktree of {ref}: {added.stderr.strip()}") 

214 return False 

215 say(f"verify: a fresh worktree of {ref} at {worktree}") 

216 try: 

217 ok = _rehearse(entry, repo, worktree, env, logs, timeout_minutes, say) 

218 finally: 

219 subprocess.run( 

220 ["git", "-C", str(repo), "worktree", "remove", "--force", str(worktree)], 

221 capture_output=True, 

222 check=False, 

223 ) 

224 shutil.rmtree(worktree.parent, ignore_errors=True) 

225 if ok: # a failure's output is named above, for the person to read 

226 shutil.rmtree(logs, ignore_errors=True) 

227 say("verify: passed" if ok else "verify: failed -- fix the entry in repos.yaml and run again") 

228 return ok

231def _rehearse(entry, repo, worktree, env, logs, timeout_minutes, say) -> bool: 

232 # The worker's own copy, with its guards: a file the repo does not 

233 # ignore, or one that would land outside the worktree, is not carried. 

234 _, refused = builtins_mod._carry_local_files(repo, worktree, list(entry.local_files)) 

235 ok = not refused 

236 for rel in refused: 

237 say(f" local_files: {rel} not carried: it is not a file, or the repo does not ignore it") 

238 baseline, installed = _changed(worktree) 

239 steps, declared = _steps(entry, say) 

240 ok = ok and declared 

241 for n, (label, argv) in enumerate(steps): 

242 shown = argv if isinstance(argv, str) else shlex.join(argv) 

243 say(f" {label}: {shown} ...") 

244 log = logs / f"{n:02d}.log" 

245 # Dispatch runs an area's setup and a test with `TEST_ENV` on top, so 

246 # the bytecode a Python test would write is no file a work item leaves. 

247 step_env = env if label == "setup" else {**env, **TEST_ENV} 

248 outcome, took = _run(argv, worktree, step_env, log, timeout_minutes * 60) 

249 say(f" {outcome if outcome == 'passed' else outcome.upper()} in {took:.1f}s") 

250 if outcome != "passed": 

251 ok = False 

252 say(_tail(log)) 

253 say(f" full output: {log}") 

254 if outcome == "timed out": 

255 say( 

256 f" it did not finish in {timeout_minutes:g} minutes, and a work item would " 

257 "wait on it too. A test runner in watch mode is the usual cause: give it its " 

258 'run-once flag, or `env: {CI: "true"}` in the repo\'s repos.yaml entry, which ' 

259 "workers get too" 

260 ) 

261 changed, installs = _changed(worktree, setup=label == "setup") 

262 left = sorted(changed - baseline - installed) 

263 for where in sorted(installs - installed): 

264 # Not a failure: Kraft's own commits leave it out. 

265 if (worktree / where).is_file(): 

266 say( 

267 f" left {where}, a lockfile the repo does not commit: Kraft's commits " 

268 "leave it out, but an agent's `git add -A` would commit it. Commit one " 

269 f"({LOCK_COMMANDS[Path(where).name]}), so every work item installs the " 

270 "same versions" 

271 ) 

272 continue 

273 say( 

274 f" left {where}/, an install the repo does not ignore: Kraft's commits " 

275 "leave it out, but an agent's `git add -A` would commit it. Ignore it " 

276 "(.gitignore)" 

277 ) 

278 installed |= installs 

279 if left: 

280 ok = False 

281 more = f" and {len(left) - 8} more" if len(left) > 8 else "" 

282 say( 

283 f" left files every work item would commit: {', '.join(left[:8])}{more}. " 

284 "Ignore them (.gitignore), or have the command not write them (commit its " 

285 "lockfile)" 

286 ) 

287 baseline |= set(left) 

288 if outcome != "passed" and label == "setup": 

289 say(" the tests were not run: nothing after a failed setup can be trusted") 

290 break 

291 return ok
