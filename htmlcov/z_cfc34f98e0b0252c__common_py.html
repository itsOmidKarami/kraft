13def write_status(conn, sql: str, params: tuple) -> bool: 

14 """Run `sql`, an `UPDATE work_items ... WHERE ...` that moves an item to a 

15 status other than an ending one, unless the item has already ended. True 

16 when it wrote; a caller appends its event only then (Kraft-y6f08). 

17 

18 Every non-ending status write in `store/` goes through here. `sql` stays a 

19 whole literal at its call site, so `dev/check_claim_handoff.py` still sees 

20 each claim. The ending writes (`mark_completed`, `MANUAL_ENDS`, 

21 `abandon_work_item`) do not: ending is the one move an item may always make. 

22 

23 A write that took also ends every hold it finds overtaken 

24 (`_end_overtaken_holds`): normally the one item this write just moved out 

25 of `queued` or `blocked` without the start queue, whose `work_item_dequeued` 

26 then comes before the caller's own event. It looks at the whole table, so a 

27 row left that way by anything else is ended here too. 

28 """ 

29 wrote = ( 

30 conn.execute(f"{sql} AND status NOT IN ({marks(ENDED)})", (*params, *ENDED)).rowcount == 1 

31 ) 

32 if wrote: 

33 _end_overtaken_holds(conn) 

34 return wrote

37def _end_overtaken_holds(conn) -> None: 

38 """A row that carries a start request and is not `queued` or `blocked` was 

39 held, and something other than the start queue has just moved it: a stop, 

40 a gate review's verdict, a skip. The start it held is over, so the request 

41 goes and its timeline says so, ahead of the mover's own event. 

42 

43 Here, not at each write: every non-ending status write passes through 

44 `write_status`, the next one written included. The queue's own moves clear 

45 the request in the same `UPDATE` (`work_items._PUT_BACK`), so they never 

46 match. An ended row is left alone: its ending event already says what 

47 happened. One scan of `work_items`, a row per work item, per status write.""" 

48 apart = (WorkItemStatus.QUEUED, WorkItemStatus.BLOCKED, *ENDED) 

49 for wid, status, saved in conn.execute( 

50 "SELECT id, status, queued_request FROM work_items " 

51 f"WHERE queued_request IS NOT NULL AND status NOT IN ({marks(apart)})", 

52 apart, 

53 ).fetchall(): 

54 conn.execute("UPDATE work_items SET queued_request = NULL WHERE id = ?", (wid,)) 

55 events.append( 

56 conn, 

57 wid, 

58 WorkItemEvent.DEQUEUED, 

59 {"why": "superseded", "detail": _dropped(saved), "to": status}, 

60 )

63def _dropped(saved: str) -> str: 

64 """What an overtaken hold was holding, for its timeline: the verb, and the 

65 steer a person attached to it, which reaches no agent now.""" 

66 request = json.loads(saved) 

67 steer = (request.get("body") or {}).get("steer") 

68 said = f"the queued {request.get('verb', 'start')} was dropped" 

69 return f"{said}, with its steer: {steer}" if steer else said

76def _span_ms(start: str | None, end: str) -> int | None: 

77 """Milliseconds between two ISO timestamps, or None if either is unusable. 

78 

79 TypeError alongside ValueError: a naive and an aware timestamp can't be 

80 subtracted, and a row written outside `_now()`'s own format (a raw SQL 

81 fixture using sqlite's `datetime('now')`, say) can carry one. Same 

82 contract either way -- unusable is unusable, not a crash. 

83 """ 

84 if not start: 

85 return None 

86 try: 

87 return int( 

88 (datetime.fromisoformat(end) - datetime.fromisoformat(start)).total_seconds() * 1000 

89 ) 

90 except (ValueError, TypeError): 

91 return None

94def session_wall_ms(row, now: str | None = None) -> int | None: 

95 """A session's wall time: the recorded one, or derived from its stamps. 

96 

97 `wall_ms` is only ever written by `session_exited`, and a session that is 

98 paused, skipped, capped, lost or still running never reaches it -- 82 rows 

99 on this machine, every one of them carrying the two stamps `session_exited` 

100 would itself have used (Kraft-s7c04.18). The column is a cache of a value 

101 the row already holds, so a missing one is derived rather than counted as 

102 zero. That is also why this is one read-side derivation and not a `wall_ms` 

103 write bolted onto each of the five paths that reach a terminal status: a 

104 write fixes nothing retroactively, and no write at all can describe a 

105 session that has not stopped. 

106 

107 A still-running row has no `exited_at` and derives against `now`, which is 

108 what makes a live node report the time it has actually spent instead of 

109 `0s` (Kraft-s7c04.47's totals half). 

110 

111 None, never 0, when there is no usable start: zero is an answer, and it 

112 would be a wrong one. 

113 """ 

114 if row["wall_ms"] is not None: 

115 return row["wall_ms"] 

116 return _span_ms(row["started_at"] or row["created_at"], row["exited_at"] or now or _now())

119def wait_sessions(conn, work_item_ids) -> set[str]: 

120 """The sessions among `work_item_ids`' that ran an external wait (a forge 

121 task watching CI, a review, a merge): each observation it made names it in 

122 an `external_wait_observed` record (`kraft.waits`). Such a session waits; 

123 it does no work of its own.""" 

124 ids = list(work_item_ids) 

125 if not ids: 

126 return set() 

127 rows = conn.execute( 

128 "SELECT DISTINCT json_extract(payload, '$.session_id') AS sid FROM events " 

129 f"WHERE work_item_id IN ({','.join('?' * len(ids))}) " 

130 "AND type = ?", 

131 (*ids, ForgeEvent.EXTERNAL_WAIT_OBSERVED), 

132 ).fetchall() 

133 return {r["sid"] for r in rows if r["sid"]}

136def wait_timed_out_sessions(conn, work_item_ids) -> set[str]: 

137 """The sessions among `work_item_ids`' that ended because an external 

138 wait timed out. Such a session exits `capped_out`, the status a fix loop's 

139 cap also writes; the wait's own `external_wait_ended` record (`kraft.waits`) 

140 is what tells the two apart (Kraft-uwbc8), so no status was added.""" 

141 ids = list(work_item_ids) 

142 if not ids: 

143 return set() 

144 rows = conn.execute( 

145 "SELECT json_extract(payload, '$.session_id') AS sid FROM events " 

146 f"WHERE work_item_id IN ({','.join('?' * len(ids))}) " 

147 "AND type = ? AND json_extract(payload, '$.outcome') = 'timed_out'", 

148 (*ids, ForgeEvent.EXTERNAL_WAIT_ENDED), 

149 ).fetchall() 

150 return {r["sid"] for r in rows if r["sid"]}
