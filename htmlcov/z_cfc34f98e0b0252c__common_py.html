10def write_status(conn, sql: str, params: tuple) -> bool: 

11 """Run `sql`, an `UPDATE work_items ... WHERE ...` that moves an item to a 

12 status other than an ending one, unless the item has already ended. True 

13 when it wrote; a caller appends its event only then (Kraft-y6f08). 

14 

15 Every non-ending status write in `store/` goes through here. `sql` stays a 

16 whole literal at its call site, so `dev/check_claim_handoff.py` still sees 

17 each claim. The ending writes (`mark_completed`, `MANUAL_ENDS`, 

18 `abandon_work_item`) do not: ending is the one move an item may always make. 

19 """ 

20 return conn.execute(f"{sql} AND status NOT IN (?, ?)", (*params, *ENDED)).rowcount == 1

27def _span_ms(start: str | None, end: str) -> int | None: 

28 """Milliseconds between two ISO timestamps, or None if either is unusable. 

29 

30 TypeError alongside ValueError: a naive and an aware timestamp can't be 

31 subtracted, and a row written outside `_now()`'s own format (a raw SQL 

32 fixture using sqlite's `datetime('now')`, say) can carry one. Same 

33 contract either way -- unusable is unusable, not a crash. 

34 """ 

35 if not start: 

36 return None 

37 try: 

38 return int( 

39 (datetime.fromisoformat(end) - datetime.fromisoformat(start)).total_seconds() * 1000 

40 ) 

41 except ValueError, TypeError: 

42 return None

45def session_wall_ms(row, now: str | None = None) -> int | None: 

46 """A session's wall time: the recorded one, or derived from its stamps. 

47 

48 `wall_ms` is only ever written by `session_exited`, and a session that is 

49 paused, skipped, capped, lost or still running never reaches it -- 82 rows 

50 on this machine, every one of them carrying the two stamps `session_exited` 

51 would itself have used (Kraft-s7c04.18). The column is a cache of a value 

52 the row already holds, so a missing one is derived rather than counted as 

53 zero. That is also why this is one read-side derivation and not a `wall_ms` 

54 write bolted onto each of the five paths that reach a terminal status: a 

55 write fixes nothing retroactively, and no write at all can describe a 

56 session that has not stopped. 

57 

58 A still-running row has no `exited_at` and derives against `now`, which is 

59 what makes a live node report the time it has actually spent instead of 

60 `0s` (Kraft-s7c04.47's totals half). 

61 

62 None, never 0, when there is no usable start: zero is an answer, and it 

63 would be a wrong one. 

64 """ 

65 if row["wall_ms"] is not None: 

66 return row["wall_ms"] 

67 return _span_ms(row["started_at"] or row["created_at"], row["exited_at"] or now or _now())

70def wait_timed_out_sessions(conn, work_item_ids) -> set[str]: 

71 """The sessions among `work_item_ids`' that ended because an external 

72 wait timed out. Such a session exits `capped_out`, the status a fix loop's 

73 cap also writes; the wait's own `external_wait_ended` record (`kraft.waits`) 

74 is what tells the two apart (Kraft-uwbc8), so no status was added.""" 

75 ids = list(work_item_ids) 

76 if not ids: 

77 return set() 

78 rows = conn.execute( 

79 "SELECT json_extract(payload, '$.session_id') AS sid FROM events " 

80 f"WHERE work_item_id IN ({','.join('?' * len(ids))}) " 

81 "AND type = 'external_wait_ended' AND json_extract(payload, '$.outcome') = 'timed_out'", 

82 ids, 

83 ).fetchall() 

84 return {r["sid"] for r in rows if r["sid"]}
