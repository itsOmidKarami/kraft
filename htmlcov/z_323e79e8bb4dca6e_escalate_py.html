1"""Escalate to Kraft-Agent (docs/superpowers/specs/2026-09-10-escalate-to-kraft-agent-design.md). 

2 

3One escalation turn dispatches through the same `_subprocess.run_task` 

4engine every chain node uses, `--resume`d once a thread exists. The 

5dispatch is deliberately not a Kraft worker (`identify_as_worker=False`): 

6`client.resolve_context()` then resolves it the same as a human's own 

7session standing in the worktree, which is what lets it call 

8`kraft item retry` on the very item it is escalating without any change to 

9`client.context._forbid_self_action`. 

10"""

101def _last_stop(db, work_item_id: str, evts: list | None = None) -> dict: 

102 """The most recent `work_item_needs_human` event's payload, or `{}` -- the 

103 live answer to "why is this stopped", same reverse-scan idiom 

104 `kraft.executor.dispatch.last_measurement`/`needs_context_question` 

105 already use. `evts`, when given, is a timeline the caller already 

106 fetched (`gates.auto_escalate_stuck`'s single read, threaded through 

107 `dispatch`) -- read fresh only when nothing was handed in, which is 

108 every call from the manual `/escalate` route today.""" 

109 evts = evts if evts is not None else db.read(lambda c: events.read_after(c, 0, work_item_id)) 

110 for e in reversed(evts): 

111 if e["type"] == "work_item_needs_human": 

112 return e["payload"] 

113 return {}

116def _reason_line(stop: dict, status: str) -> str: 

117 """The `{reason_line}` `_STATE` slot, with the stop's `suggested_action` 

118 when it has one (Kraft-s7c04.27), so a turn starts from what the chain 

119 already concluded instead of rediscovering it. `needs_human` only -- a 

120 `paused` item's most recent `work_item_needs_human` reason (if it has one 

121 at all) belongs to whatever it stopped for *before* being paused, not to 

122 why it is paused now, so reporting it here would be stale or actively 

123 misleading (Kraft-k5ol code-review finding).""" 

124 if status != "needs_human": 

125 return "" 

126 line = f"Why it is stopped: {stop.get('reason') or '(no reason recorded)'}\n" 

127 if suggested := stop.get("suggested_action"): 

128 line += ( 

129 f"What Kraft suggests a person do: {suggested['action']} -- {suggested['reason']} " 

130 f"(`kraft item {suggested['action']}`)\n" 

131 ) 

132 return line

143def _judge_line(db, work_item_id: str, node_id: str | None, evts: list | None = None) -> str: 

144 """The `{judge_line}` slot: the fix-loop judge's own read of why this node's 

145 rounds were not converging, or "" (Kraft-s7c04.5). 

146 

147 The judge is asked to explain every verdict, its reasoning is already paid 

148 for, and nothing downstream read it -- so on 43717ee6 the escalation 

149 re-derived the root cause the judge had already named, 17 minutes and $3.72 

150 later. 

151 

152 Bounded on two axes. `_JUDGE_SPENT`, because a verdict from a closed episode 

153 is stale rather than context; and on `node_id`, because `judge_verdict` 

154 payloads name their node and an earlier node's fix-loop trend says nothing 

155 about the node this escalation is actually about. 

156 """ 

157 if node_id is None: 

158 return "" 

159 evts = evts if evts is not None else db.read(lambda c: events.read_after(c, 0, work_item_id)) 

160 for e in reversed(evts): 

161 if e["type"] in _JUDGE_SPENT: 

162 return "" 

163 if e["type"] == "judge_verdict" and e["payload"].get("node_id") == node_id: 

164 reasoning = (e["payload"].get("reasoning") or "").strip() 

165 if not reasoning: 

166 return "" 

167 return f"What the fix-loop judge made of it: {reasoning}\n" 

168 return ""

171def escalation_running(db, work_item_id: str) -> str | None: 

172 """The id of a `pending`/`running` escalation session for 

173 `work_item_id`, or None. 

174 

175 Moved out of `kraft.api.routes.lifecycle` (Kraft-lpdd) so 

176 `kraft.executor.gates.auto_escalate_stuck` can run the same check 

177 `retry`, `resume`, `escalate`, and `skip` already do before 

178 dispatching a fresh agent into a worktree an escalation turn may 

179 already be live in. 

180 

181 `retry` (lifecycle.py) also uses this to tell a caller apart from an 

182 *unrelated* second escalation turn: it compares the id this returns 

183 against the caller's own `X-Kraft-Session-Id` rather than excluding 

184 that session from the search here -- a genuine self-retry still has to 

185 be recognised as "an escalation turn is live", just handled differently 

186 (deferred, not refused) than a stranger's call landing mid-turn. 

187 """ 

188 row = db.read( 

189 lambda c: c.execute( 

190 "SELECT id FROM worker_sessions WHERE work_item_id = ? AND hook_point = 'escalation' " 

191 "AND status IN ('pending', 'running') LIMIT 1", 

192 (work_item_id,), 

193 ).fetchone() 

194 ) 

195 return row["id"] if row else None

199 """The status of `work_item_id`'s most recently *created* escalation 

200 session, finished or still running -- unlike `escalation_running`, which 

201 only ever returns an id for a `pending`/`running` one. Lets 

202 `gates.auto_escalate_stuck` tell "the last turn asked a question and is

210 "SELECT status FROM worker_sessions WHERE work_item_id = ? " 

211 "AND hook_point = 'escalation' ORDER BY created_at DESC LIMIT 1", 

212 (work_item_id,), 

213 ).fetchone() 

214 ) 

215 return row["status"] if row else None

291def _thread_runtime(db, work_item_id: str, thread: int) -> dict | None: 

292 """The runtime escalation `thread` started on, off its first turn's 

293 `escalation_message`; None for a thread that predates the record.""" 

294 evts = db.read(lambda c: events.read_after(c, 0, work_item_id)) 

295 for e in evts: 

296 if e["type"] == "escalation_message" and e["payload"].get("thread") == thread: 

297 runtime = e["payload"].get("runtime") 

298 return {k: runtime[k] for k in _RUNTIME if k in runtime} if runtime else None 

299 return None

302async def dispatch( 

303 db, 

304 run_dirs, 

305 *, 

306 work_item_id: str, 

307 message: str, 

308 launch: LaunchContext, 

309 auto: bool = False, 

310 #: Start a fresh agent session instead of continuing the item's latest 

311 #: escalation thread (ESCALATION_THREADS_SPEC.md §1). No-op on an item's 

312 #: very first escalation -- there is no prior thread to be "new" from, 

313 #: and `row["escalation_session_id"]` is already `None` either way. 

314 new_thread: bool = False, 

315 #: A timeline the caller already fetched, reused instead of a fresh 

316 #: `events.read_after` here. `None` (the manual `/escalate` route's 

317 #: only call site) falls back to reading it fresh -- that path calls 

318 #: this once per human action, not once per `run()`/`resume()`, so the 

319 #: extra read there costs nothing worth threading a parameter through 

320 #: `deps.spawn` for. 

321 evts: list | None = None, 

322) -> str: 

323 """One escalation turn: send `message` into `work_item_id`'s escalation 

324 thread, resuming it if `work_items.escalation_session_id` is already set 

325 (unless `new_thread` starts a fresh one instead). 

326 

327 Assumes its caller already checked the item is `needs_human` and that no 

328 escalation turn is currently running for it (`escalation_running`) -- 

329 `kraft.api.routes.lifecycle`'s manual `/escalate` route and 

330 `kraft.executor.gates.auto_escalate_stuck`'s unattended trigger both do, 

331 the same separation `executor.run`/`resume` keep from their own 

332 preconditions. 

333 """ 

334 row = db.read( 

335 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (work_item_id,)).fetchone() 

336 ) 

337 if row is None: 

338 raise LookupError(f"unknown work_item {work_item_id!r}") 

339 

340 latest_thread = db.read(lambda c: store.latest_escalation_thread(c, work_item_id)) 

341 thread = latest_thread + 1 if new_thread else (latest_thread or 1) 

342 resume_session_id = None if new_thread else row["escalation_session_id"] 

343 turn = db.read(lambda c: store.escalation_thread_turn_count(c, work_item_id, thread)) + 1 

344 

345 session_id = uuid.uuid4().hex 

346 worktree = run_dirs.worktrees / work_item_id 

347 

348 task_instruction = _STATE.format( 

349 opening=_AUTO_OPENING if auto else _MANUAL_OPENING, 

350 status=row["status"], 

351 node_id=row["current_node_id"], 

352 reason_line=_reason_line(_last_stop(db, work_item_id, evts), row["status"]), 

353 judge_line=_judge_line(db, work_item_id, row["current_node_id"], evts=evts), 

354 description_line=_description_line(row), 

355 action_line=_NEEDS_HUMAN_ACTION if row["status"] == "needs_human" else _PAUSED_ACTION, 

356 hands_off=_HANDS_OFF.format(work_item_id=work_item_id), 

357 message=message, 

358 ) 

359 

360 # Node-scoped (Kraft-l8ype, `stuck-escalation-is-an-exec-node-control`): 

361 # the turn runs under the policy of the node the item stopped at, resolved 

362 # from its snapshot like every other agent launch -- instance, repository, 

363 # work item, chain and node. Its tool lists, frozen sandbox, harness 

364 # allowlist and token budget all bound it; no snapshot, or no such node in 

365 # it, is no policy anyone could know, and refused rather than run unbounded. 

366 policy = _node_policy(row) 

367 breach = ( 

368 stops.budget_breach( 

369 db, work_item_id, _policy.NO_BUDGET, row=row, path=row["current_node_id"] 

370 ) 

371 if policy is not None 

372 else None 

373 ) 

374 if policy is None or breach is not None: 

375 await _record_message(db, work_item_id, session_id, message, auto, thread, turn, None) 

376 return await _refused( 

377 db, 

378 run_dirs, 

379 session_id=session_id, 

380 row=row, 

381 log=( 

382 stops.budget_reason(breach) 

383 if breach is not None 

384 else f"escalation has no policy to run under: work item {work_item_id} has " 

385 f"no node {row['current_node_id']!r} in a materialized chain" 

386 ) 

387 + "\n", 

388 ) 

389 

390 # Its own `escalate:` kwarg (left at the `False` default) is the fix loop's 

391 # unrelated "buy a stronger model" bump -- same word, different feature; 

392 # not to be confused with this module. 

393 # 

394 # `harness:`, **not** `command: "claude"`. Same shape Task 4b removed from 

395 # `gate_review.py`, and removed here for the same reason: a hardcoded 

396 # `command` bypasses the harness declaration entirely, so an operator who 

397 # overlays `~/.kraft/templates/harnesses/claude.yaml` is ignored and a test 

398 # fixture cannot substitute a fake. `run_agent_task` falls through to the 

399 # harness's own declared command when `command` is empty 

400 # (`adapters/agent.py`'s `command=command or None`), which is what every 

401 # other V1 agent launch already does. 

402 try: 

403 # The item's sandbox, as for every launch of it (Ruling 189). 

404 sandbox = executor.item_sandbox(row, launch) 

405 except executor.SandboxUnresolved as exc: 

406 await _record_message(db, work_item_id, session_id, message, auto, thread, turn, None) 

407 return await _refused(db, run_dirs, session_id=session_id, row=row, log=f"{exc}\n") 

408 try: 

409 inv = _agent.resolve_agent_task( 

410 ESCALATION_TASK, 

411 launch.repo_entry, 

412 launch.library_steering, 

413 skills_dir=launch.skills_dir, 

414 # The repository's steering, frozen with the item at intake. 

415 **executor.frozen_steering(row), 

416 policy=policy, 

417 ) 

418 except _agent.HarnessUnavailable as exc: 

419 await _record_message(db, work_item_id, session_id, message, auto, thread, turn, None) 

420 return await _refused( 

421 db, 

422 run_dirs, 

423 session_id=session_id, 

424 row=row, 

425 log=f"escalation selects harness {ESCALATION_TASK.harness!r}, which is not " 

426 f"available: {exc}\n", 

427 ) 

428 # A turn that resumes a thread runs on the runtime the thread started on 

429 # (`resumed-escalation-preserves-original-runtime`): the profile may have 

430 # changed since, and a resumed conversation on another model or harness is 

431 # not the session it claims to be. Only a new thread takes the current 

432 # profile. 

433 original = _thread_runtime(db, work_item_id, thread) if resume_session_id else None 

434 if original is not None: 

435 inv = inv._replace(**original) 

436 runtime = {k: getattr(inv, k) for k in _RUNTIME} 

437 # An automatic turn is its node running, so the node's time cap bounds it 

438 # (Kraft-8en38). A person's turn is a person talking: no cap. 

439 time_cap = None 

440 if auto: 

441 hit = db.read(lambda c: caps.for_turn(c, row, row["current_node_id"])) 

442 if hit is not None and hit.remaining_s <= 0: 

443 await _record_message(db, work_item_id, session_id, message, auto, thread, turn, None) 

444 return await _refused( 

445 db, run_dirs, session_id=session_id, row=row, log=f"not started: {hit.reason}\n" 

446 ) 

447 time_cap = caps.Deadline(caps.monotonic() + hit.remaining_s, hit) if hit else None 

448 await _record_message(db, work_item_id, session_id, message, auto, thread, turn, runtime) 

449 files = thread_files(work_item_id, thread) 

450 await _archive_last_turn(db, run_dirs, worktree, files) 

451 try: 

452 status = await _agent.run_agent_task( 

453 db, 

454 run_dirs, 

455 session_id=session_id, 

456 work_item_id=work_item_id, 

457 node_id=row["current_node_id"], 

458 hook_point="escalation", 

459 command=inv.command, 

460 harness=inv.harness, 

461 model=inv.model, 

462 effort=inv.effort, 

463 permission_mode=inv.permission_mode, 

464 deny_tools=inv.deny_tools, 

465 allowed_tools=inv.allowed_tools, 

466 steering_texts=inv.steering_texts, 

467 sandbox=sandbox, 

468 task_instruction=task_instruction, 

469 title=row["title"], 

470 repo_path=row["repo"], 

471 cwd=worktree, 

472 resume_session_id=resume_session_id, 

473 autocompact="auto", 

474 identify_as_worker=False, 

475 harness_id=ESCALATION_TASK.harness, 

476 repo_entry=launch.repo_entry, 

477 thread=thread, 

478 files=files, 

479 time_cap=time_cap, 

480 ) 

481 except _agent.LaunchRefused as exc: 

482 return await _refused(db, run_dirs, session_id=session_id, row=row, log=f"{exc}\n") 

483 

484 cli_session_id = _usage.READERS["claude-stream-json"].session_id( 

485 run_dirs.logs / f"{session_id}.log" 

486 ) 

487 if cli_session_id: 

488 await db.write(lambda c: store.set_escalation_session(c, work_item_id, cli_session_id)) 

489 return status

492def thread_files(work_item_id: str, thread: int) -> str: 

493 """The name every turn of one escalation thread writes its result file 

494 and session summary under (Kraft-s7c04.54, 

495 `a-resumed-escalation-turn-writes-where-it-remembers`). 

496 

497 Per thread, not per turn: a `--resume`d conversation remembers the 

498 literal paths an earlier turn wrote to, and a model that trusts that 

499 memory over the fresh instruction wrote to the earlier turn's file 

500 (b5afe84c), which `require_result_file` then recorded as `failed`. With 

501 one name per thread, the path it remembers is this turn's path. 

502 """ 

503 return f"escalation-{work_item_id}-{thread}"

506async def _archive_last_turn(db, run_dirs, worktree, files: str) -> None: 

507 """Give the thread's last turn its own names back before the next turn 

508 writes, so this turn starts with no result file to be mistaken for its 

509 own and the last turn's stays readable from its row. 

510 

511 The result family (`.json` and its `.exit`/`.cid` sidecars) moves to the 

512 last turn's session id and its row follows. The summary is copied, not 

513 moved: it sits in the worktree, where a repo that does not ignore 

514 `.engineering/` may track it. 

515 """ 

516 shared = result_path_for(run_dirs, files) 

517 summary = f"{_SESSIONS}/{files}.md" 

518 last = db.read( 

519 lambda c: c.execute( 

520 "SELECT id, session_summary_ref FROM worker_sessions WHERE result_path = ?", 

521 (str(shared),), 

522 ).fetchone() 

523 ) 

524 if last is not None: 

525 own = result_path_for(run_dirs, last["id"]) 

526 for suffix in (".json", ".exit", ".cid"): 

527 if shared.with_suffix(suffix).exists(): 

528 shared.with_suffix(suffix).replace(own.with_suffix(suffix)) 

529 ref = last["session_summary_ref"] 

530 if ref == summary and (worktree / summary).exists(): 

531 ref = f"{_SESSIONS}/{last['id']}.md" 

532 shutil.copyfile(worktree / summary, worktree / ref) 

533 await db.write( 

534 lambda c: c.execute( 

535 "UPDATE worker_sessions SET result_path = ?, session_summary_ref = ? WHERE id = ?", 

536 (str(own), ref, last["id"]), 

537 ) 

538 ) 

539 # A turn that never got a row (refused before launch) owns nothing here. 

540 for suffix in (".json", ".exit", ".cid"): 

541 shared.with_suffix(suffix).unlink(missing_ok=True)

544def _node_policy(row) -> _policy.InstancePolicy | None: 

545 """The policy of the node `row` stopped at, from its snapshot; None when 

546 there is no snapshot or no such node in it.""" 

547 snapshot = store.materialized_chain_of(row) 

548 nodes = snapshot.chain.nodes if snapshot is not None else () 

549 node = next((n for n in nodes if n.id == row["current_node_id"]), None) 

550 return executor.scope_policy(row, node) if node is not None else None
