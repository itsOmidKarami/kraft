1"""Per-repo and per-hook standards, injected through the system prompt. 

2 

3The context-injection boundary (`00_overview.md` glossary) bans `CLAUDE.md`, 

4`AGENTS.md` and any repo file as a context channel. That rule governs the 

5*channel*, not the existence of standards — this is the other half: authored, 

6Kraft-owned files under `$KRAFT_HOME/templates/steering/`, reaching the agent 

7through the per-invocation system prompt the adapter already builds. Kraft never 

8reads or writes a steering file inside a target repository, which is why a name 

9containing a path separator is rejected outright. 

10 

11`validate` and `read` are separate because the text must never be written back 

12into `registry.yaml` or `repos.yaml`: those dicts round-trip through 

13`GET /registry` and the Settings screens, so resolving into them would inline the 

14bodies into the operator's config and delete the names. 

15""" 

16 

17from __future__ import annotations 

18 

19from pathlib import Path 

20 

21#: Total injected budget, counting the header and separators, not just bodies. 

22#: An oversized system prompt degrades every launch and costs money on each. 

23MAX_BYTES = 8192 

24 

25#: What `adapters/agent.py` wraps the bodies in. Defined here so the budget and 

26#: the injection cannot drift apart; Task 5 asserts the two stay equal. 

27HEADING = "\n\n## Project standards\n\n" 

28_OVERHEAD = len(HEADING.encode()) 

29 

30#: Separator `adapters/agent.py` joins bodies with (`"\n\n".join(...)`). Bodies 

31#: contribute N-1 of these to the assembled block, which `_OVERHEAD` alone 

32#: does not count — `assembled_bytes` is what makes the MAX_BYTES comment above 

33#: (header *and separators*) true rather than aspirational. 

34_SEP = "\n\n" 

35 

36 

37class SteeringError(Exception): 

38 pass 

39 

40 

41def _path(steering_dir: Path, name: str, where: str) -> Path: 

42 if "/" in name or "\\" in name or name in ("", ".", "..") or name.startswith("."): 

43 raise SteeringError( 

44 f"{where}: steering name {name!r} must be a bare file name — Kraft " 

45 "never reads a steering file from outside its own templates directory" 

46 ) 

47 return Path(steering_dir) / f"{name}.md" 

48 

49 

50def path_for(steering_dir: Path, name: str, *, where: str) -> Path: 

51 """The file a steering name resolves to, or raise. 

52 

53 The public form of the same rule `validate` applies, so the Settings editor 

54 cannot accept a name the config loaders would then reject. 

55 """ 

56 return _path(steering_dir, name, where) 

57

59def assembled_bytes(bodies: list[str] | tuple[str, ...]) -> int: 

60 """Bytes of the actual injected block for these bodies: `HEADING` plus the 

61 bodies joined by `_SEP` — exactly what `adapters/agent.py` builds. Used both 

62 by `validate` (one config file's own names) and by `resolve_invocation` 

63 (the repo names and hook names it combines), so a union that blows the 

64 shared budget is caught wherever it is actually assembled. 

65 """ 

66 if not bodies: 

67 return 0 

68 return _OVERHEAD + len(_SEP.encode()) * (len(bodies) - 1) + sum(len(b.encode()) for b in bodies) 

69 

70 

71def validate(steering_dir: Path, names: list[str], *, where: str) -> None: 

72 """Every name resolves and the total fits the budget, or raise. 

73 

74 Runs at config load, against the *real* steering directory — not against 

75 whatever directory a candidate file was written into for validation. 

76 """ 

77 bodies: list[str] = [] 

78 for name in names: 

79 path = _path(steering_dir, name, where) 

80 if not path.is_file(): 

81 raise SteeringError(f"{where}: steering {name!r} not found at {path}") 

82 try: 

83 bodies.append(path.read_text()) 

84 except (OSError, ValueError) as exc: 

85 raise SteeringError(f"{where}: cannot read steering {name!r} at {path}: {exc}") from exc 

86 total = assembled_bytes(bodies) 

87 if names and total > MAX_BYTES: 

88 raise SteeringError( 

89 f"{where}: steering totals {total} bytes, over the {MAX_BYTES} byte budget" 

90 ) 

91

93def read(steering_dir: Path, names: list[str]) -> tuple[str, ...]: 

94 """Bodies for `names`, in order. Called at dispatch, after `validate`. 

95 

96 Assumes `validate` already passed for these names — it does no checking of 

97 its own. A file deleted, chmod'd or corrupted between validation (config 

98 load) and this call (dispatch) is reported as a `SteeringError` naming the 

99 file, so a caller catching `SteeringError` around dispatch sees it as the 

100 steering problem it is rather than as a bare `FileNotFoundError` (Kraft-fza). 

101 

102 The budget is a different story: this function does not measure it, but 

103 `resolve_invocation` re-checks the total over the concatenation it 

104 assembles, on every dispatch. A file that grew past the budget after 

105 `validate` ran is therefore still caught — at launch rather than at save. 

106 """ 

107 bodies: list[str] = [] 

108 for name in names: 

109 path = _path(steering_dir, name, "steering") 

110 try: 

111 bodies.append(path.read_text()) 

112 except (OSError, ValueError) as exc: 

113 raise SteeringError(f"steering: cannot read {name!r} at {path}: {exc}") from exc 

114 return tuple(bodies)
