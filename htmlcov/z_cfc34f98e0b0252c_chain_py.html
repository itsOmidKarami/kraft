51def complete_node(conn: sqlite3.Connection, work_item_id, node_id) -> None: 

52 # Idempotent: resume can re-enter an already-completed node (a reconciled 

53 # non-fix node, or a fix_loop node re-measured after a crash in the 

54 # complete_node -> enter_node window) and must not emit a second 

55 # node_completed. See Kraft-gbt / Kraft-126. Within one run fork only: a 

56 # retry reruns completed work (`retry-can-target-completed-work`), and the 

57 # rerun's completion is its own. 

58 done = conn.execute( 

59 "SELECT 1 FROM events WHERE work_item_id = ? AND type = 'node_completed' " 

60 "AND json_extract(payload, '$.node_id') = ? AND seq > (" 

61 " SELECT COALESCE(MAX(after_seq), 0) FROM run_forks WHERE work_item_id = ?" 

62 ") LIMIT 1", 

63 (work_item_id, node_id, work_item_id), 

64 ).fetchone() 

65 if done: 

66 return 

67 conn.execute("UPDATE work_items SET updated_at = ? WHERE id = ?", (_now(), work_item_id)) 

68 events.append(conn, work_item_id, "node_completed", {"node_id": node_id})

98 conn: sqlite3.Connection, 

99 work_item_id, 

100 attachments: list[dict], 

101 materialized_chain: str, 

102 *, 

103 seen: tuple[str | None, str], 

104) -> None: 

105 """Replace a not-yet-started item's attachments and the chain they trim 

106 (Kraft-s7c04.28), both at once so the trim and the documents that justify 

107 it cannot disagree. A compare-and-set on `seen`, the `(attachments, 

108 materialized_chain)` the caller built this write from (Kraft-5pw3f):

120 conn.execute(

124 )

136def revise_chain( 

137 conn: sqlite3.Connection, 

138 work_item_id, 

139 revised: str, 

140 payload: dict, 

141 *, 

142 seen: tuple[str | None, str | None], 

143) -> None: 

144 """Replace the chain this item runs with an approved revision 

145 (`MaterializedChain.to_json`), and record it as `chain_revised` -- one 

146 transaction, so a reader sees the old chain or the new one and its event, 

147 never half of it. 

148 

149 The chain it runs is the run fork's copy once a retry made one 

150 (`materialized_chain_of` reads `run_chain` first), and the intake snapshot 

151 before that; the revision replaces whichever that is. The intake snapshot 

152 of a forked item stays what intake froze, and so does every `run_forks` 

153 row: those are records of earlier runs, not the chain a reader runs. A 

154 later retry forks from the revised chain, so it keeps the revision. 

155 

156 A compare-and-set on `seen`, the `(materialized_chain, run_chain)` the 

157 revision was computed from, as `set_attachments` is: `ValueError` if a 

158 write changed either since (a retry's fork, an attachment change, another 

159 approval), which a revision of the old chain must not overwrite. 

160 """ 

161 row = conn.execute( 

162 "SELECT materialized_chain, run_chain FROM work_items WHERE id = ?", (work_item_id,) 

163 ).fetchone() 

164 if (row["materialized_chain"], row["run_chain"]) != seen: 

165 raise ValueError("the chain changed while this approval was made; approve it again") 

166 column = "run_chain" if row["run_chain"] is not None else "materialized_chain" 

167 conn.execute( 

168 f"UPDATE work_items SET {column} = ?, updated_at = ? WHERE id = ?", 

169 (revised, _now(), work_item_id), 

170 ) 

171 events.append(conn, work_item_id, "chain_revised", payload)

174def skip_node( 

175 conn: sqlite3.Connection, 

176 work_item_id: str, 

177 node_id: str, 

178 gate: str | None, 

179 note: str | None, 

180 *, 

181 session_ids: list[str] | None = None, 

182) -> None: 

183 """Advance past `node_id` (its own gate `gate`, if it has one and that is 

184 what is being bypassed) without running or approving it. 

185 

186 Sets the item back to `active` the same way `approve_gate` and 

187 `retry_after_cap` do — the caller spawns `executor.run` right after this 

188 write, same as every other door onto the chain. `retry_at` is cleared for 

189 the same reason `pause_work_item` clears it: a `ci_wait`-due item skipped 

190 out from under the poller must not wake back up under the old wait. 

191 

192 `session_ids` carries the node's own running sessions when the skip 

193 interrupts a live attempt — marked `paused` here, *before* the caller's 

194 `_terminate` signals them, so the adapter's death handler reads a session 

195 it expected to stop rather than one that just failed (`pause_work_item`'s 

196 ordering, same race). 

197 """ 

198 now = _now() 

199 if not write_status( 

200 conn, 

201 "UPDATE work_items SET status = 'active', retry_at = NULL, updated_at = ? WHERE id = ?", 

202 (now, work_item_id), 

203 ): 

204 return 

205 events.append( 

206 conn, work_item_id, "node_skipped", {"node_id": node_id, "gate": gate, "note": note} 

207 ) 

208 for sid in session_ids or []: 

209 conn.execute( 

210 "UPDATE worker_sessions SET status = 'paused', exited_at = ? WHERE id = ?", 

211 (now, sid), 

212 ) 

213 events.append(conn, work_item_id, "worker_session_paused", {"session_id": sid})

255def materialized_chain_of(row): 

256 """`row["materialized_chain"]` as the model that wrote it, or None. 

257 

258 None for every row the legacy intake path wrote — and for every row written 

259 before the column existed. A caller that needs V1 has to handle that, which 

260 is why this returns None rather than raising: the legacy path is still live 

261 until Task 5 retires it. 

262 

263 Imported inside the function: `kraft.templates.models` pulls in `kraft.policy` 

264 and `kraft.harness`, and `kraft.store` is imported by both the API and the 

265 worker, neither of which should pay for the template schema to read a row. 

266 """ 

267 from kraft.templates.models import MaterializedChain 

268 

269 keys = row.keys() 

270 # The run fork's copy first (`RunFork.materialized_chain`): after a retry 

271 # that is the chain the item runs, and the intake snapshot is its oldest 

272 # ancestor's. 

273 raw = (row["run_chain"] if "run_chain" in keys else None) or ( 

274 row["materialized_chain"] if "materialized_chain" in keys else None 

275 ) 

276 if not raw: 

277 return None 

278 chain = MaterializedChain.from_json(raw) 

279 # The item's own override, layered on read (Kraft-ab1bh): a reader that 

280 # selected only some columns gets the snapshot without it, which is only 

281 # safe because no such reader asks the snapshot for policy. 

282 override = policy_override_of(row) 

283 return dataclasses.replace(chain, item_policy=override) if override is not None else chain

286def policy_override_of(row): 

287 """`row["policy_override"]` as the `WorkItemPolicy` it holds, or None.""" 

288 from kraft.policy import FROZEN, WorkItemPolicy 

289 

290 raw = row["policy_override"] if "policy_override" in row.keys() else None 

291 if not raw: 

292 return None 

293 data = json.loads(raw) 

294 if isinstance(data, dict) and data.get(RETIRED_WAIT_TIMEOUT) is not None: 

295 data = _spread_retired_wait_timeout(row, data) 

296 # Read back leniently, like a snapshot (Kraft-9ct4q): an override stored 

297 # before tool names were checked must still load. The write doors refuse 

298 # a rule; a launch refuses one that is already stored. 

299 return WorkItemPolicy.model_validate(data, context=FROZEN)

302def _spread_retired_wait_timeout(row, data: dict) -> dict: 

303 """An item-wide `wait_timeout_minutes` stored before Ruling 196 meant 

304 "every wait this item runs": it becomes each wait task's own 

305 `total_time_cap_minutes`, never a cap on the whole item. A path that 

306 already sets one keeps it.""" 

307 from kraft.templates.models import MaterializedChain 

308 

309 deprecated( 

310 "%s: an item-wide %s is deprecated (Ruling 196); read as each wait task's " 

311 "total_time_cap_minutes", 

312 row["id"] if "id" in row.keys() else "a work item", 

313 RETIRED_WAIT_TIMEOUT, 

314 ) 

315 value = data[RETIRED_WAIT_TIMEOUT] 

316 data = {k: v for k, v in data.items() if k != RETIRED_WAIT_TIMEOUT} 

317 raw_chain = (row["run_chain"] if "run_chain" in row.keys() else None) or ( 

318 row["materialized_chain"] if "materialized_chain" in row.keys() else None 

319 ) 

320 if not raw_chain: 

321 return data 

322 paths = dict(data.get("paths") or {}) 

323 for node in MaterializedChain.from_json(raw_chain).chain.nodes: 

324 for task in node.tasks(): 

325 if getattr(task.task, "target", None) is not None and task.task.target.waits: 

326 layer = dict(paths.get(task.path) or {}) 

327 layer.setdefault("total_time_cap_minutes", value) 

328 paths[task.path] = layer 

329 return {**data, "paths": paths}

344def chain_node_ids(row) -> tuple[str, ...]: 

345 """Every node id of this item's chain, in order, whichever column holds it. 

346 

347 **The one reader for an ordering question**, and the reason it exists rather 

348 than a guard per caller. `chain_definition` is `"{}"` on a V1 row, so 

349 `json.loads(row["chain_definition"])["nodes"]` raises `KeyError` there -- 

350 and every caller that asks this question (`resume`, `retry`, `skip`, 

351 `ci_wait`, `rate_limit_retry`) asks it *after* it has already written the 

352 row's new status. The raise therefore leaves the item claimed `active` with 

353 no walk behind it: it looks running and is not, which is the worst shape a 

354 failure can take here. 

355 

356 A guard in each caller would be a larger diff than this and would leave the 

357 sixth caller, written next month, broken in exactly the same way. So the 

358 shape decision lives here, once. 

359 

360 `()` for a row with neither column populated -- a caller distinguishes "no 

361 nodes" from "node not found" through `node_index`'s `default`. 

362 

363 **What it guarantees, precisely: no `KeyError` for any shape Kraft writes.** 

364 Not "cannot raise", which earlier drafts of this docstring said and the code 

365 never did. A stored `materialized_chain` that a *later* build's 

366 `MaterializedChain` no longer validates raises `TemplateLibraryError` from 

367 `materialized_chain_of`, and a hand-edited `chain_definition` that is not a 

368 mapping of node dicts with `id` keys raises `JSONDecodeError`/`AttributeError` 

369 /`KeyError`. Neither is swallowed on purpose: turning a schema mismatch into 

370 `()` would make a chain silently look empty, and the callers that read this 

371 after a status write are wrapped in `stops.claimed_or_stopped`, which turns a 

372 propagating exception into a `needs_human` stop naming the item. Loud and 

373 stopped beats quiet and empty. 

374 """ 

375 v1 = materialized_chain_of(row) 

376 if v1 is not None: 

377 return tuple(node.id for node in v1.chain.nodes) 

378 raw = row["chain_definition"] if "chain_definition" in row.keys() else None 

379 legacy = json.loads(raw) if raw else {} 

380 return tuple(n["id"] for n in (legacy.get("nodes") or []))

383def node_index(row, node_id, *, default=None):

385

387 raises a `KeyError` for a row Kraft wrote -- see there for the two inputs

397def gate_node_index(row, gate: str, *, default=None): 

398 """Where `gate` sits in this item's ordered nodes, or `default`. 

399 

400 Beside `node_index` because the answer differs by shape and the difference 

401 is exactly one line: a V1 gate **is** a node, so its id is a node id 

402 (`gate-is-an-ordered-node`); a legacy gate is a `gate_after` string on the 

403 node in front of it. Callers ask "which node does this gate sit at" and

410 nodes = (json.loads(raw) if raw else {}).get("nodes") or [] 

411 return next((i for i, n in enumerate(nodes) if n.get("gate_after") == gate), default)

414def chain_view(row) -> dict: 

415 """The item's chain in the shape the API's `chain_definition` field and the 

416 SPA's `ChainDefinition` type speak, over either column. 

417 

418 A V1 row's `chain_definition` is `"{}"`, and the board reads 

419 `chain_definition.nodes` to draw its stage bar and to name the current node 

420 -- so returning the raw column left the board blank. This projects the 

421 frozen V1 snapshot into the same `{template_id, nodes: [...]}` envelope 

422 instead, so the board is *correct* for a V1 item rather than merely not 

423 crashing. 

424 

425 **A V1 gate node reports itself under `gate_after`, and that is deliberate.** 

426 Eleven SPA consumers ask one of three questions of that field -- "is this 

427 node a gate" (`StageGraph`, `Phone`, `MiniChain`, the gate counts), "what is 

428 this gate called" (`NotStarted`'s first-gate line, `Inspector/Config`), and 

429 "which node owns gate X" (`ItemCard.gateNodeId`, `ChainReviewDiff`'s 

430 `tailStart`, `PolicyPage`'s reject-loop detection). Leaving it null answered 

431 all three with "this chain has no gates", which is worse than the blank 

432 board it replaced: `gateNodeId` returned `None` for every V1 item, so the 

433 gate's document door disappeared. 

434 

435 Teaching eleven consumers a second rule is the mistake this whole class is 

436 about, and it is not needed: for a gate node the answer to all three 

437 questions *is* its own id. A V1 gate is where its own document lives 

438 (`gates.gate_artifact` reads the gate's own `artifact`), so "which node owns 

439 gate `spec_approval`" is `spec_approval`. `kind` carries the structural fact 

440 beside it for a reader that needs to know a gate is a node rather than a 

441 flag on one (`gate-is-an-ordered-node`), and `MiniChain` uses it. 

442 

443 The rest of the projection exists because the Config tab renders these 

444 fields and blanks read as "not configured": `fix_loop` is the *cap key* 

445 (`walk._loop_key`, `<node>.fix_loop`) rather than an authored loop name, 

446 which is what `PolicyPage` matches against `policy.yaml`'s `loops:`, and 

447 `auto_escalate` is whether the gate declares an `auto_review` task, which is 

448 what `Inspector/Config`'s toggle enables itself on. 

449 """ 

450 v1 = materialized_chain_of(row) 

451 if v1 is None: 

452 raw = row["chain_definition"] if "chain_definition" in row.keys() else None 

453 legacy = json.loads(raw) if raw else {} 

454 return {**legacy, "nodes": legacy.get("nodes") or []} 

455 return {"template_id": v1.chain.id, "nodes": [node_view(n) for n in v1.chain.nodes]}

458def node_view(node) -> dict: 

459 """One resolved V1 node in the shape the SPA's `ChainNode` speaks. See 

460 `chain_view` for why a gate reports itself under `gate_after`.""" 

461 from kraft.templates.models import GateNode 

462 

463 gate = node.node if isinstance(node.node, GateNode) else None 

464 # The node's *own* steps, not `node.tasks()`: the legacy `tasks`/`steps` 

465 # pair is what the node runs, and `tasks()` also yields the recovery pass, 

466 # the fix loop, the judge and a gate's reviewer -- which the Config tab 

467 # would then render as though they were ordinary work. 

468 steps = [[task.path for task in step.tasks] for step in node.steps] 

469 return { 

470 "id": node.id, 

471 "kind": node.node.kind.value, 

472 # Canonical task paths, which is what a V1 node has instead of a list of 

473 # hook-point names. Empty on a gate, which declares no execution shape. 

474 "tasks": [path for group in steps for path in group], 

475 "steps": steps or None, 

476 "gate_after": node.id if gate is not None else None, 

477 "reject_to": gate.reject_to if gate is not None else None, 

478 # The cap key `policy.yaml`'s `loops:` would name, not an authored loop 

479 # name -- V1 has none. Mirrors `walk._loop_key`; duplicated rather than 

480 # imported because `kraft.store` must not import the executor. 

481 "fix_loop": f"{node.id}.fix_loop" if node.fix_loop else None, 

482 "auto_escalate": node.auto_review is not None if gate is not None else None, 

483 "on_failure": [task.path for step in node.on_failure for task in step.tasks] or None, 

484 # The attachment kind that would drop this node at intake -- what the 

485 # intake preview strikes through (Kraft-ene04). Declared, not decided: 

486 # on an item's own chain the covered nodes are already gone. 

487 "covered_by": node.covered_by, 

488 }

497def effective_nodes(chain, node_overrides: dict) -> tuple: 

498 """A `MaterializedChain`'s ordered nodes with `node_overrides` folded on as 

499 a **typed** overlay: `tuple[ResolvedNode, ...]`. 

500 

501 A read-time view that writes nothing, so 

502 `materialized-chain-is-immutable-work-item-input` still holds -- that 

503 requirement freezes the stored snapshot, and this never touches it. The 

504 blind `{**node, **override}` dict merge `effective_chain` does cannot apply 

505 to a typed node at all: an unknown key would be accepted silently, and a 

506 key of the wrong type would reach the runtime as one. 

507 

508 **One key is supported: `auto_escalate: bool`** -- the persisted, publicly 

509 exposed override name (`db.py`'s `work_items.node_overrides`, 

510 `set_node_overrides`, `PATCH /work-items/{id}`), which keeps its spelling 

511 even though the chain field it governs is now `GateNode.auto_review`. 

512 The split: the **chain declares** the reviewing task, the **override 

513 permits or suppresses** it. So `false` clears `auto_review`, and `true` 

514 only confirms what the gate already declares -- **a gate declaring no 

515 `auto_review` cannot be switched on by an override, because an override 

516 names no task.** Broadening the supported set is Task 8's 

517 (`retry-overrides-are-policy-bounded`). 

518 

519 Every other key in a patch is ignored here rather than rejected: the same 

520 `node_overrides` blob also carries `model`/`effort`/`attempts`, which 

521 `dispatch_node` and `_policy.resolve_cap` read for themselves from the raw 

522 dict. This function answers one question -- what the *nodes* look like. 

523 """ 

524 from kraft.templates.models import GateNode 

525 

526 if not node_overrides: 

527 return chain.chain.nodes 

528 out = [] 

529 for resolved in chain.chain.nodes: 

530 patch = node_overrides.get(resolved.id) or {} 

531 if ( 

532 isinstance(resolved.node, GateNode) 

533 and patch.get("auto_escalate") is False 

534 and resolved.node.auto_review is not None 

535 ): 

536 # `model_copy`, not a re-validated `model_dump` round trip: the one 

537 # supported patch sets a field to `None`, which its own annotation 

538 # already permits, so there is nothing a revalidation could catch 

539 # that the type does not. A second supported key would change that. 

540 # Both halves: the model's field and the `ResolvedNode`'s resolved 

541 # slot, so neither a reader holding the typed node nor one holding 

542 # the resolved task can see a reviewer this item suppressed. 

543 out.append( 

544 dataclasses.replace( 

545 resolved, 

546 node=resolved.node.model_copy(update={"auto_review": None}), 

547 auto_review=None, 

548 ) 

549 ) 

550 else: 

551 out.append(resolved) 

552 return tuple(out)

555def effective_chain(chain_definition: dict, node_overrides: dict) -> dict: 

556 """`chain_definition` with `node_overrides` folded over each node. 

557 

558 The legacy-dict sibling of `effective_nodes`, and retired with the 

559 `chain_definition` column itself (Task 5). Its three remaining readers all 

560 want a node field V1's schema does not have -- the Config tab's rendered 

561 YAML, `effective_auto_escalate_stuck` and `effective_auto_escalate_delay_s` 

562 (`auto_escalate_stuck`/`auto_escalate_delay_s` are `policy.yaml` keys in V1, 

563 not node keys) -- so there is nothing here for a typed overlay to convert 

564 *to*. Anything asking what a V1 node looks like calls `effective_nodes`. 

565 

566 A read-time view, not a write: `chain_definition` stays exactly what 

567 the legacy loader produced at intake (or the last `chain_template` 

568 switch), and `node_overrides` is a separate, always-small delta layer on 

569 top of it. Anything that decides node behaviour at run time (auto-gate 

570 review, the Config tab, the effective-chain YAML) must call this instead 

571 of reading `chain_definition["nodes"]` directly, or it sees stale config 

572 on an overridden node. 

573 """ 

574 if not node_overrides: 

575 return chain_definition 

576 nodes = [ 

577 {**node, **node_overrides[node["id"]]} if node["id"] in node_overrides else node 

578 for node in chain_definition["nodes"] 

579 ] 

580 return {**chain_definition, "nodes": nodes}

583def _v1_node_override(row, key: str): 

584 """The per-node override value for `key` on the node the item is *currently* 

585 stopped on, for a **V1** row -- or None when nothing sets it. 

586 

587 `auto_escalate_stuck` and `auto_escalate_delay_s` are the two keys this 

588 answers, and in V1 neither is a *node* field at all: `ExecNode`/`GateNode` 

589 declare neither, both live in `policy.yaml`, and the only per-node layer left 

590 is the override dict itself. So this reads the override and stops -- there is 

591 no node value beneath it to fall through to. 

592 

593 That is a deliberate narrowing of the controller's "read `effective_nodes`": 

594 the overlay is typed over `ResolvedNode`, and a typed node has no field named 

595 `auto_escalate_stuck` for an overlay to produce. Reading the override layer 

596 directly is the whole of what "the per-node value" can mean here. 

597 

598 Without this, both `effective_*` helpers below returned `default` for every V1 

599 row -- `chain_definition` is `"{}"`, so their `"nodes" not in` guard fired 

600 first -- while `PATCH /work-items/{id}` validated the override, persisted it 

601 and returned 200. The policy-level value still worked; only the per-node 

602 override was silently dead. 

603 """ 

604 return node_overrides_of(row).get(row["current_node_id"], {}).get(key)

607def effective_auto_escalate_stuck(row, default: bool) -> bool: 

608 """Per-item node override -> chain node value -> `default` (mirrors 

609 `store/budget.py:effective_budget`'s "override beats node beats 

610 caller-supplied fallback" shape). 

611 

612 `default` is the caller's already-resolved policy value -- 

613 `policy.auto_escalate_stuck`, or a conservative `False` when policy 

614 failed to load entirely -- the same way `effective_budget` takes 

615 `policy_budget` already picked out of the whole `Policy` rather than 

616 the `Policy` itself. 

617 

618 Reads the node the item is *currently* stopped on 

619 (`row["current_node_id"]`): the field on any other node in the chain 

620 has no bearing on whether *this* stop escalates. 

621 

622 The guard is on the raw `chain_definition`, before `effective_chain` 

623 is ever called -- not on its result. `effective_chain` indexes 

624 `chain_definition["nodes"]` directly whenever `node_overrides` is 

625 non-empty (`store/chain.py:146-149`), so a work item seeded with a 

626 bare `"{}"` chain_definition (as several existing tests do) plus any 

627 node override at all raises `KeyError` *inside* `effective_chain`, 

628 before a `chain.get("nodes", [])` on its return value would ever run. 

629 Checking `"nodes" in chain_definition` first and returning `default` 

630 when it's missing -- the same "no chain reads as no node value" 

631 fallback, just placed where it actually has to sit -- skips the call 

632 that would crash instead of trying to catch its result afterwards. 

633 """ 

634 if row["materialized_chain"] if "materialized_chain" in row.keys() else None: 

635 value = _v1_node_override(row, "auto_escalate_stuck") 

636 return default if value is None else value 

637 chain_definition = json.loads(row["chain_definition"]) 

638 if "nodes" not in chain_definition: 

639 return default 

640 chain = effective_chain(chain_definition, node_overrides_of(row)) 

641 node = next((n for n in chain["nodes"] if n["id"] == row["current_node_id"]), None) 

642 value = node.get("auto_escalate_stuck") if node else None 

643 return default if value is None else value

646def effective_auto_escalate_delay_s(row, default: int) -> int: 

647 """Per-item node override -> chain node value -> `default` -- the same 

648 override chain `effective_auto_escalate_stuck` resolves, for the seconds 

649 a delayed `auto_escalate`/`auto_escalate_stuck` waits after its 

650 triggering event before firing (Kraft-vyk8). 

651 

652 `default` is the caller's already-resolved `policy.auto_escalate_delay_s`, 

653 or 0 when policy failed to load entirely -- 0 is also this feature's own 

654 "immediate, unchanged" default, so a missing policy degrades to exactly 

655 today's behaviour rather than a more conservative one (unlike 

656 `effective_auto_escalate_stuck`'s conservative `False` fallback, which 

657 exists because *unset* there means "don't auto-act at all"). 

658 

659 Reads the node the item is *currently* stopped on 

660 (`row["current_node_id"]`), same reasoning as 

661 `effective_auto_escalate_stuck`: while an item sits `awaiting_gate` or 

662 `needs_human`, that is still the node the gate/stop belongs to. 

663 """ 

664 if row["materialized_chain"] if "materialized_chain" in row.keys() else None: 

665 value = _v1_node_override(row, "auto_escalate_delay_s") 

666 return default if value is None else value 

667 chain_definition = json.loads(row["chain_definition"]) 

668 if "nodes" not in chain_definition: 

669 return default 

670 chain = effective_chain(chain_definition, node_overrides_of(row)) 

671 node = next((n for n in chain["nodes"] if n["id"] == row["current_node_id"]), None) 

672 value = node.get("auto_escalate_delay_s") if node else None 

673 return default if value is None else value

676def node_started(conn: sqlite3.Connection, work_item_id: str, node_id: str) -> bool: 

677 """Whether `node_id` has ever been entered for this item. 

678 

679 `node_started` is written by `enter_node`, in the same transaction that 

680 sets `current_node_id` -- so a node that is current, or that the chain has 

681 already moved past (including a fix-loop node re-entered more than once), 

682 both show up here. This is the "a node's config is locked once it starts" 

683 check for overrides and reset-to-template (UI v2 · 04, point 1/2). 

684 """ 

685 row = conn.execute( 

686 "SELECT 1 FROM events WHERE work_item_id = ? AND type = 'node_started' " 

687 "AND json_extract(payload, '$.node_id') = ? LIMIT 1", 

688 (work_item_id, node_id), 

689 ).fetchone() 

690 return row is not None

693def set_node_overrides(conn: sqlite3.Connection, work_item_id: str, patch: dict[str, dict]) -> dict: 

694 """Merge `patch` into the item's stored `node_overrides` and return the new 

695 whole object. 

696 

697 `patch == {}` (the top-level object itself, not a node inside it) clears 

698 every override -- "Reset to template" (point 2). A non-empty `patch` is 

699 per-node: `{node_id: {}}` drops just that node's overrides, `{node_id: 

700 {field: value}}` sets fields on it. The caller (the PATCH route) has 

701 already validated node ids, field names and `node_started` locking. 

702 """ 

703 row = conn.execute( 

704 "SELECT node_overrides FROM work_items WHERE id = ?", (work_item_id,) 

705 ).fetchone() 

706 current = json.loads(row["node_overrides"]) if row and row["node_overrides"] else {} 

707 if not patch: 

708 new = {} 

709 else: 

710 new = dict(current) 

711 for node_id, fields in patch.items(): 

712 if fields: 

713 new[node_id] = {**new.get(node_id, {}), **fields} 

714 else: 

715 new.pop(node_id, None) 

716 conn.execute( 

717 "UPDATE work_items SET node_overrides = ?, updated_at = ? WHERE id = ?", 

718 (json.dumps(new) if new else None, _now(), work_item_id), 

719 ) 

720 events.append(conn, work_item_id, "node_overrides_changed", {"overrides": new}) 

721 return new
