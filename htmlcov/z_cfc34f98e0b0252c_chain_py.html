43def complete_node(conn: sqlite3.Connection, work_item_id, node_id) -> None: 

44 # Idempotent: resume can re-enter an already-completed node (a reconciled 

45 # non-fix node, or a fix_loop node re-measured after a crash in the 

46 # complete_node -> enter_node window) and must not emit a second 

47 # node_completed. See Kraft-gbt / Kraft-126. 

48 done = conn.execute( 

49 "SELECT 1 FROM events WHERE work_item_id = ? AND type = 'node_completed' " 

50 "AND json_extract(payload, '$.node_id') = ? LIMIT 1", 

51 (work_item_id, node_id), 

52 ).fetchone() 

53 if done: 

54 return 

55 conn.execute("UPDATE work_items SET updated_at = ? WHERE id = ?", (_now(), work_item_id)) 

56 events.append(conn, work_item_id, "node_completed", {"node_id": node_id})

73def set_chain_template( 

74 conn: sqlite3.Connection, work_item_id, template_id: str, chain_definition: str 

75) -> None: 

76 """Switch a not-yet-started item onto a different chain template 

77 (Kraft-gwn6): the caller has already 404'd an unknown template and 409'd a

98def skip_node( 

99 conn: sqlite3.Connection, 

100 work_item_id: str, 

101 node_id: str, 

102 gate: str | None, 

103 note: str | None, 

104 *, 

105 session_ids: list[str] | None = None, 

106) -> None: 

107 """Advance past `node_id` (its own gate `gate`, if it has one and that is 

108 what is being bypassed) without running or approving it. 

109 

110 Sets the item back to `active` the same way `approve_gate` and 

111 `retry_after_cap` do — the caller spawns `executor.run` right after this 

112 write, same as every other door onto the chain. `retry_at` is cleared for 

113 the same reason `pause_work_item` clears it: a `ci_wait`-due item skipped 

114 out from under the poller must not wake back up under the old wait. 

115 

116 `session_ids` carries the node's own running sessions when the skip 

117 interrupts a live attempt — marked `paused` here, *before* the caller's 

118 `_terminate` signals them, so the adapter's death handler reads a session 

119 it expected to stop rather than one that just failed (`pause_work_item`'s 

120 ordering, same race). 

121 """ 

122 now = _now() 

123 conn.execute( 

124 "UPDATE work_items SET status = 'active', retry_at = NULL, updated_at = ? WHERE id = ?", 

125 (now, work_item_id), 

126 ) 

127 events.append( 

128 conn, work_item_id, "node_skipped", {"node_id": node_id, "gate": gate, "note": note} 

129 ) 

130 for sid in session_ids or []: 

131 conn.execute( 

132 "UPDATE worker_sessions SET status = 'paused', exited_at = ? WHERE id = ?", 

133 (now, sid), 

134 ) 

135 events.append(conn, work_item_id, "worker_session_paused", {"session_id": sid})

144def effective_chain(chain_definition: dict, node_overrides: dict) -> dict: 

145 """`chain_definition` with `node_overrides` folded over each node. 

146 

147 A read-time view, not a write: `chain_definition` stays exactly what 

148 `templates.materialize` produced at intake (or the last `chain_template` 

149 switch), and `node_overrides` is a separate, always-small delta layer on 

150 top of it. Anything that decides node behaviour at run time (auto-gate 

151 review, the Config tab, the effective-chain YAML) must call this instead 

152 of reading `chain_definition["nodes"]` directly, or it sees stale config 

153 on an overridden node. 

154 """ 

155 if not node_overrides: 

156 return chain_definition 

157 nodes = [ 

158 {**node, **node_overrides[node["id"]]} if node["id"] in node_overrides else node 

159 for node in chain_definition["nodes"] 

160 ] 

161 return {**chain_definition, "nodes": nodes}

164def effective_auto_escalate_stuck(row, default: bool) -> bool: 

165 """Per-item node override -> chain node value -> `default` (mirrors 

166 `store/budget.py:effective_budget`'s "override beats node beats 

167 caller-supplied fallback" shape). 

168 

169 `default` is the caller's already-resolved policy value -- 

170 `policy.auto_escalate_stuck`, or a conservative `False` when policy 

171 failed to load entirely -- the same way `effective_budget` takes 

172 `policy_budget` already picked out of the whole `Policy` rather than 

173 the `Policy` itself. 

174 

175 Reads the node the item is *currently* stopped on 

176 (`row["current_node_id"]`): the field on any other node in the chain 

177 has no bearing on whether *this* stop escalates. 

178 

179 The guard is on the raw `chain_definition`, before `effective_chain` 

180 is ever called -- not on its result. `effective_chain` indexes 

181 `chain_definition["nodes"]` directly whenever `node_overrides` is 

182 non-empty (`store/chain.py:146-149`), so a work item seeded with a 

183 bare `"{}"` chain_definition (as several existing tests do) plus any 

184 node override at all raises `KeyError` *inside* `effective_chain`, 

185 before a `chain.get("nodes", [])` on its return value would ever run. 

186 Checking `"nodes" in chain_definition` first and returning `default` 

187 when it's missing -- the same "no chain reads as no node value" 

188 fallback, just placed where it actually has to sit -- skips the call 

189 that would crash instead of trying to catch its result afterwards. 

190 """ 

191 chain_definition = json.loads(row["chain_definition"]) 

192 if "nodes" not in chain_definition: 

193 return default 

194 chain = effective_chain(chain_definition, node_overrides_of(row)) 

195 node = next((n for n in chain["nodes"] if n["id"] == row["current_node_id"]), None) 

196 value = node.get("auto_escalate_stuck") if node else None 

197 return default if value is None else value

200def effective_auto_escalate_delay_s(row, default: int) -> int: 

201 """Per-item node override -> chain node value -> `default` -- the same 

202 override chain `effective_auto_escalate_stuck` resolves, for the seconds 

203 a delayed `auto_escalate`/`auto_escalate_stuck` waits after its 

204 triggering event before firing (Kraft-vyk8). 

205 

206 `default` is the caller's already-resolved `policy.auto_escalate_delay_s`, 

207 or 0 when policy failed to load entirely -- 0 is also this feature's own 

208 "immediate, unchanged" default, so a missing policy degrades to exactly 

209 today's behaviour rather than a more conservative one (unlike 

210 `effective_auto_escalate_stuck`'s conservative `False` fallback, which 

211 exists because *unset* there means "don't auto-act at all"). 

212 

213 Reads the node the item is *currently* stopped on 

214 (`row["current_node_id"]`), same reasoning as 

215 `effective_auto_escalate_stuck`: while an item sits `awaiting_gate` or 

216 `needs_human`, that is still the node the gate/stop belongs to. 

217 """ 

218 chain_definition = json.loads(row["chain_definition"]) 

219 if "nodes" not in chain_definition: 

220 return default 

221 chain = effective_chain(chain_definition, node_overrides_of(row)) 

222 node = next((n for n in chain["nodes"] if n["id"] == row["current_node_id"]), None) 

223 value = node.get("auto_escalate_delay_s") if node else None 

224 return default if value is None else value

227def node_started(conn: sqlite3.Connection, work_item_id: str, node_id: str) -> bool: 

228 """Whether `node_id` has ever been entered for this item. 

229 

230 `node_started` is written by `enter_node`, in the same transaction that 

231 sets `current_node_id` -- so a node that is current, or that the chain has 

232 already moved past (including a fix-loop node re-entered more than once), 

233 both show up here. This is the "a node's config is locked once it starts" 

234 check for overrides and reset-to-template (UI v2 · 04, point 1/2). 

235 """ 

236 row = conn.execute( 

237 "SELECT 1 FROM events WHERE work_item_id = ? AND type = 'node_started' " 

238 "AND json_extract(payload, '$.node_id') = ? LIMIT 1", 

239 (work_item_id, node_id), 

240 ).fetchone() 

241 return row is not None

244def set_node_overrides(conn: sqlite3.Connection, work_item_id: str, patch: dict[str, dict]) -> dict: 

245 """Merge `patch` into the item's stored `node_overrides` and return the new 

246 whole object. 

247 

248 `patch == {}` (the top-level object itself, not a node inside it) clears 

249 every override -- "Reset to template" (point 2). A non-empty `patch` is 

250 per-node: `{node_id: {}}` drops just that node's overrides, `{node_id: 

251 {field: value}}` sets fields on it. The caller (the PATCH route) has 

252 already validated node ids, field names and `node_started` locking. 

253 """ 

254 row = conn.execute( 

255 "SELECT node_overrides FROM work_items WHERE id = ?", (work_item_id,) 

256 ).fetchone() 

257 current = json.loads(row["node_overrides"]) if row and row["node_overrides"] else {} 

258 if not patch: 

259 new = {} 

260 else: 

261 new = dict(current) 

262 for node_id, fields in patch.items(): 

263 if fields: 

264 new[node_id] = {**new.get(node_id, {}), **fields} 

265 else: 

266 new.pop(node_id, None) 

267 conn.execute( 

268 "UPDATE work_items SET node_overrides = ?, updated_at = ? WHERE id = ?", 

269 (json.dumps(new) if new else None, _now(), work_item_id), 

270 ) 

271 events.append(conn, work_item_id, "node_overrides_changed", {"overrides": new}) 

272 return new
