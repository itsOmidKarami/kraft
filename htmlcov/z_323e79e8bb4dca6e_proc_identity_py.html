18def pid_alive(pid: int) -> bool: 

19 """True only if pid names a live process (a zombie/defunct child is NOT alive).""" 

20 try: 

21 return psutil.Process(pid).status() != psutil.STATUS_ZOMBIE 

22 except psutil.NoSuchProcess: 

23 return False 

24 except psutil.Error: 

25 return False

28def identity_ok(pid: int, pid_start_time) -> bool: 

29 if pid_start_time is None or not pid_alive(pid): 

30 return False 

31 try: 

32 return same_start(psutil.Process(pid).create_time(), pid_start_time) 

33 except psutil.Error: 

34 return False

37def same_start(observed: float, recorded: float) -> bool: 

38 """Whether a process's start time is the one recorded at its launch: 

39 exactly, or one whole second off. Linux adds a process's ticks since boot 

40 to the boot time in whole seconds (`btime`), which moves by one when the 

41 wall clock is stepped across a second, a leap second included. A reused 

42 pid would have to start a second apart to the tick (10 ms) as well. A 

43 larger step reads as another process, which is never signalled.""" 

44 shift = observed - recorded 

45 return abs(shift - round(shift)) < 1e-6 and abs(round(shift)) <= 1

48async def kill_group(pgid: int, grace: float, reap: Callable[[], object] | None = None) -> None: 

49 """SIGTERM `pgid` (again at the first liveness check), then SIGKILL 

50 whatever is still there after `grace` seconds. `frontend/e2e/serve.py`'s 

51 `_terminate` is the model this copies. The server's sessions and the local 

52 worker's (`remote.local_jobs`) both stop through it. 

53 

54 The session's own child is already gone by the time this runs — this is 

55 for what it backgrounded into the same group and never waited on itself 

56 (a fixture server, `just dev`), which `start_new_session=True` put in the 

57 same process group as the child but this coroutine never touched 

58 (Kraft-1nye). `os.killpg` raises `ProcessLookupError` the moment the group 

59 has no members left, so the common case (nothing backgrounded) returns 

60 almost immediately rather than paying `grace`. 

61 

62 `reap` is the leader's own `Popen.poll`, called before every liveness 

63 check. On a pause or skip this runs from `run_task`'s `finally` while the 

64 leader is still alive, so the SIGTERM above kills it but nothing reaps it 

65 -- the poll loop that normally would was cancelled out from under it. On 

66 Linux an unreaped zombie still counts as a group member, so `killpg(pgid, 

67 0)` kept succeeding and every cancel paid the full `grace` (10s by 

68 default) before falling through to SIGKILL. macOS raises 

69 `PermissionError` for a zombie-only group instead, which is why this 

70 passed locally and failed pause/resume, SIGTERM shutdown and the e2e 

71 pause flow in CI (Kraft-rki). 

72 

73 The first liveness check is a second SIGTERM, not a probe: a member forked 

74 while the first one was being delivered (what `_wrap_with_exit_file`'s `sh` 

75 forks for the worker's command, on a cancel straight after launch) misses it 

76 and would otherwise outlive the whole `grace` -- a pause or skip then paid 

77 it, flaking under load. 

78 """ 

79 try: 

80 os.killpg(pgid, signal.SIGTERM) 

81 except (ProcessLookupError, PermissionError): 

82 return 

83 deadline = time.monotonic() + grace 

84 probe = signal.SIGTERM 

85 while time.monotonic() < deadline: 

86 await asyncio.sleep(0.2) 

87 if reap is not None: 

88 reap() 

89 try: 

90 os.killpg(pgid, probe) 

91 probe = 0 

92 except (ProcessLookupError, PermissionError): 

93 return 

94 try: 

95 os.killpg(pgid, signal.SIGKILL) 

96 except (ProcessLookupError, PermissionError): 

97 pass
