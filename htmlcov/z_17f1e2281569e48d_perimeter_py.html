21def _is_api_path(path: str) -> bool: 

22 """True for `/api` itself or anything under it — one predicate for the 

23 boundary every middleware and 404 path below has to agree on. `path == 

24 "/api"` matters even though nothing is mounted at the router root today: 

25 `.startswith("/api/")` alone lets the bare prefix slip through as if it 

26 were an ordinary SPA path.""" 

27 return path == "/api" or path.startswith("/api/")

30async def _spa_navigation(request: Request, call_next): 

31 # A browser deep-link / refresh on a client-side route (e.g. /work-items/<id>) 

32 # would otherwise reach the same catch-all any GET falls through to anyway 

33 # — this is a fast path, not the only path. Any top-level navigation 

34 # carries Sec-Fetch-Dest: document; hand those the SPA shell directly and 

35 # skip routing. Static assets are dest=script/style, XHR is dest=empty, so 

36 # only real navigations are caught. Excluded for /api/: that prefix is 

37 # unambiguously JSON, so a forged header there must not stand in for a 

38 # real 401/404/200. 

39 dist = getattr(request.app.state, "frontend_dist", None) 

40 if ( 

41 dist is not None 

42 and request.method == "GET" 

43 and request.headers.get("sec-fetch-dest") == "document" 

44 and not _is_api_path(request.url.path) 

45 ): 

46 # The browser caches by URL, so without no-store a refresh could answer 

47 # from a stale cached shell. `vary` says the same thing to caches that 

48 # honour it. 

49 return FileResponse( 

50 dist / "index.html", 

51 headers={"cache-control": "no-store", "vary": "sec-fetch-dest"}, 

52 ) 

53 return await call_next(request)

58#: and the login screen reads the bind address from it. 

59_PUBLIC_PATHS = {"/api/login", "/api/health"}

62def _is_static_asset(app: FastAPI, path: str) -> bool: 

63 """True for a file that ships with the SPA bundle. 

64 

65 The login page cannot render if its own JS and CSS come back 401, so the 

66 bundle is served before a session exists. These are build artifacts, not 

67 data — nothing about the running system leaks through them. 

68 """ 

69 dist = getattr(app.state, "frontend_dist", None) 

70 if dist is None or not path or path == "/": 

71 return False 

72 candidate = (dist / path.lstrip("/")).resolve() 

73 return dist.resolve() in candidate.parents and candidate.is_file()

76#: Hostnames that can only mean this machine. `urlsplit().hostname` strips the 

77#: brackets off an IPv6 literal, so "::1" covers "[::1]" as well. 

78_LOCAL_HOSTS = {"localhost", "127.0.0.1", "[::1]", "::1"}

81def _client_is_local(request: Request | WebSocket) -> bool: 

82 """True when the peer address is a loopback IP. 

83 

84 Fails closed: an absent peer, or one that is not an IP at all (starlette's 

85 TestClient reports the literal "testclient"), is not local. 

86 

87 uvicorn resolves the peer through `proxy_headers`, on by default with 

88 `forwarded_allow_ips="127.0.0.1"` — so a reverse proxy on this box surfaces 

89 the real client here, and a forged `X-Forwarded-For` from a remote peer is 

90 ignored because that peer is not trusted. Both directions only make this 

91 stricter than reading the socket would be. 

92 """ 

93 client = request.client 

94 if client is None: 

95 return False 

96 try: 

97 return ipaddress.ip_address(client.host).is_loopback 

98 except ValueError: 

99 return False

102def _requires_auth(app: FastAPI, request: Request | WebSocket) -> bool: 

103 """Auth is off on localhost and on for anything else (design 5e). 

104 

105 Three details this depends on: 

106 

107 * it is keyed to the address the process actually bound at startup, not the 

108 one saved in `access.yaml` — a bind change "takes effect on restart", so 

109 saving one must not lock the operator out of a server still on loopback; 

110 * the short-circuit needs the *peer* to be loopback too. `bound_host` is a 

111 claim, and a launch that widened the bind without going through 

112 `cli.admin._bind` (`uvicorn kraft.api:app --host 0.0.0.0`) does not update it — 

113 so a remote caller reaching a server that believes it is on loopback still 

114 has to log in; 

115 * with no password set there is nothing to authenticate *against*, so the 

116 gate stays open rather than bricking the API into a state where even 

117 setting the first password is refused. `__main__` will not start a 

118 non-loopback bind in that state, and the Access screen refuses to save one. 

119 `_perimeter` is what stops that open gate from being reachable remotely. 

120 """ 

121 local_bind = getattr(app.state, "bound_host", "127.0.0.1") in config_mod.LOOPBACK 

122 if local_bind and _client_is_local(request): 

123 return False 

124 return bool((getattr(app.state, "access", None) or {}).get("password_hash"))

127async def _authenticate(request: Request, call_next): 

128 app = request.app 

129 if ( 

130 not _requires_auth(app, request) 

131 or request.url.path in _PUBLIC_PATHS 

132 or _is_static_asset(app, request.url.path) 

133 or (request.method == "GET" and not _is_api_path(request.url.path)) 

134 ): 

135 return await call_next(request) 

136 # After the SPA-shell branch on purpose: that branch answers any non-/api 

137 # GET, so a bearer check ahead of it would leave that path reachable, and 

138 # one inside it would hand an MCP client HTML not JSON. 

139 bearer = request.headers.get("authorization", "") 

140 expected = getattr(app.state, "mcp_token", None) 

141 if expected and bearer.startswith("Bearer ") and hmac.compare_digest(bearer[7:], expected): 

142 return await call_next(request) 

143 token = request.cookies.get(auth_mod.COOKIE) 

144 if not token or not await app.state.db.write( 

145 lambda c, token=token: auth_mod.touch_session(c, token) 

146 ): 

147 return JSONResponse({"detail": "authentication required"}, status_code=401) 

148 return await call_next(request)

151async def _perimeter(request: Request, call_next): 

152 """Who may talk to this server at all, before any question of a session. 

153 

154 Declared *after* `_authenticate` on purpose. Starlette inserts each added 

155 middleware at the front of the stack, so the last one declared is the 

156 outermost and runs first — this has to be entered before the auth gate and 

157 before the SPA-shell branch, or a refused request gets answered by them. 

158 """ 

159 st = request.app.state 

160 peer = request.client.host if request.client else "an unknown peer" 

161 has_password = bool((getattr(st, "access", None) or {}).get("password_hash")) 

162 

163 # 1. The bind in access.yaml is a claim; the peer address is the fact. A 

164 # launch that widened the bind without going through `cli.admin._bind` would 

165 # otherwise serve the whole API to the LAN with the auth gate wide open, 

166 # because `_requires_auth` has no password to demand. 

167 if not _client_is_local(request) and not has_password: 

168 return JSONResponse( 

169 {"detail": f"this server is configured for a loopback bind; refusing {peer}"}, 

170 status_code=403, 

171 ) 

172 

173 # 2. DNS rebinding: a page on evil.com whose name flips to 127.0.0.1 is 

174 # same-origin with a local Kraft and can read every response, ids 

175 # included, then drive any route. A server that bound loopback may only 

176 # be addressed by a loopback name; a server bound off loopback may only 

177 # be addressed by a name the operator put in `allowed_hosts` (Kraft-cdy) 

178 # -- there is no bound address to compare against there, so without an 

179 # allowlist this fails closed rather than skipping the check. 

180 # 

181 # Keyed on `sec-fetch-site` because only a browser can be rebound, and 

182 # every other client — the CLI, MCP, httpx, curl — would otherwise need a 

183 # Host allowlist for no gain. Browsers older than the Fetch Metadata 

184 # rollout (pre-2020) do not send it and are not covered. 

185 if request.headers.get("sec-fetch-site"): 

186 hostname = urlsplit(f"//{request.headers.get('host', '')}").hostname 

187 bound_host = getattr(st, "bound_host", "127.0.0.1") 

188 if bound_host in config_mod.LOOPBACK: 

189 allowed = _LOCAL_HOSTS 

190 else: 

191 allowed = set((getattr(st, "access", None) or {}).get("allowed_hosts") or []) 

192 if hostname not in allowed: 

193 return JSONResponse( 

194 {"detail": f"unexpected Host for a server bound to {bound_host}"}, 

195 status_code=403, 

196 ) 

197 

198 # 3. Cross-site write. `_origin_ok` plus one clause, so that a LAN instance 

199 # serving its own SPA (Origin and Host both 192.168.1.5:8765) is not 

200 # locked out of its own board. Reads are left alone: this is about the 

201 # mutating routes that take no JSON body and so need no preflight. 

202 origin = request.headers.get("origin") 

203 if ( 

204 request.method not in ("GET", "HEAD") 

205 and origin 

206 and not _origin_ok(origin) 

207 and urlsplit(origin).netloc != request.headers.get("host") 

208 ): 

209 return JSONResponse({"detail": "cross-site request refused"}, status_code=403) 

210 

211 return await call_next(request)

214def _origin_ok(origin: str | None) -> bool: 

215 if not origin: 

216 return True # non-browser client 

217 return urlsplit(origin).hostname in _LOCAL_HOSTS
