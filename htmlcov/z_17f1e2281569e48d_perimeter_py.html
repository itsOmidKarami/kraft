25def _is_api_path(path: str) -> bool: 

26 """True for `/api` itself or anything under it — one predicate for the 

27 boundary every middleware and 404 path below has to agree on. `path == 

28 "/api"` matters even though nothing is mounted at the router root today: 

29 `.startswith("/api/")` alone lets the bare prefix slip through as if it 

30 were an ordinary SPA path.""" 

31 return path == "/api" or path.startswith("/api/")

42async def _spa_navigation(request: Request, call_next): 

43 # A browser deep-link / refresh on a client-side route (e.g. /work-items/<id>) 

44 # would otherwise reach the same catch-all any GET falls through to anyway 

45 # — this is a fast path, not the only path. Any top-level navigation 

46 # carries Sec-Fetch-Dest: document; hand those the SPA shell directly and 

47 # skip routing. Static assets are dest=script/style, XHR is dest=empty, so 

48 # only real navigations are caught. Excluded for /api/: that prefix is 

49 # unambiguously JSON, so a forged header there must not stand in for a 

50 # real 401/404/200. Nor for FastAPI's /docs and /redoc, which a browser 

51 # would otherwise never see. 

52 dist = getattr(request.app.state, "frontend_dist", None) 

53 if ( 

54 dist is not None 

55 and request.method == "GET" 

56 and request.headers.get("sec-fetch-dest") == "document" 

57 and not _is_api_path(request.url.path) 

58 and not _is_fastapi_docs_path(request.app, request.url.path) 

59 ): 

60 # The browser caches by URL, so without no-store a refresh could answer 

61 # from a stale cached shell. `vary` says the same thing to caches that 

62 # honour it. 

63 return spa_shell(request, dist, {"vary": "sec-fetch-dest"}) 

64 return await call_next(request)

67def spa_shell(request: Request, dist, headers: dict[str, str] | None = None): 

68 """The SPA's `index.html`, or a page saying to restart when the bundle on 

69 disk is no longer this server's. 

70 

71 `kraft admin update` replaces the installed package, `_bundled/web` 

72 included, under a server that keeps running until it is restarted. The new 

73 interface against the old API misreads it (a 1.4 server's board, drawn by 

74 2.0's interface, put every item under RUNNING), so the shell is refused. A 

75 hashed asset is still served as is: its name is its content, and a tab 

76 already open on this server's own interface keeps working.""" 

77 headers = {"cache-control": "no-store", **(headers or {})} 

78 running = getattr(request.app.state, "version", None) 

79 installed = update_mod.installed() 

80 if running and installed != running: 

81 return HTMLResponse(_restart_page(running, installed), status_code=503, headers=headers) 

82 return FileResponse(dist / "index.html", headers=headers)

133def _is_static_asset(app: FastAPI, path: str) -> bool: 

134 """True for a file that ships with the SPA bundle. 

135 

136 The login page cannot render if its own JS and CSS come back 401, so the 

137 bundle is served before a session exists. These are build artifacts, not 

138 data — nothing about the running system leaks through them. 

139 """ 

140 dist = getattr(app.state, "frontend_dist", None) 

141 if dist is None or not path or path == "/": 

142 return False 

143 candidate = (dist / path.lstrip("/")).resolve() 

144 return dist.resolve() in candidate.parents and candidate.is_file()

153def _client_is_local(request: Request | WebSocket) -> bool: 

154 """True when the peer address is a loopback IP. 

155 

156 Fails closed: an absent peer, or one that is not an IP at all (starlette's 

157 TestClient reports the literal "testclient"), is not local. 

158 

159 uvicorn resolves the peer through `proxy_headers`, on by default with 

160 `forwarded_allow_ips="127.0.0.1"` — so a reverse proxy on this box surfaces 

161 the real client here, and a forged `X-Forwarded-For` from a remote peer is 

162 ignored because that peer is not trusted. Both directions only make this 

163 stricter than reading the socket would be. 

164 """ 

165 client = request.client 

166 if client is None: 

167 return False 

168 try: 

169 return ipaddress.ip_address(client.host).is_loopback 

170 except ValueError: 

171 return False

174def _requires_auth(app: FastAPI, request: Request | WebSocket) -> bool: 

175 """Auth is off on localhost and on for anything else (design 5e). 

176 

177 Three details this depends on: 

178 

179 * it is keyed to the address the process actually bound at startup, not the 

180 one saved in `access.yaml` — a bind change "takes effect on restart", so 

181 saving one must not lock the operator out of a server still on loopback; 

182 * the short-circuit needs the *peer* to be loopback too. `bound_host` is a 

183 claim, and a launch that widened the bind without going through 

184 `cli.admin._bind` (`uvicorn kraft.api:app --host 0.0.0.0`) does not update it — 

185 so a remote caller reaching a server that believes it is on loopback still 

186 has to log in; 

187 * with no password set there is nothing to authenticate *against*, so the 

188 gate stays open rather than bricking the API into a state where even 

189 setting the first password is refused. `__main__` will not start a 

190 non-loopback bind in that state, and the Access screen refuses to save one. 

191 `_perimeter` is what stops that open gate from being reachable remotely. 

192 """ 

193 local_bind = getattr(app.state, "bound_host", "127.0.0.1") in config_mod.LOOPBACK 

194 if local_bind and _client_is_local(request): 

195 return False 

196 return bool((getattr(app.state, "access", None) or {}).get("password_hash"))

199async def _authenticate(request: Request, call_next): 

200 app = request.app 

201 if ( 

202 not _requires_auth(app, request) 

203 or request.url.path in _PUBLIC_PATHS 

204 or _is_static_asset(app, request.url.path) 

205 or (request.method == "GET" and not _is_api_path(request.url.path)) 

206 ): 

207 return await call_next(request) 

208 # After the SPA-shell branch on purpose: that branch answers any non-/api 

209 # GET, so a bearer check ahead of it would leave that path reachable, and 

210 # one inside it would hand an MCP client HTML not JSON. 

211 bearer = request.headers.get("authorization", "") 

212 presented = bearer[7:] if bearer.startswith("Bearer ") else "" 

213 expected = getattr(app.state, "mcp_token", None) 

214 if presented and expected and hmac.compare_digest(presented, expected): 

215 return await call_next(request) 

216 # The trigger token files work and nothing else; anywhere else it falls 

217 # through to the cookie check like no credential at all. 

218 trigger = getattr(app.state, "trigger_token", None) 

219 if ( 

220 presented 

221 and trigger 

222 and request.method == "POST" 

223 and request.url.path == "/api/triggers" 

224 and hmac.compare_digest(presented, trigger) 

225 ): 

226 return await call_next(request) 

227 token = request.cookies.get(auth_mod.COOKIE) 

228 if not token or not await app.state.db.write( 

229 lambda c, token=token: auth_mod.touch_session(c, token) 

230 ): 

231 return JSONResponse({"detail": "authentication required"}, status_code=401) 

232 return await call_next(request)

235async def _perimeter(request: Request, call_next): 

236 """Who may talk to this server at all, before any question of a session. 

237 

238 Declared *after* `_authenticate` on purpose. Starlette inserts each added 

239 middleware at the front of the stack, so the last one declared is the 

240 outermost and runs first — this has to be entered before the auth gate and 

241 before the SPA-shell branch, or a refused request gets answered by them. 

242 """ 

243 refused = _refusal(request, check_origin=request.method not in ("GET", "HEAD")) 

244 if refused is None: 

245 return await call_next(request) 

246 bound_host = getattr(request.app.state, "bound_host", "127.0.0.1") 

247 if refused == _unexpected_host(bound_host) and _is_page_navigation(request): 

248 return HTMLResponse(_unexpected_host_page(request, bound_host), status_code=403) 

249 return JSONResponse({"detail": refused}, status_code=403)

266def _raw_host(request: Request) -> str: 

267 """The `Host` header as the browser sent it, UTF-8 with replacement.""" 

268 for key, value in request.scope.get("headers", []):

274def _unexpected_host_page(request: Request, bound_host: str) -> str: 

275 """What a browser shows for a name this server does not answer to: the 

276 name it got and how to allow it, not a bare JSON line. A 1.4 server never 

277 checked the Host of a plain-http browser, so after an upgrade this is the 

278 first a LAN user hears of `allowed_hosts`. The Host is the requester's 

279 own text, so it is escaped, after decoding it as UTF-8: Starlette's 

280 latin-1 turns a non-ASCII name into mojibake. 

281 

282 The link back is the bound address, or 127.0.0.1 for a wildcard bind, 

283 which listens there too.""" 

284 raw = _raw_host(request) 

285 name = html.escape(config_mod.host_name(raw) or raw or "(none)") 

286 port = getattr(request.app.state, "bound_port", None) or request.url.port or 8765 

287 here = "127.0.0.1" if bound_host in ("0.0.0.0", "::") else bound_host 

288 local = html.escape(f"http://{config_mod.url_host(here)}:{port}/") 

289 if bound_host in config_mod.LOOPBACK: 

290 how = ( 

291 f"<p>It is bound to {html.escape(bound_host)}, so it answers only at " 

292 f"<code>localhost</code> or <code>127.0.0.1</code>. On this machine, open " 

293 f'<a href="{local}">{local}</a>. To reach it from another device, bind it to ' 

294 "the network on Settings &rsaquo; Access first.</p>" 

295 ) 

296 else: 

297 how = ( 

298 "<p>On a network bind, Kraft answers a browser only at a name listed in " 

299 "<code>allowed_hosts</code>. To allow this one, open Kraft on the machine it " 

300 f'runs on, at <a href="{local}">{local}</a>, and add <code>{name}</code> on ' 

301 "Settings &rsaquo; Access, where it applies at once. Or add it to " 

302 "<code>allowed_hosts</code> in <code>access.yaml</code> and restart Kraft. " 

303 "Enter the exact name: a wildcard is refused.</p>" 

304 ) 

305 return ( 

306 '<!doctype html><html lang="en"><head><meta charset="utf-8">' 

307 '<meta name="viewport" content="width=device-width, initial-scale=1">' 

308 "<title>Kraft: address not allowed</title>" 

309 "<style>body{font:16px/1.5 system-ui,sans-serif;max-width:36rem;margin:3rem auto;" 

310 "padding:0 16px;color:#1d1d1f;background:#fff}" 

311 "@media (prefers-color-scheme:dark){body{color:#e8e8ea;background:#161618}" 

312 "a{color:#8ab4ff}}code{font-size:.95em}</style></head><body>" 

313 f"<h1>Kraft does not answer to <code>{name}</code></h1>{how}</body></html>" 

314 )

317#: No other site may put Kraft in a frame. A loopback board has no login, so 

318#: a page that framed it invisibly could turn a click on its own content into 

319#: Resume or Approve (clickjacking). `X-Frame-Options` is the older spelling 

320#: of the same rule, for a browser that ignores `frame-ancestors`. 

321_FRAME_HEADERS = { 

322 "content-security-policy": "frame-ancestors 'self'", 

323 "x-frame-options": "SAMEORIGIN", 

324}

327async def _frame_guard(request: Request, call_next): 

328 """Every response, the SPA shell and the API alike, says only Kraft's own 

329 pages may frame it. Declared last, so it is the outermost middleware and 

330 a 401 or 403 from the ones inside carries the headers as well. A page 

331 with a policy of its own (`apidocs`) gets the directive added to it.""" 

332 response = await call_next(request) 

333 for name, value in _FRAME_HEADERS.items(): 

334 own = response.headers.get(name) 

335 if own is None: 

336 response.headers[name] = value 

337 elif name == "content-security-policy" and "frame-ancestors" not in own: 

338 response.headers[name] = f"{own}; {value}" 

339 return response

342def _refusal(request: Request | WebSocket, *, check_origin: bool) -> str | None: 

343 """Why `_perimeter` refuses this request, or None to let it through. 

344 

345 HTTP middleware does not run for websockets, so `/ws/events` asks this 

346 too, with `check_origin` always on: a browser opens a websocket to any 

347 site with no same-origin check, so its Origin is the only sign of which 

348 page asked. One set of rules for both, or the board's own live stream 

349 is refused on a LAN bind its HTTP calls pass. 

350 """ 

351 st = request.app.state 

352 peer = request.client.host if request.client else "an unknown peer" 

353 has_password = bool((getattr(st, "access", None) or {}).get("password_hash")) 

354 

355 # 1. The bind in access.yaml is a claim; the peer address is the fact. A 

356 # launch that widened the bind without going through `cli.admin._bind` would 

357 # otherwise serve the whole API to the LAN with the auth gate wide open, 

358 # because `_requires_auth` has no password to demand. 

359 if not _client_is_local(request) and not has_password: 

360 return f"this server is configured for a loopback bind; refusing {peer}" 

361 

362 # 2. DNS rebinding: a page on evil.com whose name flips to 127.0.0.1 is 

363 # same-origin with a local Kraft and can read every response, ids 

364 # included, then drive any route. A server that bound loopback may only 

365 # be addressed by a loopback name; a server bound off loopback may only 

366 # be addressed by a name the operator put in `allowed_hosts` (Kraft-cdy) 

367 # -- there is no bound address to compare against there, so without an 

368 # allowlist this fails closed rather than skipping the check. 

369 # 

370 # Only a browser can be rebound, but a browser cannot be told apart by 

371 # `sec-fetch-site` alone: Chromium sends no Sec-Fetch-* headers to a 

372 # plain-http origin on any name but localhost, which is exactly the 

373 # origin a rebinding page has. Nor by Origin or Referer: a same-origin 

374 # GET carries no Origin, and the page picks its own referrer policy. 

375 # So on a loopback bind the Host is checked on every request. Every 

376 # client of its own -- the CLI, MCP, the VS Code extension, the dev 

377 # proxy, a worker's in-process calls -- already dials 127.0.0.1 or 

378 # localhost. Off loopback, auth is on, and a rebound page has no 

379 # session cookie for its own name (and its login POST carries an 

380 # Origin), so the allowlist is held to requests that look like a 

381 # browser's, and the CLI, MCP and curl need no entry in it. A loopback 

382 # name is always allowed there too: a browser sends one only when it is 

383 # on this machine, so the board still opens at 127.0.0.1 on a LAN bind 

384 # without listing it, and a rebound page carries its own name instead. 

385 hostname = config_mod.host_name(request.headers.get("host", "")) 

386 bound_host = getattr(st, "bound_host", "127.0.0.1") 

387 if bound_host in config_mod.LOOPBACK: 

388 refused = hostname not in _LOCAL_HOSTS 

389 else: 

390 allowed = set((getattr(st, "access", None) or {}).get("allowed_hosts") or []) 

391 refused = _from_a_browser(request) and hostname not in allowed | _LOCAL_HOSTS 

392 if refused: 

393 return _unexpected_host(bound_host) 

394 

395 # 3. Cross-site write. `_origin_ok` plus one clause, so that a LAN instance 

396 # serving its own SPA (Origin and Host both 192.168.1.5:8765) is not 

397 # locked out of its own board. Reads are left alone: this is about the 

398 # mutating routes that take no JSON body and so need no preflight. 

399 origin = request.headers.get("origin") 

400 if ( 

401 check_origin 

402 and origin 

403 and not _origin_ok(origin) 

404 and urlsplit(origin).netloc != request.headers.get("host") 

405 ): 

406 return "cross-site request refused" 

407 

408 return None

411def _from_a_browser(request: Request | WebSocket) -> bool: 

412 """Whether a request carries anything a browser adds on its own: Fetch 

413 metadata, an Origin, a Referer, or the Accept of a page navigation. The 

414 CLI, MCP and curl send none of these.""" 

415 headers = request.headers 

416 return bool( 

417 headers.get("sec-fetch-site") 

418 or headers.get("origin") 

419 or headers.get("referer") 

420 or "text/html" in headers.get("accept", "") 

421 )

424def _origin_ok(origin: str | None) -> bool: 

425 if not origin: 

426 return True # non-browser client 

427 return urlsplit(origin).hostname in _LOCAL_HOSTS
