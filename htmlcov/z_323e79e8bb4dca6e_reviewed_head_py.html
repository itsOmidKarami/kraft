1"""The head a person's final review approved, and a merge that pushes nothing 

2else (`merge-refuses-a-head-final-review-did-not-see`). 

3 

4Approving the chain's first `chain_finalized` gate pins the item's branch tips 

5(`tips`): every checkout a later merge could push, on the `gate_approved` 

6event. So does skipping that gate, which a person does explicitly, on its 

7`node_skipped`. Only Kraft's own rewrite of that exact head, with no agent and 

8no conflict resolved -- a clean rebase, the root's pointer bump to its merged 

9members -- moves the pin (`repin`, `reviewed_head_repinned`). Right before it 

10asks the forge to merge, the merge handler reads the tip it would push against 

11the pin (`unreviewed`): anything else is code no person approved, and it does 

12not merge. 

13 

14No pin, no check: a chain with no final gate, or an item whose final gate was 

15approved before the pin existed. 

16"""

37def branch_tip(checkout, branch: str) -> str | None: 

38 """`branch`'s tip in `checkout`, or None when git cannot read it: 

39 `refs/heads/<branch>`, what the merge pushes, never `HEAD`, which an agent 

40 may have moved off the branch (or, in place, a person's).""" 

41 return _config.git_read( 

42 Path(checkout), "rev-parse", f"refs/heads/{branch}", expected_failure=True 

43 )

46def tips(db, work_item_id: str, row, worktree) -> Tips: 

47 """The item's branch tip in its own checkout and in every workspace member 

48 whose merge request has not merged: everything a later merge could push.""" 

49 checkouts = [str(worktree)] + [ 

50 r["path"] 

51 for r in db.read(lambda c: store.repos_for(c, work_item_id)) 

52 if r["role"] == "submodule" and r["state"] != "merged" 

53 ] 

54 branch = store.branch_for(row) 

55 return {path: branch_tip(path, branch) for path in dict.fromkeys(checkouts)}

58def unchanged(before: Mapping[str, str | None], after: Mapping[str, str | None]) -> bool: 

59 """Whether nothing a merge would push moved between two `tips` reads. A 

60 tip git could not read counts as moved: holding costs a review at most, 

61 going on could merge code nobody saw.""" 

62 return before == after and None not in after.values()

65def final_gate(row) -> str | None: 

66 """The id of `row`'s chain's first `chain_finalized` gate, or None.""" 

67 from kraft.templates.models import GateNode # store's reason: the schema is heavy 

68 

69 chain = store.materialized_chain_of(row) 

70 if chain is None: 

71 return None 

72 return next( 

73 ( 

74 n.id 

75 for n in chain.chain.nodes 

76 if isinstance(n.node, GateNode) and n.node.chain_finalized 

77 ), 

78 None, 

79 )

82def pin_for(db, row, node_id: str | None, worktree) -> Tips | None: 

83 """The tips approving or skipping `node_id` pins: the item's own when it 

84 is its chain's final gate, None for any other node.""" 

85 if node_id is None or node_id != final_gate(row): 

86 return None 

87 return tips(db, row["id"], row, worktree)

90def token(tips: Mapping[str, str | None]) -> str | None: 

91 """What a person's final-review decision names as the head they were shown: 

92 the sha itself for one checkout, so it can be recognised, and a 

93 `sha256:` digest over every checkout's realpath and tip for several (a 

94 workspace with unmerged members). None when a tip could not be read: there 

95 is nothing a decision could name. Clients treat it as opaque.""" 

96 if not tips or None in tips.values(): 

97 return None 

98 if len(tips) == 1: 

99 return next(iter(tips.values())) 

100 lines = sorted(f"{os.path.realpath(path)}\0{sha}" for path, sha in tips.items()) 

101 return "sha256:" + hashlib.sha256("\n".join(lines).encode()).hexdigest()

104def review_head(db, row, gate: str | None, worktree) -> str | None: 

105 """The token a decision on `gate` must carry: that of the tips it would pin 

106 (`pin_for`), so None unless `gate` is the chain's final one.""" 

107 pin = pin_for(db, row, gate, worktree) 

108 return token(pin) if pin is not None else None

111def started_over(c, work_item_id: str, session_id: str) -> Tips | None: 

112 """The tips an agent at the final gate recorded as it started (a reviewer, 

113 a reply, an escalation turn: `store.create_session(tips=...)`, on its 

114 `worker_session_created`). None when its session recorded none.""" 

115 row = c.execute( 

116 "SELECT payload FROM events WHERE work_item_id = ? AND type = ? " 

117 "AND json_extract(payload, '$.session_id') = ? ORDER BY seq DESC LIMIT 1", 

118 (work_item_id, SessionEvent.WORKER_SESSION_CREATED, session_id), 

119 ).fetchone() 

120 return json.loads(row["payload"]).get("tips") if row is not None else None

123def moved_under(db, work_item_id: str, session_ids, now: Mapping[str, str | None]) -> bool: 

124 """Whether an agent a person's approval or skip of the final gate just 

125 stopped (`session_ids`) may have moved what that decision would pin 

126 (`now`): its recorded start tips are not `now`, or it recorded none -- 

127 an agent started before they were recorded is not known to have left the 

128 branch alone, so it counts as having moved it.""" 

129 starts = db.read(lambda c: [started_over(c, work_item_id, sid) for sid in session_ids]) 

130 return any(start is None or not unchanged(start, now) for start in starts)

133def pinned(c, work_item_id: str) -> Tips | None: 

134 """The pin: the newest `tips` an approval or a skip of the final gate, or a 

135 repin, recorded. None when there is none.""" 

136 row = c.execute( 

137 "SELECT payload FROM events WHERE work_item_id = ? " 

138 f"AND type IN ({', '.join('?' * len(_PINNING))}) " 

139 "AND json_extract(payload, '$.tips') IS NOT NULL ORDER BY seq DESC LIMIT 1", 

140 (work_item_id, *_PINNING), 

141 ).fetchone() 

142 return json.loads(row["payload"])["tips"] if row is not None else None

145def same_checkout(a, b) -> bool: 

146 """Whether two checkout paths name one checkout: the pin is written by 

147 whoever approved, the merge reads by the folder the walk handed it.""" 

148 return os.path.realpath(str(a)) == os.path.realpath(str(b))

151def _entry(pin: Mapping[str, str | None], checkout) -> str | None: 

152 """The key `pin` holds `checkout` under, or None.""" 

153 return next((k for k in pin if same_checkout(k, checkout)), None)

156def unreviewed(db, work_item_id: str, checkout, branch: str) -> dict | None: 

157 """What a merge of `branch` in `checkout` would push that the final review 

158 did not approve: `{repo, reviewed, current}`, `reviewed` None for a 

159 checkout the pin does not hold. None when the tip is the pinned one, or 

160 nothing is pinned. A tip git cannot read is never the pinned one.""" 

161 pin = db.read(lambda c: pinned(c, work_item_id)) 

162 if pin is None: 

163 return None 

164 key = _entry(pin, checkout) 

165 reviewed = pin[key] if key is not None else None 

166 current = branch_tip(checkout, branch) 

167 if reviewed is not None and current == reviewed: 

168 return None 

169 return {"repo": str(checkout), "reviewed": reviewed, "current": current}

172def reviewed_tip(db, work_item_id: str, checkout) -> str | None: 

173 """The tip the final review approved for `checkout`, None when nothing is

175 pin = db.read(lambda c: pinned(c, work_item_id)) 

176 key = _entry(pin, checkout) if pin is not None else None 

177 return pin[key] if key is not None else None

180async def bring_home(checkout, branch: str, *, in_place: bool) -> bool: 

181 """Bring the commits the forge holds for `branch` into `checkout`'s own 

182 branch, so the final review that comes round again shows them. True when it 

183 moved the branch. 

184 

185 The forge's tip is fetched into `FETCH_HEAD` only (`--refmap=` keeps the 

186 shared `refs/remotes/origin/<branch>` alone, which `git.push` does not trust 

187 -- Kraft-m7ppj), and the branch is fast-forwarded only when all hold: 

188 `FETCH_HEAD` is a fast-forward of it, the checkout's HEAD is that branch (an 

189 agent may have left it elsewhere), and no tracked file is modified. Then 

190 `merge --ff-only` moves the working tree with the ref. Otherwise -- a 

191 force-push, local commits the forge lacks, a checkout in use -- nothing is 

192 touched and False says so. An in-place checkout is the person's own, and is 

193 never touched, nor even fetched into.""" 

194 if in_place: 

195 return False 

196 from kraft.adapters.forge import git # imports this module through `run` 

197 from kraft.adapters.forge.models import ForgeError 

198 from kraft.worker import sandbox 

199 

200 checkout, ref = Path(checkout), f"refs/heads/{branch}" 

201 try: 

202 await git.run_git( 

203 checkout, 

204 ["git", "fetch", "--refmap=", "origin", branch], 

205 env=sandbox.hardened_git_env(), 

206 ) 

207 except ForgeError: 

208 return False 

209 fetched = _config.git_read( 

210 checkout, "rev-parse", "--verify", "FETCH_HEAD", expected_failure=True 

211 ) 

212 before = branch_tip(checkout, branch) 

213 if ( 

214 not fetched 

215 or fetched == before 

216 or _config.git_read( 

217 checkout, "merge-base", "--is-ancestor", ref, fetched, expected_failure=True 

218 ) 

219 != "" 

220 or _config.git_read(checkout, "symbolic-ref", "-q", "HEAD", expected_failure=True) != ref 

221 or _config.git_read(checkout, "status", "--porcelain", "--untracked-files=no") != "" 

222 ): 

223 return False 

224 # Raised, not False: a planted .lfsconfig is for a person, not a quiet skip. 

225 git.assert_lfsconfig_unmoved(checkout, "bring the forge's commits in") 

226 try: 

227 await git.run_git(checkout, ["git", "merge", "--ff-only", fetched]) 

228 except ForgeError: 

229 return False 

230 return branch_tip(checkout, branch) == fetched

233async def repin( 

234 db, 

235 work_item_id: str, 

236 before: Mapping[str, str | None], 

237 after: Mapping[str, str | None], 

238 *, 

239 cause: str, 

240) -> None: 

241 """Kraft rewrote the item's branch itself, with no agent and no conflict 

242 resolved, between two reads of its tips (`cause`: `rebase`, or 

243 `pointer_bump`): each checkout whose tip `before` was the pinned one is 

244 pinned at its tip `after`. A checkout whose tip was not the pinned one 

245 keeps its pin, so a change no person saw still fails the merge.""" 

246 pin = db.read(lambda c: pinned(c, work_item_id)) 

247 if pin is None: 

248 return 

249 moved: dict[str, dict] = {} 

250 for path, was in before.items(): 

251 now = after.get(path) 

252 key = _entry(pin, path) 

253 if key is None or was is None or now is None or now == was or pin[key] != was: 

254 continue 

255 moved[key] = {"from": was, "to": now} 

256 if not moved: 

257 return 

258 new = {**pin, **{k: m["to"] for k, m in moved.items()}} 

259 await db.write( 

260 lambda c: events.append( 

261 c, 

262 work_item_id, 

263 ForgeEvent.REVIEWED_HEAD_REPINNED, 

264 {"tips": new, "moved": moved, "cause": cause}, 

265 ) 

266 )
