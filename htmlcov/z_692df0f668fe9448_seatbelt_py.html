32SANDBOX_EXEC = "/usr/bin/sandbox-exec" 

33#: Longest `end_session` waits for a sweep; the sweep itself stops after ten kill rounds. 

34SWEEP_SECONDS = 30 

35 

36#: Readable besides `host_view.system_dirs()`; none holds operator data. 

37SYSTEM_READS = ( 

38 "/private/etc", 

39 "/private/var/db/timezone", 

40 "/private/var/db/dyld", 

41 "/dev/fd", 

42 # With Xcode.app selected, `/usr/bin/git` is a shim through `xcodebuild`, which refuses 

43 # to run ("You have not agreed to the Xcode license") unless it reads this: the licence 

44 # version the machine agreed to, no operator data. 

45 "/Library/Preferences/com.apple.dt.Xcode.plist", 

46 # ...and whose plug-in scan, until Xcode's first launch has run, loads MobileDevice: 

47 # `/System/Library/PrivateFrameworks/MobileDevice.framework` is a link to this. 

48 "/Library/Apple/System/Library/PrivateFrameworks/MobileDevice.framework", 

49) 

50#: By name, never `/dev` whole: it holds the operator's other terminals, and 

51#: reading one is their keystrokes. 

52DEV_READS = ("/dev/null", "/dev/zero", "/dev/random", "/dev/urandom", "/dev/dtracehelper") 

53DEV_WRITES = ("/dev/null", "/dev/zero", "/dev/random", "/dev/urandom") 

54#: Never `(allow mach-lookup)` unfiltered: that hands the worker LaunchServices, 

55#: Apple Events, the pasteboard and the keychain. 

56MACH_SERVICES: tuple[str, ...] = ( 

57 "com.apple.system.opendirectoryd.libinfo", # getpwuid: python pwd, node os.userInfo 

58 # Trust evaluation, which Security.framework asks of trustd: a Go CLI's TLS (`gh`) and 

59 # `security verify-cert` fail with OSStatus -26276 without it. It holds no secrets. 

60 "com.apple.trustd.agent", 

61) 

62 

63#: Python's `multiprocessing.Lock` (a named POSIX semaphore) and 

64#: `shared_memory`; both fail without these, and the names are global to the 

65#: operator's processes, so only the prefixes the runtime itself makes. 

66IPC_SEM_PREFIX = "/mp-" 

67IPC_SHM_PREFIX = "/psm_" 

68 

69_NEVER = ( 

70 "com.apple.SecurityServer", 

71 "com.apple.securityd", 

72 "com.apple.coreservices.launchservicesd", 

73 "com.apple.lsd.", 

74 "com.apple.coreservices.appleevents", 

75 "com.apple.pasteboard",

80def quote(path: str | Path) -> str:

82 return '"' + str(path).replace("\\", "\\\\").replace('"', '\\"') + '"'

89def _refuse_unpinned_carve_outs(mounts: Sequence[Mount]) -> None: 

90 """`(allow file-write* (subpath A))` lets a directory between a read-write 

91 directory mount A and a read-only mount R inside it be renamed, carrying R 

92 away: only a mount's own directory is pinned or unwritable.""" 

93 dsts = {_real(m.dst) for m in mounts} 

94 for a in (_real(m.dst) for m in mounts if not m.ro and _real(m.dst).is_dir()): 

95 for r in (_real(m.dst) for m in mounts if m.ro): 

96 for between in r.parents: 

97 if between == a: 

98 break 

99 if between not in dsts and between.is_relative_to(a): 

100 raise SandboxRefused( 

101 f"kind: os on macOS cannot keep {r} read-only: {between}, between it and " 

102 f"the writable {a}, is not mounted, so the worker could rename it away" 

103 )

106def render_profile( 

107 *, 

108 mounts: Sequence[Mount], 

109 read_roots: Sequence[Path], 

110 read_literals: Sequence[Path], 

111 denied_reads: Sequence[Path], 

112 tmpdir: Path, 

113 marker: Path, 

114 relay: bool, 

115) -> str: 

116 for m in mounts: 

117 if m.src != m.dst: 

118 raise SandboxRefused(f"kind: os on macOS cannot show {m.src} at {m.dst}") 

119 # A link the worker can re-point (a lock, a reflog) must not choose what it may write. 

120 if not m.ro and os.path.islink(m.dst): 

121 raise SandboxRefused( 

122 f"kind: os on macOS will not give write access to {m.dst}: it is a symlink" 

123 ) 

124 _refuse_unpinned_carve_outs(mounts) 

125 sub = lambda p: f"(subpath {quote(_real(Path(p)))})" # noqa: E731 

126 lit = lambda p: f"(literal {quote(_real(Path(p)))})" # noqa: E731 

127 # `(allow file-write* (subpath D))` also lets D itself be renamed away (rename 

128 # is an unlink of the source), dropping the read-only carve-outs inside it. 

129 pin = lambda p: f"(deny file-write-unlink {lit(p)})" # noqa: E731 

130 out = [ 

131 "(version 1)", 

132 "(deny default)", 

133 "(allow process-exec*)", 

134 "(allow process-fork)", 

135 "(allow signal (target same-sandbox))", 

136 "(allow process-info* (target same-sandbox))", 

137 "(allow sysctl-read)", 

138 "(allow file-read-metadata)", 

139 ] 

140 out += [ 

141 f'(allow ipc-posix-sem (ipc-posix-name-prefix "{IPC_SEM_PREFIX}"))', 

142 f'(allow ipc-posix-shm (ipc-posix-name-prefix "{IPC_SHM_PREFIX}"))', 

143 ] 

144 if MACH_SERVICES: 

145 names = " ".join(f'(global-name "{s}")' for s in MACH_SERVICES) 

146 out.append(f"(allow mach-lookup {names})") 

147 # `/` itself: every dynamic loader reads it. 

148 out.append('(allow file-read* (literal "/"))') 

149 out += [f"(allow file-read* {sub(p)})" for p in (*host_view.system_dirs(), *SYSTEM_READS)] 

150 out += [f"(allow file-read* {lit(p)})" for p in DEV_READS] 

151 out += [f"(deny file-read* {sub(p)})" for p in host_view.system_excluded()] 

152 # The roots go after the deny: an Intel Homebrew Kraft lives under /usr/local. 

153 out += [f"(allow file-read* {sub(p)})" for p in read_roots] 

154 out += [f"(allow file-read* {lit(p)})" for p in read_literals] 

155 # Shallowest first, by the rendered path: a deeper mount must override the 

156 # tree it sits in. The worker's own tmpdir and devices follow the mounts so 

157 # a read-only mount holding them cannot deny them. 

158 for m in sorted(mounts, key=lambda m: len(_real(m.dst).parts)): 

159 dst = _real(m.dst) 

160 where = sub(dst) if dst.is_dir() else lit(dst) 

161 if m.ro: 

162 out += [f"(allow file-read* {where})", f"(deny file-write* {where})"] 

163 else: 

164 out.append(f"(allow file-read* file-write* {where})") 

165 if dst.is_dir(): 

166 out.append(pin(dst)) 

167 out += [f"(allow file-write* {lit(p)})" for p in DEV_WRITES] 

168 out += [ 

169 f"(allow file-read* file-write* {sub(tmpdir)})", 

170 pin(tmpdir), 

171 f"(allow network-bind network-inbound network-outbound (local unix-socket {sub(tmpdir)}))", 

172 f"(allow network-outbound (remote unix-socket {sub(tmpdir)}))", 

173 f"(allow file-read-data {lit(marker)})", 

174 ] 

175 # Writes too: a credential renamed inside a read-write tree could be read as the copy. 

176 out += [f"(deny file-read* file-write* {lit(p)})" for p in denied_reads] 

177 if relay: 

178 out.append( 

179 '(allow network-outbound (remote ip (string-append "localhost:" (param "RELAY_PORT"))))' 

180 ) 

181 return "\n".join(out) + "\n"

207def seatbelt_argv( 

208 cmd: list[str], 

209 cwd: str | Path, 

210 sandbox: Mapping, 

211 session: list[Mount], 

212 *, 

213 env: Mapping[str, str], 

214 passthrough: Iterable[str], 

215 home: Path | None, 

216 workdir: Path | None, 

217 sock_path: Path | None, 

218 started: Path | None, 

219 search_path: str, 

220 run_dirs: RunDirs, 

221 key: str, 

222 agent: str | None = None, 

223 mode: Literal["launch", "oneshot"] = "launch", 

224) -> list[str]: 

225 """The whole launch of `cmd` on macOS, as `bwrap.bwrap_argv` builds it on 

226 Linux: the launcher (`os_init.py launch`, on the host, which is also the 

227 relay), `sandbox-exec` with a profile written for this launch, and the init 

228 that runs the command inside. Nothing secret is on it: a secret crosses by 

229 name (`--keep`), through the launcher's environment.""" 

230 shown = bwrap.show_host( 

231 cmd, 

232 cwd, 

233 sandbox, 

234 passthrough=passthrough, 

235 workdir=workdir, 

236 search_path=search_path, 

237 agent=agent, 

238 ) 

239 files = session_files(run_dirs.base, key) 

240 roots = shown.roots 

241 mounts = [*session, *([Mount(home, home, False)] if home is not None else [])] 

242 text = render_profile( 

243 mounts=mounts, 

244 read_roots=[*shown.interp_roots, Path(os_init.__file__), *roots, *shown.ca, shim.HOST_DIR], 

245 # A symlink hop's own path: the real paths exist here, so nothing is recreated. 

246 read_literals=list( 

247 dict.fromkeys(link for link, _ in (*shown.first.links, *shown.resolved.links)) 

248 ), 

249 denied_reads=list( 

250 dict.fromkeys(h for root in roots for h in host_view.credential_shadows(root)) 

251 ), 

252 tmpdir=files.tmpdir, 

253 marker=files.marker, 

254 relay=sock_path is not None, 

255 ) 

256 for d in (files.profile.parent.parent, files.profile.parent): 

257 d.mkdir(parents=True, exist_ok=True, mode=0o700) 

258 d.chmod(0o700) 

259 write_whole(files.profile, text.encode(), 0o600) 

260 pins: dict[str, str] = {} 

261 _pin(pins, _HARDENED_GIT_CONFIG) 

262 literals = {"HOME": str(home or files.tmpdir / "home"), **pins, **env} 

263 init = [str(shown.interp), "-I", "-S", os_init.__file__] 

264 return [ 

265 *init, 

266 mode, 

267 *("--keep", ",".join(shown.keep)), 

268 *(a for name, value in literals.items() for a in ("--setenv", f"{name}={value}")), 

269 # os_init refuses a oneshot that records a start. 

270 *(("--started", str(started)) if started and mode == "launch" else ()), 

271 *("--marker", str(files.marker), "--control", str(files.control)), 

272 *("--tmpdir", str(files.tmpdir)), 

273 *(("--relay-sock", str(sock_path)) if sock_path is not None else ()), 

274 "--", 

275 SANDBOX_EXEC, 

276 *("-f", str(files.profile)), 

277 *init, 

278 "init", 

279 *("--chdir", str(shown.chdir)), 

280 "--", 

281 *cmd, 

282 ]

285def probe_darwin() -> str | None: 

286 """Why `kind: os` cannot run on this Mac, with its fix, or None.""" 

287 if (machine := platform.machine()) not in ("arm64", "x86_64"): 

288 return f"kind: os supports arm64 and x86_64 Macs, not {machine}" 

289 if os.geteuid() == 0: 

290 return "kind: os does not run while Kraft runs as root" 

291 if not os.access(SANDBOX_EXEC, os.X_OK): 

292 return f"{SANDBOX_EXEC} is missing (it ships with macOS): use kind: docker on this Mac" 

293 if problem := bwrap.missing_tools(): 

294 return problem 

295 # The real profile builder's output for an empty session: catches a profile 

296 # that does not parse as well as a Kraft that is sandboxed itself. 

297 with tempfile.TemporaryDirectory(prefix="kraft-probe-") as tmp: 

298 root = Path(os.path.realpath(tmp)) 

299 profile = root / "probe.sb" 

300 profile.write_text( 

301 render_profile( 

302 mounts=[], 

303 read_roots=[], 

304 read_literals=[], 

305 denied_reads=[], 

306 tmpdir=root / "tmp", 

307 marker=root / "marker", 

308 relay=True, 

309 ) 

310 ) 

311 trial = bwrap._run( 

312 [SANDBOX_EXEC, "-f", str(profile), "-D", "RELAY_PORT=1", "/usr/bin/true"] 

313 ) 

314 if not isinstance(trial, str) and trial.returncode == 0: 

315 return None 

316 # 65 is also how a profile that fails to parse exits; only this text means nesting. 

317 if ( 

318 not isinstance(trial, str) 

319 and trial.returncode == 65 

320 and "Operation not permitted" in trial.stderr 

321 ): 

322 return ( 

323 "Kraft is itself running inside a sandbox, so it cannot start one: " 

324 "run the Kraft server outside it" 

325 ) 

326 err = trial if isinstance(trial, str) else trial.stderr.strip() or f"exit {trial.returncode}" 

327 return f"sandbox-exec cannot start a sandbox here: {err}"

330async def end_session(files: SessionFiles, name: str, *, work_item_id: str | None) -> bool: 

331 """End every process a session left (`os_init.sweep`, in a thread, bounded), 

332 then remove its temporary directory and profile. False when processes 

333 remain or the sweep failed: the marker stays, so a later sweep still finds 

334 them, and so does the temporary directory, and the item (when known) gets a 

335 `sandbox_session_outlived`. `name` is the marker's, for the log and the event.""" 

336 if not files.marker.exists() and not files.tmpdir.exists(): 

337 files.profile.unlink(missing_ok=True) 

338 return True 

339 try: 

340 left = await asyncio.wait_for( 

341 asyncio.to_thread(os_init.sweep, str(files.marker), str(files.control)), SWEEP_SECONDS 

342 ) 

343 except (OSError, TimeoutError) as exc: 

344 error = str(exc) or "timed out" 

345 logger.warning("session %s: its processes could not be swept: %s", name, error) 

346 await _record_outlived(work_item_id, {"session": name, "pids": [], "error": error}) 

347 return False 

348 if left: 

349 logger.warning("session %s: processes survived the sweep, pids %s", name, left) 

350 await _record_outlived(work_item_id, {"session": name, "pids": left}) 

351 return False 

352 shutil.rmtree(files.tmpdir, ignore_errors=True) 

353 files.profile.unlink(missing_ok=True) 

354 return True

357async def _record_outlived(work_item_id: str | None, payload: dict) -> None: 

358 """The event, through the server's channel registry; none outside a server.""" 

359 if work_item_id is None or (registry := _channel.current()) is None: 

360 return 

361 try: 

362 await registry.record(work_item_id, SandboxEvent.SESSION_OUTLIVED, payload) 

363 # A swept marker's name may look like an item's and name none, or the server is 

364 # stopping: a close never fails on the event, and the warning above stands. 

365 except (sqlite3.Error, RuntimeError) as exc: 

366 name = payload["session"] 

367 logger.warning("session %s: its sandbox event was not recorded: %s", name, exc)
