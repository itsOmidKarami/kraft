1"""The local worker: the process that starts, reaps and holds this machine's local sessions. 

2 

3`kraft admin worker local` runs it, started by the server and found again after a server 

4restart. It listens on a unix socket under the run directory (file permissions are the 

5auth) and speaks the remote-worker framing to the server: `spawn` starts a session exactly 

6as the server built it, `hold` says whether a session is still held, `stop` ends one. 

7The server keeps supervising a running session by pid and by the files it writes; this 

8process starts it, reaps it and remembers it, so a restarted worker (or server) still finds it. 

9 

10Only leaf modules are imported, all at the top: `kraft admin update` replaces the package 

11under a running worker, and a lazy import would load the new code into this old process.""" 

12

32def install_id(kraft_version: str) -> str: 

33 """Names one installed kraft: its version, and a hash of where its package is and when it 

34 was last put there. Two installs of the same version string (a dev build reinstalled) 

35 get different ids, so an update always starts a worker of its own. 

36 

37 Each install has its own worker (socket, lock and pid file): a worker is never asked to 

38 run the code of a newer kraft than it loaded, and an old one gets no new sessions.""" 

39 init = Path(kraft.__file__).resolve() 

40 try: 

41 stamp = str(init.stat().st_mtime_ns) 

42 except OSError: 

43 stamp = "" 

44 digest = hashlib.sha1(f"{init.parent}\0{stamp}".encode()).hexdigest()[:8] 

45 return f"{re.sub(r'[^0-9A-Za-z._-]', '_', kraft_version)}-{digest}"

65def answering(run_base: Path) -> list[str]: 

66 """The installs whose worker under `run_base` answers on its socket right now.""" 

67 found = [] 

68 for pid_file in sorted(state_dir(run_base).glob("worker-*.pid")): 

69 install = pid_file.name.removeprefix("worker-").removesuffix(".pid") 

70 with socket.socket(socket.AF_UNIX) as probe: 

71 probe.settimeout(0.5) 

72 try: 

73 probe.connect(str(socket_path(run_base, install))) 

74 except OSError: 

75 continue 

76 found.append(install) 

77 return found

80class _Worker: 

81 def __init__(self, jobs: LocalJobs, kraft_version: str) -> None: 

82 self.jobs, self.kraft_version = jobs, kraft_version 

83 self.connections = 0 

84 self.writers: set[asyncio.StreamWriter] = set() 

85 self.wake = asyncio.Event() 

86 self._stopping: set[asyncio.Task] = set() 

87 

88 async def handle(self, reader: asyncio.StreamReader, writer: asyncio.StreamWriter) -> None: 

89 self.connections += 1 

90 self.writers.add(writer) 

91 t = StreamTransport(reader, writer) 

92 try: 

93 await protocol.machine_handshake(t, kraft_version=self.kraft_version) 

94 await protocol.pump(t, on_frame=lambda frame: self._frame(t, frame)) 

95 except (protocol.Closed, protocol.Refused, OSError, ValueError): 

96 pass # the server went away; sessions run on 

97 except Exception: 

98 log.exception("local worker connection failed") 

99 finally: 

100 self.connections -= 1 

101 self.writers.discard(writer) 

102 writer.close() 

103 self.wake.set() 

104 

105 async def _frame(self, t: protocol.Transport, frame: dict) -> None: 

106 kind, key = frame.get("t"), frame.get("id") 

107 if kind == "spawn": 

108 await t.send(self._spawn(key, frame)) 

109 elif kind == "hold": 

110 held = self.jobs.hold( 

111 key, pid=int(frame["pid"]), pid_start_time=frame.get("pid_start_time") 

112 ) 

113 await t.send({"t": "held", "id": key, "running": held}) 

114 elif kind == "stop": 

115 task = asyncio.create_task(self._stop(t, key)) 

116 self._stopping.add(task) 

117 task.add_done_callback(self._stopping.discard) 

118 else: 

119 log.warning("local worker: ignoring a %r frame", kind) 

120 

121 def _spawn(self, key: str, frame: dict) -> dict: 

122 try: 

123 spawned = self.jobs.spawn( 

124 key, 

125 list(frame["argv"]), 

126 cwd=frame["cwd"], 

127 env=dict(frame["env"]), 

128 log_path=frame["log_path"], 

129 ) 

130 except SpawnRefused as exc: 

131 return {"t": "refused", "id": key, "missing": exc.missing, "reason": str(exc)} 

132 return { 

133 "t": "spawned", 

134 "id": key, 

135 "pid": spawned.pid, 

136 "pid_start_time": spawned.pid_start_time, 

137 } 

138 

139 async def _stop(self, t: protocol.Transport, key: str) -> None: 

140 had = await self.jobs.stop(key) 

141 with contextlib.suppress(protocol.Closed): 

142 await t.send({"t": "stopped", "id": key, "had": had}) 

143 

144 async def run(self, stop: asyncio.Event, idle_exit_s: float, reap_s: float) -> None: 

145 loop = asyncio.get_running_loop() 

146 idle_since = loop.time() 

147 while not stop.is_set(): 

148 self.jobs.reap() 

149 now = loop.time() 

150 if self.connections or self.jobs.running(): 

151 idle_since = now 

152 elif now - idle_since >= idle_exit_s: 

153 return 

154 self.wake.clear() 

155 with contextlib.suppress(TimeoutError): 

156 await asyncio.wait_for(self.wake.wait(), reap_s) 

157 for task in list(self._stopping): 

158 task.cancel() 

159 for writer in list(self.writers): 

160 writer.close() # the server learns at once that this worker is gone

163async def serve( 

164 run_base: Path, 

165 *, 

166 kraft_version: str, 

167 stop: asyncio.Event, 

168 install: str | None = None, 

169 idle_exit_s: float = 5.0, 

170 reap_s: float = 0.2, 

171) -> None: 

172 """Run the worker of install `install` (this kraft's own, by default) until `stop` is set, 

173 or until no server is connected and no session runs for `idle_exit_s`. Returns at once when 

174 another worker of that install holds the lock.""" 

175 install = install or install_id(kraft_version) 

176 directory, path = state_dir(run_base), socket_path(run_base, install) 

177 sockets.private_dir(directory) 

178 if path.parent != directory: 

179 sockets.private_dir(path.parent) 

180 lock = sockets.take_lock(lock_path(run_base, install)) 

181 if lock is None: 

182 return 

183 try: 

184 with contextlib.suppress(FileNotFoundError): 

185 path.unlink() # left by a worker that died 

186 jobs = LocalJobs(directory) 

187 worker = _Worker(jobs, kraft_version) 

188 jobs.recover() 

189 server = await asyncio.start_unix_server(worker.handle, path=str(path), limit=FRAME_LIMIT) 

190 path.chmod(0o600) 

191 pid_file = pid_path(run_base, install) 

192 pid_file.write_text(str(os.getpid())) 

193 try: 

194 await worker.run(stop, idle_exit_s, reap_s) 

195 finally: 

196 server.close() 

197 with contextlib.suppress(FileNotFoundError): 

198 path.unlink() 

199 with contextlib.suppress(FileNotFoundError): 

200 pid_file.unlink() 

201 finally: 

202 os.close(lock) # last: until now no other worker could bind or remove the socket
