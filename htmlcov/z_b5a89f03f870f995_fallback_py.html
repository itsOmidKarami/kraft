1"""Launch fallback (Kraft-0a3h8): an agent task's launch that is rate-limited, 

2or whose harness is unavailable, moves to the next entry of its `fallback:` 

3list, and every skip or switch is logged as one `launch_fallback` event. 

4 

5`dispatch.dispatch_node` owns the loop; this module owns what it walks (the 

6candidates), what it remembers (a harness+model limited until its reset, read 

7back from `rate_limit_hit` events across every item) and what it writes. 

8 

9A task with no list never reaches any of this beyond `candidates` returning 

10its own launch alone (`fallback-is-opt-in`).""" 

11

25#: The note a fallback launch gets after an earlier candidate actually ran and 

26#: was rate-limited: its partial work may be in the worktree. 

27LIMITED_NOTE = ( 

28 "\n\nAn earlier attempt at this task on `{who}` was stopped by an API rate limit. " 

29 "The worktree may contain its partial work: check `git status` and `git diff`, " 

30 "and continue from it rather than starting over." 

31)

34def fallback_list(task: AgentTask, table=None) -> tuple[list, str | None]: 

35 """The entries that apply to `task`, and where they come from: its own 

36 `fallback:` when set (`[]` is none, overriding a profile's), else its 

37 agent profile's in `table` (a live `HarnessProfileTable`), else none.""" 

38 if task.fallback is not None: 

39 return list(task.fallback), "the task's list" 

40 profile = table.agent_profiles.get(task.profile) if table and task.profile else None 

41 if profile is not None and profile.fallback: 

42 return list(profile.fallback), f"profile {task.profile!r}'s list" 

43 return [], None

46def apply(task: AgentTask, entry) -> AgentTask: 

47 """`task` launched as `entry` says. The entry's route replaces the task's 

48 whole when either is a profile; two field routes merge per field. Its 

49 list is dropped, so lists never chain.""" 

50 update: dict = {"fallback": None} 

51 if entry.harness is not None: 

52 update["harness"] = entry.harness 

53 if entry.profile is not None: 

54 update |= {"profile": entry.profile, "model": None, "effort": None} 

55 elif entry.model is not None or entry.effort is not None: 

56 if task.profile is not None: 

57 update |= {"profile": None, "model": entry.model, "effort": entry.effort} 

58 else: 

59 update |= {k: getattr(entry, k) for k in ("model", "effort") if getattr(entry, k)} 

60 return task.model_copy(update=update)

63def candidates(task: AgentTask, table=None) -> list[AgentTask]: 

64 """The task's own launch, then one per entry of its list, in order.""" 

65 return [task] + [apply(task, e) for e in fallback_list(task, table)[0]]

68class Unavailable(Exception): 

69 """The candidate's executable is not on `PATH`.""" 

70 

71 

72def require_on_path(inv, harnesses) -> None: 

73 """The check `kraft admin doctor` makes: the harness's `executable`, else 

74 its provider's command, found on `PATH`. Not under a sandbox, where the 

75 executable lives in the container.""" 

76 exe = shlex.split(inv.command)[0] if inv.command else harnesses.valid[inv.harness].command[0] 

77 if shutil.which(exe) is None: 

78 raise Unavailable(f"`{exe}` is not on PATH")

81def known_limited(conn, harness: str, model: str | None) -> str | None: 

82 """`resets_at_iso` of the newest `rate_limit_hit` for `harness`+`model` 

83 across every work item, when that reset is still ahead; else None 

84 (`known-limited-candidate-is-skipped-until-reset`). An event written 

85 before the hit carried a harness never matches.""" 

86 row = conn.execute( 

87 "SELECT payload FROM events WHERE type = 'rate_limit_hit' " 

88 "AND json_extract(payload, '$.harness') = ? AND json_extract(payload, '$.model') IS ? " 

89 "ORDER BY seq DESC LIMIT 1", 

90 (harness, model), 

91 ).fetchone() 

92 if row is None: 

93 return None 

94 hit = json.loads(row["payload"]) 

95 resets_at = hit.get("resets_at") 

96 if not isinstance(resets_at, int | float) or resets_at <= time.time(): 

97 return None 

98 return hit.get("resets_at_iso")

101def describe(task: AgentTask, inv=None) -> dict: 

102 """What a candidate runs on, for the event: resolved when `inv` is in hand, 

103 else as the task spells it (a candidate that never resolved).""" 

104 return { 

105 "harness": task.harness, 

106 "profile": task.profile, 

107 "model": inv.model if inv is not None else task.model, 

108 "effort": inv.effort if inv is not None else task.effort, 

109 }

112def label(d: dict | None) -> str:

118class Attempts: 

119 """One task's walk down its candidates: the skip or switch waiting for 

120 the next candidate to say where it went, and the resets seen.""" 

121 

122 def __init__(self, db, work_item_id: str, node_id: str, task: str, overridden: bool): 

123 self.db, self.wid, self.node_id, self.task = db, work_item_id, node_id, task 

124 self.overridden = overridden 

125 self.pending: dict | None = None 

126 self.resets: list[str] = [] 

127 self.unavailable: list[str] = [] 

128 self.limited_run: dict | None = None 

129 

130 async def skip(self, frm: dict, reason: str, **extra) -> None: 

131 """`frm` is not launched (or its launch was limited): the event is 

132 written once the next candidate is known (`launching`/`flush`), or 

133 here, when `frm` is that next candidate and is skipped too.""" 

134 await self._write(frm, None) 

135 self.pending = {"from": frm, "reason": reason, **extra} 

136 if extra.get("resets_at_iso"): 

137 self.resets.append(extra["resets_at_iso"]) 

138 if reason == "unavailable": 

139 self.unavailable.append(f"{label(frm)}: {extra.get('detail')}") 

140 if reason == "rate_limit_hit": 

141 self.limited_run = frm 

142 

143 @property 

144 def note(self) -> str: 

145 """`LIMITED_NOTE`, only once a candidate actually ran and was limited.""" 

146 return LIMITED_NOTE.format(who=label(self.limited_run)) if self.limited_run else "" 

147 

148 async def launching(self, to: dict, session_id: str) -> None: 

149 await self._write(to, session_id) 

150 

151 async def flush(self) -> None: 

152 await self._write(None, None) 

153 

154 async def _write(self, to: dict | None, session_id: str | None) -> None: 

155 if self.pending is None: 

156 return 

157 payload = { 

158 "node_id": self.node_id, 

159 "task": self.task, 

160 **self.pending, 

161 "to": to, 

162 "session_id": session_id, 

163 "override_not_carried": self.overridden, 

164 } 

165 payload.setdefault("resets_at_iso", None) 

166 self.pending = None 

167 # `every-fallback-switch-is-logged`: the event and the server log. 

168 logger.info( 

169 "launch_fallback %s %s: %s -> %s (%s%s)", 

170 self.wid, 

171 self.task, 

172 label(payload["from"]), 

173 label(to), 

174 payload["reason"], 

175 f", until {payload['resets_at_iso']}" if payload["resets_at_iso"] else "", 

176 ) 

177 await self.db.write(lambda c: events.append(c, self.wid, "launch_fallback", payload))
