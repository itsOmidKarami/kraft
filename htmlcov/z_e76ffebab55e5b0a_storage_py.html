27async def _measured(request: Request, force: bool = False) -> storage.Usage: 

28 """The cached measurement; a walk when there is none yet or `force`.""" 

29 used = None if force else storage.usage(request.app.state)

33@api_router.get("/storage") 

34async def get_storage(request: Request, refresh: bool = False): 

35 """The disk the worktrees use, against the quota and limit when a limit is 

36 set, with one entry per measured worktree (largest first) and the folders 

37 that belong to no work item. It answers from the last measurement, and 

38 measures first when there is none yet or `refresh` is set.""" 

39 st = request.app.state 

40 used = await _measured(request, refresh) 

41 rows = st.db.read(lambda c: c.execute(_ITEM_COLUMNS).fetchall()) 

42 return storage.report(st.policy, used, {row["id"]: row for row in rows})

45def _preview_one(st, wid: str, row, size: int, members: list[Path]) -> dict: 

46 """One id of the preview, with the check `lifecycle._archive_one` makes 

47 (`_branches_to_keep`) and a count of what its worktree would lose. Blocking 

48 git calls: run it in a thread.""" 

49 why = storage.refusal(row) 

50 item = { 

51 "id": wid, 

52 "title": row["title"] if row is not None else None, 

53 "bytes": size, 

54 "archivable": why is None, 

55 "refusal": why, 

56 "uncommitted_files": None, 

57 "unpushed_commits": 0, 

58 "branch_kept": False, 

59 } 

60 if why is not None: 

61 return item 

62 worktree = st.run_dirs.worktrees / wid 

63 if worktree.is_dir(): 

64 # null when git cannot say, as when there is no worktree 

65 status = git_read(worktree, "status", "--porcelain") 

66 item["uncommitted_files"] = None if status is None else len(status.splitlines()) 

67 kept = lifecycle._branches_to_keep(Path(row["repo"]), store.branch_for(row), members) 

68 item["unpushed_commits"] = sum(kept.values()) 

69 item["branch_kept"] = bool(kept) 

70 return item

73@api_router.post("/storage/preview") 

74async def preview_storage_cleanup(body: StoragePreview, request: Request): 

75 """What archiving these items would do, without doing it: per item the 

76 size of its worktree, the uncommitted files that would be lost, the 

77 commits only its branch holds and whether the branch stays; and in total 

78 the bytes freed and the usage and state they would leave. An id that is 

79 unknown, not completed or abandoned, or already archived is answered with 

80 `archivable: false` and the reason, and frees nothing. An id given twice 

81 counts once. The bytes freed assume each worktree goes: an archive whose 

82 detached-HEAD rescue fails keeps it and frees nothing.""" 

83 st = request.app.state 

84 used = await _measured(request) 

85 ids = list(dict.fromkeys(body.ids)) 

86 rows = st.db.read( 

87 lambda c: { 

88 wid: c.execute("SELECT * FROM work_items WHERE id = ?", (wid,)).fetchone() 

89 for wid in ids 

90 } 

91 ) 

92 # On the loop, as `_archive_one` does: it reads the launch cache. 

93 members = { 

94 wid: lifecycle._connected_members(st, row) 

95 for wid, row in rows.items() 

96 if storage.reclaimable(row) 

97 } 

98 items = await asyncio.to_thread( 

99 lambda: [ 

100 _preview_one(st, wid, rows[wid], used.items.get(wid, 0), members.get(wid, [])) 

101 for wid in ids 

102 ] 

103 ) 

104 freed = sum(item["bytes"] for item in items if item["archivable"]) 

105 left, state = storage.after(st.policy, used, freed) 

106 return {"freed_bytes": freed, "used_after_bytes": left, "state_after": state, "items": items}
