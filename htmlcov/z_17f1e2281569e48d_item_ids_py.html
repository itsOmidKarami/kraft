1"""A work item named by the start of its id, made whole before routing. 

2 

3Imports only `store` and FastAPI: `kraft.api.__init__` registers the 

4middleware before the route modules load, and `api.deps` pulls in the 

5executor.""" 

6 

7from __future__ import annotations 

8 

9import re 

10 

11from fastapi import Request 

12from fastapi.responses import JSONResponse 

13 

14from kraft import store 

15 

16#: The one place a work item id arrives in a path, and the literal routes 

17#: that share its prefix. 

18_ITEM_PATH = re.compile(r"^/api/work-items/([^/]+)") 

19_LITERAL_SEGMENTS = frozenset({"bulk"}) 

20 

21 

22def ambiguous(given: str, ids: list[str]) -> str: 

23 """Why `given` names no one work item: it is the start of every one of `ids`.""" 

24 return f"{given!r} is the start of {len(ids)} work items: {', '.join(ids)}" 

25 

26 

27async def whole_work_item_id(request: Request, call_next): 

28 """Rewrite a unique start of a work item id in the path to the whole id, 

29 before routing. A handler goes on using its `wid` argument after its 

30 lookup (spend, drafts, viewed marks, the worktree path), and 

31 `deps.forbid_self_action` compares it with the caller's own item before 

32 any lookup, so resolving inside `deps._work_item_row` alone would act on 

33 the prefix. 

34 

35 Several matches answer 409 here. No match passes through untouched, so 

36 the handler's own 404 is unchanged. Registered innermost 

37 (`kraft.api.__init__`), after the perimeter and authentication admitted 

38 the caller, so a refused caller learns nothing about which ids exist.""" 

39 m = _ITEM_PATH.match(request.scope["path"]) 

40 db = getattr(request.app.state, "db", None) 

41 if m is None or m.group(1) in _LITERAL_SEGMENTS or db is None: 

42 return await call_next(request) 

43 given = m.group(1) 

44 ids = db.read(lambda c: store.match_work_item_ids(c, given)) 

45 if len(ids) > 1: 

46 return JSONResponse({"detail": ambiguous(given, ids)}, status_code=409) 

47 if len(ids) == 1 and ids[0] != given: 

48 whole, path = ids[0], request.scope["path"] 

49 request.scope["path"] = path[: m.start(1)] + whole + path[m.end(1) :] 

50 raw = request.scope.get("raw_path") 

51 if raw: 

52 # Best effort: Starlette routes and builds `request.url` from 

53 # `scope["path"]`, and nothing in Kraft reads `raw_path`; this only 

54 # keeps the two alike for whatever might. A client that 

55 # percent-encoded a character of the segment sent other bytes, and 

56 # its `raw_path` is left as sent. 

57 head = b"/api/work-items/" 

58 request.scope["raw_path"] = raw.replace(head + given.encode(), head + whole.encode(), 1) 

59 return await call_next(request)
