1"""Where a work item's files are. 

2 

3Everything an item produces, except its worktree, is written under 

4`run/items/<id>/`: 

5 

6 docs/<kind>.md the worker's documents (a sandbox mounts it read-write) 

7 history/<kind>.<n>.md earlier versions of them: Kraft's, never mounted 

8 filed/<kind>[-tag].md intake's attachment copies: Kraft's, never mounted 

9 sessions/<sid>/ one session's files: Kraft's, never mounted 

10 out/ the two files the worker writes, and all it is granted 

11 sandbox-home/ the item's sandboxes' `HOME` 

12 test-scopes/<node>.<hook>.changed-files 

13 the paths a targeted test command is told changed: 

14 Kraft's, mounted read-only into a sandbox 

15 carried.json what was copied into the worktree from the repo's carried 

16 files, and their sizes and mtimes then: Kraft's, never mounted 

17 

18Nothing is moved at upgrade: an older item's files stay in `run/logs/`, 

19`run/results/`, `run/attachments/`, `run/sandbox-home/` and its worktree's 

20`.engineering/`, and are read there. Each kind of file has one rule: 

21 

22- a session's files: written in its own folder, read where its row says; 

23- a gate document: written to `docs/`, read from there first and from the 

24 worktree's `.engineering/` second; 

25- a sandbox home: the old one while it exists, else the new one. 

26 

27A document is named by its path relative to where it lives, so the name says 

28which base it is under: `items/<id>/...` under the run folder, anything else 

29under the worktree the caller is working in. `locate` turns a name into the 

30folder a read starts at and the part of the name walked from there. 

31 

32`docs/`, a session's `out/` and the worktree are writable by a worker, so 

33nothing here trusts a name in them. Every read goes through 

34`worker.worktree_read`, which follows no symlink at any component below where 

35it starts and does not block on a FIFO; every write opens its folder through 

36`open_dir_no_symlinks` and its file with `O_CREAT | O_EXCL | O_NOFOLLOW`, so a 

37name a worker planted is neither followed nor replaced. A walk into an item's 

38folder starts at `item_dir`, opened as any path is: `run/items` and the item's 

39own folder are Kraft's, and may be links (a `run/items` moved to another 

40disk), and only what is below them is a worker's to plant a link in. 

41"""

124def filed_copy(run_dirs, wid: str, source: str | None) -> tuple[Path, str] | None: 

125 """The folder and file name of the attachment copy intake stored, found 

126 by the file name of the `source` its row records: in `filed/` first, then 

127 in `run/attachments/<id>/` for an item filed before the upgrade. Only the 

128 name is taken from the row, never its folder, so a `KRAFT_RUN_DIR` moved 

129 or spelled another way since filing still finds the copy, and a row can 

130 name nothing outside Kraft's own copies. None when neither folder holds 

131 a file by that name. The caller reads the name inside the folder with no 

132 symlink followed.""" 

133 if not source: 

134 return None 

135 name = PurePosixPath(source).name 

136 if name in ("", ".", ".."): 

137 return None 

138 for folder in filed_dirs(run_dirs, wid): 

139 if os.path.lexists(folder / name): 

140 return folder, name 

141 return None

144def stored_attachment(run_dirs, wid: str, attachment: dict, repo: str | Path) -> Path: 

145 """Where starting an item reads an attachment from. A `source` shaped 

146 like one of intake's copies (in a `filed/` folder, or in 

147 `attachments/<id>/`) is looked for first in this item's own copy 

148 folders, by its name (`filed_copy`): a `KRAFT_RUN_DIR` moved or spelled 

149 another way since filing leaves the row naming the old place, which may 

150 still hold an older copy. Otherwise, or when this item has no copy of 

151 that name, the recorded `source` when a file is there; else the copy of 

152 that name in the item's copy folders; else the recorded path as it is, 

153 for the caller to report missing. With no `source`, the path in the 

154 repo.""" 

155 source = attachment.get("source") 

156 if not source: 

157 return Path(repo) / attachment["path"] 

158 recorded = Path(source) 

159 if recorded.parent.name == "filed" or recorded.parent.parent.name == "attachments": 

160 found = filed_copy(run_dirs, wid, source) 

161 if found: 

162 return found[0] / found[1] 

163 if recorded.is_file(): 

164 return recorded 

165 found = filed_copy(run_dirs, wid, source) 

166 return found[0] / found[1] if found else recorded

187def locate(run_dirs, wid: str, name: str, *, worktree: Path | None = None) -> tuple[Path, str]: 

188 """`name` as `(root, rel)`: the folder a no-follow read starts at, and the 

189 part of the name walked from there. A name under this item's 

190 `items/<id>/` starts at `item_dir`, which is Kraft's and is opened as any 

191 path is (a link there, such as a `run/items` moved to another disk, is 

192 followed), and walks the rest; any other name starts at the worktree the 

193 caller is working in (the item's own when it gives none). A name under 

194 another item's folder is not this item's, so it resolves inside the 

195 worktree, where it reaches nothing of the run folder.""" 

196 parts = PurePosixPath(name).parts 

197 if len(parts) > 2 and parts[0] == ITEMS and parts[1] == wid: 

198 return item_dir(run_dirs, wid), PurePosixPath(*parts[2:]).as_posix() 

199 return (Path(worktree) if worktree is not None else run_dirs.worktrees / wid), name

401def remove_tree(path: str | os.PathLike[str]) -> None: 

402 """Remove the folder `path` and everything under it, best-effort, never 

403 following a link, never raising. 

404 

405 Only `path`'s own folders are resolved as a path, and they are Kraft's; 

406 below them the walk is `tree_walk.walk`'s, by descriptors: each name is 

407 opened, unlinked or removed relative to the descriptor of the folder it 

408 was listed in, with no link followed, so a link is removed rather than 

409 what it points at, and a worker swapping a folder for a link (to the 

410 daemon's home) while Kraft removes it redirects nothing. A link or a 

411 file at `path` itself is left, as `shutil.rmtree` refuses it. 

412 

413 A worker, or a tool it ran, can leave a folder its owner cannot write or 

414 list (`chmod 555`, `chmod 000`, a Go module cache), which a daemon that 

415 is not root cannot empty. Such a folder is set to mode 0700 through a 

416 descriptor of the folder itself, never a path (`_open_to_remove`, 

417 `_unlock`), and the removal retried. The mode is that fixed one, never 

418 the folder's own bits plus the owner's: should a chmod ever land on 

419 something else, it narrows access rather than widening it. A folder 

420 Kraft cannot fix that way (another user's, or a mode-0 one on a platform 

421 without `O_PATH`) is left, as is everything above it.""" 

422 path = os.path.abspath(os.fspath(path)) 

423 parent, name = os.path.split(path) 

424 if name in ("", ".", ".."): 

425 return 

426 try: 

427 top = os.open(parent, os.O_RDONLY | os.O_DIRECTORY | getattr(os, "O_CLOEXEC", 0)) 

428 except OSError: 

429 return

431 def leave(dir_fd: int, entry: str) -> None: 

432 # `top` is Kraft's and stays: a retry there keeps its own bits. 

433 _as_owner(dir_fd, lambda: os.rmdir(entry, dir_fd=dir_fd), kept=dir_fd == top) 

434 

435 try: 

436 tree_walk.walk( 

437 top, 

438 lambda dir_fd, entry, st: _remove_entry_at(dir_fd, entry, st, at_top=dir_fd == top), 

439 open_dir=_open_to_remove, 

440 leave=leave, 

441 names=[name], 

442 ) 

443 except OSError: 

444 pass 

445 finally: 

446 os.close(top)

449def _remove_entry_at(dir_fd: int, name: str, st: os.stat_result, *, at_top: bool) -> bool: 

450 """`remove_tree`'s step for one entry of the folder `dir_fd`: a folder is 

451 walked (True); anything else, a link included, is unlinked by its name 

452 there, never followed. At the top only a folder is removed.""" 

453 if stat.S_ISDIR(st.st_mode): 

454 return True 

455 if at_top: 

456 return False 

457 if _as_owner(dir_fd, lambda: os.unlink(name, dir_fd=dir_fd)): 

458 return False 

459 # Swapped for a folder since its `lstat`? Then walk that. 

460 try: 

461 return stat.S_ISDIR(os.stat(name, dir_fd=dir_fd, follow_symlinks=False).st_mode) 

462 except OSError: 

463 return False

466def _as_owner(dir_fd: int, op, *, kept: bool = False) -> bool: 

467 """Run `op`, an unlink or rmdir in the folder `dir_fd`, and when it is 

468 refused, set that folder to mode 0700 through `dir_fd` itself (`fchmod`) 

469 and run it once more; a folder that is `kept` (Kraft's, above the tree) 

470 gets its owner's rwx added to its own bits instead. True when the name 

471 is gone.""" 

472 try: 

473 op() 

474 return True 

475 except FileNotFoundError: 

476 return True 

477 except PermissionError: 

478 pass 

479 except OSError: 

480 return False 

481 try: 

482 mode = stat.S_IRWXU 

483 if kept: 

484 mode |= stat.S_IMODE(os.fstat(dir_fd).st_mode) 

485 os.fchmod(dir_fd, mode) 

486 op() 

487 except FileNotFoundError: 

488 return True 

489 except OSError: 

490 return False 

491 return True

494def _open_to_remove(dir_fd: int, name: str) -> int | None: 

495 """The folder `name` in `dir_fd`, opened with no link followed and set to 

496 mode 0700 when its owner lacks any of read, write and search, so what is 

497 in it can be listed and removed: a mode-0 folder through `_unlock` 

498 first. None, and the folder left, when Kraft cannot open it that way.""" 

499 fd = tree_walk.open_child(dir_fd, name) 

500 if fd is None: 

501 if not _unlock(dir_fd, name): 

502 return None 

503 fd = tree_walk.open_child(dir_fd, name) 

504 if fd is None: 

505 return None 

506 try: 

507 if stat.S_IMODE(os.fstat(fd).st_mode) & stat.S_IRWXU != stat.S_IRWXU: 

508 os.fchmod(fd, stat.S_IRWXU) 

509 except OSError: 

510 pass # another user's: what can be removed in it still is 

511 return fd

514def _unlock(dir_fd: int, name: str) -> bool: 

515 """Set the folder `name` in `dir_fd`, one Kraft cannot open (mode 0), to 

516 mode 0700 without following a link and without resolving any name a 

517 worker could swap between a check and the change. True when it did. 

518 

519 The folder is opened with `O_PATH | O_NOFOLLOW | O_DIRECTORY`, which needs 

520 no permission on it, checked to be a folder, and changed through 

521 `/proc/self/fd/<n>`, which names the object that descriptor holds rather 

522 than any path (glibc's `lchmod` does the same). Linux only: elsewhere, 

523 and where `/proc` is not mounted, the folder is left. A `fchmodat` by 

524 name, even with `AT_SYMLINK_NOFOLLOW`, would land on whatever the worker 

525 put at the name after Kraft looked, a hard link to a daemon's file 

526 included.""" 

527 o_path = getattr(os, "O_PATH", 0) 

528 if not o_path: 

529 return False 

530 try: 

531 pfd = os.open(name, o_path | os.O_NOFOLLOW | os.O_DIRECTORY | os.O_CLOEXEC, dir_fd=dir_fd) 

532 except OSError: 

533 return False 

534 try: 

535 if not stat.S_ISDIR(os.fstat(pfd).st_mode): 

536 return False 

537 os.chmod(f"/proc/self/fd/{pfd}", stat.S_IRWXU) 

538 return True 

539 except OSError: 

540 return False 

541 finally: 

542 os.close(pfd)

553def _copy_fd(src: int, dest_base: Path, dest_name: str) -> None: 

554 """Copy the open regular file `src` to a new file `dest_base/dest_name`, 

555 its folders made as needed. Every folder below `dest_base` is opened 

556 without following a link (`open_dir_no_symlinks`), and the file itself 

557 with `O_CREAT | O_EXCL | O_NOFOLLOW` relative to its folder: a name 

558 already there, a planted symlink included, raises `FileExistsError` and 

559 is left as it is. A copy that fails partway is removed, through the same 

560 folder handle, so no half version is left behind.""" 

561 parts = PurePosixPath(dest_name).parts 

562 if not parts: 

563 raise OSError(errno.EINVAL, f"refusing destination {dest_name!r}") 

564 leaf = _component(parts[-1], "file name") 

565 dir_fd = open_dir_no_symlinks(dest_base, parts[:-1], create=True) 

566 try: 

567 out = os.open( 

568 leaf, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o644, dir_fd=dir_fd 

569 ) 

570 try: 

571 with os.fdopen(out, "wb") as w, os.fdopen(os.dup(src), "rb") as r: 

572 while chunk := r.read(65536): 

573 w.write(chunk) 

574 except BaseException: 

575 with contextlib.suppress(OSError): 

576 os.unlink(leaf, dir_fd=dir_fd) 

577 raise 

578 finally: 

579 os.close(dir_fd)

582def copy_in(source_base: Path, source_name: str, dest_base: Path, dest_name: str) -> bool: 

583 """Copy a file a worker could have written to a new name: the source read 

584 with no symlink followed at any component below `source_base` 

585 (`open_no_symlinks`, which takes only a regular file), the destination 

586 created as `_copy_fd` does. False, leaving the destination as it was, 

587 when the source is not a regular file reached that way, is larger than 

588 `DOC_MAX_BYTES` (a worker's file is not copied at any size), the 

589 destination name is taken (a name a worker planted, or an earlier copy), 

590 a folder on the way to it is a link, or the write fails.""" 

591 try: 

592 src = open_no_symlinks(source_base, source_name) 

593 except (OSError, ValueError, IndexError): 

594 return False 

595 try: 

596 if os.fstat(src).st_size > DOC_MAX_BYTES: 

597 logger.warning( 

598 "not copying %s: larger than %d bytes", source_base / source_name, DOC_MAX_BYTES 

599 ) 

600 return False 

601 _copy_fd(src, dest_base, dest_name) 

602 except (OSError, ValueError): 

603 return False 

604 finally: 

605 os.close(src) 

606 return True

609def _next_version(history: Path, kind: str) -> int: 

610 """One past the highest `<kind>.<n>.md` in `history`, 1 when there is none.""" 

611 pattern = re.compile(rf"{re.escape(kind)}\.([0-9]+)\.md") 

612 try: 

613 names = os.listdir(history) 

614 except FileNotFoundError: 

615 names = [] 

616 taken = [int(m[1]) for n in names if (m := pattern.fullmatch(n))] 

617 return max(taken, default=0) + 1

620def keep_version(run_dirs, wid: str, kind: str, *, worktree: Path | None = None) -> Path | None: 

621 """Before a task producing `kind` is dispatched again, copy the item's 

622 current `kind` document to `history/<kind>.<n>.md`, `n` one past the 

623 highest there, and return that path. The document stays where it is, so 

624 the agent revises it in place and a failed re-run still leaves one. 

625 

626 An old document, in the worktree's `.engineering/`, is also seeded into 

627 `docs/`, so the revise note and the "write to" path name the same file; 

628 a name a worker already planted there is left alone. None, writing 

629 nothing, when there is no document. 

630 

631 A convenience, never a reason to stop a dispatch: a document larger than 

632 `DOC_MAX_BYTES` (a worker's to write at any size, and copied again on 

633 every retry) is not kept, and any failure to read or write is logged and 

634 leaves None.""" 

635 try: 

636 return _keep_version(run_dirs, wid, kind, worktree) 

637 except (OSError, ValueError) as exc: 

638 logger.warning("could not keep a version of %s's %s: %s", wid, kind, exc) 

639 return None

642def _keep_version(run_dirs, wid: str, kind: str, worktree: Path | None) -> Path | None: 

643 name = document(run_dirs, wid, kind, worktree=worktree) 

644 if name is None: 

645 return None 

646 try: 

647 src = open_no_symlinks(*locate(run_dirs, wid, name, worktree=worktree)) 

648 except (OSError, ValueError, IndexError): 

649 return None # gone, or swapped for a link, since `document` looked 

650 try: 

651 size = os.fstat(src).st_size 

652 if size > DOC_MAX_BYTES: 

653 logger.warning( 

654 "not keeping a version of %s's %s: %d bytes, over %d", 

655 wid, 

656 kind, 

657 size, 

658 DOC_MAX_BYTES, 

659 ) 

660 return None 

661 root = item_dir(run_dirs, wid) 

662 root.mkdir(parents=True, exist_ok=True) # Kraft's own; may be reached by a link 

663 history = history_dir(run_dirs, wid) 

664 kept = None 

665 n = _next_version(history, kind) 

666 for candidate in range(n, n + _VERSION_TRIES): 

667 try: 

668 os.lseek(src, 0, os.SEEK_SET) 

669 _copy_fd(src, root, f"history/{kind}.{candidate}.md") 

670 except FileExistsError: 

671 continue # another dispatch took this number first 

672 kept = history / f"{kind}.{candidate}.md" 

673 break 

674 if kept is not None and name != document_name(wid, kind): 

675 os.lseek(src, 0, os.SEEK_SET) 

676 # A convenience: whatever already holds the name, a planted link 

677 # included, is left as it is and the revise note names it anyway. 

678 with contextlib.suppress(OSError): 

679 _copy_fd(src, root, f"docs/{_component(kind, 'kind')}.md") 

680 return kept 

681 finally: 

682 os.close(src)
