1"""Agent review of a gate before a human sees it 

2(docs/superpowers/specs/2026-09-10-auto-escalate-gates-design.md, Kraft-zr3s). 

3 

4Dispatched through the same `run_agent_task` engine every chain node uses, and 

5deliberately *as a worker*: `identify_as_worker=True` means 

6`client.context._forbid_self_action` already refuses to let this session approve or 

7reject its own item. The agent therefore cannot clear the gate however it is 

8prompted -- it must report a verdict, and `kraft.executor.gates.review_gates` 

9applies it. 

10That is the opposite choice from `escalate.py`, which is not a worker precisely 

11so that a human's escalation agent *can* act. 

12""" 

13 

14from __future__ import annotations 

15 

16import json 

17import uuid 

18 

19from kraft import caps, events, executor 

20from kraft import harness as _harness 

21from kraft.adapters import agent as _agent 

22from kraft.adapters import subprocess as _subprocess 

23from kraft.adapters.profiles import HarnessUnavailable, harness_table 

24from kraft.executor.fallback import fallback_list 

25from kraft.templates.models import AgentTask, ResolvedNode 

26from kraft.worker import steering as _steering 

27 

28#: The only strings a verdict may be. Anything else -- a typo, a sentence, a 

29#: missing key -- is `undecided`, which leaves the gate pending for a human. 

30VERDICTS = frozenset({"approve", "reject", "fixed", "undecided"}) 

31 

32#: Statuses that mean the session did not finish thinking. A verdict written 

33#: beside one of these is not a judgement, whatever it says. 

34_UNTRUSTWORTHY = frozenset({"failed", "needs_context", "rate_limited"}) 

35 

36_PROMPT = ( 

37 "This is a gate review. A Kraft work item has reached the gate " 

38 "{gate!r} on node {node_id!r}, and is waiting for a decision. You are " 

39 "deciding whether it needs a person.\n" 

40 "\n" 

41 "Title: {title}\n" 

42 "{description_line}" 

43 "{artifact_line}" 

44 "\n" 

45 "You are a Kraft worker for this launch, so you cannot approve or reject " 

46 "the gate yourself -- and must not try. Report your decision by writing " 

47 "`verdict` into your result file, with your reasoning in `concerns`:\n" 

48 "\n" 

49 ' "verdict": "approve" -- the gate should clear; you are confident.\n' 

50 ' "verdict": "reject" -- send it back to be redone. `concerns` becomes ' 

51 "the steering note for the agent that redoes it, so write it as an " 

52 "instruction, not a complaint. Required: a rejection with no reasoning is " 

53 "discarded.\n" 

54 ' "verdict": "fixed" -- you repaired something in this worktree. You ' 

55 "may edit and commit here, but you may not then approve your own edit: the " 

56 "node re-runs and is measured again. `concerns` says what you changed. Note " 

57 "what that costs: the whole node runs again and is reviewed again, so this " 

58 "is the right call for a defect you would otherwise have rejected for, and " 

59 "the wrong one for a nit you could have let through.\n" 

60 ' "verdict": "undecided" -- anything else. A person looks at it, with your ' 

61 "`concerns` in front of them.\n" 

62 "\n" 

63 "Prefer `undecided` whenever you are unsure. A gate wrongly cleared is " 

64 "expensive and a gate wrongly held costs a human one glance." 

65)

68def _artifact_line(rel: str | None) -> str: 

69 if not rel: 

70 return "There is no artifact document for this gate; read the worktree.\n" 

71 return f"The document this gate is about: {rel} (relative to this worktree)\n"

74def _profile_fallback_refusal(task: AgentTask, gate: str) -> str | None: 

75 """None, or the operator-facing reason `task` must not launch (Kraft-t4y8g): 

76 its `profile:` names an agent profile that itself carries a `fallback:` 

77 list. 

78 

79 `GateNode._no_handler` already refuses a gate whose `auto_review` task 

80 declares its own `fallback:`, at template load -- "a gate review never 

81 falls back" (this module launches its reviewer once; the loop lives in 

82 `dispatch.dispatch_node` alone, and stays there: making `gate_review` run 

83 it is a feature, not this fix). A profile's list is invisible to that 

84 check: `profile:` is read live from `harnesses.yaml` 

85 (`kraft.adapters.profiles`, Kraft-ps1ao), not at template load, so a 

86 profile with no fallback list today can grow one tomorrow without the 

87 chain that selects it ever being re-validated. That is also why the 

88 refusal lives here, at the gate's actual launch, rather than in template 

89 lint or `kraft admin doctor`: both would read `harnesses.yaml` once and 

90 could go stale the moment an operator edits it, while a launch-time check 

91 is the one place it is always current. 

92 

93 Reuses `executor.fallback.fallback_list`, the same function 

94 `dispatch.dispatch_node` walks -- it already knows the precedence (the 

95 task's own list wins, else its profile's) and already names which one it 

96 found. 

97 """ 

98 if task.profile is None: 

99 return None 

100 try: 

101 table, _path = harness_table(_harness.load(None)) 

102 except HarnessUnavailable: 

103 # The launch a moment later hits the identical problem and reports 

104 # it; nothing here needs to say it twice. 

105 return None 

106 entries, source = fallback_list(task, table) 

107 if not entries or source == "the task's list": 

108 # The task's own list is `GateNode._no_handler`'s refusal, already 

109 # caught before this chain could ever be materialized. 

110 return None 

111 return ( 

112 f"gate {gate!r}'s auto_review task selects profile {task.profile!r}, whose " 

113 f"{source} is not empty; a gate review launches its reviewer once and never " 

114 "runs the fallback loop, so it is refused rather than silently ignored" 

115 )

118async def review( 

119 db, 

120 run_dirs, 

121 *, 

122 work_item_id: str, 

123 gate: str, 

124 node: ResolvedNode, 

125 launch: executor.LaunchContext, 

126) -> tuple[str, str]: 

127 """Dispatch one gate review. Returns `(verdict, note)`. 

128 

129 `node` is the gate itself, and **the gate declares the reviewing task**: 

130 `GateNode.auto_review` is an ordinary `AnyTask`, resolved and launched like 

131 any other (`gate-auto-review-is-explicit-and-bounded`). This module used to 

132 hardwire `{"command": "claude", "skill": "gate-review"}` -- a reviewer 

133 declared nowhere, which no chain could change and no template could see. 

134 Only the *prompt* is Kraft's now, because the output contract 

135 (`verdict`/`concerns`) is what the caller applies. 

136 

137 Never raises for an agent that misbehaved: every degraded outcome is 

138 `("undecided", note)`, which the caller turns into "leave the gate pending". 

139 """ 

140 auto_review = node.auto_review 

141 if auto_review is None or not isinstance(auto_review.task, AgentTask): 

142 # Defence in depth, and unreachable from a validated chain: 

143 # `GateNode.auto_review` is typed `AgentTask | None`, so a non-agent 

144 # reviewer cannot be represented, and `auto_check_due` refuses an 

145 # *undeclared* one before this is called. Note what `auto_check_due` does 

146 # **not** do -- it tests `node.auto_review is None` and never the task's 

147 # kind, which is why this arm existed at all. It stays for a 

148 # `MaterializedChain` round-tripped out of a row an older build wrote, 

149 # which the new type cannot retroactively police. 

150 # 

151 # **The event is not optional.** `_gate_review_attempts` counts an 

152 # `gate_auto_review_skipped` that is not the tail of a 

153 # `gate_auto_review_started`, and this was the one early return that 

154 # wrote neither -- so the attempt tally stayed at zero, `auto_check_due` 

155 # stayed True, and the delay poller re-armed the same dead gate every 

156 # tick forever: a `max_concurrent` slot burned and a human's `retry` 

157 # 409'd, which is the exact failure `auto_check_due` exists to prevent. 

158 await db.write( 

159 lambda c: events.append( 

160 c, 

161 work_item_id, 

162 "gate_auto_review_skipped", 

163 {"gate": gate, "reason": "unreviewable"}, 

164 ) 

165 ) 

166 return "undecided", (f"gate {gate!r} declares no agent task to review it; a person decides") 

167 if (refusal := _profile_fallback_refusal(auto_review.task, gate)) is not None: 

168 # Same posture as the kind-check above: an unpaired skip, no 

169 # `_started`, so `_gate_review_attempts` still counts it as one 

170 # attempt and the delay poller does not re-arm a dead gate forever. 

171 await db.write( 

172 lambda c: events.append( 

173 c, 

174 work_item_id, 

175 "gate_auto_review_skipped", 

176 {"gate": gate, "reason": "profile_fallback"}, 

177 ) 

178 ) 

179 return "undecided", refusal 

180 row = db.read( 

181 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (work_item_id,)).fetchone() 

182 ) 

183 if row is None: 

184 raise LookupError(f"unknown work_item {work_item_id!r}") 

185 

186 session_id = uuid.uuid4().hex 

187 rel = executor.gate_artifact(run_dirs, row, gate) 

188 task_instruction = _PROMPT.format( 

189 gate=gate, 

190 node_id=node.id, 

191 title=row["title"], 

192 description_line=f"Brief: {row['description']}\n" if row["description"] else "", 

193 artifact_line=_artifact_line(rel), 

194 ) 

195 

196 await db.write( 

197 lambda c: events.append( 

198 c, work_item_id, "gate_auto_review_started", {"gate": gate, "session_id": session_id} 

199 ) 

200 ) 

201 

202 try: 

203 # The item's sandbox, as for every launch of it (Ruling 189). 

204 sandbox = executor.item_sandbox(row, launch) 

205 except executor.SandboxUnresolved as exc: 

206 # Recorded as the reviewer's own session, so the pending gate names 

207 # why no review ran (Kraft-9t2dp); nothing launches. 

208 await executor.config_error_session( 

209 db, 

210 run_dirs, 

211 dict( 

212 session_id=session_id, 

213 work_item_id=work_item_id, 

214 node_id=node.id, 

215 hook_point=auto_review.path, 

216 round=0, 

217 head_sha=None, 

218 ), 

219 f"{auto_review.path}: {exc}\n", 

220 ) 

221 return "undecided", str(exc) 

222 # The gate's time cap bounds its review, as a node's does its tasks 

223 # (Kraft-8en38): spent refuses the launch; otherwise the one run is killed 

224 # at the deadline. 

225 hit = db.read(lambda c: caps.at_launch(c, row, auto_review)) 

226 if hit is not None and hit.remaining_s <= 0: 

227 await executor.time_capped_session( 

228 db, 

229 run_dirs, 

230 dict( 

231 session_id=session_id, 

232 work_item_id=work_item_id, 

233 node_id=node.id, 

234 hook_point=auto_review.path, 

235 round=0, 

236 head_sha=None, 

237 ), 

238 hit, 

239 ) 

240 return "undecided", hit.reason 

241 time_cap = caps.Deadline(caps.monotonic() + hit.remaining_s, hit) if hit else None 

242 try: 

243 inv = _agent.resolve_agent_task( 

244 auto_review.task, 

245 launch.repo_entry, 

246 launch.library_steering, 

247 skills_dir=launch.skills_dir, 

248 # Item-wide overrides only -- an item marked cheap stays cheap for its 

249 # gate reviews too (Kraft-ui79: no per-gate floor). Deliberately *not* 

250 # the per-node `node_overrides` model/effort dial that 

251 # `dispatch.dispatch_node` also merges (Kraft-df4tc): a node dialed to 

252 # a different model does not carry that dial into its own gate review. 

253 item_override=json.loads(row["agent_overrides"]) if row["agent_overrides"] else None, 

254 # The reviewer's and the repository's steering, frozen at intake. 

255 **executor.frozen_steering(row), 

256 # The item's own repo as recorded at intake (Kraft-jzdyp): a gate 

257 # review runs in the item's own checkout, never a fanned-out member's. 

258 item_repo=row["repo"], 

259 # The gate scope's policy: its tool lists and sandbox, as for any task. 

260 policy=executor.scope_policy(row, auto_review), 

261 ) 

262 except _steering.SteeringError as exc: 

263 # Modelled on the `SandboxUnresolved` handler above: recorded as the 

264 # reviewer's own session, so the pending gate names why no review ran, 

265 # and `gate_auto_review_started` above is not left dangling with no 

266 # terminal event for `_gate_review_attempts` to count. 

267 await executor.config_error_session( 

268 db, 

269 run_dirs, 

270 dict( 

271 session_id=session_id, 

272 work_item_id=work_item_id, 

273 node_id=node.id, 

274 hook_point=auto_review.path, 

275 round=0, 

276 head_sha=None, 

277 ), 

278 f"{auto_review.path}: {exc}\n", 

279 ) 

280 return "undecided", str(exc) 

281 status = await _agent.run_agent_task( 

282 db, 

283 run_dirs, 

284 session_id=session_id, 

285 work_item_id=work_item_id, 

286 node_id=node.id, 

287 hook_point=auto_review.path, 

288 command=inv.command, 

289 harness=inv.harness, 

290 model=inv.model, 

291 effort=inv.effort, 

292 deny_tools=inv.deny_tools, 

293 allowed_tools=inv.allowed_tools, 

294 permission_mode=inv.permission_mode, 

295 method_text=inv.method_text, 

296 steering_texts=inv.steering_texts, 

297 sandbox=sandbox, 

298 task_instruction=task_instruction, 

299 title=row["title"], 

300 repo_path=row["repo"], 

301 cwd=run_dirs.worktrees / work_item_id, 

302 repo_entry=launch.repo_entry, 

303 time_cap=time_cap, 

304 harness_id=auto_review.task.harness, 

305 ) 

306 

307 result_path = run_dirs.results / f"{session_id}.json" 

308 note = _subprocess.read_concerns(result_path) or "" 

309 verdict = _subprocess.read_verdict(result_path) 

310 if status in _UNTRUSTWORTHY or verdict not in VERDICTS: 

311 return "undecided", note 

312 if verdict in ("reject", "fixed") and not note.strip(): 

313 # The note is the steering instruction for the re-run. Without one there 

314 # is nothing to send back with, so it goes to a human instead. 

315 return "undecided", note 

316 return verdict, note
