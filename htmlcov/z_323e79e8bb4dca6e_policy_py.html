474def _cron_values(name: str, part: str, label: str, lo: int, hi: int) -> frozenset[int]: 

475 """The values one cron field matches: `*`, an integer, a range `A-B`, a 

476 step `*/N`, `A-B/N` or `A/N` (A to the field's top), or a comma-separated 

477 list of those. Anything else, or a value outside `lo`-`hi`, is refused.""" 

478 shape = ( 

479 f"{name}: field {part!r} must be '*', an integer, a range (1-5), a step (*/15), " 

480 "or a comma-separated list of those" 

481 ) 

482 out: set[int] = set() 

483 for token in part.split(","): 

484 if token.isdigit() and not _DIGITS.fullmatch(token): 

485 raise CronDigitError(f"{name}: field {part!r}: {token!r} is not the digits 0-9") 

486 base, slash, step_text = token.partition("/") 

487 if slash and not (_DIGITS.fullmatch(step_text) and int(step_text) > 0): 

488 raise PolicyError(f"{name}: field {part!r}: a step must be a positive integer") 

489 step = int(step_text) if slash else 1 

490 start_text, dash, end_text = base.partition("-") 

491 if base == "*": 

492 start, end = lo, hi 

493 elif dash and _DIGITS.fullmatch(start_text) and _DIGITS.fullmatch(end_text): 

494 start, end = int(start_text), int(end_text) 

495 elif _DIGITS.fullmatch(base): 

496 start = int(base) 

497 end = hi if slash else start 

498 else: 

499 raise PolicyError(shape) 

500 for value in (start, end): 

501 if not lo <= value <= hi: 

502 raise CronRangeError( 

503 f"{name}: field {part!r}: {label} {value} is outside {lo}-{hi}" 

504 ) 

505 if start > end: 

506 raise PolicyError(f"{name}: field {part!r}: the range {start}-{end} runs backwards") 

507 out.update(range(start, end + 1, step)) 

508 return frozenset(out)

511def _cron_sets(name: str, expr: str) -> tuple[frozenset[int], ...]: 

512 """Each field's matching values, in field order (`_CRON_FIELDS`). Day of

858 authored in one file, so this is coherence rather than privilege

1457 def apply_to( 

1458 self, policy: InstancePolicy, path: str, scopes: Iterable[TaskPolicyOverride] = () 

1459 ) -> InstancePolicy: 

1460 """`policy` -- a scope's, every authored layer already applied -- with 

1461 this item's layers at `path` on top (Ruling 188). Operational fields 

1462 apply in order, last, so the item's value wins over the template's 

1463 within the maxima. Safety fields combine in no order: an allowlist 

1464 intersects, a deny list unions, a budget takes the minimum, and a 

1465 sandbox locks -- so an item's safety value only ever tightens, and is 

1466 never refused because a narrower scope already narrowed it. 

1467 

1468 A time cap is the item's own at its level (Ruling 198): item-wide it 

1469 is the work item's cap, replacing the one the repository or chain 

1470 gave it -- and so every level's default (Ruling 211) -- for every 

1471 scope in `scopes`, the chain's layers over `path`, that set none, and 

1472 meeting any scope's own; on a path it only tightens.""" 

1473 own = {n for n in SCOPE_CAP_FIELDS if any(getattr(s, n) is not None for s in scopes)} 

1474 for where, layer in self.layers_at(path): 

1475 if where == "policy": 

1476 for name in SCOPE_CAP_FIELDS: 

1477 value = _cap(getattr(layer, name)) 

1478 if value is not None and name not in own: 

1479 policy = dataclasses.replace(policy, **{name: value}) 

1480 policy = policy.apply_template_override( 

1481 layer.model_copy(update=dict.fromkeys(_ORDERLESS_SAFETY_FIELDS)) 

1482 ) 

1483 if layer.allowed_tools is not None: 

1484 allowed = policy.allowed_tools 

1485 policy = dataclasses.replace( 

1486 policy, 

1487 allowed_tools=tuple( 

1488 t 

1489 for t in (allowed if allowed is not None else layer.allowed_tools) 

1490 if t in layer.allowed_tools 

1491 ), 

1492 ) 

1493 # Grants widen, so an item's own layer may only drop them, never 

1494 # add one (Kraft-4in7z.7): what a task is granted is authored. 

1495 if layer.grants is not None: 

1496 policy = dataclasses.replace( 

1497 policy, grants=tuple(g for g in policy.grants if g in layer.grants) 

1498 ) 

1499 for name in (*BUDGET_FIELDS, *CAP_FIELDS): 

1500 value, current = _cap(getattr(layer, name)), getattr(policy, name) 

1501 if value is not None: 

1502 policy = dataclasses.replace( 

1503 policy, **{name: min(current, value) if current is not None else value} 

1504 ) 

1505 return policy

1586 def at_level(self, level: str) -> InstancePolicy: 

1587 """This policy's caps as a scope of `level` runs under them (Ruling 

1588 211): the value a layer set -- its own or one it inherits from the 

1589 chain, an enclosing node, step or task, or the work item's override -- 

1590 else `level`'s default, else its maximum; and never past that 

1591 maximum. A scope's own value past it is refused when the chain is 

1592 checked (`ResolvedChain._check_caps`); an inherited one is held to it 

1593 here.""" 

1594 updates = {} 

1595 for name in SCOPE_CAP_FIELDS: 

1596 value = getattr(self, name) 

1597 if value is None: 

1598 value = getattr(getattr(self.cap_defaults, level), name) 

1599 bound = self.maxima.nearest(level, name) 

1600 if bound is not None: 

1601 value = bound[1] if value is None else min(value, bound[1]) 

1602 updates[name] = value 

1603 return dataclasses.replace(self, **updates)

1613 def apply_template_override(self, raw: TaskPolicyOverride | dict) -> InstancePolicy: 

1614 """Layer `raw` onto this policy, field by field 

1615 (`policy-override-rules-are-field-specific`): a ratchet-only safety 

1616 field (`allowed_tools`, `token_budget`) may only narrow the inherited 

1617 value; an operational field (`timeout_minutes`, `max_attempts`, 

1618 `allowed_harnesses`) may move either way but not past an explicitly 

1619 configured administrator maximum 

1620 (`work-item-policy-may-exceed-default-ceilings-within-admin-maximum`) 

1621 -- for `allowed_harnesses` that bound is always `maxima`, so a 

1622 `defaults:` value narrower than `maxima:` can still be widened back 

1623 up to it. `deny_tools` only accumulates, and a `sandbox` a broader 

1624 layer set cannot be changed (Ruling 105). The same method resolves a 

1625 repository override over the instance policy, a work-item override 

1626 over that, and so on (`policy-is-layered-by-execution-scope`)."""

1628 raw 

1629 if isinstance(raw, TaskPolicyOverride) 

1630 else TemplatePolicyOverride.model_validate(raw) 

1631 )

1634 if getattr(override, "escalation_harness", None) is not None: 

1635 updates["escalation_harness"] = override.escalation_harness 

1636

1638 updates["deny_tools"] = tuple(dict.fromkeys((*self.deny_tools, *override.deny_tools))) 

1639 if override.grants is not None: 

1640 updates["grants"] = tuple(dict.fromkeys((*self.grants, *override.grants)))

1644 raise PolicyError( 

1645 f"'sandbox' cannot change the inherited sandbox {self.sandbox.model_dump()!r}", 

1646 field="sandbox", 

1647 ) 

1648 updates["sandbox"] = override.sandbox

1652 if value is None:

1733class CarriedPolicy: 

1734 """A pre-V1 `policy.yaml` moved onto the V1 seed's (Ruling 172): the seed, 

1735 with the operator's value for every key the V1 schema still has, and every 

1736 key it no longer has -- or whose value it refuses -- dropped and named with 

1737 the value it had, so a major update never resets a spend cap or a timeout 

1738 without saying so. 

1739 

1740 `loops:` is the one section whose *keys* changed: a legacy fix-loop cap 

1741 (`verify_fix_loop`, `rebase_bounce`, ...) names a loop V1 never runs, so an 

1742 entry is carried only when the V1 seed names the same loop.""" 

1743 

1744 data: dict 

1745 #: Dotted key -> the value it had. 

1746 dropped: dict[str, object] 

1747 

1748 @classmethod 

1749 def from_legacy(cls, legacy: dict, seed: dict) -> CarriedPolicy: 

1750 data = {key: value for key, value in seed.items()} 

1751 dropped: dict[str, object] = {} 

1752 live_loops = set(seed.get("loops") or {}) 

1753 for key, value in legacy.items(): 

1754 if value is None: 

1755 continue 

1756 if key not in PolicyInput.model_fields: 

1757 dropped[key] = value 

1758 elif key == "loops" and isinstance(value, dict): 

1759 loops = dict(data.get("loops") or {}) 

1760 for name, cap in value.items(): 

1761 if name in live_loops: 

1762 loops[name] = cap 

1763 else: 

1764 dropped[f"loops.{name}"] = cap 

1765 data["loops"] = loops 

1766 else: 

1767 data[key] = value 

1768 # A carried value V1 refuses goes too: the result always loads. 

1769 while True: 

1770 try: 

1771 PolicyInput.model_validate(data) 

1772 return cls(data, dropped) 

1773 except ValidationError as exc: 

1774 loc = exc.errors()[0]["loc"] 

1775 if not loc or loc[0] not in legacy: 

1776 raise 

1777 key, *rest = loc 

1778 if key == "loops" and rest: 

1779 # One loop's cap, not the section: put back the seed's. 

1780 name = rest[0] 

1781 dropped[f"loops.{name}"] = data["loops"].pop(name) 

1782 if name in live_loops: 

1783 data["loops"][name] = seed["loops"][name] 

1784 continue 

1785 dropped[key] = legacy[key] 

1786 if key in seed: 

1787 data[key] = seed[key] 

1788 else: 

1789 del data[key]
