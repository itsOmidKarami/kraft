1"""Optional Docker isolation for a worker subprocess (Kraft-rki). 

2 

3`--disallowed-tools` is the agent CLI policing itself at Kraft's request, not 

4a sandbox: it does not stop a worker that spawns a shell directly, and a 

5command-scoped allowlist does not help either -- an allowlisted `pytest` 

6still executes a `conftest.py` the implementation node itself wrote, in the 

7same host process, as the invoking user. This module is the one place that 

8knows what a `sandbox:` value looks like, which of a binding's and a repo's 

9wins, and how to wrap a command to actually run inside one. 

10 

11Isolation here is two halves, and the second is the load-bearing one. The 

12mounts (`docker_argv`) keep the container out of everything it has no 

13business writing. But a worker's own worktree gitdir must stay writable for 

14it to commit at all, and that is where every file git redirects hooks and 

15config through lives -- so `harden_host_git_env` pins `core.hooksPath` (and 

16the other program-valued config keys) for every git Kraft runs, which makes 

17what a worker writes there unable to execute regardless of which redirect 

18file it used. 

19 

20Read docs/superpowers/specs/2026-09-13-worker-sandbox-docker-design.md for 

21the design this implements. 

22 

23Off by default: nothing calls `resolve` unless a binding or a repo entry sets 

24`sandbox:` at all, and `docker_argv` is only ever called with a resolved, 

25already-validated value. 

26"""

36#: Only kind implemented. A binding naming any other kind is rejected at 

37#: config load, not at dispatch -- the same "unknown X" treatment an unknown 

38#: agent profile or forge backend already gets (`templates.py`). 

39_KNOWN_KINDS = {"docker"}

41#: Kraft's own vars, plus whichever auth var this install's `claude` CLI 

42#: actually uses, forwarded bare (`-e NAME`, no value) so docker copies each 

43#: from its own process env -- never written as a literal value into 

44#: registry.yaml or repos.yaml. A var absent from that env is simply not set 

45#: in the container; `docker run -e NAME` for an unset NAME is not an error. 

46#: `CLAUDE_CODE_OAUTH_TOKEN` is what a subscription install's headless 

47#: `claude -p` authenticates with (`claude setup-token`) -- the interactive 

48#: login session itself lives in the OS keychain, which a Linux container 

49#: cannot reach. `ANTHROPIC_API_KEY` covers an API-key install the same way; 

50#: the two are not mutually exclusive to forward, only one is ever actually 

51#: set on a given install. 

52FORWARDED_ENV = ( 

53 "KRAFT_RESULT_PATH", 

54 "KRAFT_WORK_ITEM_ID", 

55 "KRAFT_SESSION_ID", 

56 "KRAFT_REVIEW_PACKAGE", 

57 "CLAUDE_CODE_OAUTH_TOKEN",

66def validate(sandbox: object, *, where: str) -> None: 

67 """A binding's or a repo's own `sandbox:` value, or raise. 

68 

69 Only called with a value that is neither `None` nor `False` -- both of 

70 those are meaningful ("no sandbox" / "explicitly off") and have no shape 

71 to check, so callers filter them out before reaching here. 

72 """ 

73 if not isinstance(sandbox, dict): 

74 raise SandboxError(f"{where}: 'sandbox' must be a mapping, not {sandbox!r}") 

75 kind = sandbox.get("kind") 

76 if kind not in _KNOWN_KINDS: 

77 raise SandboxError( 

78 f"{where}: sandbox kind {kind!r} is not supported; known: {sorted(_KNOWN_KINDS)}" 

79 ) 

80 image = sandbox.get("image") 

81 if not isinstance(image, str) or not image: 

82 raise SandboxError(f"{where}: sandbox needs a non-empty string 'image'")

85def resolve(binding: dict, repo_entry: dict | None) -> dict | None: 

86 """The sandbox a hook launches under. 

87 

88 A repo's own `sandbox` wins over the binding's wholesale -- the same 

89 override-wins-over-registry rule `test_scopes`/`test_command` already use 

90 for `command` (`executor/dispatch.py`), not a second rule and not a 

91 merge. `None` on the repo entry (no `sandbox` key at all, the common 

92 case) falls through to the binding's; `False` turns off a binding that 

93 turned sandboxing on. 

94 """ 

95 repo_sandbox = (repo_entry or {}).get("sandbox") 

96 sandbox = repo_sandbox if repo_sandbox is not None else binding.get("sandbox") 

97 return sandbox or None

100def container_name(session_id: str) -> str: 

101 """The `--name` a sandboxed session's container runs under. 

102 

103 A handle a caller can `docker kill`/`docker rm -f` by, independent of 

104 whatever became of the `docker run` client's own pid -- that client is 

105 the process Kraft's group-kill signals, but the container itself is 

106 parented by the docker daemon, not that group, and does not die with it. 

107 `session_id` is already a Kraft-minted id (safe for docker's 

108 `[a-zA-Z0-9_.-]` name charset); prefixed so a stray `kraft-*` container is 

109 obviously this tool's to clean up. 

110 """ 

111 return f"kraft-{session_id}"

114def docker_argv( 

115 cmd: list[str], 

116 cwd: str | Path, 

117 sandbox: dict, 

118 results_dir: str | Path, 

119 env: dict | None = None, 

120 name: str | None = None, 

121 result_path: str | Path | None = None, 

122) -> list[str]: 

123 """Wrap `cmd` to run inside `sandbox['image']` instead of directly on the host. 

124 

125 Mounts `cwd` (the git worktree) at its original host path, so every 

126 relative path the agent's own prompt already names stays meaningful 

127 unchanged inside the container -- nothing downstream of this has to know 

128 the process ran in one. Runs as the invoking host uid:gid so a container 

129 default of root does not leave root-owned files behind in the worktree 

130 for a later host-side step to trip over -- `--security-opt=no-new-privileges` 

131 and `--cap-drop=ALL` are what keep that true: without them, a setuid-root 

132 binary in the operator's image (`su`, `mount`, `sudo` all ship in most base 

133 images) lets the worker regain root inside the container and write 

134 root-owned or setuid-root files into the bind-mounted worktree anyway. 

135 

136 Everything else the container can reach is mounted read-only with exactly 

137 the paths this session must write carved back out read-write, rather than 

138 mounted read-write with the dangerous paths shadowed read-only: a 

139 shadow list is a list of the escapes someone thought of (`.git/hooks`, 

140 then `.git/modules/<sub>/hooks`, then `.git/worktrees/<id>/modules/...`), 

141 and the container only has to find the one nobody listed. 

142 

143 `results_dir`: read-only, because a hook legitimately *reads* its 

144 neighbours there (its own `<session>.review.md`, the previous fix 

145 session's result file), but a session writing another session's 

146 `<other>.json` would forge that session's status and findings. Only this 

147 session's own `result_path` is mounted back read-write. 

148 

149 `<repo>/.git`: read-only, with `objects/`, `refs/`, `logs/` and this 

150 worktree's own gitdir (`<repo>/.git/worktrees/<id>`) read-write -- what a 

151 commit from inside a linked worktree actually writes. That worktree 

152 gitdir has to be read-write wholesale: a commit rewrites HEAD, the index 

153 and per-worktree refs through lockfiles beside them, so the directory 

154 itself must be writable, and every file git redirects config and hooks 

155 through (`commondir`, `config.worktree`, `modules/<sm>/config`) lives in 

156 it. Those are not enumerated here -- `harden_host_git_env` is what makes 

157 them harmless, by pinning `core.hooksPath` and friends on every git Kraft 

158 itself runs, so whatever a worker plants in a file nobody listed still 

159 has nothing to execute it. 

160 

161 A Kraft worktree's `.git` is a file pointing at that gitdir (`git 

162 worktree`'s own layout), outside `cwd` -- without the mount every git 

163 command inside the container fails to find it. That file lives inside 

164 the read-write `cwd` mount, though, so it is shadowed read-only on top: 

165 otherwise a sandboxed worker rewrites it to point at a gitdir of its own 

166 making, complete with a `hooks/` it populates itself, and the next 

167 host-side git command run in the worktree executes that hook as the 

168 invoking host user. 

169 

170 `env` is `run_task`'s own `env=` argument -- e.g. `PYTHONDONTWRITEBYTECODE` 

171 for a fix-loop re-measure -- which is otherwise silently dropped: only 

172 `FORWARDED_ENV` crosses into the container by default. Passed through as 

173 literal `-e NAME=VALUE`, unlike `FORWARDED_ENV`'s bare `-e NAME` (which 

174 copies from docker's own process env, never written to disk). 

175 """ 

176 cwd = str(cwd) 

177 results_dir = str(results_dir) 

178 argv = [ 

179 "docker", 

180 "run", 

181 "--rm", 

182 "-u", 

183 f"{os.getuid()}:{os.getgid()}", 

184 "--security-opt=no-new-privileges", 

185 "--cap-drop=ALL", 

186 "-v", 

187 f"{cwd}:{cwd}", 

188 "-w", 

189 cwd, 

190 "-v", 

191 f"{results_dir}:{results_dir}:ro", 

192 ] 

193 if result_path is not None: 

194 argv += ["-v", f"{result_path}:{result_path}"] 

195 argv += _gitdir_mounts(Path(cwd)) 

196 if name is not None: 

197 argv += ["--name", name] 

198 for env_name in FORWARDED_ENV: 

199 argv += ["-e", env_name] 

200 for env_name, value in (env or {}).items(): 

201 argv += ["-e", f"{env_name}={value}"] 

202 argv.append(sandbox["image"]) 

203 argv += cmd 

204 return argv

207#: The only paths under a repo's common gitdir a commit made from inside a 

208#: linked worktree writes. None of them is ever executed by git (objects and 

209#: refs are data, `logs/` is the reflog), so read-write here does not hand the 

210#: container host code execution the way `hooks/` or `config` would. 

211_WRITABLE_COMMON_GITDIR = ("objects", "refs", "logs")

214def _gitdir_mounts(cwd: Path) -> list[str]: 

215 """`-v` arguments for the repo gitdir a worktree's `.git` file points at. 

216 

217 Empty when `cwd` has no `.git` at all (never expected for a real 

218 invocation, but this is not the place to raise over it) or already has an 

219 ordinary `.git` directory of its own -- that one is inside `cwd`, already 

220 mounted with it, and nothing extra is needed. 

221 

222 The `is_dir()` guards below are read-write *carve-outs*, not read-only 

223 shadows: a path that is missing stays covered by the read-only mount of 

224 its parent, so a container that creates one gains nothing. (An earlier 

225 revision guarded read-only shadows this way, where a missing path meant 

226 no shadow at all and the container could just create it.) 

227 """ 

228 git_path = cwd / ".git" 

229 if git_path.is_dir(): 

230 return [] 

231 try: 

232 text = git_path.read_text() 

233 except OSError: 

234 return [] 

235 if not text.startswith("gitdir:"): 

236 return [] 

237 # `.git` itself lives inside the read-write `cwd` mount. Left alone, a 

238 # sandboxed worker repoints it at a gitdir it builds inside the 

239 # worktree -- HEAD, objects/, refs/, a config with hooks -- and the next 

240 # host-side git command run there executes the worker's hook as the 

241 # invoking host user. It never legitimately changes for the life of the 

242 # worktree, so shadowing it read-only costs nothing. 

243 mounts = ["-v", f"{git_path}:{git_path}:ro"] 

244 gitdir = Path(text.removeprefix("gitdir:").strip()) 

245 # `<repo>/.git/worktrees/<id>` -> common gitdir `<repo>/.git`. Only that 

246 # one subdirectory is this session's to write; every sibling worktree's 

247 # gitdir stays read-only under the common mount. 

248 common, worktree_gitdir = ( 

249 (gitdir.parent.parent, gitdir) if gitdir.parent.name == "worktrees" else (gitdir, None) 

250 ) 

251 # `logs/` does not exist in a freshly cloned repo, and git creates it on 

252 # the first ref update -- which it cannot do under the read-only mount, 

253 # failing the commit. Creating it here is what git itself would do a 

254 # moment later. 

255 try: 

256 (common / "logs").mkdir(exist_ok=True) 

257 except OSError: 

258 pass 

259 mounts += ["-v", f"{common}:{common}:ro"] 

260 for name in _WRITABLE_COMMON_GITDIR: 

261 path = common / name 

262 if path.is_dir(): 

263 mounts += ["-v", f"{path}:{path}"] 

264 if worktree_gitdir is not None: 

265 # A commit from this worktree writes all over its own gitdir (index, 

266 # HEAD, COMMIT_EDITMSG, per-worktree refs, lockfiles), so this one is 

267 # read-write wholesale -- with the exceptions below. 

268 mounts += ["-v", f"{worktree_gitdir}:{worktree_gitdir}"] 

269 # `commondir` is a file inside that read-write gitdir naming where 

270 # `hooks/` and `config` actually resolve to (the common gitdir 

271 # mounted above). Left writable, a worker repoints it at a directory 

272 # of its own inside the worktree, carrying its own `config` and 

273 # `hooks/pre-commit`, and the next host-side git command run here 

274 # runs that hook as the invoking host user -- the same escape the 

275 # read-only `hooks/`/`config` mounts exist to close, reached through 

276 # the one file that redirects to them. It always exists once a 

277 # linked worktree is created, but is created here too rather than 

278 # guarded with `exists()`, so the shadow can never be dodged by a 

279 # worker that deletes it first. 

280 commondir = worktree_gitdir / "commondir" 

281 try: 

282 commondir.touch(exist_ok=True) 

283 except OSError: 

284 pass 

285 mounts += ["-v", f"{commondir}:{commondir}:ro"] 

286 # `config.worktree` is the other file inside this read-write gitdir git 

287 # reads as part of its config stack -- whenever 

288 # `extensions.worktreeConfig = true` (`git sparse-checkout set` turns 

289 # this on), git reads it in addition to the common gitdir's `config`. 

290 # Left writable, a worker plants `core.hooksPath` there just as easily 

291 # as through `commondir`, and the next host-side `git commit` this 

292 # worktree runs executes that hook as the invoking host user -- the 

293 # same escape, a second file wide. Created here rather than guarded 

294 # with `exists()` for the same reason as `commondir`: it must not be 

295 # dodged by a worker that deletes it first. 

296 worktree_config = worktree_gitdir / "config.worktree" 

297 try: 

298 worktree_config.touch(exist_ok=True) 

299 except OSError: 

300 pass 

301 mounts += ["-v", f"{worktree_config}:{worktree_config}:ro"] 

302 return mounts

305#: Config a repository must not be able to set for a git *Kraft itself* runs, 

306#: because each of these values is a program git executes. 

307#: 

308#: A worker's worktree gitdir is read-write by construction -- a commit writes 

309#: its index, HEAD, lockfiles and per-worktree refs, so the directory holding 

310#: them cannot be read-only, and every file git redirects through lives in 

311#: that same directory (`commondir`, `config.worktree`, `modules/<sm>/config`, 

312#: and whichever one the next git version adds). Enumerating them one shadow 

313#: mount at a time is a list of the escapes someone thought of; this is the 

314#: other side of the same problem, and it does not need the list: whatever a 

315#: worker plants, the host-side git that would have run it is launched with 

316#: these pinned instead. 

317#: 

318#: `/dev/null` as a hooks *directory* resolves every hook to `/dev/null/<name>`, 

319#: which is not a file, so git finds no hook to run. `core.fsmonitor` names a 

320#: program git spawns on `status`; empty is "unset", git's own default. 

321#: 

322#: `diff.external` is deliberately *not* here even though it is the same kind 

323#: of vector: git reads an empty value as the empty command and dies 

324#: ("cannot run : No such file or directory") on every `git diff`, which is 

325#: half of what Kraft does with a worktree. Nothing safe to pin it to exists, 

326#: so that one stays covered by the mounts -- the config files git could read 

327#: it from are the common gitdir's `config` (read-only mount), `config.worktree` 

328#: and whatever `commondir` redirects to (both shadowed read-only). 

329_HARDENED_GIT_CONFIG = ( 

330 ("core.hooksPath", os.devnull), 

331 ("core.fsmonitor", ""), 

332)

335def harden_host_git_env(env: MutableMapping[str, str] | None = None) -> None: 

336 """Pin `_HARDENED_GIT_CONFIG` for every git this process ever spawns. 

337 

338 `GIT_CONFIG_COUNT`/`_KEY_n`/`_VALUE_n` is git's own environment form of 

339 `git -c key=value`, and carries `-c`'s precedence: it beats the system, 

340 global, repo *and* worktree config files, so a `core.hooksPath` a worker 

341 plants in any of them loses. Setting it on the server's own environment 

342 once covers every git Kraft runs -- `builtins`, `adapters.forge.git`, 

343 `adapters.forge.run`, `config.git_read`, `index.ingest`, the worktree 

344 teardown in `api.routes.lifecycle` -- and every git those spawn in turn 

345 (a submodule's, a rebase's), without each call site having to remember. 

346 Doing it per call site is what left `git worktree prune` and 

347 `git submodule update` running a worker's hook while `git commit` had 

348 `--no-verify`. 

349 

350 Kraft's own worker sessions inherit it too (they are children of this 

351 process), so a repo hook does not fire on an agent's commits either -- 

352 the same thing `adapters.forge.git` already asked for explicitly with 

353 `--no-verify`. 

354 

355 `adapters.forge.git.push` is the one exception, running with 

356 `unhardened_git_env()` instead: see that function for why. 

357 """ 

358 env = os.environ if env is None else env 

359 env["GIT_CONFIG_COUNT"] = str(len(_HARDENED_GIT_CONFIG)) 

360 for i, (key, value) in enumerate(_HARDENED_GIT_CONFIG): 

361 env[f"GIT_CONFIG_KEY_{i}"] = key 

362 env[f"GIT_CONFIG_VALUE_{i}"] = value

365def unhardened_git_env() -> dict[str, str]: 

366 """A copy of the process env with `harden_host_git_env`'s override lifted. 

367 

368 For `git push` alone: by the time Kraft pushes, the commit is already 

369 made and the worktree gitdir's hooks are either read-only-mounted 

370 (sandboxed) or none of a worker's business to have tampered with in the 

371 first place (unsandboxed) -- so the pinned `core.hooksPath=/dev/null` 

372 is no longer buying anything there, only breaking real hooks a human 

373 installed, like git-lfs's pre-push. Popping `GIT_CONFIG_COUNT` alone is 

374 enough: git reads none of the paired `GIT_CONFIG_KEY_n`/`_VALUE_n` 

375 without it (Kraft-rki). 

376 """ 

377 env = dict(os.environ) 

378 env.pop("GIT_CONFIG_COUNT", None) 

379 return env

382async def teardown(session_id: str) -> None: 

383 """`docker rm -f` this session's container, best-effort. 

384 

385 The one place a session's container is torn down, called from every path 

386 a session can *end* on -- `run_task`'s `finally` (normal exit, pause, 

387 skip, timeout) and `reattach`'s (a session adopted or resolved after a 

388 Kraft restart). The pid Kraft signals is `docker run`'s own client 

389 process, not the container: the container is parented by the docker 

390 daemon, so a killed -- or restarted-away-from -- client leaves it running 

391 with the worktree still mounted (Kraft-rki). 

392 

393 Safe to call for a session that never had a sandbox at all: the name 

394 (`container_name`) simply matches nothing, and `docker rm -f` on an 

395 unknown name is a no-op this swallows. That is deliberate -- the callers 

396 are "a session ended" points, and making them each first work out whether 

397 this one was sandboxed is how a path gets missed. Errors (already gone, 

398 docker not installed) are not this function's to raise: a container that 

399 is already down is the success case. 

400 """ 

401 try: 

402 proc = await asyncio.create_subprocess_exec( 

403 "docker", 

404 "rm", 

405 "-f", 

406 container_name(session_id), 

407 stdout=subprocess.DEVNULL, 

408 stderr=subprocess.DEVNULL, 

409 ) 

410 await proc.wait() 

411 except OSError: 

412 pass
