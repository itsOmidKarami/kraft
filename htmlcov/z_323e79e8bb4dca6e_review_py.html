1"""One producer for "what changed", read by two consumers. 

2 

3`GET /work-items/{wid}/diff` renders the change for the human at a gate; 

4`write_package` writes it to a file a review agent is handed by path. Sub-project 

5F spec §9 makes that one function on purpose: the human approving a review and 

6the reviewer that produced it must be looking at the same change, and two 

7separately-written producers are two things that eventually disagree about what 

8the change is. 

9 

10Two ranges, not one. `read_change(wt, base)` is the working tree against `base` 

11-- an agent that wrote files without committing them is the normal mid-chain 

12state, and a committed-only diff would show an empty change set while the work 

13sat on disk. `read_change(wt, base, head="HEAD")` is `base..HEAD`, what earlier 

14nodes finished and committed. The gate viewer shows both, apart (Kraft-nceo). 

15 

16`write_package` hands a review agent one range, and from the second round of a 

17fix loop onward that range starts at the head the previous review was taken at 

18rather than at `base` (`since`, Kraft-s7c04.1). Re-reading the whole branch 

19every round is what made `verify` half of all Kraft spend: 43717ee6 measured 

20the same 38-file diff seven times. The package says which range it is showing 

21and names the git command for the rest, so nothing is hidden -- only unpasted. 

22""" 

23 

24from __future__ import annotations 

25 

26from pathlib import Path 

27from typing import NamedTuple 

28 

29from kraft import config as _config 

30 

31#: Wider than git's default 3. A reviewer holding ten lines of context per hunk 

32#: can judge most changes without opening the file: "the diff's context lines 

33#: ARE the changed files". Only the package uses it -- the gate viewer renders 

34#: in a browser, where the extra context is scrolling, not evidence. 

35PACKAGE_CONTEXT = 10 

36 

37 

38class Change(NamedTuple): 

39 commits: list[str] 

40 files: list[dict] 

41 diff: str 

42 untracked: list[str] 

43 

44 

45def read_change( 

46 worktree: Path, base: str, *, head: str | None = None, context: int | None = None 

47) -> Change | None: 

48 """The change in `worktree` against `base`, or None if git itself failed. 

49 

50 With `head` set the range is `base..head` and nothing uncommitted is in it, 

51 so `untracked` is `[]`: `git status --porcelain` describes the working tree 

52 and has no meaning for a committed range. 

53 

54 None is not "no changes": returning an empty diff for a git failure is 

55 indistinguishable from a clean tree to whoever is about to approve it. 

56 """ 

57 rev = [base] + ([head] if head else []) 

58 diff_args = ["diff"] + ([f"-U{context}"] if context is not None else []) + rev 

59 # strip=False: a diff whose last line is blank context is still that diff 

60 body = _config.git_read(worktree, *diff_args, strip=False) 

61 numstat = _config.git_read(worktree, "diff", "--numstat", *rev) 

62 if body is None or numstat is None: 

63 return None 

64 untracked: list[str] = [] 

65 if head is None: 

66 status = _config.git_read(worktree, "status", "--porcelain", "-uall") 

67 if status is None: 

68 return None 

69 untracked = [ln[3:] for ln in status.splitlines() if ln.startswith("?? ")] 

70 # A repo with no commits past `base` is normal, not a failure, so an empty 

71 # log is not folded in with the None checks above. 

72 log = _config.git_read(worktree, "log", "--oneline", f"{base}..{head or 'HEAD'}") or "" 

73 

74 files = [] 

75 for line in numstat.splitlines(): 

76 parts = line.split("\t") 

77 if len(parts) == 3: 

78 ins, dels, path = parts 

79 files.append( 

80 { 

81 "path": path, 

82 "insertions": int(ins) if ins.isdigit() else 0, 

83 "deletions": int(dels) if dels.isdigit() else 0, 

84 } 

85 ) 

86 return Change(commits=log.splitlines(), files=files, diff=body, untracked=untracked) 

87 

88 

89#: What a narrowed package tells the reviewer, in place of the plain header. 

90#: Both halves matter: which range it is looking at, and that the rest of the 

91#: branch is still reachable. Without the second half a reviewer that obeys 

92#: "read that file first" has been quietly cut off from the change as a whole, 

93#: which is the failure this narrowing must not cause (Kraft-s7c04.1). 

94_SINCE_HEADER = ( 

95 "# Review package for {since}..working tree\n\n" 

96 "This is the change **since your last review of this branch**, not the whole " 

97 "branch. Everything before {since} was reviewed in an earlier round, and the " 

98 "findings that review left are listed in your instruction.\n\n" 

99 "For the full branch diff, run `git diff {base}...HEAD` in this worktree. Do " 

100 "that when you need to judge the change as a whole -- whether it does what " 

101 "the work item asked, whether it breaks a caller outside these hunks -- and " 

102 "not otherwise.\n" 

103) 

104 

105 

106def render_package(change: Change, base: str, *, since: str | None = None) -> str: 

107 stat = "\n".join(f"{f['path']} | +{f['insertions']} -{f['deletions']}" for f in change.files) 

108 header = ( 

109 _SINCE_HEADER.format(since=since, base=base) 

110 if since 

111 else f"# Review package for {base}..working tree\n" 

112 ) 

113 return ( 

114 f"{header}\n" 

115 f"## Commits\n{chr(10).join(change.commits) or '(none yet — uncommitted work)'}\n\n" 

116 f"## Files changed\n{stat or '(none)'}\n\n" 

117 f"## Untracked\n{chr(10).join(change.untracked) or '(none)'}\n\n" 

118 f"## Diff\n{change.diff}" 

119 ) 

120 

121 

122def write_package( 

123 results_dir: Path, worktree: Path, base: str, session_id: str, *, since: str | None = None 

124) -> Path | None: 

125 """Write the package into `results_dir` and return its path, or None. 

126 

127 `since`, when given, narrows the range to `since..working tree` -- the head 

128 the previous review of this hook was taken at (Kraft-s7c04.1). `Commits` and 

129 `Files changed` narrow with it: a package describing one range and listing 

130 another is a trap for whoever reads it next. `None` is the whole branch, 

131 exactly as before, and is what round 0 of every work item gets. 

132 

133 Named by the session, the way the result file beside it already is. 

134 Naming it by the `base..HEAD` range instead looks like it distinguishes 

135 more, and distinguishes less: `results_dir` is one directory for every work 

136 item, and uncommitted work -- the normal mid-chain state this module exists 

137 to show -- leaves HEAD equal to base, so two items branched from the same 

138 commit collide, and one reviewer is handed the other item's diff. A session 

139 id is unique per dispatch, so a re-measure after a fix cycle also keeps the 

140 evidence the previous cycle was reviewed against. 

141 """ 

142 change = read_change(worktree, since or base, context=PACKAGE_CONTEXT) 

143 if change is None: 

144 return None 

145 path = results_dir / f"{session_id}.review.md" 

146 path.write_text(render_package(change, base, since=since)) 

147 return path
