51def _check( 

52 name: str, ok: bool, detail: str = "", *, skipped: bool = False, warn: bool = False 

53) -> dict: 

54 """A skipped check is `ok`: it did not fail, it did not run. Only a real 

55 failure may set the exit code, or one dead server reads as five problems. 

56 

57 `warn` is also `ok` for the same reason, but it is not a skip: the check 

58 ran and could not confirm the answer (a release feed that didn't 

59 respond, say), which is worth a human noticing even though `doctor && 

60 deploy` must not break on it. A skip and a warn look the same to the exit 

61 code and different to the human reading the output. 

62 """ 

63 return {"name": name, "ok": ok, "detail": detail, "skipped": skipped, "warn": warn}

115def _storage_line(payload: dict) -> tuple[str, str] | None: 

116 """`/health`'s `storage` as (state, one sentence), or None without a limit.""" 

117 s = payload.get("storage")

121 used = f"worktrees use {size(s['used_bytes'])} of {size(s['limit_bytes'])}"

134 return "ok", f"{used}; kraft view storage lists it by item"

137def _health_checks(payload: dict) -> list[dict]: 

138 """`/health`, one line per degraded reason. 

139 

140 A single line reading "degraded" sends you to the browser anyway, which is 

141 the thing doctor exists to avoid. 

142 """ 

143 index = payload.get("index") or {} 

144 embeddings = index.get("embeddings") or {} 

145 reasons = [ 

146 f"invalid template {name}: {why}" 

147 for name, why in (payload.get("invalid_templates") or {}).items() 

148 ] 

149 if payload.get("invalid_policy"): 

150 reasons.append(f"invalid policy: {payload['invalid_policy']}") 

151 if payload.get("invalid_intake"): 

152 reasons.append( 

153 f"invalid intake.yaml, auto-intake and its schedules are off (a trigger left in " 

154 f"policy.yaml still fires): {payload['invalid_intake']}" 

155 if payload.get("intake_off") 

156 else f"invalid intake.yaml, not applied: the running auto-intake and schedules are " 

157 f"kept; fix the file and reload: {payload['invalid_intake']}" 

158 ) 

159 reasons.extend(f"index: {error}" for error in index.get("errors") or []) 

160 orphaned = (payload.get("reattach_summary") or {}).get("unknown") or [] 

161 if orphaned: 

162 reasons.append(f"orphaned agent sessions: {', '.join(orphaned)}") 

163 stored = _storage_line(payload) 

164 if stored and stored[0] == "held": 

165 reasons.append(f"storage: {stored[1]}") 

166 # Semantic search is an opt-in extra (`available` is whether it imports), 

167 # so /health stays `ok` without it and so does doctor: an ok line carrying 

168 # the advice, never a FAIL on something the operator never asked for 

169 # (Kraft-rj8cn). Installed but failing to load or encode is a FAIL: the 

170 # operator did ask for it (Kraft-pm2rj). 

171 if not embeddings.get("available"): 

172 extra = _check( 

173 "embeddings", True, f"not installed: {embeddings.get('reason') or 'no reason given'}" 

174 ) 

175 elif embeddings.get("reason"): 

176 extra = _check("embeddings", False, f"installed but broken: {embeddings['reason']}") 

177 else: 

178 extra = _check("embeddings", True, f"available ({embeddings.get('model')})") 

179 tail = ( 

180 [extra, _check("storage", True, stored[1], warn=stored[0] == "over_quota")] 

181 if stored and stored[0] != "held" 

182 else [extra] 

183 ) 

184 if not reasons: 

185 status = str(payload.get("status", "?")) 

186 return [_check("health", payload.get("status") == "ok", status), *tail] 

187 return [*(_check("health", False, reason) for reason in reasons), *tail]

190def _restart_check(payload: dict) -> dict: 

191 """Is the server running the version installed on disk? 

192 

193 `kraft admin update` without `--restart` replaces the package under a 

194 server that keeps running the old code, and the old server's interface is 

195 replaced with it. A warning, not a failure: nothing is broken, but nothing 

196 the board shows can be trusted until the restart. A server that does not 

197 report `installed` predates this check, so it is older than this install. 

198 """ 

199 from kraft import update 

200 

201 here = update.installed() 

202 running = payload.get("version") 

203 pid = payload.get("pid") 

204 who = f"the server (pid {pid})" if pid is not None else "the server" 

205 if "installed" not in payload: 

206 return _check( 

207 "restart", 

208 True, 

209 f"{who} runs a release older than the installed {here}: restart it to finish " 

210 "the update - kraft admin restart", 

211 warn=True, 

212 ) 

213 if running != here: 

214 return _check( 

215 "restart", 

216 True, 

217 f"{who} runs {running}, but {here} is installed: restart it to finish the " 

218 "update - kraft admin restart", 

219 warn=True, 

220 ) 

221 return _check("restart", True, f"{who} runs the installed version ({here})")

243def _config_checks() -> list[dict]: 

244 templates = config_dir() 

245 if not templates.is_dir(): 

246 # Every row still reports, as a skip: every check this function can emit 

247 # emits on every path, so a caller reading the run by name never has to 

248 # ask whether a row is missing because it passed, because it was skipped, 

249 # or because an earlier branch returned before reaching it. 

250 return [ 

251 _check("config", False, f"{templates} does not exist — start `kraft` once to seed it"), 

252 _chain_templates_check(), 

253 _check("chains", True, "skipped: no config dir", skipped=True), 

254 # Nothing seeded yet, so nothing to upgrade: the first start seeds 

255 # every capability and stamps the version. 

256 _check("capabilities", True, "skipped: no config dir", skipped=True), 

257 _check("detectors.yaml", True, "skipped: no config dir", skipped=True), 

258 _check("notify.yaml", True, "skipped: no config dir", skipped=True), 

259 _check("plugins", True, "skipped: no config dir", skipped=True), 

260 _check("plugin updates", True, "skipped: no config dir", skipped=True), 

261 _token_check(), 

262 _token_check("trigger token", auth.TRIGGER_TOKEN_FILE), 

263 ] 

264 checks = [_config_row(templates)] 

265 try: 

266 access = config.Access.load(templates / "access.yaml") 

267 # Loaded as written, so this is the only place a bad one shows. 

268 bad = [why for h in access.allowed_hosts if (why := config.host_entry_problem(h))] 

269 bind = os.environ.get("KRAFT_HOST") or access.bind 

270 if bad and bind in config.LOOPBACK: 

271 # A loopback bind never reads the list, so a leftover entry 

272 # breaks nothing until the bind changes: a warning, not a failure 

273 # a `doctor && ...` script trips on after an upgrade. 

274 detail = "; ".join(bad) + f" (unused while bound to {bind})" 

275 checks.append(_check("access.yaml", True, detail, warn=True)) 

276 elif bad: 

277 checks.append(_check("access.yaml", False, "; ".join(bad))) 

278 elif bind not in config.LOOPBACK and not access.allowed_hosts: 

279 # Every browser on another device is refused, and 1.4 let a 

280 # plain-http one through: the first an upgraded LAN user hears of it. 

281 checks.append( 

282 _check( 

283 "access.yaml", 

284 True, 

285 f"bound to {bind} with no allowed_hosts: a browser on another " 

286 "device is refused; add each name it uses on Settings › Access", 

287 warn=True, 

288 ) 

289 ) 

290 else: 

291 checks.append(_check("access.yaml", True, "parses")) 

292 except config.ConfigError as exc: 

293 checks.append(_check("access.yaml", False, str(exc))) 

294 checks.append(_detectors_check(templates)) 

295 checks.append(_notify_check(templates)) 

296 checks.append(_moved_keys_check(templates)) 

297 checks.append(_chain_templates_check()) 

298 checks.append(_chains_check(templates)) 

299 checks += _plugin_checks(templates) 

300 checks.append(_capabilities_check()) 

301 checks.append(_token_check()) 

302 checks.append(_token_check("trigger token", auth.TRIGGER_TOKEN_FILE)) 

303 return checks

341def _why_it_stayed(key: str, data: dict) -> str: 

342 """Why the start did not move `policy.yaml`'s triggers (`carry_moved_keys`), 

343 which still fire but which Settings › Auto-intake does not show: a 

344 re-made copy there fires twice (R12c-02).""" 

345 triggers = data.get("triggers") 

346 if key != "triggers" or not isinstance(triggers, list): 

347 return "" 

348 refused = config.schedule_refusals(triggers, "triggers") 

349 if not refused: 

350 return "; the next start moves them" 

351 moving = len(triggers) - len(refused) 

352 out = f"; the next start moves {moving} of them" if moving else "" 

353 for index, why in refused.items(): 

354 note = policy_mod.skipped_trigger(triggers[index]) 

355 skipped = f" and is skipped, {note}" if note else "" 

356 out += ( 

357 f"; triggers.{index} stays because intake.yaml's schedules refuse {why}{skipped}" 

358 ": fix it and restart" 

359 ) 

360 return out

440def _chain_template_files(directory: Path) -> dict[str, set[str]]: 

441 """Every chain file under `directory/chains/` -- one with a top-level 

442 `nodes` list -- mapped to the ids of its nodes. By structure, not by 

443 filename, so a chain added in a later version is picked up without a code 

444 change here. Authored node ids: a chain's nodes are listed in the chain 

445 file itself even when a node `extends` a library node.""" 

446 found: dict[str, set[str]] = {} 

447 chains = directory / CHAINS_DIR 

448 if not chains.is_dir(): 

449 return found 

450 for path in sorted(chains.glob("*.yaml")): 

451 try: 

452 data = yaml.safe_load(path.read_text()) 

453 except (OSError, ValueError, yaml.YAMLError): 

454 continue 

455 if not isinstance(data, dict) or not isinstance(data.get("nodes"), list): 

456 continue 

457 found[path.name] = { 

458 node["id"] for node in data["nodes"] if isinstance(node, dict) and "id" in node 

459 } 

460 return found

463def _chain_templates_check() -> dict: 

464 """Nodes this version's chain templates ship, missing from the live

466 overwritten (`cli.seed_home`), so a template shipped or changed after an 

467 operator's copy was seeded keeps missing the new node forever with 

468 nothing to say so. 

469

475 return _check("chain_templates", True, "skipped: not an installed Kraft", skipped=True)

478 return _check(

519def _plugin_checks(templates: Path) -> list[dict]: 

520 """The installed plugins as the files say, with no server running: what 

521 does not load and why, what waits for a person, and each auto-updating 

522 plugin's last outcome.""" 

523 from kraft.plugins import fetch 

524 from kraft.plugins import update as plugin_update 

525 from kraft.plugins.config import PluginsConfig, PluginsLock 

526 

527 if why := plugins_load.config_problem(templates): 

528 return [ 

529 _check("plugins", False, why), 

530 _check("plugin updates", True, "skipped: the plugin files do not read", skipped=True), 

531 ] 

532 root = plugins_load.plugins_dir() 

533 listed = PluginsConfig.load(templates / PluginsConfig.FILE) 

534 lock = PluginsLock.load(templates / PluginsLock.FILE) 

535 found = plugins_load.installed(templates, root, verify=True) 

536 broken = [f"{p.id}: {p.left_out}" for p in found if p.left_out is not None and not p.quiet] 

537 notes = [f"{p.id} {p.left_out}" for p in found if p.quiet and p.left_out != "is disabled"] 

538 auto = [] 

539 for plugin_id, locked in lock.plugins.items(): 

540 if plugin_id not in listed.plugins: 

541 continue 

542 name = plugin_id.split("@", 1)[1] 

543 collection = listed.collections[name] 

544 if listed.namespace(plugin_id) != locked.namespace: 

545 notes.append(f"{plugin_id}: alias change pending; run kraft admin plugin update") 

546 if collection.ref != locked.ref: 

547 notes.append(f"{plugin_id}: ref change pending; run kraft admin plugin update") 

548 if plugin_update._moved(collection, locked): 

549 notes.append(

865async def _mcp_check(server_up: bool) -> dict: 

866 """Is the Kraft MCP server registered with Claude Code? 

867 

868 Next to `_agent_checks` because it answers the same question: can this 

869 machine actually launch a worker. Every unsandboxed Claude launch names 

870 Kraft's permission tool, by the name `registration.permission_tool` 

871 resolves -- direct (`kraft admin init`, a committed `.mcp.json`) or the 

872 Kraft plugin's -- and a launch with none registered is refused by name. A 

873 plugin-only install passes: no `kraft admin init` needed. 

874 

875 `ok=False`, unlike `bd` or shell completion: those are choices an operator 

876 made, this one stops every Claude worker. 

877 

878 Rejected, as the bead records: passing `--mcp-config` on every launch, which 

879 would make Kraft write the operator's agent config. 

880 """ 

881 found = registration.permission_tool(None) 

882 asking = sorted(_direct_askers(harness.load(None))) 

883 needed = True 

884 if server_up: 

885 # Each connected repo on its own: its committed settings can switch 

886 # the plugin off for its workers, and its local scope or `.mcp.json` 

887 # can register the server where the user scope does not. 

888 entries = [ 

889 config.RepoEntry.model_validate(r, context={"unrecognised_keys_reported": True}) 

890 for r in await client.repos() 

891 ] 

892 per_repo = {Path(e.path): registration.permission_tool(Path(e.path)) for e in entries} 

893 refused = [str(repo) for repo, tool in per_repo.items() if tool is None] 

894 needed = _launches_direct_asker(entries) 

895 # Every repo refused and nothing at user scope is no registration at 

896 # all: it falls through to the same failure as no repo connected, 

897 # since the first such worker anywhere is refused -- when the chains 

898 # launch one at all. 

899 if refused and (found or len(refused) < len(per_repo) or not needed): 

900 # A warning, not a failure (Kraft-9efnk.31): the launch refuses 

901 # only an unsandboxed task on a harness asking through the direct 

902 # tool, and whether a repo's items ever launch one depends on 

903 # chain, sandbox and item overrides doctor does not resolve. 

904 return _check( 

905 "mcp server", 

906 True, 

907 f"no kraft MCP server registered for Claude workers in {', '.join(refused)}: " 

908 f"an unsandboxed task there on harness {' or '.join(asking)} is refused; " 

909 f"{registration.FIX} ({_MCP_SCOPE})", 

910 warn=True, 

911 ) 

912 found = found or next(iter(per_repo.values()), None) 

913 elif found is None: 

914 # The repo-scope half needs `GET /repos`, like every other 

915 # server-dependent check here. 

916 return _check("mcp server", True, "skipped: no server", skipped=True) 

917 if found: 

918 return _check("mcp server", True, f"{found[0]}, registered in {found[1]} ({_MCP_SCOPE})") 

919 if not needed: 

920 return _check( 

921 "mcp server", 

922 True, 

923 f"no kraft MCP server registered: no chain launches an unsandboxed task on harness " 

924 f"{' or '.join(asking)} here, which is the launch that needs it; {registration.FIX} " 

925 f"before one does ({_MCP_SCOPE})", 

926 warn=True, 

927 ) 

928 return _check( 

929 "mcp server", 

930 False, 

931 f"no kraft MCP server registered, so Claude workers are refused; {registration.FIX} " 

932 f"({_MCP_SCOPE})", 

933 )

1194def _credential_check(repo: config.RepoEntry, policy) -> dict | None: 

1195 """Which names this sandbox's egress proxy holds, on which 

1196 hosts (as each harness resolves them), and which 

1197 names a harness declares still pass through. A managed name with no value 

1198 fails: the proxy refuses every request that should carry it. The value is 

1199 looked for as a launch reads it, `worker_env` of this entry (or, for one 

1200 bound to a `source`, that name alone), in doctor's own environment, which 

1201 is the daemon's when both were started from the same shell.""" 

1202 if not policy.credentials: 

1203 return None 

1204 harnesses = harness.load(None).valid.values() 

1205 hosts: dict[str, set[str]] = {c.env: set() for c in policy.credentials} 

1206 for h in harnesses: 

1207 for cred in h.managed_credentials(policy.credentials): 

1208 hosts[cred.env].update(rule.domain for rule in cred.inject) 

1209 passing = sorted({c.env for h in harnesses for c in h.credentials} - hosts.keys()) 

1210 environ = worker_env(repo) 

1211 # A bound credential's value is the daemon's under its `source` alone. 

1212 missing = list( 

1213 dict.fromkeys( 

1214 c.source or c.env 

1215 for c in policy.credentials 

1216 # Set but empty is missing too: the launch refuses it the same. 

1217 if (not os.environ.get(c.source) if c.source else c.env not in environ) 

1218 ) 

1219 ) 

1220 phases: dict[str, set[str]] = {c.env: set() for c in policy.credentials} 

1221 for c in policy.credentials: 

1222 phases[c.env].update(c.phases()) 

1223 only = {name: f" ({p} only)" for name, ps in phases.items() if len(ps) == 1 for p in ps} 

1224 detail = "proxy-managed: " + "; ".join( 

1225 f"{name}{only.get(name, '')} on {', '.join(sorted(on)) or 'no host (sentinel only)'}" 

1226 for name, on in hosts.items() 

1227 ) 

1228 if passing: 

1229 detail += f"; passes through: {', '.join(passing)}" 

1230 # A name alone that no harness declares: nothing injects it, silently. 

1231 declared = {i: [c.env for c in h.credentials] for i, h in harness.load(None).valid.items()} 

1232 orphans = [ 

1233 c.env 

1234 for c in policy.credentials 

1235 if not c.inject and not any(c.env in names for names in declared.values()) 

1236 ] 

1237 if orphans: 

1238 by = "; ".join(f"{i} declares {', '.join(names)}" for i, names in declared.items() if names) 

1239 detail = ( 

1240 f"{', '.join(orphans)} is managed but no harness declares how to inject it, " 

1241 f"so it is only a sentinel; {by}; {detail}" 

1242 ) 

1243 if missing: 

1244 detail = f"no value for {', '.join(missing)} here, so its requests are refused; {detail}" 

1245 return _check(f"credentials {_label(repo)}", not missing, detail, warn=bool(orphans))

1327 # about the install, and every repo is measured against the same answer. 

1328 auto = _runs_forge_tasks()

1330 # instructions, the same shape `api.deps.library_steering` hands 

1331 # `steering.select`. `None` when the library itself does not load -- 

1332 # `_chains_check` already reports that failure. 

1333 live = config_dir() 

1334 profiles = _library_steering(live) 

1335 # Retyped from the wire: the checks below read the entry the loader 

1336 # models, not a dict whose keys each one must spell right. The loader's 

1337 # own unrecognised-key warning stays quiet: doctor fails a row on them. 

1338 repos = [ 

1339 config.RepoEntry.model_validate(r, context={"unrecognised_keys_reported": True}) 

1340 for r in await client.repos() 

1341 ] 

1342 networked = False 

1343 asked: set[str] = set() 

1344 for repo in repos: 

1345 path = Path(repo.path) 

1346 name = f"repo {_label(repo)}" 

1347 if not path.is_dir(): 

1348 checks.append(_check(name, False, f"{path} no longer exists")) 

1349 elif not (path / ".git").exists(): 

1350 # a file in a linked worktree, a directory in a normal clone 

1351 checks.append(_check(name, False, f"{path} is no longer a git repo")) 

1352 elif not (path / ".beads").is_dir():

1354 checks.append(_check(name, True, f"{path} — no .beads: work items here file no bead"))

1366 # deliberate; being told here rather than by a parked item is what

1384 # nothing is exactly what a typo looks like (Kraft-4hn34). A key an 

1385 # older Kraft wrote is not a typo: a warning naming what replaced
