104def neutral_run( 

105 argv: list[str], *, input: str | None = None, timeout: float 

106) -> subprocess.CompletedProcess[str]: 

107 """Run `argv` in an empty directory, outside the repo, with every prompt turned off, in 

108 its own session so a timeout kills what a credential helper started too. Output past 

109 MAX_OUTPUT kills it (OutputTooLarge).""" 

110 with tempfile.TemporaryDirectory(prefix="kraft-cred-") as cwd: 

111 proc = subprocess.Popen( 

112 argv, 

113 cwd=cwd, 

114 env=_neutral_env(cwd), 

115 stdin=subprocess.PIPE if input is not None else subprocess.DEVNULL, 

116 stdout=subprocess.PIPE, 

117 stderr=subprocess.PIPE, 

118 start_new_session=True, 

119 ) 

120 over = threading.Event() 

121 bufs = (bytearray(), bytearray()) 

122 

123 def drain(pipe, buf: bytearray) -> None: 

124 try: 

125 while chunk := os.read(pipe.fileno(), 65536): 

126 buf += chunk 

127 if len(buf) > MAX_OUTPUT: 

128 over.set() 

129 _kill(proc) 

130 return 

131 except (OSError, ValueError): 

132 pass 

133 

134 readers = [ 

135 threading.Thread(target=drain, args=(p, b), daemon=True) 

136 for p, b in zip((proc.stdout, proc.stderr), bufs, strict=True) 

137 ] 

138 for t in readers: 

139 t.start() 

140 if input is not None: 

141 try: 

142 proc.stdin.write(input.encode()) 

143 proc.stdin.close() 

144 except OSError: 

145 pass 

146 deadline = time.monotonic() + timeout 

147 try: 

148 for t in readers: 

149 t.join(max(0.0, deadline - time.monotonic())) 

150 if t.is_alive(): 

151 raise subprocess.TimeoutExpired(argv, timeout) 

152 proc.wait(max(0.0, deadline - time.monotonic())) 

153 except subprocess.TimeoutExpired: 

154 _kill(proc) 

155 for t in readers: 

156 t.join(1) # a descendant that left the session may still hold a pipe 

157 for pipe in (proc.stdin, proc.stdout, proc.stderr): 

158 if pipe: 

159 pipe.close() 

160 try: 

161 proc.wait(timeout=1) # reap it 

162 except subprocess.TimeoutExpired: 

163 pass 

164 if over.is_set(): 

165 raise OutputTooLarge from None 

166 raise 

167 if over.is_set(): 

168 proc.wait() 

169 raise OutputTooLarge 

170 out, err = (bytes(b).decode("utf-8", "replace") for b in bufs) 

171 return subprocess.CompletedProcess(argv, proc.returncode, out, err)

183async def _run(what: str, argv: list[str], input: str | None = None): 

184 try: 

185 return await asyncio.to_thread(neutral_run, argv, input=input, timeout=CREDENTIAL_TIMEOUT_S) 

186 except subprocess.TimeoutExpired:

188 except OutputTooLarge:

194async def git_credential(web_url: str) -> str | None: 

195 """The password git has saved for the host, or None when it has none.""" 

196 netloc = urlsplit(web_url).netloc 

197 done = await _run( 

198 "git credential fill", 

199 ["git", "credential", "fill"], 

200 f"protocol=https\nhost={netloc}\n\n", 

201 ) 

202 # With prompts off, real git says "terminal prompts disabled" and exits 128 

203 # when it has nothing saved; only that fixed phrase is read from stderr. 

204 if done.returncode == 128 and "terminal prompts disabled" in done.stderr: 

205 return None 

206 if done.returncode != 0: 

207 raise NoToken(f"git credential fill exited {done.returncode}; its output is not shown") 

208 return next( 

209 ( 

210 line[len("password=") :] 

211 for line in done.stdout.splitlines() 

212 if line.startswith("password=") 

213 ), 

214 None, 

215 )

218async def run_command(command: str) -> str: 

219 try: 

220 argv = shlex.split(command) 

221 except ValueError: 

222 raise NoToken("the token command is not valid shell-style quoting") from None 

223 if not argv: 

224 raise NoToken("the token command is empty") 

225 done = await _run("the token command", argv) 

226 if done.returncode != 0: 

227 raise NoToken(f"command exited {done.returncode}; its output is not shown") 

228 lines = done.stdout.strip().splitlines() 

229 if not lines: 

230 raise NoToken("the command printed nothing") 

231 if len(lines) > 1: 

232 raise NoToken("the command printed more than one line; its output is not shown") 

233 return lines[0].strip()

236def read_file( 

237 spec: str, *, repos: Sequence[RepoEntry], worktrees: Path, expose: Sequence[Path] 

238) -> str: 

239 # Inside: host_view imports credentials. 

240 from kraft.worker.host_view import system_dirs 

241 

242 expanded = os.path.expandvars(os.path.expanduser(spec)) 

243 if unset := re.findall(r"\$\{?(\w+)", expanded): 

244 raise NoToken(f"the token file path names ${unset[0]}, a variable that is not set") 

245 if "$" in expanded: 

246 raise NoToken("the token file path names a variable that is not set") 

247 path = Path(expanded) 

248 # Where a worker, a sandbox or the system could read it. 

249 real = Path(os.path.realpath(path)) 

250 roots = [ 

251 *(Path(r.path).expanduser() for r in repos), 

252 worktrees, 

253 *expose, 

254 Path("/etc"), 

255 *system_dirs(), 

256 ] 

257 for root in roots: 

258 for form in (root, Path(os.path.realpath(root))): 

259 if path == form or path.is_relative_to(form) or real.is_relative_to(form): 

260 raise NoToken(f"the token file is inside {form}, which a worker can read") 

261 # One descriptor for the checks and the read: no swap in between. 

262 try: 

263 fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK) 

264 except OSError as exc: 

265 if exc.errno == errno.ELOOP: 

266 raise NoToken("the token file is a symlink; name the file itself") from None 

267 raise NoToken("the token file does not exist or cannot be read") from None 

268 try: 

269 st = os.fstat(fd) 

270 if not stat.S_ISREG(st.st_mode): 

271 raise NoToken("the token file is not a regular file") 

272 if st.st_uid != os.getuid(): 

273 raise NoToken("the token file is owned by another user") 

274 if st.st_mode & 0o077: 

275 raise NoToken("the token file is readable by group or others; chmod 600 it") 

276 with os.fdopen(fd, encoding="utf-8") as f: 

277 fd = -1 

278 return f.read().strip() 

279 except (OSError, UnicodeDecodeError): 

280 raise NoToken("the token file cannot be read") from None 

281 finally: 

282 if fd >= 0: 

283 os.close(fd)

296async def read_token( 

297 account: ForgeAccount, 

298 *, 

299 repos: Sequence[RepoEntry], 

300 worktrees: Path, 

301 expose: Sequence[Path], 

302 ssh_hint: bool = False, 

303) -> str: 

304 source = account.source() 

305 if source.from_ == "git-credential": 

306 token = await git_credential(account.web_url) 

307 if token is None: 

308 host = urlsplit(account.web_url).netloc 

309 hint = "; your remote is SSH, so git never saved one" if ssh_hint else "" 

310 raise NoToken(f"git has no saved credential for {host}{hint}") 

311 elif source.from_ == "command": 

312 token = await run_command(source.command or "") 

313 elif source.from_ == "env": 

314 token = env_token(source.env or "") 

315 else: 

316 token = read_file(source.file or "", repos=repos, worktrees=worktrees, expose=expose) 

317 if any(c.isspace() or not c.isprintable() for c in token): 

318 raise NoToken("the token contains whitespace or a control character; it is not shown") 

319 if len(token) < redact.MIN_LEN: 

320 raise NoToken(f"the token is shorter than {redact.MIN_LEN} characters") 

321 redact.register(token) 

322 redact.register(basic_form(token, source.kind)) 

323 return token

326def take_env(names: Iterable[str]) -> list[str]: 

327 """Move each variable out of the process environment, so no child inherits it.""" 

328 taken = [] 

329 for name in names: 

330 if name in os.environ: 

331 _TAKEN[name] = os.environ.pop(name) 

332 if len(_TAKEN[name]) >= redact.MIN_LEN: 

333 redact.register(_TAKEN[name]) 

334 taken.append(name) 

335 return taken

371def set_secret_files(forges_yaml: Path, running: ForgesFile | None = None) -> None: 

372 """Token files stay out of every sandbox. As with env sources, a file that does not 

373 load is scanned as text. `running` are the accounts still in use: a refused reload 

374 must not unprotect their files.""" 

375 global _SECRET_FILES 

376 try: 

377 file = ForgesFile.load(forges_yaml) 

378 specs = [a.token.file for a in file.accounts.values() if a.token and a.token.file] 

379 except ConfigError: 

380 specs = _scan(forges_yaml, _FILE_PATH) 

381 if running: 

382 specs += [a.token.file for a in running.accounts.values() if a.token and a.token.file] 

383 # A directory is never a token (read_file refuses it); protecting one would hide all of it. 

384 # os.path.isdir, not Path.is_dir: a directory the server cannot enter must not stop boot. 

385 _SECRET_FILES = tuple(p for p in dict.fromkeys(map(_expand, specs)) if not os.path.isdir(p))
