47def fix_target(st, row, gate: str, node: str | None = None) -> dict: 

48 """Where a rejection at `gate` restarts the chain, for the review page's sentence. 

49 

50 `node` is the index `executor.reject_target` picks, the function 

51 `apply_rejection` runs, so the sentence and the action cannot disagree. 

52 `then` is the nodes between it and the gate; `round` is the one a rejection 

53 now would start, against the cap the counter snapshotted (the policy's when 

54 no rejection has happened yet). Raises ValueError for a `node` not before 

55 the gate, as `reject_target` does. 

56 """ 

57 nodes = store.effective_nodes(executor.chain_of(row), store.node_overrides_of(row)) 

58 idx = executor.gate_node_index(nodes, gate) 

59 target = executor.reject_target(nodes, idx, node) 

60 key = store.reject_loop_key(gate) 

61 counter = st.db.read(lambda c: store.read_counter(c, row["id"], key)) 

62 if counter: 

63 cap = counter["cap_attempts"] 

64 else: 

65 cap = st.policy.cap_for(key).attempts if st.policy else None 

66 return { 

67 "gate": gate, 

68 "node": nodes[target].id, 

69 "then": [n.id for n in nodes[target + 1 : idx]], 

70 "round": None 

71 if cap is None 

72 else {"n": (counter["count"] if counter else 0) + 1, "max": cap}, 

73 }

80#: The event types that bound a `work_item_needs_human` stop, newest wins — 

81#: the same shape of boundary `_pending_gate` models. A stop needs one because 

82#: `store.request_gate` also sets status 'needs_human' while appending no 

83#: `work_item_needs_human`: without this set a first-match scan reads an 

84#: answered-and-resumed stop, or a pending gate, as a live one forever.

87 GateEvent.REQUESTED,

96#: How long an `escalation_message` with no session row yet still reads as a 

97#: turn that is starting: the message is recorded before the turn launches, and 

98#: a launch that fails writes its own (failed) session row. 

99_LAUNCH_WINDOW_S = 60.0

102def _turn_live(session_status: str | None, message_at: str) -> bool: 

103 """Whether the escalation turn an `escalation_message` announced is still 

104 working: its session is pending or running, or has no row yet and the 

105 message is new. A turn that exited (failed, done, asked a question) is 

106 not live: the item is back with the person.""" 

107 if session_status is not None: 

108 return session_status in LIVE 

109 age = datetime.fromisoformat(store._now()) - datetime.fromisoformat(message_at) 

110 return age.total_seconds() < _LAUNCH_WINDOW_S

113def _stop_episode(st, wid: str) -> tuple[dict | None, bool]: 

114 """The `work_item_needs_human` payload of the stop the item is *currently* 

115 sitting on (or None if anything in `_STOP_BOUNDARY` superseded it), and 

116 whether an escalation turn is working on it: an `escalation_message` landed 

117 after that boundary event (an escalation from an earlier, already-superseded 

118 stop must not read as live) and that turn's session has not exited. One 

119 event scan for both, so `display_status` costs the detail route nothing 

120 beyond what `_current_stop` already read.""" 

121 evs = st.db.read(lambda c: events.read_after(c, 0, wid)) 

122 boundary = None 

123 for e in reversed(evs): 

124 if e["type"] in _STOP_BOUNDARY: 

125 boundary = e 

126 break 

127 stop = ( 

128 boundary["payload"] 

129 if boundary is not None and boundary["type"] == WorkItemEvent.NEEDS_HUMAN 

130 else None 

131 ) 

132 last = next((e for e in reversed(evs) if e["type"] == EscalationEvent.ESCALATION_MESSAGE), None) 

133 escalated = False 

134 if boundary is not None and last is not None and last["seq"] > boundary["seq"]: 

135 session = st.db.read( 

136 lambda c: c.execute( 

137 "SELECT status FROM worker_sessions WHERE id = ?", 

138 (last["payload"].get("session_id"),), 

139 ).fetchone() 

140 ) 

141 escalated = _turn_live(session["status"] if session else None, last["created_at"]) 

142 return stop, escalated

151def display_status( 

152 row, stop_kind: str | None, escalated: bool, pending_gate: str | None 

153) -> DisplayStatus: 

154 """The design vocabulary's status badge (Kraft Design Decisions §1, §14): 

155 exactly one of `archived`, `done`, `cancelled`, `paused`, `running`, 

156 `waiting`, `needs_you`, `escalated`, `failed`. The stored `status` column 

157 (`kraft.store.work_items`) is unchanged by this -- `display_status` is a 

158 read-side view over it, `stop_kind`, whether an escalation is live, and 

159 whether a gate is pending. 

160 

161 `infra` shows as `failed`, not `needs_you`: a daemon restart 

162 that lost a session, a git refresh that failed, or a forge that 403'd 

163 three times in a row (Decisions §14's own example) are all stops where no 

164 handler applies and no timer will change that -- auto-escalation did not 

165 or cannot run for them (Item States requirement 1). A `stuck` stop still 

166 counts as `needs_you` until its escalation turn actually starts, at which 

167 point `escalated` takes over. 

168 """ 

169 if row["archived_at"] is not None: 

170 return DisplayStatus.ARCHIVED 

171 shown = TRAITS[WorkItemStatus(row["status"])].display 

172 if shown is not None: 

173 return shown 

174 # needs_human: the badge depends on the stop. An unrecognised stored kind 

175 # (a downgrade, a future kind) is simply not a failure. 

176 if pending_gate: 

177 return DisplayStatus.NEEDS_YOU 

178 if escalated: 

179 return DisplayStatus.ESCALATED 

180 return DisplayStatus.FAILED if stop_kind in FAILURE_KINDS else DisplayStatus.NEEDS_YOU

223@api_router.get("/work-items") 

224async def list_work_items(request: Request): 

225 st = request.app.state 

226 # Read outside `_read`: that closure runs on the database thread and has no 

227 # request to ask. 

228 include_abandoned = request.query_params.get("include_abandoned") == "true" 

229 # Board counts and the default list exclude archived items entirely 

230 # (UI v2 · 03); ?archived=true flips to *only* archived, for the 

231 # Archived view. There is no third state ("both") -- nothing needs it. 

232 archived = request.query_params.get("archived") == "true" 

233 

234 def _read(c): 

235 rows = c.execute( 

236 "SELECT * FROM work_items WHERE (? OR status != ?) " 

237 "AND (archived_at IS NOT NULL) = ? ORDER BY created_at", 

238 (include_abandoned, WorkItemStatus.ABANDONED, archived), 

239 ).fetchall() 

240 cursor = c.execute("SELECT COALESCE(MAX(seq), 0) FROM events").fetchone()[0] 

241 # The latest gate request or gate closer per item, in one pass — the 

242 # board renders a gate prompt per row and must not offer Approve on a 

243 # rejected gate. The closers are `executor.pending_gate`'s, so a gate 

244 # `kraft item skip` passed, or one left by an item that ended, reads 

245 # as closed here too, as it does on the item's own page. 

246 gate_types = (GateEvent.REQUESTED, *GATE_CLOSED) 

247 gates = c.execute( 

248 "SELECT work_item_id, type, payload FROM events WHERE seq IN (" 

249 " SELECT MAX(seq) FROM events" 

250 f" WHERE type IN ({','.join('?' * len(gate_types))})" 

251 " GROUP BY work_item_id)", 

252 gate_types, 

253 ).fetchall() 

254 # The latest fallback switch and session start per item: the card 

255 # marks an item whose last launch ran on a fallback (Kraft-0a3h8). 

256 launches = c.execute( 

257 "SELECT work_item_id, type, payload FROM events WHERE seq IN (" 

258 " SELECT MAX(seq) FROM events" 

259 " WHERE type IN (?, ?)" 

260 " GROUP BY work_item_id, type)", 

261 (LimitEvent.LAUNCH_FALLBACK, SessionEvent.WORKER_SESSION_STARTED), 

262 ).fetchall() 

263 # The latest `_STOP_BOUNDARY` event per item, and the latest 

264 # `escalation_message` per item -- `display_status`/`stop` (B.4) need 

265 # both, and the list reads no sessions per row, so this is the one 

266 # grouped query that stands in for `_stop_episode`'s per-item scan. 

267 boundaries = c.execute( 

268 "SELECT work_item_id, type, payload, seq FROM events WHERE seq IN (" 

269 f" SELECT MAX(seq) FROM events WHERE type IN ({','.join('?' * len(_STOP_BOUNDARY))})" 

270 " GROUP BY work_item_id)", 

271 _STOP_BOUNDARY, 

272 ).fetchall() 

273 escalations = c.execute( 

274 "SELECT e.work_item_id, e.seq, e.created_at, s.status AS session_status FROM events e " 

275 "LEFT JOIN worker_sessions s ON s.id = json_extract(e.payload, '$.session_id') " 

276 "WHERE e.seq IN (SELECT MAX(seq) FROM events WHERE type = ? " 

277 "GROUP BY work_item_id)", 

278 (EscalationEvent.ESCALATION_MESSAGE,), 

279 ).fetchall() 

280 # `mr_ref` and the step's task: the same grouped-query trade as the 

281 # rest, so the list stays one pass instead of `_mr_ref`'s per-item scan. 

282 mr_events = c.execute( 

283 "SELECT work_item_id, payload FROM events WHERE seq IN (" 

284 " SELECT MAX(seq) FROM events WHERE type = ? GROUP BY work_item_id)", 

285 (ForgeEvent.MR_OPENED,), 

286 ).fetchall() 

287 roots = c.execute( 

288 "SELECT work_item_id, mr_ref FROM work_item_repos WHERE role = 'root'" 

289 ).fetchall() 

290 tasks = c.execute( 

291 "SELECT s.work_item_id, s.hook_point FROM worker_sessions s" 

292 " JOIN work_items w ON w.id = s.work_item_id AND w.current_node_id = s.node_id" 

293 " WHERE s.hook_point != 'escalation' ORDER BY s.created_at" 

294 ).fetchall() 

295 return rows, cursor, gates, launches, boundaries, escalations, mr_events, roots, tasks

442def _judge_stop_notes(st, wid: str) -> list[dict]: 

443 """Findings a judge chose to stop chasing (`stop_downgrade`), for the 

444 human at review. 

445 

446 Distinct from `_deferred_findings`: these are `critical`/`important` 

447 findings a judge decided not to keep looping on, not the genuinely 

448 `minor` ones that never entered the loop at all -- conflating the two 

449 would make a judge-skipped real bug look like routine minor-finding 

450 triage. 

451 

452 Bounded at the last resolved gate, the same way `_concerns` below is and 

453 for the same reason (Kraft-ub2): once a human approves the gate that 

454 showed this note, re-posing it at every later gate the item reaches asks 

455 them to answer for it again. 

456 """ 

457 out: list[dict] = [] 

458 for e in reversed(st.db.read(lambda c: events.read_after(c, 0, wid))): 

459 if e["type"] in (GateEvent.APPROVED, GateEvent.REJECTED): 

460 break 

461 if ( 

462 e["type"] == ChainEvent.JUDGE_VERDICT 

463 and e["payload"].get("verdict") == "stop_downgrade" 

464 ): 

465 out.append( 

466 { 

467 "node_id": e["payload"].get("node_id"), 

468 "reasoning": e["payload"].get("reasoning", ""), 

469 "findings": e["payload"].get("findings", []), 

470 } 

471 ) 

472 out.reverse() # oldest first 

473 return out

476def _concerns(st, wid: str) -> list[str]: 

477 """`done_with_concerns` text from every session that reported one, oldest 

478 first — the same shape of thing as `_deferred_findings` (something a 

479 machine noticed and owes a human before approval), read from the event log 

480 `adapters.subprocess.run_task` stamps at session exit, never from 

481 `result_path` on disk. 

482 

483 Bounded at the last resolved gate, the way `_stop_reason` is: spec §2 puts a 

484 concern at the *next* gate, and the detail screen now passes concerns to 

485 every gate rather than only `human_review_approval`. Without a boundary one 

486 concern would be re-posed at every later gate the item reaches, long after 

487 the human who approved that gate already answered for it (Kraft-ub2). 

488 

489 Excludes judge sessions the same way `walk._diagnosis_bundle` does: a judge 

490 is told `concerns` is required on every verdict (JUDGE_PROMPT/SKILL.md), so 

491 a plain `continue` verdict's reasoning would otherwise show up here as a 

492 concern the human owes an answer for, rather than the judge's own routine 

493 chatter. 

494 """ 

495 row = st.db.read( 

496 lambda c: c.execute("SELECT * FROM work_items WHERE id = ?", (wid,)).fetchone() 

497 ) 

498 snapshot = store.materialized_chain_of(row) if row is not None else None 

499 # A judge's session is recorded under its canonical path 

500 # (`<node>.fix_loop.judge`), the key `walk._diagnosis_bundle` skips too. 

501 judges = [n.judge.path for n in snapshot.chain.nodes if n.judge] if snapshot else [] 

502 judge_session_ids = st.db.read( 

503 lambda c: { 

504 r["id"] 

505 for r in c.execute( 

506 "SELECT id FROM worker_sessions WHERE work_item_id = ? AND hook_point IN " 

507 f"({','.join('?' * len(judges))})", 

508 (wid, *judges), 

509 ).fetchall() 

510 } 

511 ) 

512 out: list[str] = [] 

513 for e in reversed(st.db.read(lambda c: events.read_after(c, 0, wid))): 

514 if e["type"] in (GateEvent.APPROVED, GateEvent.REJECTED): 

515 break 

516 if ( 

517 e["type"] == SessionEvent.WORKER_SESSION_EXITED 

518 and e["payload"].get("concerns") 

519 and e["payload"].get("session_id") not in judge_session_ids 

520 ): 

521 out.append(e["payload"]["concerns"]) 

522 out.reverse() # oldest first 

523 return out

526def _mr_ref(st, wid: str) -> dict | None: 

527 """The most recent `mr_opened` event's `{number, url}`, or None before 

528 `open_mr` has ever run. 

529 

530 A single-repo item gets no `work_item_repos` row (`repos_for`'s own 

531 docstring), so its merge request has nowhere to live but the event log -- 

532 `adapters.forge.run_task` emits `mr_opened` for exactly this reason 

533 (Kraft-d2sq: "an event, not a work-item column"). Reversed scan, first hit 

534 wins, same as `_stop_reason`: a retried `open_mr` that reused the existing 

535 MR still logs a fresh event, so this always names the current one, not a 

536 stale first-open URL for a since-force-pushed branch. 

537 

538 A multi-repo item's `open_mr` node emits one `mr_opened` per target repo 

539 (root and every declared submodule), and the event carries no repo 

540 identifier to tell them apart, so the events are never read for one: the 

541 latest would as often name a submodule's merge request as the root's. 

542 Its root's own row records the root's merge request (Kraft-mjsf), and 

543 that is the item's; a root with none has no one merge request to link. 

544 """ 

545 repo_rows = st.db.read(lambda c: store.repos_for(c, wid)) 

546 root = next((r["mr_ref"] for r in repo_rows if r["role"] == "root"), None) 

547 last = next( 

548 ( 

549 e["payload"] 

550 for e in reversed(st.db.read(lambda c: events.read_after(c, 0, wid))) 

551 if e["type"] == ForgeEvent.MR_OPENED 

552 ), 

553 None, 

554 ) 

555 return _pick_mr_ref(bool(repo_rows), root, last)

727def _step_of(row, chain: dict, task: str | None, *, with_task: bool = False) -> dict | None: 

728 """The current node's step (1-based) out of its steps, when it declares 

729 more than one and `task` (a `node.step.task` path) is in one of them; 

730 `with_task` adds the step's id and the task's, which the list row draws 

731 as `step › task` (the detail's `summary.step` does not).""" 

732 node = next((n for n in chain.get("nodes") or [] if n["id"] == row["current_node_id"]), None) 

733 steps = node.get("steps") if node else None 

734 if not steps or len(steps) < 2: 

735 return None 

736 index = next((i for i, group in enumerate(steps) if task in group), None) 

737 if index is None: 

738 return None 

739 step = {"index": index + 1, "count": len(steps)} 

740 if not with_task: 

741 return step 

742 parts = task.split(".", 2) 

743 # A path with no step or task segment (a hand-written chain) names neither. 

744 return {**step, "name": parts[1], "task": parts[2]} if len(parts) == 3 else step

1022def _sessions_out(sessions, skipped: frozenset[str], nodes: dict[str, str]) -> list[dict]: 

1023 """The item's sessions as the API gives them. `skipped` marks the newest 

1024 session of a task under a skipped path, or of a node skipped after it 

1025 began, unless it had already finished: its stored status is only where the 

1026 skip found it. A fanned-out task has a newest session per repository, and 

1027 an escalation turn is no task of the node: a node skip does not mark it.""" 

1028 newest = {(s["hook_point"], s["repository"]): s["id"] for s in sessions} 

1029 

1030 def cut_short(s) -> bool: 

1031 hook = s["hook_point"] 

1032 return ( 

1033 newest[hook, s["repository"]] == s["id"] 

1034 and not s["status"].startswith("done") 

1035 and ( 

1036 any(hook == p or hook.startswith(p + ".") for p in skipped) 

1037 or ( 

1038 hook.startswith(s["node_id"] + ".") 

1039 and not hook.endswith(".escalation")
