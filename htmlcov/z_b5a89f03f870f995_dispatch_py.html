81def _path_matches(path: str, patterns: list[str]) -> bool: 

82 """Gitignore-style precedence over one scope's own `paths` list: the last 

83 pattern `path` matches wins, so a later `!excl/**` overrides an earlier 

84 inclusive glob (test-scope design §3.4).""" 

85 matched = False 

86 for pattern in patterns: 

87 if pattern.startswith("!"): 

88 if fnmatch.fnmatchcase(path, pattern[1:]): 

89 matched = False 

90 elif fnmatch.fnmatchcase(path, pattern): 

91 matched = True 

92 return matched

95def _matched_scopes(scopes: list[dict], changed_paths: list[str]) -> list[dict]: 

96 """Scopes covering `changed_paths`, deduped and in `scopes`' own order. 

97 

98 Fails open in both directions the design calls out: a changed path 

99 matching no scope at all, or an empty diff to begin with, runs every 

100 scope rather than guess. Under-testing is the bug this exists to close; 

101 it must never reopen it here (test-scope design §3.5). 

102 """ 

103 hit: set[int] = set() 

104 for path in changed_paths: 

105 path_hit = {i for i, scope in enumerate(scopes) if _path_matches(path, scope["paths"])} 

106 if not path_hit: 

107 return list(scopes) 

108 hit |= path_hit 

109 return [scopes[i] for i in sorted(hit)] if hit else list(scopes)

123def _last_own_round_head( 

124 db, work_item_id: str, node_id: str, task_hook: str, round: int 

125) -> str | None: 

126 """The head `round - 1`'s dispatch of `task_hook` on this exact `node_id` 

127 ran at, if every scope in that dispatch finished `done` -- otherwise 

128 `None` (C7 review fix, Kraft-s7c04.14). 

129 

130 `prompts.last_review_session` is the wrong source for this: it is keyed 

131 on `(work_item_id, hook_point)` alone, not `node_id`, and takes the 

132 latest `done` row regardless of its siblings. For a while (C1, 

133 Kraft-s7c04.8, reverted 2026-09-16) `on.test.run` was dispatched by both 

134 `implementation` and `verify` under that one hook_point, with one 

135 session row per scope -- so that query would have let one node's head 

136 stand in for the other's (verify's round 0 reading C1's head and seeing 

137 an empty diff, breaking the spec's "first round selects from the full 

138 branch diff"). That reason for scoping to `node_id` is gone with C1, but 

139 the other one is independent and still live: it also stops one passing 

140 scope's row from standing in for a dispatch whose sibling scope failed 

141 (the `retry_after_cap` path: scope A failed, scope B passed, the counter 

142 reset wipes the round, and a later HEAD move would otherwise be missed 

143 because a passing B's row still reads as "reviewed"). Scoping to this 

144 node's own previous round, and requiring every row in it to be `done` 

145 with the same `head_sha`, closes both. Coarser than tracking exactly 

146 which scope(s) failed: precise per-scope identity would need scope 

147 identity persisted on the session row, which this batch does not add. 

148 Treating *any* failure (or an inconsistent head across siblings) as "not 

149 clean" can only over-select, matching `_matched_scopes`' own fail-open 

150 posture. 

151 """ 

152 rows = [ 

153 r 

154 for r in db.read(lambda c: store.sessions_for_round(c, work_item_id, node_id, round - 1)) 

155 if r["hook_point"] == task_hook 

156 ] 

157 if not rows or any(r["status"] != "done" for r in rows): 

158 return None 

159 heads = {r["head_sha"] for r in rows} 

160 return heads.pop() if len(heads) == 1 else None

163def _select_scopes( 

164 db, 

165 work_item_id: str, 

166 worktree, 

167 node_id: str, 

168 task_hook: str, 

169 round: int, 

170 repo_entry: RepoEntry | None, 

171) -> list[dict]: 

172 """The scopes the `kraft.verify_changed_test_scopes` builtin should run 

173 (test-scope design §3.2-3.3, C7 Kraft-s7c04.14). Their sandbox is the 

174 item's (`item_sandbox`). 

175 

176 Pulled out of `dispatch_node`'s subprocess branch (batch-c1 spec Task 3). 

177 C1 briefly gave `implementation` its own `on.test.run` dispatch through 

178 this same function, so the two computed scopes identically by 

179 construction; C1 was reverted (Kraft-s7c04.8, 2026-09-16) because a gate 

180 dispatched after the session exits cannot be acted on by it. `verify` is 

181 this function's only caller now, and the same-selection property C1 was 

182 after is instead conveyed to the implementation agent as a prompt note 

183 (`prompts.scope_note`) built from the repo's scope table, so it can apply 

184 the same rule to its own diff before it finishes. 

185 

186 The commands are the repo's, and only the repo's. V1 has no registry to 

187 fall back to and deliberately no command on the task: the task names an 

188 action Kraft owns, and what that action runs is a property of the 

189 repository (`repository-area-can-declare-setup-and-test-scopes`). A 

190 hardcoded single command is how verify ends up running something CI does 

191 not, or the wrong stack's suite entirely (Kraft-579, Kraft-9wzy). 

192 A bare `test_command` is read as a single `["**"]` scope, so this is one 

193 shape regardless of which field an operator set; a repo that declares 

194 neither returns no scopes at all and the caller stops for a human rather 

195 than inventing a command. 

196 

197 C7: round 0 (and a round-0 re-entry, e.g. after `retry_after_cap` wipes 

198 the round counter) always selects from the whole branch diff since 

199 `base_ref` -- the spec's first acceptance criterion, and the only choice 

200 that can't mistake a different node's gate dispatch, or a still-open 

201 round, for "already reviewed". Only a round that follows this exact 

202 `(node_id, task_hook)`'s own fully clean previous round (`done`, every 

203 sibling scope, one shared `head_sha` -- `_last_own_round_head`) narrows 

204 the diff to what changed since then, because `6c712ea8` ran a 

205 13-minute `just ci-test` for a frontend-only fix. That alone would 

206 under-test once C2 lets a round go red on more than one scope: a round 

207 following any failure at this hook does not trust the incremental diff 

208 at all and falls back to the whole-branch diff (`_last_own_round_head` 

209 itself returns `None` for an unclean round, which collapses to 

210 `base_ref` below), unioning in every scope from that round rather than 

211 only the one(s) that failed. Selecting less is only ever safe for a 

212 scope that passed. 

213 """ 

214 if repo_entry is None: 

215 return [] 

216 repo_scopes = [s.model_dump() for s in repo_entry.test_scopes or ()] 

217 if not repo_scopes and repo_entry.test_command: 

218 # Wrapped here, at the point of use, never in the loaded entry: a 

219 # re-save would persist it (`config.TestScope`, Kraft-9wzy). 

220 repo_scopes = [{"paths": ["**"], "command": repo_entry.test_command}] 

221 # One table after resolution: the repository's scopes, then each area's, 

222 # every area scope carrying the setup it needs first 

223 # (`repository-area-can-declare-setup-and-test-scopes`). 

224 area_scopes = [ 

225 {**scope, "area": name, "setup": area.get("setup")} 

226 for name, area in repo_entry.areas.items() 

227 for scope in (area.get("verification") or {}).get("test_scopes") or [] 

228 ] 

229 if not repo_scopes and not area_scopes: 

230 return [] 

231 scopes = [ 

232 { 

233 "paths": s["paths"], 

234 "cmd": shlex.split(s["command"]), 

235 "area": s.get("area"), 

236 "setup": shlex.split(s["setup"]) if s.get("setup") else None, 

237 } 

238 for s in [*(repo_scopes or []), *area_scopes] 

239 ] 

240 base_ref = _current_base_ref(db, work_item_id) 

241 # `since` is the diff's lower bound: round <= 0 always measures the whole 

242 # branch (never reads another round or another node's head), and only a 

243 # round that follows this exact node's own fully clean previous round at 

244 # this hook narrows it -- everything else (an unclean previous round, a 

245 # `since` `git` no longer knows about, no previous round at all) falls 

246 # through to the whole branch. 

247 since = _last_own_round_head(db, work_item_id, node_id, task_hook, round) if round > 0 else None 

248 if ( 

249 since 

250 and _config.git_read(Path(worktree), "rev-parse", "--verify", f"{since}^{{commit}}") is None 

251 ): 

252 since = None 

253 since = since or base_ref 

254 diff = ( 

255 _config.git_read(Path(worktree), "diff", "--name-only", f"{since}...HEAD") 

256 if since 

257 else None 

258 ) 

259 to_run = scopes if diff is None else _matched_scopes(scopes, diff.splitlines()) 

260 return to_run

331def item_sandbox(row, launch: LaunchContext | None) -> dict | None: 

332 """The sandbox every project-controlled launch of this item runs in, or 

333 None for an item nothing sandboxes -- the one resolution (Ruling 189). 

334 Every task, recovery, judge, escalation turn, gate review, test scope, 

335 area setup and the repository's `setup_command` reads it here: a sandbox 

336 wraps the item, not the scope that set it, because once one task has run 

337 in it the worktree is the worker's to write (Kraft-p8nem, Kraft-h10e5). 

338 

339 Whichever scope of the snapshot froze one wins, and the entry's live value 

340 -- `false` included -- cannot turn it off (Ruling 105: `sandbox` only 

341 tightens); without one, the entry's live value applies. 

342 `SandboxUnresolved` when that cannot be told: a snapshot freezing two 

343 (filed before the ruling), repositories setting two live, or a poisoned 

344 `repos.yaml` -- unreadable is never "no sandbox".""" 

345 snapshot = store.materialized_chain_of(row) 

346 try: 

347 frozen = snapshot.item_sandbox() if snapshot is not None else None 

348 if frozen is not None: 

349 return frozen.model_dump() 

350 # Every repository the item launches against, members included: a 

351 # member's live sandbox wraps the root's runs too. 

352 entries = [launch.repo_entry, *launch.repositories.values()] if launch else [] 

353 live = list(dict.fromkeys(s for e in entries if e and (s := e.effective_sandbox))) 

354 except (_policy.PolicyError, _config.ConfigError) as exc: 

355 raise SandboxUnresolved(f"cannot tell whether {row['id']} runs sandboxed: {exc}") from exc 

356 if len(live) > 1: 

357 raise SandboxUnresolved( 

358 f"{row['id']}'s repositories set different sandboxes " 

359 f"{[s.model_dump() for s in live]!r} in repos.yaml: " 

360 "a sandbox wraps the whole work item (Ruling 189), so they must agree" 

361 ) 

362 return live[0].model_dump() if live else None

450 return await _subprocess.run_task( 

451 db, 

452 run_dirs, 

453 cmd=cmd, 

454 cwd=worktree,

457 # the same worktree. A .pyc written on an earlier cycle has the same 

458 # second-resolution mtime and (often) size as the fixed source, so 

459 # CPython would import the stale bytecode and the re-measure would

467 # An area's setup runs once, before the first of its scopes -- including 

468 # an area nobody chose at intake that the changed paths selected anyway

474 if scope["setup"]: 

475 if scope["area"] not in setups: 

476 setups[scope["area"]] = asyncio.ensure_future(_run(scope["setup"])) 

477 ready = await setups[scope["area"]] 

478 if ready != "done": 

479 return ready 

480 return await _run(scope["cmd"]) 

481 

482 if execution is ExecutionMode.PARALLEL: 

483 results = list(await asyncio.gather(*(_scope(s) for s in to_run))) 

484 else: 

485 results = [] 

486 for scope in to_run: 

487 status = await _scope(scope) 

488 results.append(status) 

489 if status in _SCOPE_STOP_STATUSES: 

490 break

492 if stopped is not None: 

493 return stopped 

494 return next((s for s in results if s != "done"), "done")

1369 async def _recover(step: ResolvedStep, results: list) -> tuple[list, tuple | None]: 

1370 """`results` after this step's own handlers ran, or the handler's own 

1371 stop when one could not finish (a pause, a config error, a rate limit 

1372 ...): that is the verdict, not the task failure it was answering."""

1377 if not failed:

1380 if not task.on_failure or task.path in spent: 

1381 continue 

1382 spent.add(task.path) 

1383 verdict, h_failed, h_excs = await _handle(task.on_failure, "task", [task]) 

1384 if verdict == "ok": 

1385 (status[task.path],) = await _settle([task], step, retry=True) 

1386 elif verdict != "failed": 

1387 return results, (verdict, h_failed, h_excs) 

1388 rest = [t for t in failed if not t.on_failure] 

1389 if rest and step.on_failure and step.path not in spent: 

1390 spent.add(step.path) 

1391 verdict, h_failed, h_excs = await _handle(step.on_failure, "step", rest) 

1392 if verdict == "ok":

1394 return await _settle(step.tasks, step, retry=True), None 

1395 if verdict != "failed": 

1396 return results, (verdict, h_failed, h_excs) 

1397 return [status[t.path] for t in step.tasks], None

1419 # precisely because it must not run against an unsettled earlier one. 

1420 # `_ADVANCING` (executor/context.py) is the existing definition of 

1421 # "this task moved the node forward": ("done", "done_with_concerns").

1636def collect_findings( 

1637 db, 

1638 work_item_id: str, 

1639 node: ResolvedNode, 

1640 round: int, 

1641 *, 

1642 steps: tuple[ResolvedStep, ...] | None = None, 

1643): 

1644 """(findings, task paths that reported at least one) for one cycle. 

1645 

1646 Only the node's own measuring tasks: the fix task is dispatched with 

1647 `round=count` and the next measuring pass runs at that same round, so an 

1648 unfiltered query folds the fix agent's result file into the cycle. Only the 

1649 most recent row per hook point, because a re-entry can measure at a round a 

1650 previous pass already used -- `round` seeds from the persisted fix-loop 

1651 counter, which a gate rejection or a wait re-entry does not clear, and a 

1652 `/retry` deletes the counter row so the next pass restarts at 1 instead. 

1653 Either way stale rows sit at the same number. 

1654 

1655 A task that failed without writing a findings file at all -- the 

1656 changed-test-scope builtin is the common case, a subprocess with no findings 

1657 schema to write to -- gets a synthesized `Finding` via `_findings.from_blind_failure` instead 

1658 of vanishing (traced live on a work item that spun for 7 cycles on an 

1659 identical, invisible test failure). `reported` is not extended for it: 

1660 that set means "wrote a real, parseable result file", which a synthesized 

1661 finding does not change -- `walk.py`'s `blind_failures` still computes the 

1662 same way and still forces the loop open, now redundantly with `eligible`, 

1663 which is harmless. 

1664 

1665 One task path can carry more than one row in a single round -- the 

1666 changed-test-scope builtin mints one session per matching scope, all under 

1667 this same path (the identity problem, batch-c1 spec's "The identity 

1668 problem underneath all three"). A bare "last wins" read would let a later 

1669 scope's pass silently overwrite an earlier scope's real failure once C2 

1670 stops the loop from short-circuiting on it. Every row for the *latest* 

1671 `head_sha` this hook_point dispatched at is read instead of just the last 

1672 one -- every scope from one dispatch shares that head_sha (stamped once, 

1673 before the loop, in `dispatch_node`'s subprocess branch), so this still 

1674 collapses to the single most recent row for a hook that only ever mints 

1675 one (a re-entry at a *different* head still keeps last-wins there). 

1676 

1677 That multi-row read is scoped to builtin tasks -- the only ones a scope 

1678 loop mints more than one session for -- and not to an agent reviewer. A 

1679 reviewer that exits `needs_context` stops before 

1680 `bump_counter`, so a resume re-enters the same round at the same head: 

1681 without this filter, the stale `needs_context` row (no findings file) 

1682 would still share the latest head_sha, hit the `_FAILING_STATUSES` branch 

1683 below, and mint a bogus `from_blind_failure` critical finding alongside 

1684 the real, later row. An agent hook keeps a plain last-wins read instead. 

1685 

1686 Every finding this returns -- parsed or synthesized -- carries a 

1687 `Finding.jobs` entry per session that produced it, so a fix agent can 

1688 always pull a job's full session output directly (Kraft-s7c04.34/.35 

1689 brainstorm). A hook's several failing rows in one round (the scope-loop 

1690 case above) become ONE synthesized `Finding` with one `JobRef` per 

1691 failing job, not several near-identical findings -- the fix agent gets 

1692 one coherent notice about the hook, not a fragmented list. 

1693 """ 

1694 tasks = measured_tasks(node, steps) 

1695 rows = db.read(lambda c: store.sessions_for_round(c, work_item_id, node.id, round)) 

1696 by_hook: dict[str, list[sqlite3.Row]] = {} 

1697 for row in rows: 

1698 if row["hook_point"] in tasks: 

1699 by_hook.setdefault(row["hook_point"], []).append(row) # ordered by created_at 

1700 found: list[_findings.Finding] = [] 

1701 reported: set[str] = set() 

1702 for hook, hook_rows in by_hook.items(): 

1703 is_scope_loop = isinstance(tasks[hook].task, BuiltinTask) 

1704 latest_head = hook_rows[-1]["head_sha"] 

1705 rows_to_read = ( 

1706 [row for row in hook_rows if row["head_sha"] == latest_head] 

1707 if is_scope_loop 

1708 else hook_rows[-1:] 

1709 ) 

1710 blind_jobs: list[_findings.BlindJob] = [] 

1711 for row in rows_to_read: 

1712 parsed = _findings.parse(row["result_path"]) 

1713 if parsed: 

1714 reported.add(hook) 

1715 job = _findings.JobRef( 

1716 label=hook, log_ref=_findings.session_log_ref(work_item_id, row["id"]) 

1717 ) 

1718 found.extend(replace(f, jobs=(job,)) for f in parsed) 

1719 elif row["status"] in _FAILING_STATUSES: 

1720 task = tasks[hook].task 

1721 # An agent's recorded command is its whole argv, prompt and all: 

1722 # nothing to re-run, and a prompt that carries last round's 

1723 # findings (`inputs: [carried_findings]`) would fold into this 

1724 # finding and grow it every round. 

1725 command = ( 

1726 None 

1727 if isinstance(task, AgentTask) 

1728 else row["command"] 

1729 or (task.command if isinstance(task, SubprocessTask) else None) 

1730 ) 

1731 blind_jobs.append( 

1732 _findings.BlindJob( 

1733 session_id=row["id"], log_path=row["log_path"], command=command 

1734 ) 

1735 ) 

1736 if blind_jobs: 

1737 found.append(_findings.from_blind_failure(hook, work_item_id, blind_jobs)) 

1738 return found, reported

1741def needs_context_question( 

1742 db, work_item_id: str, node: ResolvedNode, round: int, *, first_iteration: bool = False 

1743) -> str | None: 

1744 """The question from a `needs_context` row in this round, or None. 

1745 

1746 Same latest-row-per-hook-point read as `collect_findings` (a resume or 

1747 `/retry` re-enters with stale rows still sitting there, so a first-match 

1748 scan could re-stop the item on a historical row forever) but deliberately 

1749 NOT its "only the measured tasks" filter: the fix loop's own tasks are 

1750 dispatched with this same round, and including them is exactly how a fix 

1751 task's own `needs_context` is meant to surface, one iteration later. 

1752 

1753 `first_iteration` is the exception to that, and only that: on the entry's 

1754 very first pass there is by definition no fix from *this* entry yet, so 

1755 any fix row at this round belongs to a bygone one. It matters because 

1756 `walk_node` now seeds `round` from the persisted counter, and it is also 

1757 re-entered on paths that are not resumes -- a gate rejection walking back 

1758 to a fix_loop node, and the wait scheduler -- where the `retry_counters` 

1759 row survives (only `retry_after_cap` deletes it). Without this, the last 

1760 pass's fix question would stop the new pass before it ran a single cycle: 

1761 the same stranding `ESCALATION_HOOK` below closes, through a third door. 

1762 Defaults off so every other caller keeps today's behaviour. 

1763 

1764 The fix loop's judge is the one exception. The judge is a brake bolted onto the 

1765 cap and never a second way to get stuck (`_judge_result` fails open 

1766 in-process), but its session row persists -- so without this filter a 

1767 judge that exited `needs_context` would strand the item on the judge's 

1768 own question at the next re-entry, which is exactly the design's 

1769 forbidden case arriving one iteration late. 

1770 

1771 `ESCALATION_HOOK` is excluded for the same reason, through a different 

1772 door. An escalation is a conversation with a human, not a measurement: 

1773 its `needs_context` *is* the question that opened the conversation, so 

1774 re-reading it once the human has answered re-stops the node on a question 

1775 that has already been settled -- forever, because nothing ever rewrites 

1776 that row. On e983d85c that cost $4.86 and 38 minutes, the human's answer 

1777 reviewed and discarded twice with no code changed, and the item was then 

1778 skipped with a CRITICAL finding still open (Kraft-7itv follow-up). 

1779 """ 

1780 skip = {ESCALATION_HOOK} | ({node.judge.path} if node.judge is not None else set()) 

1781 if first_iteration: 

1782 skip |= {t.path for step in node.fix_loop for t in step.tasks} 

1783 rows = db.read(lambda c: store.sessions_for_round(c, work_item_id, node.id, round)) 

1784 latest: dict[str, sqlite3.Row] = {} 

1785 for row in rows: 

1786 if row["hook_point"] in skip: 

1787 continue 

1788 latest[row["hook_point"]] = row # ordered by created_at, so last wins 

1789 for row in latest.values(): 

1790 if row["status"] == "needs_context": 

1791 return _subprocess.read_question(Path(row["result_path"])) or "(no question given)" 

1792 return None

1805def previous_fix_session(db, work_item_id: str, node: ResolvedNode) -> sqlite3.Row | None: 

1806 """The most recently dispatched fix task for this node, or None if none 

1807 has run yet. 

1808 

1809 Deliberately NOT scoped to a `round` passed in by the caller: `round` is 

1810 `walk_node`'s own local counter, and on a fresh entry into that function it 

1811 seeds from the persisted `retry_counters` row -- so it lands back on a 

1812 number a *previous* pass over this node already used (a gate rejection 

1813 walking back here, or the wait scheduler, neither of which clears the 

1814 counter), or on 0 after a `/retry` that deleted the row so the next 

1815 `bump_counter` restarts at 1. Either way the `worker_sessions` rows from 

1816 before that re-entry are still in the table. A lookup keyed on the caller's 

1817 local round can therefore collide with an abandoned attempt that happens to 

1818 land on the same round number (worse than nothing: it hands over a 

1819 plausible-looking file from a cycle that was already exhausted), or, after 

1820 a retry reset, miss every previous attempt outright. Ordering by 

1821 `created_at` and taking the last row sidesteps both: whichever fix task 

1822 actually ran most recently for this node is always the right one to hand 

1823 forward, regardless of what round it or the caller's local counter think 

1824 they're at. 

1825 """ 

1826 paths = fix_task_paths(node) 

1827 if not paths: 

1828 return None 

1829 rows = db.read( 

1830 lambda c: c.execute( 

1831 "SELECT * FROM worker_sessions WHERE work_item_id = ? AND node_id = ? " 

1832 f"AND hook_point IN ({','.join('?' * len(paths))}) ORDER BY created_at", 

1833 (work_item_id, node.id, *paths), 

1834 ).fetchall() 

1835 ) 

1836 return rows[-1] if rows else None

1840 db, work_item_id: str, node_id: str 

1841) -> tuple[list[_findings.Finding] | None, bool, str | None]: 

1842 """This node's most recent `findings_measured` findings, whether a 

1843 `fix_cycle_started` for the node followed it, and the head it was taken at. 

1844 

1845 Three answers to three different questions, which is why they are not 

1846 collapsed. The findings are what the next reviewer is handed 

1847 (Kraft-s7c04.1); `fix_ran` is no-progress escalation's extra condition; the 

1848 head is how `walk._carry_severity` tells a re-rating of an untouched tree 

1849 from a genuine reduction (Kraft-s7c04.3). `None` for the head on any event 

1850 written before that field existed, which floors nothing rather than 

1851 guessing. 

1852 

1853 The whole findings, not their fingerprints: the caller derives the tags it 

1854 used to get, and the messages and severities are what `walk._carry_severity` 

1855 floors a repeat's rating against (Kraft-s7c04.3) -- and what 

1856 `prompts.carried_findings_note` hands the next reviewer that declares 

1857 `inputs: [carried_findings]` (Kraft-s7c04.1). 

1858 Returning both would make this the second reader of `findings_measured` in 

1859 this module, which `unresolved_findings_steer`'s docstring forbids for good 

1860 reason -- three readers of one event is three things to keep in step. 

1861 

1862 Deliberately **not** bounded at a `work_item_retried`/`gate_*` boundary, the 

1863 way `store.last_rejection` is. A `/retry` does not change the tree, and 

1864 Kraft-m2q exists precisely because the first fix cycle after a steered retry 

1865 lost the REPEAT tag on the very finding that caused the stop -- "the part 

1866 that matters more than the findings themselves". Only the *most recent* 

1867 measurement is ever returned, so nothing older than one round can leak into 

1868 `resolve_identity`'s `known` however long the item's history is. 

1869 

1870 Read from the event log rather than carried in a local: crash resume and 

1871 `/resume` re-enter the node through `kraft.executor.walk.run_once` with the 

1872 counter intact, and a loop holding its 

1873 history in the stack frame forgets everything it has seen — on exactly the 

1874 path that motivates escalation. 

1875 

1876 REPEAT marking asks only "was this finding in the 

1877 last measurement", so it uses the fingerprints unconditionally. No-progress 

1878 escalation additionally requires the fix flag: spec §4's "no progress" means 

1879 a fix cycle ran and changed nothing — not merely that the same code was 

1880 measured twice in a row. A `POST /retry` on a no-progress stop (or a 

1881 crash/resume between the escalating findings_measured and the fix it never 

1882 got to dispatch) re-enters at round 0 and measures before it fixes; without

1884 same fingerprints and the loop would escalate straight back to needs_human 

1885 without ever giving the steered retry a chance to run. 

1886 """

1896 refunded.add(e["payload"].get("cycle"))

1901 [_findings.from_payload(f) for f in e["payload"].get("findings", [])],

1903 e["payload"].get("head_sha"),

1934def judge_history( 

1935 db, 

1936 work_item_id: str, 

1937 node_id: str, 

1938 loop_severities: frozenset[str], 

1939 evts: list | None = None, 

1940 fix_paths: Sequence[str] = (), 

1941) -> list[dict]: 

1942 """Every round measured so far for this node, oldest first: the judge's 

1943 cross-round view (spec's "cheap pointers, not full transcripts" input) -- 

1944 each round's eligible findings, with the fingerprint stable across 

1945 rounds already computed, and that round's fix session's own result-file 

1946 path. 

1947 

1948 Reconstructed from the event log and `worker_sessions` rather than 

1949 carried in a local across calls, the same resume-safety reasoning 

1950 `last_measurement` gives for reading its own history back from events 

1951 instead of a stack frame. 

1952 """ 

1953 # Kept in event order in a list, never keyed by the event's own `cycle`: 

1954 # `walk_node` seeds `round` from the loop counter on every re-entry 

1955 # (wait scheduler, crash resume, /retry), and that counter persists across 

1956 # the re-entries that are not resumes -- so a new entry's first measurement 

1957 # lands on a cycle number the previous entry already measured at, and a 

1958 # /retry that cleared the counter lands back on 0 where the first entry 

1959 # was. A dict keyed on that number silently dropped the older one and then 

1960 # `sorted()` re-labelled the *newest* measurement as the oldest -- handing 

1961 # the judge a truncated trend pointing the wrong way, on 

1962 # mr_checks/on.ci.poll every time. Seeding narrowed the collision; it did 

1963 # not remove it, so this stays a list. 

1964 # _REPAIR_ROUND (-1) is a repair pass, not a paid fix cycle -- it has 

1965 # nothing to do with the budget the judge is weighing. 

1966 #: `evts`, when given, is a timeline the caller already fetched -- the same 

1967 #: shape `escalate._reason` accepts. Read fresh when nothing is handed in, 

1968 #: which is every existing caller including the judge's own. 

1969 measured: list[tuple[int, list[_findings.Finding]]] = [] 

1970 for e in evts if evts is not None else db.read(lambda c: events.read_after(c, 0, work_item_id)): 

1971 if e["type"] == "findings_measured" and e["payload"].get("node_id") == node_id: 

1972 cycle = e["payload"]["cycle"] 

1973 if cycle < 0: 

1974 continue 

1975 measured.append( 

1976 ( 

1977 cycle, 

1978 [ 

1979 _findings.from_payload(f) 

1980 for f in e["payload"].get("findings", []) 

1981 if f.get("severity") in loop_severities 

1982 ], 

1983 ) 

1984 ) 

1985 #: `fix_paths` is the node's own fix-loop task paths (`fix_task_paths`). 

1986 #: A caller that only wants the per-round findings -- `/retry`'s seeded 

1987 #: steer, which has a node id and no resolved node -- passes none and gets 

1988 #: no fix pointers rather than a second chain read it has no use for. 

1989 fix_rows = ( 

1990 db.read( 

1991 lambda c: c.execute( 

1992 "SELECT round, result_path FROM worker_sessions WHERE work_item_id = ? " 

1993 f"AND node_id = ? AND hook_point IN ({','.join('?' * len(fix_paths))}) " 

1994 "ORDER BY round", 

1995 (work_item_id, node_id, *fix_paths), 

1996 ).fetchall() 

1997 ) 

1998 if fix_paths 

1999 else [] 

2000 ) 

2001 # Fix rounds come from `bump_counter`, which does not reset across 

2002 # re-entries, so unlike the measurements these are collision-free. 

2003 fix_by_round = {r["round"]: r["result_path"] for r in fix_rows} 

2004 return [ 

2005 {"round": i, "findings": found, "fix_result_path": fix_by_round.get(cycle)} 

2006 for i, (cycle, found) in enumerate(measured) 

2007 ]

2010def stuck_fingerprint(history: list[dict], min_repeats: int) -> str | None: 

2011 """The first fingerprint (of the latest round's) that has survived 

2012 `min_repeats` consecutive fix attempts unchanged, or None. 

2013 

2014 Generalizes `walk.walk_node`'s whole-set stuck check (`prints == 

2015 previous_prints`) to a single recurring fingerprint: a blind-failure 

2016 fingerprint (Kraft-0i6z4) can persist for many rounds while a 

2017 co-occurring review finding keeps changing shape, so the set as a whole 

2018 never repeats -- but this one thing never moved. A round only extends a 

2019 streak from the one before it if its OWN `fix_result_path` is set -- 

2020 that is the fix which ran and produced this measurement (see 

2021 `judge_history`'s round/cycle alignment); two measurements taken back to 

2022 back with no fix in between (crash/resume) never had a chance to change 

2023 and must not count as "no progress". 

2024 """ 

2025 streaks: dict[str, int] = {} 

2026 for round_ in history: 

2027 prints = {f.fingerprint for f in round_["findings"]} 

2028 fix_ran = round_["fix_result_path"] is not None 

2029 streaks = {fp: streaks.get(fp, 0) + 1 if fix_ran and fp in streaks else 1 for fp in prints} 

2030 return next((fp for fp, n in streaks.items() if n >= min_repeats), None)

2033def deferred_findings(db, work_item_id: str, loop_severities: frozenset[str]) -> list[dict]: 

2034 """Every finding this item measured that never entered the fix loop, as raw 

2035 payload dicts, deduped by fingerprint and oldest-first. 

2036 

2037 Two readers, one computation: the human at the gate 

2038 (`api.routes.board._deferred_findings`) and the review brief written for 

2039 them (`prompts.deferred_findings_note`, Kraft-s7c04.4). A brief that listed

2043 Payload dicts rather than `Finding`s, because the board serialises these 

2044 straight to JSON and has since they existed. 

2045 """ 

2046 seen: dict[str, dict] = {} 

2047 for e in db.read(lambda c: events.read_after(c, 0, work_item_id)): 

2048 if e["type"] != "findings_measured":

2057def regressed_fingerprints(history: list[dict]) -> list[str]: 

2058 """Tags in the latest round that an earlier round had and a round between 

2059 them did not -- a fix in this loop undoing a fix from earlier in the same 

2060 loop (Kraft-s7c04.7). 

2061 

2062 Distinct from `stuck_fingerprint`, which finds what was never fixed. This 

2063 finds what was fixed and then broken again, and the two need opposite 

2064 instructions: stop repeating an approach, versus stop alternating between 

2065 two and root-cause the conflict. 

2066 

2067 Every round in the gap must carry a `fix_result_path` -- the same guard 

2068 `stuck_fingerprint` applies, for the same reason. A finding that vanished 

2069 because a round crashed or a resume re-measured never had a chance to be 

2070 fixed, so its reappearance reverts nothing, and telling a fixer otherwise 

2071 would steer it away from the correct fix on a loop that is converging. 

2072 

2073 Only a fingerprint comparison, so none of this was visible before 

2074 Kraft-s7c04.2 made a reworded repeat carry the same tag: seven reports of 

2075 one oscillating defect read as seven unrelated findings. 

2076 """ 

2077 if len(history) < 3: 

2078 return [] 

2079 per_round = [{f.fingerprint for f in r["findings"]} for r in history] 

2080 out = [] 

2081 for fp in sorted(per_round[-1]): 

2082 seen = [i for i, prints in enumerate(per_round) if fp in prints] 

2083 if len(seen) < 2 or seen[-1] - seen[0] == len(seen) - 1: 

2084 continue # never absent in between: persistent, not regressed 

2085 gap = [i for i in range(seen[0] + 1, seen[-1]) if fp not in per_round[i]] 

2086 if all(history[i]["fix_result_path"] is not None for i in gap): 

2087 out.append(fp) 

2088 return out
