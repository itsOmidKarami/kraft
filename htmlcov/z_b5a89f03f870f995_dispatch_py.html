48def _path_matches(path: str, patterns: list[str]) -> bool: 

49 """Gitignore-style precedence over one scope's own `paths` list: the last 

50 pattern `path` matches wins, so a later `!excl/**` overrides an earlier 

51 inclusive glob (test-scope design §3.4).""" 

52 matched = False 

53 for pattern in patterns: 

54 if pattern.startswith("!"): 

55 if fnmatch.fnmatchcase(path, pattern[1:]): 

56 matched = False 

57 elif fnmatch.fnmatchcase(path, pattern): 

58 matched = True 

59 return matched

62def _matched_scopes(scopes: list[dict], changed_paths: list[str]) -> list[dict]: 

63 """Scopes covering `changed_paths`, deduped and in `scopes`' own order. 

64 

65 Fails open in both directions the design calls out: a changed path 

66 matching no scope at all, or an empty diff to begin with, runs every 

67 scope rather than guess. Under-testing is the bug this exists to close; 

68 it must never reopen it here (test-scope design §3.5). 

69 """ 

70 hit: set[int] = set() 

71 for path in changed_paths: 

72 path_hit = {i for i, scope in enumerate(scopes) if _path_matches(path, scope["paths"])} 

73 if not path_hit: 

74 return list(scopes) 

75 hit |= path_hit 

76 return [scopes[i] for i in sorted(hit)] if hit else list(scopes)

90def _last_own_round_head( 

91 db, work_item_id: str, node_id: str, task_hook: str, round: int 

92) -> str | None: 

93 """The head `round - 1`'s dispatch of `task_hook` on this exact `node_id` 

94 ran at, if every scope in that dispatch finished `done` -- otherwise 

95 `None` (C7 review fix, Kraft-s7c04.14). 

96 

97 `prompts._last_reviewed_head` is the wrong source for this: it is keyed 

98 on `(work_item_id, hook_point)` alone, not `node_id`, and takes the 

99 latest `done` row regardless of its siblings. For a while (C1, 

100 Kraft-s7c04.8, reverted 2026-09-16) `on.test.run` was dispatched by both 

101 `implementation` and `verify` under that one hook_point, with one 

102 session row per scope -- so that query would have let one node's head 

103 stand in for the other's (verify's round 0 reading C1's head and seeing 

104 an empty diff, breaking the spec's "first round selects from the full 

105 branch diff"). That reason for scoping to `node_id` is gone with C1, but 

106 the other one is independent and still live: it also stops one passing 

107 scope's row from standing in for a dispatch whose sibling scope failed 

108 (the `retry_after_cap` path: scope A failed, scope B passed, the counter 

109 reset wipes the round, and a later HEAD move would otherwise be missed 

110 because a passing B's row still reads as "reviewed"). Scoping to this 

111 node's own previous round, and requiring every row in it to be `done` 

112 with the same `head_sha`, closes both. Coarser than tracking exactly 

113 which scope(s) failed: precise per-scope identity would need scope 

114 identity persisted on the session row, which this batch does not add. 

115 Treating *any* failure (or an inconsistent head across siblings) as "not 

116 clean" can only over-select, matching `_matched_scopes`' own fail-open 

117 posture. 

118 """ 

119 rows = [ 

120 r 

121 for r in db.read(lambda c: store.sessions_for_round(c, work_item_id, node_id, round - 1)) 

122 if r["hook_point"] == task_hook 

123 ] 

124 if not rows or any(r["status"] != "done" for r in rows): 

125 return None 

126 heads = {r["head_sha"] for r in rows} 

127 return heads.pop() if len(heads) == 1 else None

130def _select_scopes( 

131 db, 

132 work_item_id: str, 

133 worktree, 

134 node_id: str, 

135 task_hook: str, 

136 round: int, 

137 binding: dict, 

138 repo_entry: dict, 

139) -> tuple[list[dict], dict | None]: 

140 """The scopes a `kind: subprocess` `on.test.run`-style dispatch should 

141 run, and the sandbox to run them under (test-scope design §3.2-3.3, C7 

142 Kraft-s7c04.14). 

143 

144 Pulled out of `dispatch_node`'s subprocess branch (batch-c1 spec Task 3). 

145 C1 briefly gave `implementation` its own `on.test.run` dispatch through 

146 this same function, so the two computed scopes identically by 

147 construction; C1 was reverted (Kraft-s7c04.8, 2026-09-16) because a gate 

148 dispatched after the session exits cannot be acted on by it. `verify` is 

149 this function's only caller now, and the same-selection property C1 was 

150 after is instead conveyed to the implementation agent as a prompt note 

151 (`prompts.scope_note`) built from the repo's scope table, so it can apply 

152 the same rule to its own diff before it finishes. 

153 

154 The repo's own command(s) win over the registry's. The registry is per 

155 install and one command for every repo on it; test_scopes is a property 

156 of the repo, and a hardcoded single command is how verify ends up running 

157 something CI does not, or the wrong stack's suite entirely (Kraft-579, 

158 Kraft-9wzy). `config.load_repos` already wraps a legacy `test_command` 

159 into a single `["**"]` scope, so this is one shape regardless of which 

160 field an operator set. 

161 

162 C7: round 0 (and a round-0 re-entry, e.g. after `retry_after_cap` wipes 

163 the round counter) always selects from the whole branch diff since 

164 `base_ref` -- the spec's first acceptance criterion, and the only choice 

165 that can't mistake a different node's gate dispatch, or a still-open 

166 round, for "already reviewed". Only a round that follows this exact 

167 `(node_id, task_hook)`'s own fully clean previous round (`done`, every 

168 sibling scope, one shared `head_sha` -- `_last_own_round_head`) narrows 

169 the diff to what changed since then, because `6c712ea8` ran a 

170 13-minute `just ci-test` for a frontend-only fix. That alone would 

171 under-test once C2 lets a round go red on more than one scope: a round 

172 following any failure at this hook does not trust the incremental diff 

173 at all and falls back to the whole-branch diff (`_last_own_round_head` 

174 itself returns `None` for an unclean round, which collapses to 

175 `base_ref` below), unioning in every scope from that round rather than 

176 only the one(s) that failed. Selecting less is only ever safe for a 

177 scope that passed. 

178 """ 

179 sandbox = _sandbox.resolve(binding, repo_entry) 

180 repo_scopes = repo_entry.get("test_scopes") 

181 if not repo_scopes and repo_entry.get("test_command"): 

182 # `config.load_repos` already wraps a bare `test_command` into a 

183 # `test_scopes` entry for any repo it reads off disk -- this mirrors 

184 # that for a `LaunchContext` built by hand (tests, or any future 

185 # caller that skips the yaml round-trip). 

186 repo_scopes = [{"paths": ["**"], "command": repo_entry["test_command"]}] 

187 scopes = ( 

188 [{"paths": s["paths"], "cmd": shlex.split(s["command"])} for s in repo_scopes] 

189 if repo_scopes 

190 else [{"paths": ["**"], "cmd": list(binding["command"])}] 

191 ) 

192 base_ref = _current_base_ref(db, work_item_id) 

193 # `since` is the diff's lower bound: round <= 0 always measures the whole 

194 # branch (never reads another round or another node's head), and only a 

195 # round that follows this exact node's own fully clean previous round at 

196 # this hook narrows it -- everything else (an unclean previous round, a 

197 # `since` `git` no longer knows about, no previous round at all) falls 

198 # through to the whole branch. 

199 since = _last_own_round_head(db, work_item_id, node_id, task_hook, round) if round > 0 else None 

200 if ( 

201 since 

202 and _config.git_read(Path(worktree), "rev-parse", "--verify", f"{since}^{{commit}}") is None 

203 ): 

204 since = None 

205 since = since or base_ref 

206 diff = ( 

207 _config.git_read(Path(worktree), "diff", "--name-only", f"{since}...HEAD") 

208 if since 

209 else None 

210 ) 

211 to_run = scopes if diff is None else _matched_scopes(scopes, diff.splitlines()) 

212 return to_run, sandbox

413 # When it does not, the work is still on disk -- so `verify` passes, 

414 # and only `_assert_clean` two nodes later notices, by which point the 

415 # failure names a hook rather than the cause and a human has to type 

416 # `git commit` in someone else's worktree (Kraft-7fip). Kraft owns the 

417 # worktree, so it takes the work rather than reporting it missing. 

418 # 

419 # Never at the cost of the run itself: an index lock a co-task holds, a

449 )

451 return status 

452 if kind == "subprocess":

458 # before the first failure. On 49c0cefd the third scope (just 

459 # ci-test -- ruff + 2,294 tests + intent) never ran until 5h49m into 

460 # verify because an earlier scope's failure short-circuited it, and 

461 # every commit before that point was silently unverified. The 

462 # aggregate below still fails the node on any scope's failure -- it 

463 # just no longer costs a whole extra round to find out about a 

464 # second, unrelated failure a first scope's break used to hide. 

465 # 

466 # Infra-level statuses are the one exception: they say nothing about 

467 # the code, and running further scopes after one can't be trusted 

468 # either (a human paused the item; the next scope's own binary might 

469 # be missing too; a rate limit applies to every scope alike). 

470 status = "done" 

471 for scope in to_run: 

472 scope_status = await _subprocess.run_task( 

473 db, 

474 run_dirs, 

475 hook_point=task_hook, 

476 cmd=scope["cmd"], 

477 cwd=worktree, 

478 repo_entry=repo_entry, 

479 # The fix loop re-runs the test command after an agent edits source in 

480 # the same worktree. A .pyc written on an earlier cycle has the same 

481 # second-resolution mtime and (often) size as the fixed source, so 

482 # CPython would import the stale bytecode and the re-measure would 

483 # never see the fix. Never writing bytecode keeps every cycle honest. 

484 env={"PYTHONDONTWRITEBYTECODE": "1"}, 

485 sandbox=sandbox, 

486 **{**common, "session_id": uuid.uuid4().hex},

493 if kind == "forge":

500 hook_point=task_hook, 

501 handler=binding["handler"],

522 has_rebase_bounce=bool(node.get("rebase_bounce_to")), 

523 **poll, 

524 **common,

531async def measure_node( 

532 db, 

533 run_dirs, 

534 work_item_id, 

535 node, 

536 row, 

537 registry, 

538 worktree, 

539 *, 

540 round: int = 0, 

541 steer: Steer | None = None, 

542 launch: LaunchContext | None = None, 

543 budget: _policy.Budget = _policy.NO_BUDGET, 

544 loop_severities: frozenset[str] = _policy.DEFAULT_LOOP_SEVERITIES, 

545) -> tuple[str, list[str], list[BaseException]]: 

546 await db.write(lambda c, node=node: store.enter_node(c, work_item_id, node["id"])) 

547 tasks = node["tasks"] 

548 # Kraft-gl9d: a crash/resume re-entry into this same (node, round) must not 

549 # re-spend an agent session on a task whose session already reached 'done' 

550 # against the worktree as it stands right now. Read once, ahead of the 

551 # per-task loop below -- the worktree's HEAD does not move while this 

552 # node's own tasks are still being measured. `worktree` is None only in a 

553 # unit test that stubs `dispatch_node` out entirely (no git to read); a 

554 # null head_sha just means "never reusable", same as any other. 

555 head_sha = _config.git_read(Path(worktree), "rev-parse", "HEAD") if worktree else None 

556 

557 async def _measure(t: str) -> str: 

558 # A null head_sha is never reusable (`reusable_session` itself would 

559 # say so) -- skip the read entirely rather than asking a test double 

560 # that has no worktree, and thus no HEAD, to answer it. 

561 reused = ( 

562 db.read( 

563 lambda c: store.reusable_session(c, work_item_id, node["id"], t, round, head_sha) 

564 ) 

565 if head_sha is not None 

566 else None 

567 ) 

568 if reused is not None: 

569 return reused["status"] 

570 return await dispatch_node( 

571 db, 

572 run_dirs, 

573 t, 

574 node, 

575 row, 

576 registry, 

577 worktree, 

578 loop_severities=loop_severities, 

579 round=round, 

580 steer=steer, 

581 launch=launch, 

582 budget=budget, 

583 ) 

584 

585 results = await asyncio.gather(*(_measure(t) for t in tasks), return_exceptions=True) 

586 # A pause stops the walk where it stands: the node is neither done nor failed, 

587 # and resume relaunches it. It outranks a co-task's failure, which was almost 

588 # certainly the same SIGTERM arriving on a different row. 

589 if any(r == "paused" for r in results): 

590 return "paused", [], [] 

591 # A human's interruption still outranks this, but a task that never 

592 # launched outranks a rate limit, a budget breach and a co-task's failure: 

593 # none of those are evidence about anything while a task in this node 

594 # could not even start (Kraft-579). 

595 if any(r == CONFIG_ERROR for r in results): 

596 failed = [tasks[i] for i, r in enumerate(results) if r == CONFIG_ERROR] 

597 return CONFIG_ERROR, failed, [] 

598 if any(r == RATE_LIMITED for r in results): 

599 return RATE_LIMITED, [], [] 

600 if any(r == WAITING for r in results): 

601 return WAITING, [], [] 

602 if any(r == INFRA_STOP for r in results): 

603 return INFRA_STOP, [], [] 

604 # Logged before the BUDGET rung returns: a co-task can raise in the same node 

605 # as a budget-refused agent, and that traceback is the only record of it. 

606 excs = [r for r in results if isinstance(r, BaseException)] 

607 for exc in excs: 

608 logger.error("measuring task raised in node %s: %r", node["id"], exc) 

609 # paused > rate_limited > budget > failed. A pause is a human's instruction 

610 # and outranks everything. A rate limit and a budget breach both outrank a 

611 # co-task's failure because the agent's "failure" is not evidence about the 

612 # code; a rate limit outranks a budget breach because it is Kraft's own 

613 # spend policy refusing to start, not an external constraint the agent hit. 

614 if any(r == BUDGET for r in results): 

615 return BUDGET, [], [] 

616 failed = [ 

617 tasks[i] 

618 for i, r in enumerate(results) 

619 if isinstance(r, BaseException) or r in ("failed", "needs_context", "conflict") 

620 ] 

621 if failed: 

622 return "failed", failed, excs 

623 return "ok", [], []

627#: `measure_node` treats as failed. 

628_FAILING_STATUSES = ("failed", "needs_context", "conflict")

631def collect_findings(db, work_item_id: str, node: dict, round: int, registry: Registry): 

632 """(findings, hook points that reported at least one) for one cycle. 

633 

634 Only the node's own measuring tasks: the fix task is dispatched with 

635 `round=count` and the next measuring pass runs at that same round, so an 

636 unfiltered query folds the fix agent's result file into the cycle. Only the 

637 most recent row per hook point, because a re-entry can measure at a round a 

638 previous pass already used -- `round` seeds from the persisted fix-loop 

639 counter, which a gate rejection or a `ci_wait` poll does not clear, and a 

640 `/retry` deletes the counter row so the next pass restarts at 1 instead. 

641 Either way stale rows sit at the same number. 

642 

643 A hook that failed without writing a findings file at all -- `on.test.run` 

644 is the common case, `kind: subprocess` with no findings schema to write to 

645 -- gets a synthesized `Finding` via `_findings.from_blind_failure` instead 

646 of vanishing (traced live on a work item that spun for 7 cycles on an 

647 identical, invisible test failure). `reported` is not extended for it: 

648 that set means "wrote a real, parseable result file", which a synthesized 

649 finding does not change -- `walk.py`'s `blind_failures` still computes the 

650 same way and still forces the loop open, now redundantly with `eligible`, 

651 which is harmless. 

652 

653 One hook_point can carry more than one row in a single round -- 

654 `on.test.run`'s scope loop mints one session per matching scope, all under 

655 this same hook_point (the identity problem, batch-c1 spec's "The identity 

656 problem underneath all three"). A bare "last wins" read would let a later 

657 scope's pass silently overwrite an earlier scope's real failure once C2 

658 stops the loop from short-circuiting on it. Every row for the *latest* 

659 `head_sha` this hook_point dispatched at is read instead of just the last 

660 one -- every scope from one dispatch shares that head_sha (stamped once, 

661 before the loop, in `dispatch_node`'s subprocess branch), so this still 

662 collapses to the single most recent row for a hook that only ever mints 

663 one (a re-entry at a *different* head still keeps last-wins there). 

664 

665 That multi-row read is scoped to `kind: subprocess` hooks -- the only 

666 ones a scope loop mints more than one session for -- and not to an agent 

667 reviewer. A reviewer that exits `needs_context` stops before 

668 `bump_counter`, so a resume re-enters the same round at the same head: 

669 without this filter, the stale `needs_context` row (no findings file) 

670 would still share the latest head_sha, hit the `_FAILING_STATUSES` branch 

671 below, and mint a bogus `from_blind_failure` critical finding alongside 

672 the real, later row. An agent hook keeps a plain last-wins read instead. 

673 

674 Every finding this returns -- parsed or synthesized -- carries a 

675 `Finding.jobs` entry per session that produced it, so a fix agent can 

676 always pull a job's full session output directly (Kraft-s7c04.34/.35 

677 brainstorm). A hook's several failing rows in one round (the scope-loop 

678 case above) become ONE synthesized `Finding` with one `JobRef` per 

679 failing job, not several near-identical findings -- the fix agent gets 

680 one coherent notice about the hook, not a fragmented list. 

681 """ 

682 rows = db.read(lambda c: store.sessions_for_round(c, work_item_id, node["id"], round)) 

683 by_hook: dict[str, list[sqlite3.Row]] = {} 

684 for row in rows: 

685 if row["hook_point"] in node["tasks"]: 

686 by_hook.setdefault(row["hook_point"], []).append(row) # ordered by created_at 

687 found: list[_findings.Finding] = [] 

688 reported: set[str] = set() 

689 for hook, hook_rows in by_hook.items(): 

690 is_scope_loop = registry.hooks.get(hook, {}).get("kind") == "subprocess" 

691 latest_head = hook_rows[-1]["head_sha"] 

692 rows_to_read = ( 

693 [row for row in hook_rows if row["head_sha"] == latest_head] 

694 if is_scope_loop 

695 else hook_rows[-1:] 

696 ) 

697 blind_jobs: list[_findings.BlindJob] = [] 

698 for row in rows_to_read: 

699 parsed = _findings.parse(row["result_path"]) 

700 if parsed: 

701 reported.add(hook) 

702 job = _findings.JobRef( 

703 label=hook, log_ref=_findings.session_log_ref(work_item_id, row["id"]) 

704 ) 

705 found.extend(replace(f, jobs=(job,)) for f in parsed) 

706 elif row["status"] in _FAILING_STATUSES: 

707 binding = registry.hooks.get(hook, {}) 

708 command = row["command"] or ( 

709 shlex.join(binding["command"]) 

710 if binding.get("kind") == "subprocess" and binding.get("command") 

711 else None 

712 ) 

713 blind_jobs.append( 

714 _findings.BlindJob( 

715 session_id=row["id"], log_path=row["log_path"], command=command 

716 ) 

717 ) 

718 if blind_jobs: 

719 found.append(_findings.from_blind_failure(hook, work_item_id, blind_jobs)) 

720 return found, reported

723def needs_context_question( 

724 db, work_item_id: str, node: dict, round: int, *, first_iteration: bool = False 

725) -> str | None: 

726 """The question from a `needs_context` row in this round, or None. 

727 

728 Same latest-row-per-hook-point read as `collect_findings` (a resume or 

729 `/retry` re-enters with stale rows still sitting there, so a first-match 

730 scan could re-stop the item on a historical row forever) but deliberately 

731 NOT its `row["hook_point"] in node["tasks"]` filter: the fix task 

732 (`on.implementation.start`) is dispatched with this same round, and 

733 including it is exactly how a fix task's own `needs_context` is meant to 

734 surface, one iteration later. 

735 

736 `first_iteration` is the exception to that, and only that: on the entry's 

737 very first pass there is by definition no fix from *this* entry yet, so 

738 any fix row at this round belongs to a bygone one. It matters because 

739 `walk_node` now seeds `round` from the persisted counter, and it is also 

740 re-entered on paths that are not resumes -- a gate rejection walking back 

741 to a fix_loop node, and the `ci_wait` poller -- where the `retry_counters` 

742 row survives (only `retry_after_cap` deletes it). Without this, the last 

743 pass's fix question would stop the new pass before it ran a single cycle: 

744 the same stranding `ESCALATION_HOOK` below closes, through a third door. 

745 Defaults off so every other caller keeps today's behaviour. 

746 

747 `JUDGE_HOOK` is the one exception. The judge is a brake bolted onto the 

748 cap and never a second way to get stuck (`_judge_result` fails open 

749 in-process), but its session row persists -- so without this filter a 

750 judge that exited `needs_context` would strand the item on the judge's 

751 own question at the next re-entry, which is exactly the design's 

752 forbidden case arriving one iteration late. 

753 

754 `ESCALATION_HOOK` is excluded for the same reason, through a different 

755 door. An escalation is a conversation with a human, not a measurement: 

756 its `needs_context` *is* the question that opened the conversation, so 

757 re-reading it once the human has answered re-stops the node on a question 

758 that has already been settled -- forever, because nothing ever rewrites 

759 that row. On e983d85c that cost $4.86 and 38 minutes, the human's answer 

760 reviewed and discarded twice with no code changed, and the item was then 

761 skipped with a CRITICAL finding still open (Kraft-7itv follow-up). 

762 """ 

763 rows = db.read(lambda c: store.sessions_for_round(c, work_item_id, node["id"], round)) 

764 latest: dict[str, sqlite3.Row] = {} 

765 for row in rows: 

766 if row["hook_point"] in (JUDGE_HOOK, ESCALATION_HOOK): 

767 continue 

768 if first_iteration and row["hook_point"] == "on.implementation.start": 

769 continue 

770 latest[row["hook_point"]] = row # ordered by created_at, so last wins 

771 for row in latest.values(): 

772 if row["status"] == "needs_context": 

773 return _subprocess.read_question(Path(row["result_path"])) or "(no question given)" 

774 return None

777def last_measurement( 

778 db, work_item_id: str, node_id: str 

779) -> tuple[list[_findings.Finding] | None, bool, str | None]: 

780 """This node's most recent `findings_measured` findings, whether a 

781 `fix_cycle_started` for the node followed it, and the head it was taken at. 

782 

783 Three answers to three different questions, which is why they are not 

784 collapsed. The findings are what the next reviewer is handed 

785 (Kraft-s7c04.1); `fix_ran` is no-progress escalation's extra condition; the 

786 head is how `walk._carry_severity` tells a re-rating of an untouched tree 

787 from a genuine reduction (Kraft-s7c04.3). `None` for the head on any event 

788 written before that field existed, which floors nothing rather than 

789 guessing. 

790 

791 The whole findings, not their fingerprints: the caller derives the tags it 

792 used to get, and the messages and severities are what `prompts. 

793 carried_findings_note` hands the next reviewer (Kraft-s7c04.1) and what 

794 `walk._carry_severity` floors a repeat's rating against (Kraft-s7c04.3). 

795 Returning both would make this the second reader of `findings_measured` in 

796 this module, which `unresolved_findings_steer`'s docstring forbids for good 

797 reason -- three readers of one event is three things to keep in step. 

798 

799 Deliberately **not** bounded at a `work_item_retried`/`gate_*` boundary, the 

800 way `store.last_rejection` is. A `/retry` does not change the tree, and 

801 Kraft-m2q exists precisely because the first fix cycle after a steered retry 

802 lost the REPEAT tag on the very finding that caused the stop -- "the part 

803 that matters more than the findings themselves". Only the *most recent* 

804 measurement is ever returned, so nothing older than one round can leak into 

805 `resolve_identity`'s `known` however long the item's history is. 

806 

807 Read from the event log rather than carried in a local: `kraft.executor. 

808 resuming.reconcile_current_node` re-enters `kraft.executor.walk.walk_node` 

809 after a crash or a resume with the counter intact, and a loop holding its 

810 history in the stack frame forgets everything it has seen — on exactly the 

811 path that motivates escalation. 

812 

813 REPEAT marking asks only "was this finding in the 

814 last measurement", so it uses the fingerprints unconditionally. No-progress 

815 escalation additionally requires the fix flag: spec §4's "no progress" means 

816 a fix cycle ran and changed nothing — not merely that the same code was 

817 measured twice in a row. A `POST /retry` on a no-progress stop (or a 

818 crash/resume between the escalating findings_measured and the fix it never 

819 got to dispatch) re-enters at round 0 and measures before it fixes; without 

820 the flag, a deterministic reviewer seeing unchanged code would report the 

821 same fingerprints and the loop would escalate straight back to needs_human 

822 without ever giving the steered retry a chance to run. 

823 """ 

824 evts = db.read(lambda c: events.read_after(c, 0, work_item_id)) 

825 fix_seen = False 

826 for e in reversed(evts): 

827 if e["type"] == "fix_cycle_started" and e["payload"].get("node_id") == node_id: 

828 fix_seen = True 

829 elif e["type"] == "findings_measured" and e["payload"].get("node_id") == node_id: 

830 return ( 

831 [_findings.from_payload(f) for f in e["payload"].get("findings", [])], 

832 fix_seen, 

833 e["payload"].get("head_sha"), 

834 ) 

835 return None, False, None

837 

838#: The fix loop's judge hook (2026-09-12-verify-fix-loop-judge-design): 

839#: dispatched directly by `kraft.executor.walk.walk_node`, the same way 

840#: `on.implementation.start` is -- not from a node's own `tasks` list, so it 

841#: never contaminates `collect_findings`/`needs_context_question`'s per-task 

842#: reads, and every node with a `fix_loop` gets it unconditionally (spec 

843#: decision 4: no new policy field). 

844JUDGE_HOOK = "on.fix_loop.judge" 

845 

846#: The escalation hook's own `hook_point`. Not a node task and not dispatched 

847#: from a `tasks` list -- `escalate.dispatch` writes it directly -- so, like 

848#: `JUDGE_HOOK`, its session rows sit in the same `(node, round)` scan that 

849#: `needs_context_question` reads and must be skipped there. Named rather than 

850#: spelled out at each site because `reattach` compares against it too. 

851ESCALATION_HOOK = "escalation" 

852 

853#: Statuses that mean the judge session actually finished thinking -- the 

854#: same "was this a real judgement" gate `gate_review._UNTRUSTWORTHY` 

855#: applies, phrased as the allowlist its own `VERDICTS` check mirrors. 

856_JUDGE_TRUSTED_STATUS = frozenset({"done", "done_with_concerns"}) 

857_JUDGE_VERDICTS = frozenset({"continue", "stop_needs_human", "stop_downgrade"})

872def judge_history( 

873 db, 

874 work_item_id: str, 

875 node_id: str, 

876 loop_severities: frozenset[str], 

877 evts: list | None = None, 

878) -> list[dict]: 

879 """Every round measured so far for this node, oldest first: the judge's 

880 cross-round view (spec's "cheap pointers, not full transcripts" input) -- 

881 each round's eligible findings, with the fingerprint stable across 

882 rounds already computed, and that round's fix session's own result-file 

883 path. 

884 

885 Reconstructed from the event log and `worker_sessions` rather than 

886 carried in a local across calls, the same resume-safety reasoning 

887 `last_measurement` gives for reading its own history back from events 

888 instead of a stack frame. 

889 """ 

890 # Kept in event order in a list, never keyed by the event's own `cycle`: 

891 # `walk_node` seeds `round` from the loop counter on every re-entry 

892 # (ci_wait poller, crash resume, /retry), and that counter persists across 

893 # the re-entries that are not resumes -- so a new entry's first measurement 

894 # lands on a cycle number the previous entry already measured at, and a 

895 # /retry that cleared the counter lands back on 0 where the first entry 

896 # was. A dict keyed on that number silently dropped the older one and then 

897 # `sorted()` re-labelled the *newest* measurement as the oldest -- handing 

898 # the judge a truncated trend pointing the wrong way, on 

899 # mr_checks/on.ci.poll every time. Seeding narrowed the collision; it did 

900 # not remove it, so this stays a list. 

901 # _REPAIR_ROUND (-1) is a repair pass, not a paid fix cycle -- it has 

902 # nothing to do with the budget the judge is weighing. 

903 #: `evts`, when given, is a timeline the caller already fetched -- the same 

904 #: shape `escalate._reason` accepts. Read fresh when nothing is handed in, 

905 #: which is every existing caller including the judge's own. 

906 measured: list[tuple[int, list[_findings.Finding]]] = [] 

907 for e in evts if evts is not None else db.read(lambda c: events.read_after(c, 0, work_item_id)): 

908 if e["type"] == "findings_measured" and e["payload"].get("node_id") == node_id: 

909 cycle = e["payload"]["cycle"] 

910 if cycle < 0: 

911 continue 

912 measured.append( 

913 ( 

914 cycle, 

915 [ 

916 _findings.from_payload(f) 

917 for f in e["payload"].get("findings", []) 

918 if f.get("severity") in loop_severities 

919 ], 

920 ) 

921 ) 

922 fix_rows = db.read( 

923 lambda c: c.execute( 

924 "SELECT round, result_path FROM worker_sessions WHERE work_item_id = ? " 

925 "AND node_id = ? AND hook_point = 'on.implementation.start' ORDER BY round", 

926 (work_item_id, node_id), 

927 ).fetchall() 

928 ) 

929 # Fix rounds come from `bump_counter`, which does not reset across 

930 # re-entries, so unlike the measurements these are collision-free. 

931 fix_by_round = {r["round"]: r["result_path"] for r in fix_rows} 

932 return [ 

933 {"round": i, "findings": found, "fix_result_path": fix_by_round.get(cycle)} 

934 for i, (cycle, found) in enumerate(measured) 

935 ]

938def stuck_fingerprint(history: list[dict], min_repeats: int) -> str | None: 

939 """The first fingerprint (of the latest round's) that has survived 

940 `min_repeats` consecutive fix attempts unchanged, or None. 

941 

942 Generalizes `walk.walk_node`'s whole-set stuck check (`prints == 

943 previous_prints`) to a single recurring fingerprint: a blind-failure 

944 fingerprint (Kraft-0i6z4) can persist for many rounds while a 

945 co-occurring review finding keeps changing shape, so the set as a whole 

946 never repeats -- but this one thing never moved. A round only extends a 

947 streak from the one before it if its OWN `fix_result_path` is set -- 

948 that is the fix which ran and produced this measurement (see 

949 `judge_history`'s round/cycle alignment); two measurements taken back to 

950 back with no fix in between (crash/resume) never had a chance to change 

951 and must not count as "no progress". 

952 """ 

953 streaks: dict[str, int] = {} 

954 for round_ in history: 

955 prints = {f.fingerprint for f in round_["findings"]} 

956 fix_ran = round_["fix_result_path"] is not None 

957 streaks = {fp: streaks.get(fp, 0) + 1 if fix_ran and fp in streaks else 1 for fp in prints} 

958 return next((fp for fp, n in streaks.items() if n >= min_repeats), None)

961def deferred_findings(db, work_item_id: str, loop_severities: frozenset[str]) -> list[dict]: 

962 """Every finding this item measured that never entered the fix loop, as raw 

963 payload dicts, deduped by fingerprint and oldest-first. 

964 

965 Two readers, one computation: the human at the gate 

966 (`api.routes.board._deferred_findings`) and the review brief written for 

967 them (`prompts.deferred_findings_note`, Kraft-s7c04.4). A brief that listed 

968 a different set from the card above it would be worse than one that listed 

969 nothing. 

970 

971 Payload dicts rather than `Finding`s, because the board serialises these 

972 straight to JSON and has since they existed. 

973 """ 

974 seen: dict[str, dict] = {} 

975 for e in db.read(lambda c: events.read_after(c, 0, work_item_id)): 

976 if e["type"] != "findings_measured": 

977 continue 

978 for raw in e["payload"].get("findings", []): 

979 if raw.get("severity") in loop_severities: 

980 continue 

981 seen.setdefault(_findings.from_payload(raw).fingerprint, raw) 

982 return list(seen.values())

985def regressed_fingerprints(history: list[dict]) -> list[str]: 

986 """Tags in the latest round that an earlier round had and a round between 

987 them did not -- a fix in this loop undoing a fix from earlier in the same 

988 loop (Kraft-s7c04.7). 

989 

990 Distinct from `stuck_fingerprint`, which finds what was never fixed. This 

991 finds what was fixed and then broken again, and the two need opposite 

992 instructions: stop repeating an approach, versus stop alternating between 

993 two and root-cause the conflict. 

994 

995 Every round in the gap must carry a `fix_result_path` -- the same guard 

996 `stuck_fingerprint` applies, for the same reason. A finding that vanished 

997 because a round crashed or a resume re-measured never had a chance to be 

998 fixed, so its reappearance reverts nothing, and telling a fixer otherwise 

999 would steer it away from the correct fix on a loop that is converging. 

1000 

1001 Only a fingerprint comparison, so none of this was visible before 

1002 Kraft-s7c04.2 made a reworded repeat carry the same tag: seven reports of 

1003 one oscillating defect read as seven unrelated findings. 

1004 """ 

1005 if len(history) < 3: 

1006 return [] 

1007 per_round = [{f.fingerprint for f in r["findings"]} for r in history] 

1008 out = [] 

1009 for fp in sorted(per_round[-1]): 

1010 seen = [i for i, prints in enumerate(per_round) if fp in prints] 

1011 if len(seen) < 2 or seen[-1] - seen[0] == len(seen) - 1: 

1012 continue # never absent in between: persistent, not regressed 

1013 gap = [i for i in range(seen[0] + 1, seen[-1]) if fp not in per_round[i]] 

1014 if all(history[i]["fix_result_path"] is not None for i in gap): 

1015 out.append(fp) 

1016 return out

1048async def judge_verdict( 

1049 db, 

1050 run_dirs, 

1051 work_item_id: str, 

1052 node: dict, 

1053 row, 

1054 registry: Registry, 

1055 worktree, 

1056 *, 

1057 round: int, 

1058 key: str, 

1059 eligible: list[_findings.Finding], 

1060 policy: _policy.Policy, 

1061 launch: LaunchContext | None, 

1062 budget: _policy.Budget, 

1063) -> tuple[str, str]: 

1064 """Ask the fix loop's judge whether the cycle about to be dispatched is 

1065 still worth it (2026-09-12-verify-fix-loop-judge-design). Returns 

1066 `(verdict, reasoning)`; `verdict` is `"paused"` (propagate, no event) or 

1067 one of `"continue"`/`"stop_needs_human"`/`"stop_downgrade"`, never 

1068 anything else -- `_judge_result` is the only place that decides which. 

1069 

1070 A registry with no `JUDGE_HOOK` binding at all -- a hand-built test 

1071 registry, or a real install's `registry.yaml` from before this feature 

1072 landed -- fails open the same as any other untrusted outcome, rather 

1073 than a `KeyError` out of `dispatch_node`'s own lookup. 

1074 """ 

1075 if JUDGE_HOOK not in registry.hooks: 

1076 return "continue", "" 

1077 cap = _policy.resolve_cap(policy, key, store.node_overrides_of(row).get(node["id"])) 

1078 counter = db.read(lambda c: store.read_counter(c, work_item_id, key)) 

1079 attempts_used = counter["count"] if counter else 0 

1080 started_at = counter["started_at"] if counter else _now() 

1081 elapsed = (datetime.fromisoformat(_now()) - datetime.fromisoformat(started_at)).total_seconds() 

1082 history = judge_history(db, work_item_id, node["id"], policy.loop_severities) 

1083 instruction = prompts.JUDGE_PROMPT.format( 

1084 node_id=node["id"], 

1085 history=prompts.format_judge_history(history), 

1086 attempts_used=attempts_used, 

1087 cap_attempts=cap.attempts, 

1088 elapsed_s=int(elapsed), 

1089 cap_wall_clock_s=cap.wall_clock_s, 

1090 ) 

1091 status = await dispatch_node( 

1092 db, 

1093 run_dirs, 

1094 JUDGE_HOOK, 

1095 node, 

1096 row, 

1097 registry, 

1098 worktree, 

1099 instruction_override=instruction, 

1100 round=round, 

1101 launch=launch, 

1102 budget=budget, 

1103 ) 

1104 if status == "paused": 

1105 return "paused", "" 

1106 session = None 

1107 for s in db.read(lambda c: store.sessions_for_round(c, work_item_id, node["id"], round)): 

1108 if s["hook_point"] == JUDGE_HOOK: 

1109 session = s # ordered by created_at -- last one wins 

1110 if session is None: 

1111 return "continue", "" 

1112 result_path = Path(session["result_path"]) 

1113 verdict = _judge_result(status, _subprocess.read_verdict(result_path)) 

1114 reasoning = _subprocess.read_concerns(result_path) or "" 

1115 return verdict, reasoning
