16#: How long a `ci_poll` node waits for a pipeline to settle, and how long it 

17#: sleeps between checks. Both are overridable per node in the registry; a 

18#: pipeline slower than this needs a human whatever the number is. 

19DEFAULT_POLL_TIMEOUT = 1800.0 

20DEFAULT_POLL_INTERVAL = 5.0 

21_MAX_POLL_INTERVAL = 60.0 

22 

23#: How long the merge node waits for the forge to report the merge request 

24#: actually merged, and how long it sleeps between reads. Five minutes rather 

25#: than DEFAULT_POLL_TIMEOUT's thirty: by the time this wait starts, `merge` 

26#: has just re-validated CI itself (`wait_for_ci`, same wait `ci_poll` uses), 

27#: so a merge that has not landed by now is waiting on something a person has 

28#: to see, not on a pipeline `mr_sync`'s push may have re-armed (Kraft-266b, 

29#: Kraft-x10m). Deliberately not registry-tunable — nothing has asked, and 

30#: these are one edit away if something does. 

31MERGE_VERIFY_TIMEOUT = 300.0 

32MERGE_VERIFY_INTERVAL = 5.0 

33 

34 

35#: Kraft-x92: one flaky/rate-limited `ci_status` call used to fail the whole 

36#: wait -- an error at minute 12 of a 30-minute poll read the same as a red 

37#: pipeline. Tolerate a short run of consecutive errors before giving up; 

38#: reset the count the moment a call succeeds, so this bounds a burst, not 

39#: the wait's total error budget. 

40_MAX_CONSECUTIVE_POLL_ERRORS = 3

43async def poll_ci( 

44 forge: Forge, *, repo: Path, branch: str, timeout: float, interval: float 

45) -> tuple[CIStatus, bool]: 

46 """Wait for a pipeline to settle. 

47 

48 Returns the last status seen and whether the wait ran out with it still 

49 pending. Backoff rather than a fixed interval: a thirty-minute pipeline 

50 should not cost three hundred CLI invocations, and the early checks are the 

51 ones worth making promptly. 

52 """ 

53 loop = asyncio.get_running_loop() 

54 deadline = loop.time() + timeout 

55 # The cap bounds how far the backoff *grows*, not what the caller asked 

56 # for: a registry that sets a 300s interval for a rate-limited forge must 

57 # not silently get 60s and five times the CLI calls. 

58 cap = max(_MAX_POLL_INTERVAL, interval) 

59 consecutive_errors = 0 

60 while True: 

61 try: 

62 ci = await forge.ci_status(repo=repo, mr=MR(number=0, url=""), branch=branch) 

63 except ForgeError: 

64 consecutive_errors += 1 

65 if consecutive_errors > _MAX_CONSECUTIVE_POLL_ERRORS: 

66 raise 

67 logger.warning( 

68 "ci_status failed mid-poll (%d/%d consecutive), retrying", 

69 consecutive_errors, 

70 _MAX_CONSECUTIVE_POLL_ERRORS, 

71 ) 

72 remaining = deadline - loop.time() 

73 if remaining <= 0: 

74 raise 

75 await asyncio.sleep(min(interval, remaining)) 

76 interval = min(interval * 2, cap) 

77 continue 

78 consecutive_errors = 0 

79 # A branch that cannot merge is an answer, not a wait: a conflict will 

80 # not resolve itself in thirty minutes (Kraft-ejj9). 

81 if ci.state != "pending" or ci.mergeable is False: 

82 return ci, False 

83 remaining = deadline - loop.time() 

84 if remaining <= 0: 

85 return ci, True 

86 await asyncio.sleep(min(interval, remaining)) 

87 interval = min(interval * 2, cap)

90async def render_ci( 

91 ci: CIStatus, 

92 *, 

93 forge: Forge, 

94 repo: Path, 

95 branch: str, 

96 head_sha: str | None, 

97 _retried: bool = False, 

98) -> tuple[str, str]: 

99 """Render the log line and verdict for one CI read. 

100 

101 Order matters and is the fix (Kraft-bjjm): a pipeline not for this head, 

102 or still pending, is a wait — never a failure — however the merge status 

103 reads, because GitLab recomputes merge status asynchronously after a push 

104 and a head pushed seconds ago is routinely reported unmergeable for a 

105 moment. Only a *settled* green pipeline's unmergeable reading is trusted, 

106 and only after a second read confirms it (Kraft-ejj9 stays true, just 

107 later). A settled red pipeline is `"infra"` when every failed job is the 

108 forge's own fault, `"failed"` (code-red) otherwise, and green+confirmed- 

109 unmergeable is `"conflict"`. 

110 """ 

111 log = f"pipeline {ci.state}: {ci.url}\n" + "".join(f" {j}\n" for j in ci.jobs) 

112 if ci.sha and head_sha and ci.sha != head_sha: 

113 return log, "waiting" 

114 if ci.state == "pending": 

115 return log, "waiting" 

116 if ci.state == "success" and ci.mergeable is False: 

117 if _retried: 

118 log += f"merge request is not mergeable: {ci.merge_detail or 'unknown'}\n" 

119 return log, "conflict" 

120 fresh = await forge.ci_status(repo=repo, mr=MR(number=0, url=""), branch=branch) 

121 return await render_ci( 

122 fresh, forge=forge, repo=repo, branch=branch, head_sha=head_sha, _retried=True 

123 ) 

124 if ci.state == "success": 

125 return log, "done" 

126 if is_infra_red(ci): 

127 return log, "infra" 

128 return log, "failed" 

129 

130 

131#: Kraft-h81i, Kraft-s8ul, Kraft-ddxn: a failed job whose own `failure_reason` 

132#: names one of these is the forge's own infrastructure, not the branch's 

133#: code. `script_failure`, an unrecognised reason, or a red pipeline that 

134#: named no reason at all, are code-red — the failure a fix loop is for. 

135#: 

136#: Deliberately NOT `"cancelled"`. By the time a job's `failure_reason` is read 

137#: here, `render_ci`'s sha guard has already confirmed this pipeline belongs to 

138#: the current head, so nothing newer could have superseded it through 

139#: GitLab's redundant-pipeline auto-cancel — the only thing left that could 

140#: have cancelled *this* pipeline is a human clicking Cancel. Auto-retrying 

141#: that through the forge would silently override a person's own decision, 

142#: which is worse than spending one code-red fix cycle that finds nothing to 

143#: change and reports back quickly with the cancellation named in the finding. 

144_INFRA_REASONS = frozenset( 

145 { 

146 "runner_system_failure", 

147 "stuck_or_timeout_failure", 

148 "job_execution_timeout", 

149 "scheduler_failure", 

150 } 

151)

154def is_infra_red(ci: CIStatus) -> bool: 

155 """A settled red pipeline with no jobs at all (a config error stopped 

156 anything from running) is infra-shaped by definition — there is no job to 

157 blame, so there is nothing for a fix loop to fix. 

158 

159 `not ci.failed_jobs` must mean *the forge confirmed* zero jobs, never "we 

160 could not find out" — that ambiguity is exactly Kraft-h81i's real 

161 regression (caught on review of this plan): a `glab ci get` call that 

162 itself fails (`ForgeError`, unparseable JSON) has nothing to report either, 

163 and treating that the same as a genuine zero-job pipeline reads a 

164 perfectly ordinary `script_failure` this fix loop exists to catch as 

165 infra instead, self-retrying twice and stopping at `needs_human` with no 

166 fix loop ever run. `glab.py`'s `_failure_detail` is the only caller that 

167 can produce an empty `failed_jobs` tuple, and it is written so the 

168 unreadable case never does — it returns a placeholder job with 

169 `failure_reason=None` instead, which is not a member of `_INFRA_REASONS`, 

170 so the `all(...)` below reads it as code-red by default. Only a 

171 *successful* read that confirms no job ran returns the real empty tuple 

172 this function treats as infra. 

173 """ 

174 if not ci.failed_jobs: 

175 return True 

176 return all(j.failure_reason in _INFRA_REASONS for j in ci.failed_jobs) 

177 

178 

179#: A runner failure or a config error resolves on its own or it does not -- 

180#: two retries, not thirty minutes of them (Kraft-h81i). Widen this if a real 

181#: install's infra is flakier than that. The count that matters is run.py's 

182#: persisted `ci_infra:<node_id>` counter, not a variable in this module -- 

183#: see `retry_infra_once`'s docstring for why. 

184_INFRA_RETRY_CAP = 2 

185#: Generous on purpose: this counter is bookkeeping across `ci_poll` entries, 

186#: not a user-facing loop with its own `policy.yaml` entry, so its wall-clock 

187#: side of the cap should essentially never be what trips it -- the attempts 

188#: side always will be, first. 

189_INFRA_WALL_CLOCK_S = 3600 

190 

191#: Kraft-43kw: merge_watch's own budget for a target-branch pipeline that 

192#: never settles at all -- plain module constants, not a policy.yaml key, 

193#: same as the pair above (`_run_one` has no `policy` object to resolve one 

194#: from). Deliberately smaller than `ci_wait.py`'s shared `ci_wait` cap 

195#: (1800s/60 attempts) so this always resolves first: the merged work is 

196#: already on the target branch by the time this node runs, so giving up on 

197#: watching and finishing is a smaller consequence than paging a human over 

198#: a pipeline this item cannot fix either way. 

199_POST_MERGE_WAIT_CAP = 40 

200_POST_MERGE_WAIT_WALL_CLOCK_S = 900

203async def retry_infra_once( 

204 forge: Forge, 

205 *, 

206 repo: Path, 

207 branch: str, 

208 head_sha: str | None, 

209 first: CIStatus, 

210 branch_only: bool = False, 

211) -> tuple[str, str]: 

212 """Kick one retry of a settled infra-red pipeline through the forge, then 

213 read it back once and hand back whatever verdict that read gives. 

214 

215 Deliberately not a loop over `_INFRA_RETRY_CAP`: `forge.retry_jobs` only 

216 *starts* the job again -- the pipeline is not settled the instant it 

217 returns, so a synchronous re-read immediately after almost always still 

218 sees `"pending"` (verdict `"waiting"`). A loop with an immediate re-read 

219 (this function's previous shape) therefore only ever spent one kick 

220 before returning `"waiting"`, whatever its cap said, and reset that 

221 "budget" to zero on every fresh call -- once per `ci_wait` re-entry, 

222 forever. Real settlement takes real CI minutes, which only elapse 

223 *between* separate calls to this node, so the retry budget is counted by 

224 `run.py`'s persisted `ci_infra:<node_id>` counter instead, bumped once per 

225 `ci_poll` entry that is still infra-red -- this function just does the one 

226 kick-and-read that counter's caller decided was still within budget. 

227 

228 `branch_only` picks which read the kick is followed by, and exists because 

229 `merge_watch` calls this after its own merge request is gone: the default 

230 `ci_status` resolves an MR first (`gh pr view` / `glab mr view`), which on 

231 a just-merged branch has nothing left to resolve and raises `ForgeError`, 

232 failing the terminal node -- the same MR-vs-branch split the caller's 

233 first read already makes with `branch_ci_status`. Both real backends 

234 raise there; `FakeForge` does not, so the double that holds this down 

235 (`tests/test_forge_run.py`) raises on the MR path on purpose. 

236 """ 

237 await forge.retry_jobs(repo=repo, ci=first) 

238 ci_status = ( 

239 await forge.branch_ci_status(repo=repo, branch=branch, head_sha=head_sha or "") 

240 if branch_only 

241 else await forge.ci_status(repo=repo, mr=MR(number=0, url=""), branch=branch) 

242 ) 

243 return await render_ci(ci_status, forge=forge, repo=repo, branch=branch, head_sha=head_sha)

246async def wait_for_ci( 

247 forge: Forge, *, repo: Path, branch: str, timeout: float, interval: float 

248) -> tuple[str, str]: 

249 """Wait for CI to settle, and render the log line and the mergeable/red 

250 verdict `merge`'s own CI gate reports (Kraft-266b, Kraft-x10m): 

251 `mr_sync`'s push after `human_review` can re-arm a project's 

252 required-pipeline rule for a head `mr_checks` never watched, so `merge` 

253 re-runs this exact wait, with the same timeout/interval, right before it 

254 calls `forge.merge()` — not a second, differently-tuned approximation of 

255 it. `ci_poll` no longer calls this (Kraft-ru98): it hands an unsettled 

256 pipeline back to the scheduler instead of blocking here. 

257 

258 Returns the log text and "done"/"failed", the same vocabulary a node's 

259 status already uses. A timeout here still maps to "failed" -- `merge`'s 

260 caller has no scheduler to hand a wait back to. 

261 """ 

262 ci, timed_out = await poll_ci( 

263 forge, repo=repo, branch=branch, timeout=timeout, interval=interval 

264 ) 

265 if timed_out: 

266 # A timeout and a red pipeline are both a failed node, but a reviewer 

267 # -- and any fix loop built on this node (Kraft-cbr) -- has to tell 

268 # "finished red" from "never finished". 

269 log = f"pipeline timed out after {timeout:g}s, still pending: {ci.url}\n" + "".join( 

270 f" {j}\n" for j in ci.jobs 

271 ) 

272 return log, "failed" 

273 head_sha = await git._head_sha(repo) 

274 return await render_ci(ci, forge=forge, repo=repo, branch=branch, head_sha=head_sha)

277async def poll_merged( 

278 forge: Forge, *, repo: Path, branch: str, timeout: float, interval: float 

279) -> MRRef | None: 

280 """Read the merge request back until it stops being open. 

281 

282 `glab mr merge --yes` exits 0 both for "merged" and for "merge when all 

283 merge checks pass", and the second one merges nothing (Kraft-79x3, MR !76); 

284 `gh pr merge` has the same shape. The exit code is not the answer — the 

285 merge request's own state is. Returns the last `MRRef` read, or None if the 

286 branch has no merge request at all any more; the caller decides from the 

287 state. Same backoff as `poll_ci`, for the same reason. 

288 """ 

289 loop = asyncio.get_running_loop() 

290 deadline = loop.time() + timeout 

291 cap = max(_MAX_POLL_INTERVAL, interval) 

292 while True: 

293 ref = await forge.find_mr(repo=repo, branch=branch) 

294 if ref is None or ref.state != "open": 

295 return ref 

296 remaining = deadline - loop.time() 

297 if remaining <= 0: 

298 return ref 

299 await asyncio.sleep(min(interval, remaining)) 

300 interval = min(interval * 2, cap)
