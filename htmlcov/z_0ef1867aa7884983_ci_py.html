1"""Reading a pipeline once: `render_ci`'s verdict on one read, what counts as 

2infrastructure-red, and the one self-retry of an infra-red pipeline. Waiting 

3for it is `kraft.waits`' -- nothing here sleeps. 

4"""

6from __future__ import annotations 

7 

8from datetime import UTC, datetime, timedelta 

9from pathlib import Path 

10 

11from kraft.adapters.forge.models import MR, CIStatus, Forge

14async def render_ci( 

15 ci: CIStatus, 

16 *, 

17 forge: Forge, 

18 repo: Path, 

19 branch: str, 

20 head_sha: str | None, 

21 _retried: bool = False, 

22) -> tuple[str, str]: 

23 """Render the log line and verdict for one CI read. 

24 

25 Order matters and is the fix (Kraft-bjjm): a pipeline not for this head, 

26 or still pending, is a wait — never a failure — however the merge status 

27 reads, because GitLab recomputes merge status asynchronously after a push 

28 and a head pushed seconds ago is routinely reported unmergeable for a 

29 moment. Only a *settled* green pipeline's unmergeable reading is trusted, 

30 and only after a second read confirms it (Kraft-ejj9 stays true, just 

31 later). A settled red pipeline is `"infra"` when every failed job is the 

32 forge's own fault, `"failed"` (code-red) otherwise, and green+confirmed- 

33 unmergeable is `"conflict"`. 

34 

35 A pending read standing only on a run cancelled `_CANCEL_GRACE` ago or 

36 more is `"abandoned"` (Kraft-kbqmk): a successor to a relabel or a 

37 push-superseded run registers within seconds, so one that has not in 

38 that long is not coming, and a person decides now rather than at the 

39 wait's full timeout. A fresher cancel stays a wait (Kraft-zn8me). 

40 """ 

41 log = f"pipeline {ci.state}: {ci.url}\n" + "".join(f" {j}\n" for j in ci.jobs) 

42 if ci.sha and head_sha and ci.sha != head_sha: 

43 return log, "waiting" 

44 if ci.state == "pending": 

45 if _abandoned(ci.cancelled_at): 

46 log += ( 

47 f"the run for {(head_sha or ci.sha)[:7] or 'this head'} was cancelled at " 

48 f"{ci.cancelled_at} and no new run has started since; nothing about the code " 

49 "failed. re-run CI, then retry this item\n" 

50 ) 

51 return log, "abandoned" 

52 return log, "waiting" 

53 if ci.state == "success" and ci.mergeable is False: 

54 if _retried: 

55 log += f"merge request is not mergeable: {ci.merge_detail or 'unknown'}\n" 

56 return log, "conflict" 

57 fresh = await forge.ci_status(repo=repo, mr=MR(number=0, url=""), branch=branch) 

58 return await render_ci( 

59 fresh, forge=forge, repo=repo, branch=branch, head_sha=head_sha, _retried=True 

60 ) 

61 if ci.state == "success": 

62 return log, "done" 

63 if is_infra_red(ci): 

64 return log, "infra" 

65 return log, "failed" 

66 

67 

68#: How long a cancelled run may stand as the latest for its head before it 

69#: counts as one nobody followed up. Minutes, not seconds: a successor 

70#: registers within seconds, and the forge's clock is not this machine's. 

71_CANCEL_GRACE = timedelta(minutes=5)

74def _abandoned(cancelled_at: str) -> bool: 

75 """Whether a cancel is `_CANCEL_GRACE` old. A time that will not parse is 

76 not old: it waits, and the wait's own timeout still stops it.""" 

77 try: 

78 at = datetime.fromisoformat(cancelled_at) 

79 except ValueError: 

80 return False 

81 if at.tzinfo is None: 

82 return False 

83 return datetime.now(UTC) - at >= _CANCEL_GRACE 

84 

85 

86#: Kraft-h81i, Kraft-s8ul, Kraft-ddxn: a failed job whose own `failure_reason` 

87#: names one of these is the forge's own infrastructure, not the branch's 

88#: code. `script_failure`, an unrecognised reason, or a red pipeline that 

89#: named no reason at all, are code-red — the failure a fix loop is for. 

90#: 

91#: No `"cancelled"`: a cancelled run never reaches here. Both backends read it 

92#: as pending (Kraft-zn8me), because a cancelled run is never a verdict, and a 

93#: run a person cancelled with no successor is `render_ci`'s "abandoned". 

94#: It is never auto-retried here, which would override that person's decision. 

95_INFRA_REASONS = frozenset( 

96 { 

97 "runner_system_failure", 

98 "stuck_or_timeout_failure", 

99 "job_execution_timeout", 

100 "scheduler_failure", 

101 } 

102)

105def is_infra_red(ci: CIStatus) -> bool: 

106 """A settled red pipeline with no jobs at all (a config error stopped 

107 anything from running) is infra-shaped by definition — there is no job to 

108 blame, so there is nothing for a fix loop to fix. 

109 

110 `not ci.failed_jobs` must mean *the forge confirmed* zero jobs, never "we 

111 could not find out" — that ambiguity is exactly Kraft-h81i's real 

112 regression (caught on review of this plan): a `glab ci get` call that 

113 itself fails (`ForgeError`, unparseable JSON) has nothing to report either, 

114 and treating that the same as a genuine zero-job pipeline reads a 

115 perfectly ordinary `script_failure` this fix loop exists to catch as 

116 infra instead, self-retrying twice and stopping at `needs_human` with no 

117 fix loop ever run. `glab.py`'s `_failure_detail` is the only caller that 

118 can produce an empty `failed_jobs` tuple, and it is written so the 

119 unreadable case never does — it returns a placeholder job with 

120 `failure_reason=None` instead, which is not a member of `_INFRA_REASONS`, 

121 so the `all(...)` below reads it as code-red by default. Only a 

122 *successful* read that confirms no job ran returns the real empty tuple 

123 this function treats as infra. 

124 """ 

125 if not ci.failed_jobs: 

126 return True 

127 return all(j.failure_reason in _INFRA_REASONS for j in ci.failed_jobs) 

128 

129 

130#: A runner failure or a config error resolves on its own or it does not -- 

131#: two retries, not thirty minutes of them (Kraft-h81i). Widen this if a real 

132#: install's infra is flakier than that. The count that matters is run.py's 

133#: persisted `ci_infra:<node_id>` counter, not a variable in this module -- 

134#: see `retry_infra_once`'s docstring for why. 

135_INFRA_RETRY_CAP = 2 

136#: Generous on purpose: this counter is bookkeeping across `ci_poll` entries, 

137#: not a user-facing loop with its own `policy.yaml` entry, so its wall-clock 

138#: side of the cap should essentially never be what trips it -- the attempts 

139#: side always will be, first. 

140_INFRA_WALL_CLOCK_S = 3600

143async def retry_infra_once( 

144 forge: Forge, 

145 *, 

146 repo: Path, 

147 branch: str, 

148 head_sha: str | None, 

149 first: CIStatus, 

150 branch_only: bool = False, 

151) -> tuple[str, str]: 

152 """Kick one retry of a settled infra-red pipeline through the forge, then 

153 read it back once and hand back whatever verdict that read gives. 

154 

155 Deliberately not a loop over `_INFRA_RETRY_CAP`: `forge.retry_jobs` only 

156 *starts* the job again -- the pipeline is not settled the instant it 

157 returns, so a synchronous re-read immediately after almost always still 

158 sees `"pending"` (verdict `"waiting"`). A loop with an immediate re-read 

159 (this function's previous shape) therefore only ever spent one kick 

160 before returning `"waiting"`, whatever its cap said, and reset that 

161 "budget" to zero on every fresh call -- once per scheduler re-entry, 

162 forever. Real settlement takes real CI minutes, which only elapse 

163 *between* separate calls to this node, so the retry budget is counted by 

164 `run.py`'s persisted `ci_infra:<node_id>` counter instead, bumped once per 

165 `ci_poll` entry that is still infra-red -- this function just does the one 

166 kick-and-read that counter's caller decided was still within budget. 

167 

168 `branch_only` picks which read the kick is followed by, and exists because 

169 `merge_watch` calls this after its own merge request is gone: the default 

170 `ci_status` resolves an MR first (`gh pr view` / `glab mr view`), which on 

171 a just-merged branch has nothing left to resolve and raises `ForgeError`, 

172 failing the terminal node -- the same MR-vs-branch split the caller's 

173 first read already makes with `branch_ci_status`. Both real backends 

174 raise there; `FakeForge` does not, so the double that holds this down 

175 (`tests/adapters/forge/test_run.py`) raises on the MR path on purpose. 

176 """ 

177 await forge.retry_jobs(repo=repo, ci=first) 

178 ci_status = ( 

179 await forge.branch_ci_status(repo=repo, branch=branch, head_sha=head_sha or "") 

180 if branch_only 

181 else await forge.ci_status(repo=repo, mr=MR(number=0, url=""), branch=branch) 

182 ) 

183 log, verdict = await render_ci( 

184 ci_status, forge=forge, repo=repo, branch=branch, head_sha=head_sha 

185 ) 

186 # This kick just re-started the run, so an old cancel is not one nobody 

187 # followed up, and "abandoned" is no verdict any caller of this handles. 

188 return log, "waiting" if verdict == "abandoned" else verdict
