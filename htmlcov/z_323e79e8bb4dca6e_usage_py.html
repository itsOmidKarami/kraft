1"""Token, cost and wall-time capture per worker session. 

2 

3Nothing recorded any of this before: the detail control row, the log modal 

4header and the whole analytics view are all reading numbers that have to be 

5harvested when a session ends. Two shapes are accepted, because two kinds of 

6task produce them: 

7 

8* the agent envelope an agent CLI writes as the last line of its log 

9 (Claude Code with ``--output-format json``: ``usage.input_tokens`` and 

10 friends, ``total_cost_usd``, ``model``); 

11* a ``usage`` block in the result file any adapter may write at 

12 ``$KRAFT_RESULT_PATH``. 

13 

14**A finished session's settled cost is only ever the agent's own number.** 

15The agent knows what it was billed and Kraft does not, so a computed figure 

16must never pass for a finished session's true cost: that would be a guess 

17wearing the same font as a fact. A finished session with tokens and no 

18reported cost therefore keeps an estimate flagged ``cost_estimated`` when its 

19model can be priced (see `_settle_cost` below), and ``cost_usd = NULL`` when it 

20cannot, which the rollups report as unknown rather than counting as zero. 

21 

22**A running session is a different question.** Its own true cost does not 

23exist yet -- the agent has not been billed -- so there is nothing for Kraft to 

24defer to. `estimate_cost` prices its live tokens (input, cache read, cache 

25write; not output, which is unknown live -- see `Usage.tokens_out`) against 

26`prices.json`, a packaged per-model rate table snapshotted from models.dev. 

27`worker_sessions.cost_estimated` marks a row carrying this guess. The exit or 

28pause envelope overwrites it with the agent's own figure and clears the flag 

29the moment the session settles -- *when that envelope reports a cost*. When it 

30does not (a crash, or an interruption before the agent's first response), the 

31row keeps an estimate rather than losing the only spend figure Kraft has: 

32`_settle_cost` (`store.sessions`) re-estimates from the session's final tokens 

33(output included now, since it's known) and leaves `cost_estimated` set. A 

34model missing from the table still gets no estimate at all (`None`), exactly 

35as a finished session with no reported cost and no priceable model gets no 

36total. 

37"""

54@dataclass(frozen=True) 

55class Usage: 

56 #: Uncached input only. Cache writes and reads were billed as input too, 

57 #: and are kept apart so a run's cache share is visible (Ruling 211); 

58 #: `total` is what every budget counts (Decision 18). 

59 tokens_in: int = 0 

60 #: None means "not knowable yet", not zero. A claude stream-json assistant 

61 #: line carries `output_tokens` as of the *start* of that message, not its 

62 #: final count (measured: a finished session's live fold gave 64, its 

63 #: result envelope 7701), so `from_stream` reports this as unknown rather 

64 #: than publish a number that is off by two orders of magnitude while the 

65 #: session is still running. `session_exited`'s envelope fills it in. 

66 tokens_out: int | None = 0 

67 #: None means "this agent did not report a cost", never "it was free". 

68 cost_usd: float | None = None 

69 model: str | None = None 

70 tokens_cache_write: int = 0 

71 tokens_cache_read: int = 0 

72 #: The largest context one request sent (its uncached input, cache reads 

73 #: and cache writes), which picks a long-context price tier 

74 #: (`estimate_cost`). Only a log with a line per request shows it 

75 #: (claude's and Amp's stream); 0 for one that reports running totals 

76 #: (codex, cursor, an exit envelope), which prices at the base tier. 

77 peak_context: int = 0

102@functools.lru_cache(maxsize=1) 

103def _prices() -> dict: 

104 """The packaged rate table (`prices.json`), read once per process. 

105 

106 Never fetched at runtime: a live network call inside a hot progress-tick 

107 path is both slow and one more way for an estimate to fail, and this 

108 number only ever backs a live guess a real cost replaces at exit. Refresh 

109 the file itself with `just refresh-prices` (`dev/refresh_prices.py`) 

110 instead. A packaging error (the file went missing from the wheel) means 

111 no model prices, not a crash: every session then estimates as `None`, 

112 same as a model this snapshot never learned about. 

113 """ 

114 try: 

115 text = importlib.resources.files("kraft").joinpath("prices.json").read_text() 

116 return json.loads(text).get("models", {}) 

117 except (OSError, json.JSONDecodeError): 

118 return {}

127def estimate_cost(usage: Usage, model: str | None, *, include_output: bool = False) -> float | None: 

128 """A session's cost, estimated from its tokens against `prices.json`. 

129 

130 Input, cache read and cache write are always priced. `include_output` 

131 defaults False -- the live case (`session_progress`): output tokens are 

132 unknown while a claude session runs (`Usage.tokens_out` is `None`), so 

133 pricing them would fold the same wrong number `from_stream` now refuses to 

134 publish straight into a dollar figure, and the estimate is a floor on the 

135 session's eventual cost, same shape as `cost_complete` elsewhere in this 

136 module. `None` when `model` is not in `prices.json` -- a session on a 

137 model this snapshot has never priced gets no guess, not a wrong one. 

138 

139 `include_output=True` is the settled case (`session_exited`, 

140 `record_pause_usage`, via `_settle_cost`): by then `tokens_out` is the 

141 exit envelope's real, final count, not the live stream's unknown one, so 

142 there is no reason left to leave it unpriced -- doing so would still 

143 under-report a crashed or interrupted session's only spend figure by 

144 whatever it had generated. 

145 

146 `prices.json`'s `cache_write` is the 1-hour cache-creation rate (2x 

147 input), not models.dev's own 5-minute figure (1.25x input) -- Claude Code 

148 writes 1-hour entries. `dev/refresh_prices.py` derives it rather than 

149 trusting models.dev's `cache_write`; see `CACHE_WRITE_MULTIPLE` there for 

150 the proof against a real session's `total_cost_usd`. 

151 """ 

152 rates = _prices().get(model or "") 

153 if not isinstance(rates, dict): 

154 return None 

155 # A long-context tier bills the whole session once one request's context 

156 # passes it (OpenAI's pricing past 272k); the highest one passed wins. 

157 for tier in list(rates.get("tiers", ())): 

158 if usage.peak_context > tier["above"]: 

159 rates = tier 

160 per_million = ( 

161 usage.tokens_in * rates.get("input", 0) 

162 + usage.tokens_cache_read * rates.get("cache_read", 0) 

163 + usage.tokens_cache_write * rates.get("cache_write", 0) 

164 ) 

165 if include_output: 

166 per_million += (usage.tokens_out or 0) * rates.get("output", 0) 

167 return per_million / 1_000_000

174def _from_usage_block(block: object, model: object, *, live: bool = False) -> Usage | None: 

175 """One ``usage`` mapping, in either the agent's or Kraft's own field names. 

176 

177 `live=True` is `from_stream`'s case: this block is a claude stream-json 

178 `assistant` line's `usage`, whose `output_tokens` is only the count as of 

179 the start of that message, not the message's final count. Reporting it 

180 would publish a wrong number as if it were real (module docstring on 

181 `from_stream`), so `tokens_out` is forced unknown (`None`) instead. A 

182 finished envelope's `usage` block (the default, `live=False`) is the 

183 turn's actual final count and is read as reported. 

184 """ 

185 if not isinstance(block, dict): 

186 return None 

187 u = Usage( 

188 tokens_in=_int(block.get("input_tokens", block.get("tokens_in"))), 

189 tokens_out=None if live else _int(block.get("output_tokens", block.get("tokens_out"))), 

190 model=model if isinstance(model, str) else None, 

191 # Cache writes and reads are input tokens that were billed; leaving 

192 # them out would under-report a long agent run by most of its input. 

193 tokens_cache_write=_int( 

194 block.get("cache_creation_input_tokens", block.get("tokens_cache_write")) 

195 ), 

196 tokens_cache_read=_int( 

197 block.get("cache_read_input_tokens", block.get("tokens_cache_read")) 

198 ), 

199 peak_context=_int(block.get("peak_context")), 

200 ) 

201 return u if u.total else None

214def _dominant(by_model: dict) -> str | None: 

215 """The `modelUsage` entry with the most tokens against it, or None. 

216 

217 Strictly greater, so a tie keeps the first key and this stays deterministic 

218 -- and so a mapping whose values are not the shape expected (every entry 

219 scoring 0) degrades to the first key rather than to nothing. 

220 """ 

221 best, best_tokens = None, -1 

222 for name, stats in by_model.items(): 

223 if not isinstance(name, str) or not name: 

224 continue 

225 tokens = ( 

226 sum(_int(stats.get(k)) for k in _MODEL_TOKEN_KEYS) if isinstance(stats, dict) else 0 

227 ) 

228 if tokens > best_tokens: 

229 best, best_tokens = name, tokens 

230 return best

233def _from_model_usage(by_model: object, model: str | None) -> Usage | None: 

234 """Tokens from `modelUsage`: the whole CLI session's, every model summed. 

235 

236 Sums across models rather than picking one: this is "what did this session 

237 spend", not "which model did the work" -- that second question is 

238 `_model_of`/`_dominant`'s and is deliberately left alone (Kraft-s7c04.15). 

239 None when it reports nothing, so a task that spent nothing is not a 

240 zero-token run. 

241 """ 

242 if not isinstance(by_model, dict): 

243 return None 

244 stats = [s for s in by_model.values() if isinstance(s, dict)] 

245 u = Usage( 

246 tokens_in=sum(_int(s.get("inputTokens")) for s in stats), 

247 tokens_out=sum(_int(s.get("outputTokens")) for s in stats), 

248 model=model, 

249 tokens_cache_write=sum(_int(s.get("cacheCreationInputTokens")) for s in stats), 

250 tokens_cache_read=sum(_int(s.get("cacheReadInputTokens")) for s in stats), 

251 ) 

252 return u if u.total else None

255def _model_of(envelope: dict) -> str | None: 

256 """The model an agent reported, in either shape it reports it. 

257 

258 `--output-format stream-json`'s result envelope has no top-level `model`: 

259 it carries `modelUsage`, a mapping keyed by model name. Reading only 

260 `model` is why `worker_sessions.model` was NULL on every row ever written 

261 (Kraft-2r8s). A stated `model` still wins -- a result file written by a 

262 non-agent adapter names its model directly. 

263 

264 *Which* key is the second half of that bug. `modelUsage` is keyed in order 

265 of first use, and the first use is Claude Code's own haiku warm-up -- so 

266 taking the first key named haiku on every agent row ever written, and made 

267 any cost-by-model reading of `worker_sessions` invalid (Kraft-s7c04.15). 

268 The model that did the work is the one the tokens are against. 

269 """ 

270 model = envelope.get("model") 

271 if isinstance(model, str) and model: 

272 return model 

273 by_model = envelope.get("modelUsage") 

274 return _dominant(by_model) if isinstance(by_model, dict) else None

277def from_envelope(envelope: object) -> Usage | None: 

278 """Usage from a decoded agent result envelope, or None if it carries none.""" 

279 if not isinstance(envelope, dict): 

280 return None 

281 # `modelUsage` first (Kraft-s7c04.65): it is cumulative over the session, 

282 # Task-tool sub-agents and any turn killed before its result included, and 

283 # it is what `total_cost_usd` paid for. The `usage` block is the main 

284 # agent's last turn only (193475c9 read 1.80M input tokens of the 2.11M 

285 # billed). An interrupted envelope zeroes `usage` and reports only here 

286 # (Kraft-s7c04.18). `usage` stays the read for a result file, or any 

287 # envelope with no `modelUsage` to read. 

288 u = _from_model_usage(envelope.get("modelUsage"), _model_of(envelope)) 

289 if u is None: 

290 u = _from_usage_block(envelope.get("usage"), _model_of(envelope)) 

291 if u is None: 

292 return None 

293 cost = envelope.get("total_cost_usd", envelope.get("cost_usd")) 

294 if isinstance(cost, int | float) and not isinstance(cost, bool): 

295 u = replace(u, cost_usd=float(cost)) 

296 return u

307def from_stream(lines: Iterable[str], seen: dict[str, Usage], *, live: bool = True) -> Usage | None: 

308 """Fold new stream-json lines into `seen` and return the running total. 

309 

310 `seen` maps `request_id` -> that request's usage and is owned by the 

311 caller, which is what makes repeated calls over a growing log correct: the 

312 CLI emits several `assistant` lines per API request, with identical usage 

313 on each (measured: two lines, one request), so summing lines rather than 

314 requests double-counts. `tokens_in`, `tokens_cache_read` and 

315 `tokens_cache_write` converge on the number `session_exited` writes this 

316 way. 

317 

318 `tokens_out` deliberately does not, while `live=True` (the default -- this 

319 is a session still running): each claude `assistant` line's 

320 `output_tokens` is that message's count as of when the line was written, 

321 not its final count (measured on a finished session: this fold gave out=64 

322 over the run, the result envelope said 7701 -- cache read/write matched 

323 exactly). An earlier version of this docstring claimed output tokens 

324 converged too; they do not, and publishing the partial count as if it were 

325 real is the bug this fixes: `_from_usage_block(..., live=True)` forces 

326 `tokens_out` to `None` per line, and `None` taints the fold (`_sum`). The 

327 result's `tokens_out` is therefore `None` while a session runs; only 

328 `session_exited`'s envelope ever sets it. 

329 

330 `live=False` is `_envelope_amp`'s case: Amp's stream is this same 

331 claude-shaped `assistant` line, but Amp's `result` line carries no `usage` 

332 at all, so these lines are Amp's *only* record of a finished session's 

333 tokens, not just its live progress -- forcing `tokens_out` unknown there 

334 would make it permanently unrecoverable rather than merely delayed. Amp's 

335 own accuracy mid-message is unmeasured; nothing here claims it is exact, 

336 only that Amp has no better number to fall back on the way claude does. 

337 

338 Returns None only when nothing has been seen at all, so a caller can tell 

339 "no usage yet" from "zero tokens so far, model known". 

340 

341 No cost from tokens: no per-request cost is reported, and cost is only 

342 ever the agent's own number (module docstring). `session_progress` prices 

343 an *estimate* from these live tokens itself (`estimate_cost`); this 

344 function's own `cost_usd` stays `None`. 

345 """ 

346 for raw in lines: 

347 line = raw.strip() 

348 if not line.startswith("{"): 

349 continue 

350 try: 

351 obj = json.loads(line) 

352 except json.JSONDecodeError: 

353 continue 

354 if not isinstance(obj, dict): 

355 continue 

356 model = obj.get("model") 

357 if obj.get("type") == "system" and isinstance(model, str) and model: 

358 seen.setdefault(_INIT_KEY, Usage(model=model)) 

359 continue 

360 message = obj.get("message") 

361 if not isinstance(message, dict): 

362 continue 

363 u = _from_usage_block(message.get("usage"), message.get("model"), live=live) 

364 if u is None: 

365 continue 

366 request_id = obj.get("request_id") 

367 # A line with no request_id cannot be deduplicated against anything; 

368 # key it on its own position so it is counted exactly once. 

369 key = ( 

370 request_id if isinstance(request_id, str) and request_id else f"{_INIT_KEY}{len(seen)}" 

371 ) 

372 seen[key] = u 

373 if not seen: 

374 return None 

375 return replace( 

376 _sum(seen.values()), 

377 # The FIRST model seen, deliberately -- not `_model_of`'s dominant-by- 

378 # tokens read (Kraft-s7c04.15). `_INIT_KEY` is seeded from the 

379 # `system`/`init` line above and inserted before any `assistant` line, 

380 # and that line names the session's real model. This is already right 

381 # and must not be "fixed" to match the envelope path. 

382 model=next((u.model for u in seen.values() if u.model), None), 

383 peak_context=max( 

384 u.tokens_in + u.tokens_cache_read + u.tokens_cache_write for u in seen.values() 

385 ), 

386 )

389def _sum(usages: Iterable[Usage]) -> Usage: 

390 """Each kind of token summed; no cost and no model, which do not add. 

391 

392 `tokens_out` sums to `None`, not 0, the moment any contributor's is `None` 

393 -- one live (unknown) reading taints the whole total, the same way an 

394 unknown `cost_usd` does elsewhere in this module. Every other kind is 

395 always an int (never `None`), so summing them plainly is unchanged. 

396 """ 

397 usages = list(usages) 

398 values = {k: [getattr(u, k) for u in usages] for k in KINDS} 

399 return Usage( 

400 **{k: (None if any(v is None for v in vs) else sum(vs)) for k, vs in values.items()} 

401 )

404def _combine(envelopes: list[dict]) -> dict: 

405 """One envelope standing for every agent invocation a log holds. 

406 

407 A log can carry several result envelopes -- one per invocation, when the 

408 agent is re-launched into the same log (Kraft-s7c04.60). Measured against 

409 every such log on the owner's machine: within one CLI `session_id`, 

410 `total_cost_usd` and `modelUsage` are *cumulative* across invocations 

411 (5.54 -> 6.03 -> 6.08 -> 6.12 over four) while `usage` is that invocation 

412 alone. So taking the last envelope kept the cost but dropped every earlier 

413 invocation's tokens, and summing costs would double-count them. 

414 

415 Per CLI session, then, the last envelope speaks for the whole: its cost and 

416 its `modelUsage` tokens. Not "the first envelope plus `modelUsage` 

417 growth": 6c235b8c's first turn ran 2,380 lines and was killed before it 

418 wrote a result, so its first result's `usage` held 1.2M input tokens of 

419 the 123M its $34.72 paid for. Only a group with no `modelUsage` to read 

420 falls back to summing each invocation's own `usage`. Distinct CLI sessions 

421 in one log are summed. A group whose last envelope reports no cost makes 

422 the whole cost unknown, never zero. 

423 """ 

424 groups: dict[object, list[dict]] = {} 

425 for envelope in envelopes: 

426 groups.setdefault(envelope.get("session_id"), []).append(envelope) 

427 parts: list[Usage] = [] 

428 cost: float | None = 0.0 

429 for group in groups.values(): 

430 last = group[-1] 

431 total = _from_model_usage(last.get("modelUsage"), None) 

432 parts += [total] if total is not None else [from_envelope(e) for e in group] 

433 last_cost = (from_envelope(last) or Usage()).cost_usd 

434 cost = None if cost is None or last_cost is None else cost + last_cost 

435 u = _sum(parts) 

436 combined: dict = { 

437 # In Kraft's own names, which `_from_usage_block` reads as well. 

438 "usage": {k: getattr(u, k) for k in KINDS}, 

439 "model": _model_of(envelopes[-1]), 

440 } 

441 if cost is not None: 

442 combined["total_cost_usd"] = cost 

443 return combined

446def read_envelope(log_path: Path) -> dict | None: 

447 """The agent's result envelope: every invocation's, combined (`_combine`) 

448 when there is more than one. 

449 

450 An invocation is a `type: result` line (or an untyped one, the shape a 

451 result file has) that `from_envelope` can read. A `usage` block alone is 

452 not enough: a Task-tool sub-agent's `system/task_progress` line carries 

453 one of its own, `{total_tokens, tool_uses, duration_ms}`, and read as an 

454 invocation it over-counted 6c235b8c by 156,662 input tokens 

455 (Kraft-lp01z). 

456 

457 Claude Code's `stream-json` output appends a trailing `system/task_summary` 

458 line *after* the `result` line that carries `usage` and `total_cost_usd`. 

459 Taking the literal last line picked up that summary instead and silently 

460 dropped a reported cost (Kraft-xob8). Looking for `usage` finds the 

461 `result` line regardless of what follows it; a log that never has one 

462 still gets its last parseable JSON object, unchanged from before. 

463 """ 

464 try: 

465 lines = [ln for ln in log_path.read_text().splitlines() if ln.strip()] 

466 except OSError: 

467 return None 

468 fallback = None 

469 envelopes = [] 

470 for line in lines: 

471 try: 

472 envelope = json.loads(line) 

473 except json.JSONDecodeError: 

474 continue 

475 if not isinstance(envelope, dict): 

476 continue 

477 fallback = envelope 

478 if envelope.get("type", "result") == "result" and from_envelope(envelope): 

479 envelopes.append(envelope) 

480 if len(envelopes) > 1: 

481 return _combine(envelopes) 

482 return envelopes[0] if envelopes else fallback

508def _rate_limit_claude(log_path: Path) -> RateLimitInfo | None: 

509 """The rejected `rate_limit_info` from a claude stream-json log, or None. 

510 

511 The CLI emits a `rate_limit_event` line on most turns, nearly all of them 

512 `status: "allowed"` -- an `overageStatus` of "rejected" on an otherwise 

513 allowed turn means only that overage spend was refused, not that the turn 

514 itself was blocked. Only a top-level `status: "rejected"` means the launch 

515 was refused. Scanned across every line, not just the last: unlike the 

516 result envelope, this event is not guaranteed to be the final line. 

517 Best-effort like `agent._envelope_is_error`: a log Kraft cannot read yet is 

518 "no rejection seen", not a crash. 

519 """ 

520 try:

522 except OSError: 

523 return None 

524 for line in lines: 

525 try: 

526 obj = json.loads(line) 

527 except json.JSONDecodeError: 

528 continue 

529 if not isinstance(obj, dict) or obj.get("type") != "rate_limit_event": 

530 continue 

531 info = obj.get("rate_limit_info") 

532 if not isinstance(info, dict) or info.get("status") != "rejected": 

533 continue

535 if not isinstance(resets_at, int | float): 

536 continue 

537 return RateLimitInfo( 

538 rate_limit_type=info.get("rateLimitType"), 

539 resets_at=resets_at, 

540 resets_at_iso=datetime.fromtimestamp(resets_at, UTC).isoformat(), 

541 ) 

542 return None

565def net_of_earlier( 

566 own: Usage, log_path: Path, earlier_log: Path, cli: str, reader: str = "claude-stream-json" 

567) -> tuple[Usage, bool]: 

568 """`(own less what CLI session cli had already spent by the end of 

569 earlier_log, netting_failed)` -- `earlier_log` is a session this one 

570 resumed (Kraft-s7c04.62). Both logs' envelopes report the session's 

571 running totals, so the difference is this session's own spend. Unknown 

572 earlier spend makes this session's cost unknown too, never the whole 

573 running total and never zero. 

574 

575 `netting_failed` is `True` exactly when `earlier_log` has no known 

576 running totals to net against: the returned `Usage` then still carries 

577 `own`'s tokens as read -- the CLI session's *cumulative* totals, not this 

578 turn's own share, since there is nothing to subtract them from -- and its 

579 caller (`store.sessions._settle_cost`, Kraft-wz83s) must not price an 

580 unestimated cost off them, or a paused-then-resumed turn's spend is 

581 counted twice: once in the paused row's own estimate, again in the 

582 resumed row's estimate off the same cumulative tokens. `False` means 

583 either a clean net (tokens and cost both this turn's own share) or that 

584 there was nothing to net in the first place (`mine is None` -- this 

585 session's own log carries no cumulative reading of its own, so `own` is 

586 returned exactly as its caller already read it, which callers handle 

587 exactly like a fresh, unresumed session). 

588 

589 `reader` names the logs' schema (Kraft-wge0e). Claude's log can hold 

590 result envelopes of more than one CLI session, so it is read for `cli`'s; 

591 any other harness's envelope is its own running total -- codex's 

592 `turn.completed` usage is the thread's. A reader whose log counts only 

593 its own requests (`Reader.running_totals` False) has nothing to net: 

594 `own` is already this session's share, returned as read (Kraft-tugdf.23).""" 

595 if not READERS[reader].running_totals: 

596 return own, False 

597 

598 def cumulative(path: Path) -> Usage | None: 

599 if reader == "claude-stream-json": 

600 return _cumulative(path, cli) 

601 return from_envelope(READERS[reader].envelope(path)) 

602 

603 mine, before = cumulative(log_path), cumulative(earlier_log) 

604 if mine is None: 

605 return own, False 

606 if before is None: 

607 return replace(own, cost_usd=None), True 

608 cost = ( 

609 None 

610 if mine.cost_usd is None or before.cost_usd is None 

611 else max(mine.cost_usd - before.cost_usd, 0.0) 

612 ) 

613 return ( 

614 Usage( 

615 **{k: max(getattr(mine, k) - getattr(before, k), 0) for k in KINDS}, 

616 cost_usd=cost, 

617 model=own.model, 

618 peak_context=own.peak_context, 

619 ), 

620 False, 

621 )

659def _unfinished_jobs_claude(log_path: Path) -> list[str]: 

660 """What each background job still running when the agent's last turn 

661 ended was running, oldest first; `[]` when none was, or no turn ended. 

662 

663 A job is backgrounded two ways, and the log shows both: the agent asks 

664 (`run_in_background` on a Bash call), or Claude Code moves a foreground 

665 command that outlived its timeout (`task_updated` with `is_backgrounded`) 

666 -- the second is how a full-suite run left the turn in Kraft-nxqft. A 

667 job ends with its `task_notification`. Read as of the last `result` line, 

668 the turn's end: the CLI kills what is left on its way out, which the log 

669 reports after that line, and that kill is not the agent waiting on it. 

670 """ 

671 try: 

672 lines = log_path.read_text().splitlines() 

673 except OSError: 

674 return [] 

675 running: dict[object, str] = {} # tool_use_id -> the command it runs 

676 started: dict[object, tuple[object, str]] = {} # task_id -> (tool_use_id, command) 

677 at_turn_end: list[str] = [] 

678 for line in lines: 

679 try: 

680 obj = json.loads(line) 

681 except json.JSONDecodeError: 

682 continue 

683 if not isinstance(obj, dict): 

684 continue 

685 kind, sub = obj.get("type"), obj.get("subtype") 

686 if kind == "result": 

687 at_turn_end = list(running.values()) 

688 elif kind == "assistant" and isinstance(obj.get("message"), dict): 

689 for block in obj["message"].get("content") or (): 

690 args = block.get("input") if isinstance(block, dict) else None 

691 if isinstance(args, dict) and args.get("run_in_background") is True: 

692 running[block.get("id")] = str(args.get("command") or block.get("name")) 

693 elif kind != "system": 

694 continue 

695 elif sub == "task_started": 

696 job = (obj.get("tool_use_id"), str(obj.get("description"))) 

697 started[obj.get("task_id")] = job 

698 if obj.get("is_backgrounded") is True: 

699 running.setdefault(*job) 

700 elif sub == "task_updated" and isinstance(obj.get("patch"), dict): 

701 job = started.get(obj.get("task_id")) 

702 if job is not None and obj["patch"].get("is_backgrounded") is True: 

703 running.setdefault(*job) 

704 elif job is not None and obj["patch"].get("status") in _JOB_ENDED: 

705 running.pop(job[0], None) 

706 elif sub == "task_notification": 

707 running.pop(obj.get("tool_use_id"), None) 

708 return at_turn_end

1045 own `usage`; the closing `result` carries the conversation's running 

1046 total (a resumed second turn reported 24603 input, its first 12198 plus 

1047 its own 12405), so it is never read: the steps already count each 

1048 invocation in a log once. `thinking_tokens` is a share of 

1049 `output_tokens` (12225 + 407 = 12632 = `total_tokens` with 322 of 

1050 thinking). Uncached input is `total - output - cache_read`, right whether 

1051 `cache_read_tokens` is a share of `input_tokens` or beside it; every run 

1052 measured had none. No cost: the stream carries none. The model is 

1053 `init.model`, present only when `--model` was passed. 

1054 """ 

1055 for obj in _json_lines(lines): 

1056 if obj.get("event") == "init": 

1057 model = (obj.get("init") or {}).get("model") 

1058 if isinstance(model, str) and model: 

1059 seen[_AGY_MODEL] = Usage(model=model) 

1060 continue 

1061 step = obj.get("step_update") if obj.get("event") == "step_update" else None 

1062 if not isinstance(step, dict) or step.get("state") != "DONE": 

1063 continue 

1064 block = step.get("usage") 

1065 if not isinstance(block, dict): 

1066 continue 

1067 out, cache = _int(block.get("output_tokens")), _int(block.get("cache_read_tokens")) 

1068 total = _int(block.get("total_tokens")) or _int(block.get("input_tokens")) + out 

1069 seen[f"{step.get('conversation_id')}:{step.get('step_index')}"] = Usage( 

1070 tokens_in=max(total - out - cache, 0), tokens_out=out, tokens_cache_read=cache 

1071 ) 

1072 steps = [u for k, u in seen.items() if k != _AGY_MODEL] 

1073 if not steps: 

1074 return None 

1075 model = seen.get(_AGY_MODEL) 

1076 return replace(_sum(steps), model=model.model if model else None)

1258def usd(n: float) -> str: 

1259 """Dollars as the web UI's budget meter prints them (`format.ts` `usd`): 

1260 cents from $1 up, a tenth of a cent under it, so a stop under a dollar 

1261 names the same number as the meter beside it. Rounded half up on the 

1262 float's exact value, as JavaScript's `toFixed` does: Python's own format 

1263 rounds an exact binary tie to even, so $0.0625 would read $0.062 here and 

1264 $0.063 on the meter.""" 

1265 if not math.isfinite(n): 

1266 return f"${n}" 

1267 step = Decimal("0.01") if n >= 1 else Decimal("0.001") 

1268 # Enough digits for any finite float: the default 28 refuses a cap of 1e26 and up. 

1269 with localcontext(prec=400): 

1270 return f"${Decimal(n).quantize(step, rounding=ROUND_HALF_UP)}" 

1271 

1272 

1273def cap_usd(n: float) -> str: 

1274 """A dollar cap as it was set: `usd`'s digits, and more where the cap has 

1275 more, so a cap of $0.0001 never reads "$0.000" (as if it were zero) and 

1276 one just raised to $0.0705 never reads as the $0.070 it replaced.""" 

1277 if not math.isfinite(n): 

1278 return f"${n}" 

1279 exact = Decimal(repr(n)) 

1280 if exact.as_tuple().exponent < (-2 if n >= 1 else -3): 

1281 return f"${exact:f}" 

1282 return usd(n)
