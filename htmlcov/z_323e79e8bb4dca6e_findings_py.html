1"""The shared finding schema (`03_plugin_adapters.md` §4) and its identity. 

2 

3Every reader here is best-effort: a plugin that writes a malformed result file 

4loses its findings, never the session. That matches `read_summary_ref` in 

5`adapters/subprocess.py`, which swallows a broken file for the same reason. 

6"""

14from dataclasses import dataclass, replace 

15from pathlib import Path

17from kraft.render import strip_ansi

19SEVERITIES: tuple[str, ...] = ("critical", "important", "minor") 

20 

21_WS = re.compile(r"\s+") 

22 

23#: Bytes read from the tail of a failed task's log -- enough to reach a 

24#: failure summary without loading a possibly large log in full. 

25_TAIL_BYTES = 20_000 

26 

27#: Cap on the extracted failure text, before the retrieval line is appended. 

28#: Keeps a synthesized Finding's message the same order of size as a real 

29#: review finding's (~200-600 chars observed), not a log dump. 

30_MESSAGE_CAP = 400 

31 

32#: Lines worth keeping from a failed task's output. 

33_MARKER = re.compile(r"✘|FAILED|Error:|Traceback") 

34 

35#: Wall-clock noise that must not affect a finding's fingerprint: an 

36#: identical failure at a different duration or timestamp is still the same 

37#: failure. `# ponytail: heuristic, not framework-aware; upgrade to 

38#: structured test-name diffing if a real framework ever needs it.` 

39_NOISE = re.compile(r"\(\d+(?:\.\d+)?\s*(?:ms|s|m)\)|\b\d{2}:\d{2}:\d{2}\b") 

40 

41#: What a `same_as` must look like to be believed: the shape `fingerprint` 

42#: itself produces. A model asked to echo a tag will sometimes write a sentence 

43#: instead, and a sentence accepted as an identity is worse than no identity. 

44_TAG = re.compile(r"^[0-9a-f]{16}$") 

45

47@dataclass(frozen=True) 

48class Finding: 

49 severity: str 

50 message: str 

51 file: str | None 

52 line: int | None 

53 source_plugin: str 

54 #: The tag of a finding from an earlier round that this one restates, as the 

55 #: reviewer that was shown both says so. Optional and last, so every 

56 #: positional construction keeps working. 

57 same_as: str | None = None 

58 

59 @property 

60 def fingerprint(self) -> str: 

61 """Stable identity across fix cycles. 

62 

63 `line` is excluded deliberately: the fix task edits the file, so every 

64 line below the edit shifts, and including it would make every finding 

65 look new after any fix — exactly the blindness this exists to remove. 

66 Severity is excluded too: the same defect re-reported at a different 

67 severity is the same defect. 

68 

69 Those two exclusions are right and stay. What the original design did 

70 not anticipate is **rewording** — `message` is LLM prose, so the same 

71 defect restated in new words hashed differently, `stuck_fingerprint`'s 

72 streak never exceeded 1, and the stuck detector could never fire 

73 (Kraft-s7c04.2: one reattach.py defect reported in five consecutive 

74 measurements under five distinct fingerprints, then shipped). 

75 

76 No hash can see that two sentences describe one defect. The reviewer 

77 that read both can, so it is asked, and `same_as` is its answer — 

78 honoured over the hash. It is only ever believed for a tag the reviewer 

79 was actually shown; see `resolve_identity`. 

80 """ 

81 if self.same_as: 

82 return self.same_as 

83 norm = _WS.sub(" ", self.message).strip().lower() 

84 raw = "\0".join((self.source_plugin, self.file or "", norm)) 

85 return hashlib.sha256(raw.encode()).hexdigest()[:16]

88def _one(raw: object) -> Finding | None: 

89 if not isinstance(raw, dict): 

90 return None 

91 severity = raw.get("severity") 

92 message = raw.get("message") 

93 source_plugin = raw.get("source_plugin") 

94 if severity not in SEVERITIES: 

95 return None 

96 if not isinstance(message, str) or not message: 

97 return None 

98 if not isinstance(source_plugin, str) or not source_plugin: 

99 return None 

100 file = raw.get("file") 

101 line = raw.get("line") 

102 same_as = raw.get("same_as") 

103 return Finding( 

104 severity=severity, 

105 message=message, 

106 file=file if isinstance(file, str) and file else None, 

107 line=line if isinstance(line, int) and not isinstance(line, bool) else None, 

108 source_plugin=source_plugin, 

109 same_as=same_as if isinstance(same_as, str) and _TAG.match(same_as) else None, 

110 )

113def parse(result_path: str | Path) -> list[Finding]: 

114 """Findings from a result file. Never raises; a bad file yields none.""" 

115 try: 

116 data = json.loads(Path(result_path).read_text()) 

117 except OSError, json.JSONDecodeError, UnicodeDecodeError: 

118 return [] 

119 if not isinstance(data, dict): 

120 return [] 

121 raw = data.get("findings") 

122 if not isinstance(raw, list): 

123 return [] 

124 return [f for f in (_one(r) for r in raw) if f is not None]

145def resolve_identity(found: list[Finding], known: frozenset[str] | set[str]) -> list[Finding]: 

146 """`found` with every `same_as` that names a tag nobody was shown stripped. 

147 

148 A reviewer claims identity with a previous finding by echoing its tag 

149 (Kraft-s7c04.2). Trusting that unchecked lets one hallucinated or stale tag 

150 collapse two distinct defects onto a single identity, or resurrect one from 

151 an episode that is over -- and the stuck detector would then fire on a 

152 fiction, stopping a loop that is converging. 

153 

154 `known` is the set of tags this round's reviewer was actually handed, which 

155 is exactly what `prompts.carried_findings_note` listed for it. A finding 

156 whose claim fails falls back to its own prose hash, which is what it would 

157 have had before it claimed anything. 

158 """ 

159 return [f if not f.same_as or f.same_as in known else replace(f, same_as=None) for f in found]

162def _tail_text(log_path: str | Path) -> str | None: 

163 """The last `_TAIL_BYTES` of a log file, or None if it can't be read. 

164 Best-effort, matching this module's own stated philosophy.""" 

165 try: 

166 with open(log_path, "rb") as f: 

167 f.seek(0, os.SEEK_END) 

168 size = f.tell() 

169 f.seek(max(0, size - _TAIL_BYTES)) 

170 data = f.read() 

171 except OSError: 

172 return None 

173 return data.decode("utf-8", errors="replace")

176def _extract_message(text: str) -> str: 

177 text = _NOISE.sub("", strip_ansi(text)) 

178 lines = [ln.strip() for ln in text.splitlines() if ln.strip()] 

179 marked = [ln for ln in lines if _MARKER.search(ln)] 

180 chosen = marked[:5] if marked else lines[-5:] 

181 return "\n".join(chosen)[:_MESSAGE_CAP]

184def from_blind_failure( 

185 hook: str, 

186 log_path: str | Path | None, 

187 work_item_id: str, 

188 session_id: str, 

189 reproduce: str | None = None, 

190) -> Finding: 

191 """A `Finding` for a task that failed without writing its own findings 

192 file -- most commonly a `kind: subprocess` measuring task like 

193 `on.test.run`. Without this, the task's failure never reaches the fix 

194 prompt's content, the judge, or the fingerprint-based stuck-detector; it 

195 is only ever named, never shown (traced live on a work item that spun 

196 for 7 cycles on the identical test failure because of exactly that gap). 

197 

198 `reproduce`, when given, must be a fixed string that does not vary 

199 between rounds (e.g. the hook's own registry-bound command) -- it goes 

200 into `message`, which `Finding.fingerprint` hashes whole, so anything 

201 round-specific here (a session id, a log path) would fingerprint an 

202 unchanged failure differently every cycle and defeat the point of this 

203 function. 

204 """ 

205 if reproduce: 

206 pointer = f"Reproduce with: {reproduce}" 

207 else: 

208 pointer = f"Full log: kraft view logs {work_item_id} --session {session_id}" 

209 text = _tail_text(log_path) if log_path else None 

210 if not text or not text.strip(): 

211 message = f"{hook} failed; no output captured.\n\n{pointer}" 

212 else: 

213 message = f"{_extract_message(text)}\n\n{pointer}" 

214 return Finding(severity="critical", message=message, file=None, line=None, source_plugin=hook)
