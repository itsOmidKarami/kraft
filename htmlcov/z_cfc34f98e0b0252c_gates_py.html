29def approve_gate( 

30 conn: sqlite3.Connection, work_item_id, gate, *, by: str = "human", tips: dict | None = None 

31) -> bool | str: 

32 """Clear the gate and set the item running again. True when it is running. 

33 

34 `tips` is the head the chain's final gate approves (`reviewed_head.pin_for`), 

35 read before this write: recorded on `gate_approved`, it is what the merge 

36 may push. None for any other gate. 

37 

38 While another in-place item works in this item's checkout 

39 (`checkout_holder`), the approval is still recorded, the item is paused 

40 and queued to go on from the gate once the checkout frees 

41 (`hold_for_checkout`), and the holder's id is returned. False, and 

42 nothing written, for an ended item. 

43 

44 Also closes the gate node's own `node_started`/`node_completed` pair. A V1 

45 gate *is* a node (`gate-is-an-ordered-node`) and `gates.maybe_gate` enters 

46 it, but nothing ever completed it: the walk resumes at the node *after* the 

47 gate, so the board rendered an approved gate as a node still running, 

48 forever (4a's Concern 4). The gate's node id and its name are the same 

49 string in V1, which is why one argument answers both. 

50 

51 Written here rather than at each approval door so the human's `POST 

52 .../approve` and `review_gates`' `approve` verdict cannot drift -- the one 

53 rule the two doors have to keep. A rejection is deliberately *not* a 

54 completion: the gate reopens, and `gate_rejected` already says so. 

55 `complete_node` is idempotent, so re-approving writes one event. 

56 """ 

57 holder = checkout_holder(conn, work_item_id) 

58 if not write_status( 

59 conn, 

60 "UPDATE work_items SET status = 'active', updated_at = ? WHERE id = ?" 

61 if holder is None 

62 else "UPDATE work_items SET status = 'paused', updated_at = ? WHERE id = ?", 

63 (_now(), work_item_id), 

64 ): 

65 return False 

66 payload = {"gate": gate, "by": by, **({"tips": tips} if tips is not None else {})} 

67 events.append(conn, work_item_id, GateEvent.APPROVED, payload, node_id=gate) 

68 chain.complete_node(conn, work_item_id, gate) 

69 if holder is None: 

70 return True 

71 # `approved`: its replay refreshes the base first, as a direct approval does. 

72 hold_for_checkout(conn, work_item_id, holder, {"node": gate, "approved": True}) 

73 return holder

128def approval_stands( 

129 conn: sqlite3.Connection, work_item_id, gate: str, upstream: Collection[str] 

130) -> bool: 

131 """Whether `gate`'s approval still stands, so a walk passes over it. 

132 

133 Per gate: the newest of `gate_requested`, `gate_approved`, `gate_rejected` 

134 and `gate_reopened` that names `gate` must be its approval -- another 

135 gate's approval since says nothing about this one. And nothing since may 

136 have sent the chain back to a node in `upstream`, the ids of the chain's 

137 nodes up to and including `gate`: that re-run walks back through `gate` 

138 with new work, so it is asked again. Two motions do that without a 

139 `gate_reopened` for the gates they span, so their own events are the word: 

140 a rejection of any gate, re-entering at its `node` (one with none re-entered 

141 at its own gate), and a person's gateless `request_changes`, rewinding to its 

142 `target` unless it was cancelled (`review.request_rewind`). 

143 """ 

144 row = conn.execute( 

145 "SELECT seq, type FROM events WHERE work_item_id = ? " 

146 f"AND type IN ({', '.join('?' * len(_GATE_WORD))}) " 

147 "AND json_extract(payload, '$.gate') = ? ORDER BY seq DESC LIMIT 1", 

148 (work_item_id, *_GATE_WORD, gate), 

149 ).fetchone() 

150 if row is None or row["type"] != GateEvent.APPROVED: 

151 return False 

152 later = conn.execute( 

153 "SELECT type, json_extract(payload, '$.review_id') AS rid, " 

154 "COALESCE(json_extract(payload, '$.node'), json_extract(payload, '$.gate'), " 

155 "json_extract(payload, '$.target')) AS entered FROM events " 

156 "WHERE work_item_id = ? AND type IN (?, ?, ?) AND seq > ? ORDER BY seq DESC", 

157 ( 

158 work_item_id, 

159 GateEvent.REJECTED, 

160 GateEvent.REWIND_REQUESTED, 

161 GateEvent.REWIND_CANCELLED, 

162 row["seq"], 

163 ), 

164 ).fetchall() 

165 cancelled = set() 

166 for e in later: 

167 if e["type"] == GateEvent.REWIND_CANCELLED: 

168 cancelled.add(e["rid"]) 

169 elif e["entered"] in upstream and not ( 

170 e["type"] == GateEvent.REWIND_REQUESTED and e["rid"] in cancelled 

171 ): 

172 return False 

173 return True

210def reject_gate( 

211 conn: sqlite3.Connection, 

212 work_item_id, 

213 gate, 

214 note, 

215 *, 

216 reopen: bool, 

217 node: str | None = None, 

218 by: str = "human", 

219 verdict: str | None = None, 

220 rerun: Sequence[str] = (), 

221) -> str | None: 

222 """Record the rejection. `reopen` flips the item back to active for the 

223 backward-motion re-run (02 §7.2); a rejection that breached the gate's 

224 reject loop leaves it needs_human. 

225 

226 A reopen while another in-place item works in this item's checkout 

227 (`checkout_holder`) pauses the item instead and queues the re-run from 

228 `node`, with the note as its steer (`hold_for_checkout`); the holder's id 

229 is returned. None otherwise. 

230 

231 `rerun` is the nodes with a fix loop that the re-run measures again: a 

232 rejection that reopens gives them their rounds back 

233 (`counters.reset_fix_loops`), here so that it is one write with the 

234 rejection, and an item that has ended (which records none) resets nothing. 

235 

236 `node` is the chain node the re-run enters at (Kraft-ko7j). It rides the 

237 event rather than a column: the events table is already append-only and 

238 already holds the note, and `store.last_rejection` reads both back. 

239 

240 `by` records who decided (Kraft-zr3s) -- a gate cleared by an agent and a 

241 gate cleared by a person have to be tellable apart in the timeline forever 

242 after. 

243 

244 `verdict` records *what* was decided, for the same reason and one level 

245 down: `by: agent` alone cannot tell a reviewer that rejected from one that 

246 repaired the worktree and committed (`gate_review.VERDICTS`' `fixed`), and 

247 those two have very different consequences -- a `fixed` re-enters at the 

248 gate's own node and regenerates the artifact the gate is about. Without it 

249 that cycle is invisible in every aggregate and only readable by putting 

250 session logs in order by hand (Kraft-s7c04.16). Rides the event alongside 

251 `note` and `node` for the reason those do. 

252 """ 

253 holder = checkout_holder(conn, work_item_id) if reopen else None 

254 status = "status" if not reopen else "'active'" if holder is None else "'paused'" 

255 if not write_status( 

256 conn, 

257 f"UPDATE work_items SET status = {status}, updated_at = ? WHERE id = ?", 

258 (_now(), work_item_id), 

259 ): 

260 return None 

261 events.append( 

262 conn, 

263 work_item_id, 

264 GateEvent.REJECTED, 

265 {"gate": gate, "note": note, "node": node, "by": by, "verdict": verdict}, 

266 ) 

267 if reopen: 

268 reset_fix_loops(conn, work_item_id, list(rerun), by) 

269 if holder is not None: 

270 # The re-run is kept on the row, not only in the queued request, so a 

271 # person who pauses the held item and resumes it by hand gets it too: 

272 # the cursor at its target, the note as the next steer. 

273 if node is not None: 

274 conn.execute( 

275 "UPDATE work_items SET current_node_id = ?, current_step = 0 WHERE id = ?", 

276 (node, work_item_id), 

277 ) 

278 if note: 

279 set_steer(conn, work_item_id, note) 

280 # An agent's note is not a person's steer (`walk.run`'s `steer_source`). 

281 source = {"steer_source": "gate_review"} if by == "agent" else {} 

282 hold_for_checkout(conn, work_item_id, holder, {"node": node, **source}) 

283 return holder

344def gate_rounds( 

345 conn: sqlite3.Connection, work_item_id: str, *, with_contents: bool = True 

346) -> dict[str, list[dict]]: 

347 """Every gate's rounds, oldest first, read from the events (one SQL read, 

348 filtered to the types below: the item detail polls it). 

349 

350 A round is one time the chain reached a gate and something ended it. It 

351 opens at a `gate_requested` and ends at the first of a `gate_approved`, 

352 `gate_rejected` or `node_skipped` that names the gate in `payload.gate` 

353 (never the event's `node_id` column: a rejection stores its re-entry node 

354 there; a skip of a node with no gate carries `gate: null` and is no round, 

355 unless its `node_id` is a gate that already has rounds: after the reject 

356 cap `pending_gate` is None, so the skip of that gate carries `gate: null` 

357 too), or an item-level ending (`work_item_completed`, `work_item_abandoned`: 

358 outcome `closed`, for every round still open). A round with no ending yet is 

359 `pending`. 

360 

361 A `gate_requested` while the gate's round is still open (a restart asking 

362 again) is the same round: `requested_at` is the first ask. An ending with 

363 no open round is a round of its own, `requested_at` None. Three real paths 

364 write one: `pass_unchanged_revision` (a chain revision that changes 

365 nothing, approved by `kraft`), `bounce_on_feedback` (a review-thread bounce 

366 rejects before the gate is asked), and a skip of a gate whose round already 

367 ended (after the reject cap left the item on it). 

368 

369 `gate_reopened` ends and starts nothing: it marks the gate's newest ended 

370 round `reopened`. A rejection that breached the reject cap sent nothing 

371 back: the `work_item_needs_human` of `kind` cap on the gate that follows it 

372 sets `sent_to` None. `reviewer` is the round's own auto-review outcome, as 

373 `reviewOutcome` reads it in the UI: an ending `by: agent`, or a 

374 `gate_auto_review_skipped` with reason `passed` or `undecided`; the later 

375 wins. 

376 

377 With `with_contents` each round also carries its `review` and `threads` 

378 (`_attach_round_contents`): what was submitted at the gate in the round's 

379 window, `(previous round's ended_at, this round's ended_at]`. 

380 """ 

381 rows = conn.execute( 

382 "SELECT type, payload, created_at FROM events WHERE work_item_id = ? " 

383 f"AND type IN ({', '.join('?' * len(_ROUND_EVENTS))}) " 

384 "AND (type != ? OR json_extract(payload, '$.kind') = ?) ORDER BY seq", 

385 ( 

386 work_item_id, 

387 *_ROUND_EVENTS, 

388 WorkItemEvent.NEEDS_HUMAN, 

389 StopKind.CAP.value, 

390 ), 

391 ).fetchall() 

392 rounds: dict[str, list[dict]] = {} 

393 opened: dict[str, dict] = {} # gate -> its round with no ending yet 

394 

395 def end(gate: str, at: str, outcome: str, **ending) -> dict: 

396 r = opened.pop(gate, None) 

397 if r is None: 

398 r = _round(len(rounds.setdefault(gate, [])) + 1, None) 

399 rounds[gate].append(r) 

400 r.update(ended_at=at, outcome=outcome, **ending) 

401 return r 

402 

403 for row in rows: 

404 kind, at, p = row["type"], row["created_at"], json.loads(row["payload"]) 

405 gate = p.get("gate") 

406 if kind == ChainEvent.NODE_SKIPPED and not gate and p.get("node_id") in rounds: 

407 gate = p["node_id"] # a skip after the reject cap names no gate, only the node 

408 if not gate and kind in _GATE_EVENTS: 

409 continue 

410 if kind == GateEvent.REQUESTED: 

411 if gate not in opened: 

412 r = _round(len(rounds.setdefault(gate, [])) + 1, at) 

413 rounds[gate].append(r) 

414 opened[gate] = r 

415 elif kind in (GateEvent.APPROVED, GateEvent.REJECTED): 

416 rejected = kind == GateEvent.REJECTED 

417 outcome = ( 

418 ("fixed" if p.get("verdict") == "fixed" else "rejected") if rejected else "approved" 

419 ) 

420 r = end( 

421 gate, 

422 at, 

423 outcome, 

424 by=p.get("by"), 

425 note=p.get("note") if rejected else None, 

426 sent_to=p.get("node") if rejected else None, 

427 verdict=p.get("verdict") if rejected else None, 

428 ) 

429 if p.get("by") == "agent": 

430 note = p.get("note") or p.get("reason") or p.get("summary") or "" 

431 r["reviewer"] = { 

432 "kind": "reject" if rejected else "approve", 

433 "note": str(note).strip(), 

434 } 

435 elif kind == ChainEvent.NODE_SKIPPED: 

436 end(gate, at, "skipped", note=p.get("note")) 

437 elif kind in GATE_CLOSED: 

438 for g in list(opened): 

439 end(g, at, "closed") 

440 elif kind == GateEvent.REOPENED: 

441 if rounds.get(gate) and gate not in opened: 

442 rounds[gate][-1]["reopened"] = True 

443 elif kind == GateEvent.AUTO_REVIEW_SKIPPED: 

444 if gate in opened and p.get("reason") in _REVIEW_VERDICT: 

445 opened[gate]["reviewer"] = { 

446 "kind": _REVIEW_VERDICT[p["reason"]], 

447 "note": str(p.get("note") or "").strip(), 

448 } 

449 elif kind == WorkItemEvent.NEEDS_HUMAN: # only the cap's, by the query 

450 newest = (rounds.get(p.get("node_id")) or [None])[-1] 

451 if newest is not None and newest["outcome"] == "rejected": 

452 newest["sent_to"] = None 

453 if with_contents: 

454 _attach_round_contents(conn, work_item_id, rounds) 

455 return rounds

476def _windows(rounds: list[dict]) -> list[tuple[str | None, str | None]]: 

477 """Each round's window `(after, upto]` over a gate's rounds, oldest first: 

478 `after` is the previous round's `ended_at` (None for the first), `upto` its 

479 own (None while pending). The one place the rule lives: a review or thread 

480 made between two rounds belongs to the later one, and `requested_at` plays 

481 no part. Timestamps are ISO strings, so they compare as text.""" 

482 return [(rounds[i - 1]["ended_at"] if i else None, r["ended_at"]) for i, r in enumerate(rounds)]

531def _round_threads(threads: list[dict]) -> tuple[dict, list[dict]]: 

532 first = sorted(threads, key=lambda t: t["label"] != ThreadLabel.MUST_FIX)[:3] 

533 shown = [{k: v for k, v in t.items() if k != "created_at"} for t in first] 

534 return _thread_counts(threads), shown

537def _attach_round_contents( 

538 conn: sqlite3.Connection, work_item_id: str, rounds: dict[str, list[dict]] 

539) -> None: 

540 """Give each round its `review` (the newest one submitted at the gate in its 

541 window) and its `threads` / `first_threads`. Two reads for every gate.""" 

542 reviews: dict[str, list] = {} 

543 for r in conn.execute( 

544 "SELECT id, gate, outcome, summary, head_sha, submitted_at FROM reviews " 

545 "WHERE work_item_id = ? AND gate IS NOT NULL ORDER BY submitted_at, rowid", 

546 (work_item_id,), 

547 ): 

548 reviews.setdefault(r["gate"], []).append(r) 

549 threads = _gate_thread_rows(conn, work_item_id) 

550 for gate, rs in rounds.items(): 

551 for r, window in zip(rs, _windows(rs), strict=True): 

552 made = [x for x in reviews.get(gate, ()) if _in_window(window, x["submitted_at"])] 

553 if made: 

554 newest = made[-1] 

555 r["review"] = { 

556 "id": newest["id"], 

557 "outcome": newest["outcome"], 

558 "summary": newest["summary"] or "", 

559 "head_sha": newest["head_sha"], 

560 } 

561 r["threads"], r["first_threads"] = _round_threads( 

562 [t for t in threads.get(gate, ()) if _in_window(window, t["created_at"])] 

563 )
