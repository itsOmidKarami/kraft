25def approve_gate( 

26 conn: sqlite3.Connection, work_item_id, gate, *, by: str = "human", tips: dict | None = None 

27) -> bool | str: 

28 """Clear the gate and set the item running again. True when it is running. 

29 

30 `tips` is the head the chain's final gate approves (`reviewed_head.pin_for`), 

31 read before this write: recorded on `gate_approved`, it is what the merge 

32 may push. None for any other gate. 

33 

34 While another in-place item works in this item's checkout 

35 (`checkout_holder`), the approval is still recorded, the item is paused 

36 and queued to go on from the gate once the checkout frees 

37 (`hold_for_checkout`), and the holder's id is returned. False, and 

38 nothing written, for an ended item. 

39 

40 Also closes the gate node's own `node_started`/`node_completed` pair. A V1 

41 gate *is* a node (`gate-is-an-ordered-node`) and `gates.maybe_gate` enters 

42 it, but nothing ever completed it: the walk resumes at the node *after* the 

43 gate, so the board rendered an approved gate as a node still running, 

44 forever (4a's Concern 4). The gate's node id and its name are the same 

45 string in V1, which is why one argument answers both. 

46 

47 Written here rather than at each approval door so the human's `POST 

48 .../approve` and `review_gates`' `approve` verdict cannot drift -- the one 

49 rule the two doors have to keep. A rejection is deliberately *not* a 

50 completion: the gate reopens, and `gate_rejected` already says so. 

51 `complete_node` is idempotent, so re-approving writes one event. 

52 """ 

53 holder = checkout_holder(conn, work_item_id) 

54 if not write_status( 

55 conn, 

56 "UPDATE work_items SET status = 'active', updated_at = ? WHERE id = ?" 

57 if holder is None 

58 else "UPDATE work_items SET status = 'paused', updated_at = ? WHERE id = ?", 

59 (_now(), work_item_id), 

60 ): 

61 return False 

62 payload = {"gate": gate, "by": by, **({"tips": tips} if tips is not None else {})} 

63 events.append(conn, work_item_id, GateEvent.APPROVED, payload, node_id=gate) 

64 chain.complete_node(conn, work_item_id, gate) 

65 if holder is None: 

66 return True 

67 # `approved`: its replay refreshes the base first, as a direct approval does. 

68 hold_for_checkout(conn, work_item_id, holder, {"node": gate, "approved": True}) 

69 return holder

124def approval_stands( 

125 conn: sqlite3.Connection, work_item_id, gate: str, upstream: Collection[str] 

126) -> bool: 

127 """Whether `gate`'s approval still stands, so a walk passes over it. 

128 

129 Per gate: the newest of `gate_requested`, `gate_approved`, `gate_rejected` 

130 and `gate_reopened` that names `gate` must be its approval -- another 

131 gate's approval since says nothing about this one. And nothing since may 

132 have sent the chain back to a node in `upstream`, the ids of the chain's 

133 nodes up to and including `gate`: that re-run walks back through `gate` 

134 with new work, so it is asked again. Two motions do that without a 

135 `gate_reopened` for the gates they span, so their own events are the word: 

136 a rejection of any gate, re-entering at its `node` (one with none re-entered 

137 at its own gate), and a person's gateless `request_changes`, rewinding to its 

138 `target` unless it was cancelled (`review.request_rewind`). 

139 """ 

140 row = conn.execute( 

141 "SELECT seq, type FROM events WHERE work_item_id = ? " 

142 f"AND type IN ({', '.join('?' * len(_GATE_WORD))}) " 

143 "AND json_extract(payload, '$.gate') = ? ORDER BY seq DESC LIMIT 1", 

144 (work_item_id, *_GATE_WORD, gate), 

145 ).fetchone() 

146 if row is None or row["type"] != GateEvent.APPROVED: 

147 return False 

148 later = conn.execute( 

149 "SELECT type, json_extract(payload, '$.review_id') AS rid, " 

150 "COALESCE(json_extract(payload, '$.node'), json_extract(payload, '$.gate'), " 

151 "json_extract(payload, '$.target')) AS entered FROM events " 

152 "WHERE work_item_id = ? AND type IN (?, ?, ?) AND seq > ? ORDER BY seq DESC", 

153 ( 

154 work_item_id, 

155 GateEvent.REJECTED, 

156 GateEvent.REWIND_REQUESTED, 

157 GateEvent.REWIND_CANCELLED, 

158 row["seq"], 

159 ), 

160 ).fetchall() 

161 cancelled = set() 

162 for e in later: 

163 if e["type"] == GateEvent.REWIND_CANCELLED: 

164 cancelled.add(e["rid"]) 

165 elif e["entered"] in upstream and not ( 

166 e["type"] == GateEvent.REWIND_REQUESTED and e["rid"] in cancelled 

167 ): 

168 return False 

169 return True

172def _since_requested(conn: sqlite3.Connection, work_item_id, gate, type: str) -> list[dict]: 

173 """`type` events about `gate` since it was last requested, oldest first.""" 

174 found = [] 

175 for e in reversed(events.read_after(conn, 0, work_item_id)): 

176 if e["payload"].get("gate") != gate: 

177 continue 

178 if e["type"] == GateEvent.REQUESTED: 

179 break 

180 if e["type"] == type: 

181 found.insert(0, e["payload"]) 

182 return found

206def reject_gate( 

207 conn: sqlite3.Connection, 

208 work_item_id, 

209 gate, 

210 note, 

211 *, 

212 reopen: bool, 

213 node: str | None = None, 

214 by: str = "human", 

215 verdict: str | None = None, 

216 rerun: Sequence[str] = (), 

217) -> str | None: 

218 """Record the rejection. `reopen` flips the item back to active for the 

219 backward-motion re-run (02 §7.2); a rejection that breached the gate's 

220 reject loop leaves it needs_human. 

221 

222 A reopen while another in-place item works in this item's checkout 

223 (`checkout_holder`) pauses the item instead and queues the re-run from 

224 `node`, with the note as its steer (`hold_for_checkout`); the holder's id 

225 is returned. None otherwise. 

226 

227 `rerun` is the nodes with a fix loop that the re-run measures again: a 

228 rejection that reopens gives them their rounds back 

229 (`counters.reset_fix_loops`), here so that it is one write with the 

230 rejection, and an item that has ended (which records none) resets nothing. 

231 

232 `node` is the chain node the re-run enters at (Kraft-ko7j). It rides the 

233 event rather than a column: the events table is already append-only and 

234 already holds the note, and `store.last_rejection` reads both back. 

235 

236 `by` records who decided (Kraft-zr3s) -- a gate cleared by an agent and a 

237 gate cleared by a person have to be tellable apart in the timeline forever 

238 after. 

239 

240 `verdict` records *what* was decided, for the same reason and one level 

241 down: `by: agent` alone cannot tell a reviewer that rejected from one that 

242 repaired the worktree and committed (`gate_review.VERDICTS`' `fixed`), and 

243 those two have very different consequences -- a `fixed` re-enters at the 

244 gate's own node and regenerates the artifact the gate is about. Without it 

245 that cycle is invisible in every aggregate and only readable by putting 

246 session logs in order by hand (Kraft-s7c04.16). Rides the event alongside 

247 `note` and `node` for the reason those do. 

248 """ 

249 holder = checkout_holder(conn, work_item_id) if reopen else None 

250 status = "status" if not reopen else "'active'" if holder is None else "'paused'" 

251 if not write_status( 

252 conn, 

253 f"UPDATE work_items SET status = {status}, updated_at = ? WHERE id = ?", 

254 (_now(), work_item_id), 

255 ): 

256 return None 

257 events.append( 

258 conn, 

259 work_item_id, 

260 GateEvent.REJECTED, 

261 {"gate": gate, "note": note, "node": node, "by": by, "verdict": verdict}, 

262 ) 

263 if reopen: 

264 reset_fix_loops(conn, work_item_id, list(rerun), by) 

265 if holder is not None: 

266 # The re-run is kept on the row, not only in the queued request, so a 

267 # person who pauses the held item and resumes it by hand gets it too: 

268 # the cursor at its target, the note as the next steer. 

269 if node is not None: 

270 conn.execute( 

271 "UPDATE work_items SET current_node_id = ?, current_step = 0 WHERE id = ?", 

272 (node, work_item_id), 

273 ) 

274 if note: 

275 set_steer(conn, work_item_id, note) 

276 # An agent's note is not a person's steer (`walk.run`'s `steer_source`). 

277 source = {"steer_source": "gate_review"} if by == "agent" else {} 

278 hold_for_checkout(conn, work_item_id, holder, {"node": node, **source}) 

279 return holder

294def steer_source(conn: sqlite3.Connection, work_item_id: str, steer: str | None) -> str: 

295 """Who wrote `steer`, the item's pending steer (`walk.run`'s 

296 `steer_source`): `gate_review` when it is the note of an agent's rejection 

297 the item still sits on (a re-run `reject_gate` held for the checkout), 

298 `human` otherwise. Read off the rejection itself, so a person who pauses 

299 the held item and resumes it does not make the agent's note theirs.""" 

300 rejected = last_rejection(conn, work_item_id) 

301 by_agent = rejected is not None and rejected.get("by") == "agent" 

302 return "gate_review" if steer and by_agent and rejected.get("note") == steer else "human"

305def last_rejection(conn: sqlite3.Connection, work_item_id: str) -> dict | None: 

306 """The `gate_rejected` payload the item is currently sitting on, or None. 

307 

308 Read back out of the append-only events table rather than stored a second 

309 time on the row: the note is already durable there, it just had no reader 

310 (Kraft-ko7j). 

311 """ 

312 for e in reversed(events.read_after(conn, 0, work_item_id)): 

313 if e["type"] == GateEvent.REJECTED: 

314 return e["payload"] 

315 if e["type"] in _REJECTION_SPENT: 

316 return None 

317 return None
