1"""One seam under the forge backends: send a REST or GraphQL request, get an 

2`ApiResponse` or a `ForgeError` that carries the HTTP `status` and a `reason`. 

3 

4`CliTransport` does it with `gh api -i` / `glab api -i`, which print the status 

5line and headers before the body, so the status, `Retry-After` and 

6`X-GitHub-SSO` reach the same mapping (`models.reason_for`) any transport uses. 

7"""

64def _clip(text: str) -> str: 

65 text = " ".join(text.split()) 

66 return text if len(text) <= _CLIP else f"{text[:_CLIP]}…"

69def parse_included(stdout: str) -> ApiResponse | None: 

70 """The response in `api -i` output, or None when stdout holds no status line. 

71 

72 Read from the start: only a 1xx block (`100 Continue`) followed by another 

73 status block is skipped, so a body that quotes a status line stays the body.""" 

74 blocks = stdout.replace("\r\n", "\n").split("\n\n") 

75 start = 0 

76 if not _STATUS_LINE.match(blocks[0]): 

77 return None 

78 while ( 

79 start + 1 < len(blocks) 

80 and _STATUS_LINE.match(blocks[start])[1].startswith("1") # type: ignore[index] 

81 and _STATUS_LINE.match(blocks[start + 1]) 

82 ): 

83 start += 1 

84 head, *rest = blocks[start].split("\n") 

85 headers = { 

86 k.strip().lower(): v.strip() for k, _, v in (line.partition(":") for line in rest) if _ 

87 } 

88 body = "\n\n".join(blocks[start + 1 :]) 

89 return ApiResponse(int(_STATUS_LINE.match(head)[1]), headers, body) # type: ignore[index]

92def graphql_reason(errors: list[dict]) -> Reason | None: 

93 """GraphQL answers 200 with `errors[]`; the first error that maps decides.""" 

94 for err in errors: 

95 kind, msg = err.get("type"), str(err.get("message", "")).lower() 

96 if kind == "NOT_FOUND": 

97 return "not_found" 

98 if kind == "FORBIDDEN": 

99 return "forbidden" 

100 if kind == "RATE_LIMITED": 

101 return "rate_limited" 

102 if "expectedheadoid" in msg or ( 

103 "head" in msg and ("out of date" in msg or "modified" in msg) 

104 ): 

105 return "head_moved" 

106 return None

109def origin_host(repo: Path) -> str | None: 

110 """The host of the repo's `origin`, None without one.""" 

111 remote = git_read(repo, "remote", "get-url", "origin", expected_failure=True) 

112 match = next((m for rx in _REMOTE if remote and (m := rx.match(remote.strip()))), None) 

113 return match["host"].lower() if match else None

116@lru_cache(maxsize=256) 

117def _gh_host(repo: Path) -> str | None: 

118 """The host `gh api` should be told about, as gh itself reads the remote: an 

119 scp-style or `ssh://` origin goes through `ssh -G`, which turns an alias 

120 (`github-work`) into its real host, and GitHub's SSH-over-443 host is 

121 github.com. None without an origin.""" 

122 host = origin_host(repo) 

123 if host is None or host.startswith("-"): 

124 return None 

125 remote = git_read(repo, "remote", "get-url", "origin", expected_failure=True) or "" 

126 if not remote.strip().lower().startswith(("http://", "https://", "git://")): 

127 try: 

128 out = subprocess.run( 

129 ["ssh", "-G", host], 

130 capture_output=True, 

131 text=True, 

132 timeout=5, 

133 stdin=subprocess.DEVNULL, 

134 ).stdout 

135 except (OSError, subprocess.SubprocessError): 

136 out = "" # no ssh to ask: the host as written 

137 host = next( 

138 ( 

139 ln.split(None, 1)[1].strip().lower() 

140 for ln in out.splitlines() 

141 if ln.startswith("hostname ") 

142 ), 

143 host, 

144 ) 

145 return "github.com" if host == "ssh.github.com" else host

148async def gh_host(repo: Path) -> str | None: 

149 """`_gh_host`, asked off the event loop and remembered per repo: it runs 

150 git and maybe `ssh -G`, and a moved remote is a restart-level change.""" 

151 return await asyncio.to_thread(_gh_host, repo)

158def _retry_after(headers: Mapping[str, str]) -> float | None: 

159 """Seconds to wait: `Retry-After`, else the time left until GitHub's 

160 `x-ratelimit-reset` (epoch seconds) when its quota is spent.""" 

161 retry = headers.get("retry-after", "") 

162 if retry.replace(".", "", 1).isdigit(): 

163 return float(retry) 

164 reset = headers.get("x-ratelimit-reset", "") 

165 if headers.get("x-ratelimit-remaining") == "0" and reset.isdigit(): 

166 return max(0.0, int(reset) - _now()) 

167 return None

170def _flags(fields: Mapping[str, Field] | None) -> list[str]: 

171 flags: list[str] = [] 

172 for key, value in (fields or {}).items(): 

173 if isinstance(value, list | tuple): 

174 # A list is gh's repeated `key[]=v`; raw, as every string is. 

175 flags += [x for item in value for x in ("-f", f"{key}[]={item}")] 

176 continue 

177 typed = not isinstance(value, str) or value in _PLACEHOLDERS 

178 if isinstance(value, bool): 

179 value = "true" if value else "false" 

180 flags += ["-F" if typed else "-f", f"{key}={value}"] 

181 return flags

189class CliTransport: 

190 def __init__(self, cli: Literal["gh", "glab"], repo: Path): 

191 self.cli, self.repo = cli, repo 

192 

193 async def _call( 

194 self, label: str, args: list[str], *, graphql: bool = False 

195 ) -> tuple[ApiResponse, int]: 

196 """Run `<cli> api … -i`; the response and exit code, or the `ForgeError` it 

197 earned. A GraphQL 200 is returned whatever the exit code: its `errors[]` 

198 hold the reason.""" 

199 # A job log can hold bytes that are not UTF-8 (a binary cat, a Latin-1 

200 # compiler); strict decoding would raise past every ForgeError handler. 

201 done = await git.run_cli(self.repo, [self.cli, "api", *args], errors="replace") 

202 resp = parse_included(done.stdout) 

203 if resp is None: 

204 if done.returncode == 0: 

205 raise ForgeError(f"{label}: {self.cli} printed no HTTP response") 

206 # No request was made (not signed in, no network): no status, no reason, 

207 # so `failure_cause`'s text markers still decide. 

208 raise ForgeError(f"{label}: {_clip(done.stderr.strip() or done.stdout)}") 

209 if resp.status >= 400 or (done.returncode != 0 and not graphql): 

210 try: 

211 message = str(resp.json().get("message", "")) 

212 except (ForgeError, AttributeError): 

213 message = "" 

214 raise ForgeError( 

215 f"{label}: {message or _clip(resp.text or done.stderr)} (HTTP {resp.status})", 

216 status=resp.status, 

217 reason=reason_for(resp.status, resp.headers, message), 

218 retry_after=_retry_after(resp.headers), 

219 sso_url=_sso_url(resp.headers.get("x-github-sso", "")), 

220 ) 

221 return resp, done.returncode 

222 

223 async def _host(self) -> list[str]: 

224 host = await gh_host(self.repo) if self.cli == "gh" else None 

225 return ["--hostname", host] if host and host != "github.com" else []

227 async def rest( 

228 self, 

229 method: str, 

230 path: str, 

231 *, 

232 fields: Mapping[str, Field] | None = None, 

233 tail: bool = False, # the CLI hands over the whole body; only HTTP windows it 

234 ) -> ApiResponse: 

235 args = [path, "-X", method, "-i", *_flags(fields), *await self._host()] 

236 return (await self._call(f"{method} {path}", args))[0] 

237 

238 async def graphql(self, query: str, variables: Mapping[str, Field] | None = None) -> dict: 

239 args = ["graphql", "-i", "-f", f"query={query}", *_flags(variables), *await self._host()] 

240 resp, rc = await self._call("graphql", args, graphql=True) 

241 body = resp.json() 

242 if not isinstance(body, dict): 

243 raise ForgeError(f"graphql: answered without an object: {_clip(resp.text)}") 

244 if errors := body.get("errors"): 

245 message = "; ".join(str(e.get("message", e)) for e in errors) 

246 raise ForgeError( 

247 f"graphql: {_clip(message)}", 

248 status=resp.status, 

249 reason=graphql_reason(errors), 

250 retry_after=_retry_after(resp.headers), 

251 sso_url=_sso_url(resp.headers.get("x-github-sso", "")), 

252 ) 

253 if rc != 0: 

254 raise ForgeError(f"graphql: {self.cli} exited {rc}", status=resp.status) 

255 return body.get("data") or {}
