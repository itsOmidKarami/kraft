1"""What every sandbox backend shares (Kraft-rki): host git hardening, and the 

2guards on a sandboxed worktree. 

3 

4`--disallowed-tools` is the agent CLI policing itself at Kraft's request, not 

5a sandbox: it does not stop a worker that spawns a shell directly, and a 

6command-scoped allowlist does not help either -- an allowlisted `pytest` 

7still executes a `conftest.py` the implementation node itself wrote, in the 

8same host process, as the invoking user. How a command runs inside a sandbox 

9is `kraft.worker.backends`'. 

10 

11Isolation is two halves, and the second is the load-bearing one. A backend 

12keeps the sandbox out of everything it has no business writing. But a 

13worker's own worktree gitdir must stay writable for it to commit at all, and 

14that is where every file git redirects hooks and config through lives -- so 

15`harden_host_git_env` pins `core.hooksPath` (and the other program-valued 

16config keys) for every git Kraft runs, which makes what a worker writes there 

17unable to execute regardless of which redirect file it used. 

18 

19Read docs/superpowers/specs/2026-09-13-worker-sandbox-docker-design.md for 

20the design this implements. 

21"""

57def submodule_refusal(who: str) -> str: 

58 """Why a sandbox and submodule members are refused together (Kraft-dshto, 

59 Ruling 180): one sentence, the same at load, connect, intake, doctor and 

60 for an item already in flight. `who` names what set the sandbox. 

61 

62 A container must write its worktree's gitdir to commit, and a submodule's 

63 gitdir lives inside it (`<worktree gitdir>/modules/<sm>`), so a worker can 

64 plant a hook, `core.sshCommand` or a filter driver there. `push` runs 

65 unhardened, and nothing pins the other two, so host git would run them 

66 as the operator.""" 

67 return ( 

68 f"{who} sets a sandbox, and a sandboxed item cannot mount submodules: a " 

69 "sandboxed worker can write each submodule's gitdir and plant hooks or " 

70 "core.sshCommand there, which host git runs as you when Kraft pushes the " 

71 "submodule (Kraft-dshto). Drop the sandbox, or work on one repository" 

72 )

83def git_identity(cwd: Path) -> dict[str, str]: 

84 """The identity a worker commits under, as git's env form: the daemon's 

85 own `GIT_AUTHOR_*`/`GIT_COMMITTER_*` where set, else the host's 

86 `user.name`/`user.email` for this repository. The container has neither 

87 the host's `~/.gitconfig` nor a writable repo config to set one in, so 

88 without this every commit fails "unable to auto-detect email address". 

89 

90 Read through the common gitdir, never the worktree: a co-task's container 

91 may be writing that worktree right now.""" 

92 dirs = linked_gitdirs(cwd) 

93 gitdir = dirs[0] if dirs is not None else cwd / ".git" 

94 env = dict(os.environ) 

95 harden_host_git_env(env) 

96 found: dict[str, str] = {} 

97 looked_up: dict[str, str | None] = {} 

98 for name, key in _IDENTITY.items(): 

99 if os.environ.get(name): 

100 found[name] = os.environ[name] 

101 continue 

102 if key not in looked_up: 

103 done = subprocess.run( 

104 ["git", f"--git-dir={gitdir}", "config", "--get", key], 

105 capture_output=True, 

106 text=True, 

107 env=env, 

108 ) 

109 looked_up[key] = done.stdout.strip() if done.returncode == 0 else None 

110 if looked_up[key]: 

111 found[name] = looked_up[key] 

112 return found

115def _gitdir_of(cwd: Path) -> Path | None: 

116 """The gitdir `cwd/.git` names, resolved, when it is a `gitdir:` file.""" 

117 git_path = cwd / ".git" 

118 if git_path.is_dir(): 

119 return None 

120 try: 

121 text = git_path.read_text() 

122 except OSError: 

123 return None 

124 if not text.startswith("gitdir:"): 

125 return None 

126 return Path(text.removeprefix("gitdir:").strip()).resolve()

129def linked_gitdirs(cwd: Path) -> tuple[Path, Path] | None: 

130 """`(common gitdir, worktree gitdir)` for a linked worktree, resolved, or 

131 None when `cwd/.git` is a directory or does not name `<repo>/.git/worktrees/<id>`.""" 

132 gitdir = _gitdir_of(cwd) 

133 if gitdir is None or gitdir.parent.name != "worktrees": 

134 return None 

135 return gitdir.parent.parent, gitdir

147#: Config a repository must not be able to set for a git *Kraft itself* runs, 

148#: because each of these values is a program git executes. 

149#: 

150#: A worker's worktree gitdir is read-write by construction -- a commit writes 

151#: its index, HEAD, lockfiles and per-worktree refs, so the directory holding 

152#: them cannot be read-only, and every file git redirects through lives in 

153#: that same directory (`commondir`, `config.worktree`, `modules/<sm>/config`, 

154#: and whichever one the next git version adds). Enumerating them one shadow 

155#: mount at a time is a list of the escapes someone thought of; this is the 

156#: other side of the same problem, and it does not need the list: whatever a 

157#: worker plants, the host-side git that would have run it is launched with 

158#: these pinned instead. 

159#: 

160#: `/dev/null` as a hooks *directory* resolves every hook to `/dev/null/<name>`, 

161#: which is not a file, so git finds no hook to run. `core.fsmonitor` names a 

162#: program git spawns on `status`; empty is "unset", git's own default. 

163#: 

164#: `diff.external` is deliberately *not* here even though it is the same kind 

165#: of vector: git reads an empty value as the empty command and dies 

166#: ("cannot run : No such file or directory") on every `git diff`, which is 

167#: half of what Kraft does with a worktree. Nothing safe to pin it to exists, 

168#: so that one stays covered by the mounts -- the config files git could read 

169#: it from are the common gitdir's `config` (read-only mount), `config.worktree` 

170#: and whatever `commondir` redirects to (both shadowed read-only). 

171_HOOK_PINS = ( 

172 ("core.hooksPath", os.devnull), 

173 ("core.fsmonitor", ""), 

174) 

175 

176#: Host git never walks into a repository nested in a worktree (Kraft-nx4id). 

177#: 

178#: A worker can build a git repository of its own anywhere in the worktree it 

179#: writes and commit a gitlink to it -- no `.gitmodules`, no declared member. 

180#: That repository's config is the worker's: `filter.<x>.clean`, a textconv, 

181#: an `ext::` remote, anything with a name git will not let `-c` enumerate. 

182#: The main worktree's config is safe to read (the common gitdir is mounted 

183#: read-only, `config.worktree` and `commondir` are shadowed, and git reads no 

184#: `config` from `<gitdir>/worktrees/<id>` itself); a nested repository's is 

185#: not, so what has to hold is that host git never *enters* one. Each of these 

186#: is a way it would, from config the operator (not the worker) may have set: 

187#: `submodule.recurse` makes checkout, reset and rebase recurse, the 

188#: `recurseSubmodules` pair makes fetch and push run inside a submodule with 

189#: its own remotes, `diff.submodule=diff`/`log` and `status.submoduleSummary` 

190#: run diff and log inside it, and `diff.ignoreSubmodules` is the default for 

191#: every `status`/`diff` below -- `dirty` compares only the commit a gitlink 

192#: records, which git reads without spawning anything in the nested repo. 

193#: That last one is only a default: a `.gitmodules` the worker writes can 

194#: override it per path, which is why the calls that compare a worktree also 

195#: pass `SUBMODULES_UNENTERED` on the command line. 

196_NO_RECURSION_PINS = ( 

197 ("submodule.recurse", "false"), 

198 ("fetch.recurseSubmodules", "false"), 

199 ("push.recurseSubmodules", "no"), 

200 ("diff.submodule", "short"), 

201 ("status.submoduleSummary", "false"), 

202 ("diff.ignoreSubmodules", "dirty"), 

203) 

204 

205_HARDENED_GIT_CONFIG = _HOOK_PINS + _NO_RECURSION_PINS 

206 

207#: For every host `status` and worktree `diff` Kraft runs in a worktree: a 

208#: gitlink counts as changed when the commit it points at moves (Kraft-qlsf 

209#: needs that much), but git never runs `status` inside it to look for 

210#: uncommitted edits -- that child process is what loads a planted repo's 

211#: config and runs its filters (review-g1 finding 1). On the command line, 

212#: because a worker's `.gitmodules` can override `diff.ignoreSubmodules`. 

213SUBMODULES_UNENTERED = "--ignore-submodules=dirty"

216def _pin(env: MutableMapping[str, str], pins: tuple[tuple[str, str], ...]) -> None: 

217 env["GIT_CONFIG_COUNT"] = str(len(pins)) 

218 for i, (key, value) in enumerate(pins): 

219 env[f"GIT_CONFIG_KEY_{i}"] = key 

220 env[f"GIT_CONFIG_VALUE_{i}"] = value

223def harden_host_git_env(env: MutableMapping[str, str] | None = None) -> None: 

224 """Pin `_HARDENED_GIT_CONFIG` for every git this process ever spawns. 

225 

226 `GIT_CONFIG_COUNT`/`_KEY_n`/`_VALUE_n` is git's own environment form of 

227 `git -c key=value`, and carries `-c`'s precedence: it beats the system, 

228 global, repo *and* worktree config files, so a `core.hooksPath` a worker 

229 plants in any of them loses. Setting it on the server's own environment 

230 once covers every git Kraft runs -- `builtins`, `adapters.forge.git`, 

231 `adapters.forge.run`, `config.git_read`, `index.ingest`, the worktree 

232 teardown in `api.routes.lifecycle` -- and every git those spawn in turn 

233 (a submodule's, a rebase's), without each call site having to remember. 

234 Doing it per call site is what left `git worktree prune` and 

235 `git submodule update` running a worker's hook while `git commit` had 

236 `--no-verify`. 

237 

238 Kraft's own worker sessions inherit it too (they are children of this 

239 process), so a repo hook does not fire on an agent's commits either -- 

240 the same thing `adapters.forge.git` already asked for explicitly with 

241 `--no-verify`. 

242 

243 `adapters.forge.git.push` is the one exception, running with 

244 `unhardened_git_env()` instead: see that function for why. 

245 """ 

246 _pin(os.environ if env is None else env, _HARDENED_GIT_CONFIG)

249def unhardened_git_env() -> dict[str, str]: 

250 """A copy of the process env with `harden_host_git_env`'s hook pin lifted. 

251 

252 For `git push` alone: by the time Kraft pushes, the commit is already 

253 made and the worktree gitdir's hooks are either read-only-mounted 

254 (sandboxed) or none of a worker's business to have tampered with in the 

255 first place (unsandboxed) -- so the pinned `core.hooksPath=/dev/null` 

256 is no longer buying anything there, only breaking real hooks a human 

257 installed, like git-lfs's pre-push (Kraft-rki). 

258 

259 Only that pin: `_NO_RECURSION_PINS` stay, so an operator's 

260 `push.recurseSubmodules=on-demand` still cannot make this push run 

261 `git push` inside a repository the worker nested in its worktree, with 

262 that repository's remotes and `core.sshCommand` (Kraft-nx4id). 

263 """ 

264 env = dict(os.environ) 

265 _pin(env, tuple(p for p in _HARDENED_GIT_CONFIG if p[0] != "core.hooksPath")) 

266 return env

276def nested_repos(worktree: Path) -> dict[str, str | None] | None: 

277 """Every path in `worktree` host git could treat as a repository of its 

278 own, mapped to the commit its gitlink records -- None for an untracked 

279 one, which `git add -A` would stage as a gitlink. None when git cannot 

280 read the worktree at all. 

281 

282 Read without entering any of them (Kraft-nx4id): `ls-files -s` is the 

283 index alone, and `ls-files --others` lists an untracked directory holding 

284 a `.git` as `path/` without opening it -- neither parses the nested 

285 repository's config, let alone runs anything from it.""" 

286 staged = _ls_files(worktree, "-s") 

287 untracked = _ls_files(worktree, "--others", "--exclude-standard") 

288 if staged is None or untracked is None: 

289 return None 

290 found: dict[str, str | None] = {} 

291 for entry in staged: 

292 meta, _, path = entry.partition("\t") 

293 mode, sha, _ = meta.split(" ", 2) 

294 if mode == "160000": 

295 found[path] = sha 

296 for path in untracked: 

297 if path.endswith("/"): 

298 found[path.rstrip("/")] = None 

299 return found

302def _gitlinks_at(worktree: Path, rev: str) -> dict[str, str] | None: 

303 done = subprocess.run( 

304 ["git", "ls-tree", "-r", "-z", rev], cwd=worktree, capture_output=True, text=True 

305 ) 

306 if done.returncode != 0: 

307 return None 

308 links = {} 

309 for entry in done.stdout.split("\0")[:-1]: 

310 meta, _, path = entry.partition("\t") 

311 mode, _, sha = meta.split(" ", 2) 

312 if mode == "160000": 

313 links[path] = sha 

314 return links

317def planted_repos(worktree: Path, base: str | None) -> list[str] | None: 

318 """The nested repositories in a sandboxed item's `worktree` its worker 

319 made, which no host git may be let near -- or None when git cannot say. 

320 

321 A sandboxed item declares no submodules (Ruling 180 refuses the pairing), 

322 so none of these is Kraft's: an untracked nested repository, a gitlink 

323 whose directory holds a `.git` (git enters those), and a gitlink the 

324 branch added or moved since `base`. A gitlink `base` already had, left 

325 unpopulated, is the repository's own and inert -- git has nothing to 

326 enter -- so it does not stop a sandboxed item on a repo with submodules.""" 

327 found = nested_repos(worktree) 

328 at_base = _gitlinks_at(worktree, base) if base else {} 

329 if found is None or at_base is None: 

330 return None 

331 return sorted( 

332 path 

333 for path, sha in found.items() 

334 if sha is None 

335 or os.path.lexists(worktree / path / ".git") 

336 or (base and at_base.get(path) != sha) 

337 )
