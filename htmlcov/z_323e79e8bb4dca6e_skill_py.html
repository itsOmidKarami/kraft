1"""The *method* an agent hook follows, injected through the system prompt. 

2 

3Sibling of `steering.py`, and deliberately shaped like it: same bare-name rule, 

4same validate/read split, same reason for the split — the value must round-trip 

5through `GET /registry` and the Settings screens as a name, so resolving it into 

6the config dict would inline a method body into the operator's YAML. 

7 

8Where it differs is the fallback. Steering is operator-authored and exists only 

9under `$KRAFT_HOME`; a method is Kraft-authored and *shipped*, with the operator 

10directory as an optional overlay on top. A value containing `:` is not a file 

11at all: it names a skill in some other tool's plugin system, which Kraft cannot 

12read, so the agent is told to load it by name. The one exception is Kraft's own 

13qualifier: `kraft:spec` is the method Kraft ships as `spec`. 

14""" 

15 

16from __future__ import annotations 

17 

18import os 

19from collections.abc import Mapping 

20from pathlib import Path 

21 

22#: Method files that ship with Kraft. Read from the package, never from 

23#: `$KRAFT_HOME/config/`, because `cli.seed_home` copies templates once and 

24#: never again — a shipped method would then be frozen at whichever version the 

25#: operator first installed. 

26BUNDLED = Path(__file__).parent / "skills" 

27 

28#: What `adapters/agent.py` wraps the method text in. 

29HEADING = "\n\n## Method\n\n" 

30 

31#: Appended after the method text, whatever its source. A reference to another 

32#: tool's plugin skill can name a skill this agent's installation does not 

33#: have; without this the agent improvises a method and the human finds out at 

34#: the gate. 

35UNAVAILABLE = ( 

36 "\n\nIf you cannot load the method named above, stop with status " 

37 '"needs_context" and say in "question" exactly which method was missing. Do ' 

38 "not improvise a method of your own." 

39) 

40 

41#: What a `provider:name` value that names no loaded add-on becomes. Kraft 

42#: cannot read another tool's plugin cache, so the reference is handed to the 

43#: agent, which can. 

44PLUGIN_PROMPT = "Follow the {ref} skill for how to do this work. Load it before you start." 

45 

46 

47#: The qualifier of the methods Kraft itself ships. A V1 skill name is 

48#: namespace-qualified (`templates.models.AgentTask.skill`), and `kraft:spec` 

49#: names Kraft's own shipped method -- the same file a bare `spec` does 

50#: (Kraft-vhcop). 

51OWN_PLUGIN = "kraft:" 

52 

53 

54class SkillError(Exception): 

55 pass 

56 

57 

58#: Shipped methods that were renamed, old name to new. `library.yaml` is seeded 

59#: once and never overwritten, so a home seeded before the rename still names 

60#: the old one (Kraft-35u4m.1). 

61RENAMED = {"mr-checks-repair": "mr-metadata-repair"} 

62

64def _own_name(value: str) -> str: 

65 """`value` with Kraft's own `kraft:` qualifier removed, if it has one, and a 

66 renamed method's old name read as its new one.""" 

67 name = value.removeprefix(OWN_PLUGIN) 

68 return RENAMED.get(name, name) 

69 

70 

71def is_plugin_ref(value: str) -> bool: 

72 """A `:` means a namespaced skill, a Kraft add-on's or another tool's plugin's, 

73 not a file name -- except Kraft's own `kraft:` qualifier, which names a 

74 method Kraft ships.""" 

75 return not value.startswith(OWN_PLUGIN) and ":" in value 

76 

77 

78def _is_file(path: Path) -> bool: 

79 """`Path.is_file()`, but a name too long for the filesystem is "no such file" 

80 on every Python: 3.14 answers False, 3.12 and 3.13 raise ENAMETOOLONG.""" 

81 try: 

82 return path.is_file() 

83 except OSError: 

84 return False

87def _bare(name: str, where: str) -> None: 

88 if "/" in name or "\\" in name or name in ("", ".", "..") or name.startswith("."): 

89 raise SkillError( 

90 f"{where}: skill name {name!r} must be a bare directory name — Kraft " 

91 "never reads a method file from outside its own skills directories" 

92 ) 

93 

94 

95def _local_path(skills_dir: Path | None, name: str, where: str) -> Path: 

96 _bare(name, where) 

97 if skills_dir is not None: 

98 overlay = Path(skills_dir) / name / "SKILL.md" 

99 if _is_file(overlay): 

100 return overlay 

101 bundled = BUNDLED / name / "SKILL.md" 

102 if _is_file(bundled): 

103 return bundled 

104 looked = [str(bundled)] 

105 if skills_dir is not None: 

106 looked.insert(0, str(Path(skills_dir) / name / "SKILL.md")) 

107 raise SkillError(f"{where}: skill {name!r} not found at {' or '.join(looked)}") 

108 

109 

110def path_for(skills_dir: Path | None, name: str, *, where: str) -> Path: 

111 """The file a bare or `kraft:` skill name resolves to, or raise. Not for 

112 a namespaced ref (an add-on's or another tool's plugin's).""" 

113 return _local_path(skills_dir, _own_name(name), where)

116def _no_skill(where: str, value: str, path: Path) -> SkillError: 

117 qualifier, _, name = value.partition(":") 

118 return SkillError( 

119 f"{where}: add-on {qualifier!r} has no skill {name!r} at {path}; a method Kraft " 

120 f"ships is written kraft:{name}" 

121 )

124def _addon_path( 

125 addon_dirs: Mapping[str, Path | None] | None, value: str, where: str, *, present: bool = True 

126) -> tuple[Path, str] | None: 

127 """The store and the store-relative file `<namespace>:<name>` names when 

128 the namespace is a loaded Kraft add-on's; None when it is not, which 

129 leaves the reference another tool's. `present` also requires the file to 

130 be there; `read` asks the store's manifest instead. 

131 

132 A skill a child add-on inherits is its own key, `<namespace>:<name>`, to 

133 the store of the ancestor that defines it (`inherit.skill_dirs`); the 

134 namespace is still asked first, so a child that is not loaded is refused. 

135 A key with `:` is never a namespace, which is a bare name.""" 

136 dirs = addon_dirs or {} 

137 qualifier, _, name = value.partition(":") 

138 if qualifier not in dirs: 

139 return None 

140 root = dirs[qualifier] 

141 if root is not None and value in dirs: 

142 root = dirs[value] 

143 if root is None: 

144 raise SkillError( 

145 f"{where}: add-on {qualifier!r} is installed but not loaded, so its skill " 

146 f"{name!r} cannot be read; `kraft admin health` says why" 

147 ) 

148 _bare(name, where) 

149 rel = f"skills/{name}/SKILL.md" 

150 from kraft.addons.manifest import in_layout # manifest imports kraft.config, which imports this 

151 

152 # A name no store can hold (`Deploy`) is no skill, even where a 

153 # case-insensitive filesystem finds `skills/deploy/` under it. 

154 if not in_layout(rel) or (present and not _is_file(Path(root) / rel)): 

155 raise _no_skill(where, value, Path(root) / rel) 

156 return Path(root), rel

159def validate( 

160 skills_dir: Path | None, 

161 value, 

162 *, 

163 where: str, 

164 addon_dirs: Mapping[str, Path | None] | None = None, 

165) -> None: 

166 """The value is a usable skill reference, or raise. 

167 

168 Another tool's plugin reference is accepted without a lookup: whether the 

169 agent's installation has it is not knowable here, and the injected text 

170 tells the agent to stop with `needs_context` if it cannot load it. A 

171 loaded Kraft add-on's is looked up in its store (`addon_dirs`, namespace 

172 to directory). 

173 """ 

174 if not isinstance(value, str) or not value: 

175 raise SkillError(f"{where}: 'skill' must be a non-empty string") 

176 if is_plugin_ref(value): 

177 _addon_path(addon_dirs, value, where) 

178 return 

179 _local_path(skills_dir, _own_name(value), where)

182def read( 

183 skills_dir: Path | None, value: str, addon_dirs: Mapping[str, Path | None] | None = None 

184) -> str: 

185 """The method text to inject. Called at dispatch, after `validate`. An 

186 add-on's is read through its store's digest, so a store edited after it 

187 was written is refused here, before its text reaches an agent.""" 

188 if is_plugin_ref(value): 

189 found = _addon_path(addon_dirs, value, "skill", present=False) 

190 if found is None: 

191 return PLUGIN_PROMPT.format(ref=value) 

192 from kraft.addons import fetch # fetch imports kraft.config, which imports this 

193 

194 root, rel = found 

195 try: 

196 store = fetch.open_store(root) 

197 # A listed file that is gone is an edit of the store, not a missing skill. 

198 if rel not in store.listed and not os.path.lexists(root / rel): 

199 raise _no_skill("skill", value, root / rel) 

200 return (store.read(rel) or b"").decode() 

201 except fetch.StoreMismatch as exc: 

202 qualifier = value.partition(":")[0] 

203 raise SkillError(f"skill: add-on {qualifier!r} could not be read: {exc}") from exc 

204 except (OSError, ValueError) as exc: 

205 raise SkillError(f"skill: cannot read {value!r} at {root / rel}: {exc}") from exc 

206 path = _local_path(skills_dir, _own_name(value), "skill") 

207 try: 

208 return path.read_text() 

209 except (OSError, ValueError) as exc: 

210 raise SkillError(f"skill: cannot read {value!r} at {path}: {exc}") from exc
