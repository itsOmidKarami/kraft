1"""GitHub over its REST and GraphQL APIs with an account's token, never through `gh`. 

2 

3`GhCli` already sends every request as a `repos/{owner}/{repo}/...` path, or a GraphQL 

4document with `{"owner": "{owner}", "name": "{repo}"}` variables, which `gh` expands. So 

5this is `GhCli` on the shared account transport (`api_account`), filling the same 

6placeholders from the repo's push URL. Every request shape stays in `gh.py`.""" 

7 

8from __future__ import annotations 

9 

10import re 

11from pathlib import Path 

12from urllib.parse import quote 

13 

14from kraft.adapters.forge.api_account import ApiAccount 

15from kraft.adapters.forge.gh import GhCli 

16from kraft.adapters.forge.models import ForgeError 

17from kraft.forges import github_owner_name 

18 

19_PLACEHOLDER = re.compile(r"\{owner\}|\{repo\}") 

20 

21 

22class GitHubApi(ApiAccount, GhCli): 

23 """`client: api` on a GitHub account (github.com, GHES or `SUBDOMAIN.ghe.com`).""" 

24 

25 _PLACEHOLDER = _PLACEHOLDER 

26 _VALUE_PLACEHOLDER = _PLACEHOLDER # as `gh api -F`: the whole value, nothing in a string 

27 _PROJECT_WORD = "a GitHub repository" 

28 

29 async def _project(self, repo: Path) -> str: 

30 project = await super()._project(repo) 

31 try: 

32 github_owner_name(project, repo.name) 

33 except ValueError as exc: 

34 raise ForgeError(f"forge account {self.name}: {exc}") from None 

35 return project 

36 

37 def _segment(self, project: str, placeholder: str) -> str: 

38 owner, name = project.split("/") 

39 return owner if placeholder == "{owner}" else name 

40 

41 def _fill(self, project: str, path: str) -> str: 

42 return _PLACEHOLDER.sub(lambda m: quote(self._segment(project, m[0]), safe=""), path) 

43 

44 def _fill_value(self, project: str, value: str) -> str: 

45 return self._segment(project, value) # JSON, not a path: the raw segment
