19#: What runs `kraft`: this interpreter, `-P` so a file in the cwd named like a 

20#: module Kraft imports cannot run instead. A test swaps it for a `-c` script. 

21COMMAND = (sys.executable, "-P", "-m", "kraft") 

22#: How long a quitting console waits for a command to stop on SIGINT before 

23#: it kills it. 

24QUIT_GRACE_S = 3.0

30 ("console",): "This is the console.",

33#: would run inside the console and stop when it quits. 

34ATTACHED = "The server would stop with the console; `/admin start --detach` starts it on its own."

68def _without_yes(word: str) -> str: 

69 """`word` with `-y` (also bundled: `-yy`, `-yh`) and `--yes` or any 

70 abbreviation of it taken out; empty when nothing is left.""" 

71 if _given([word], "--yes"): 

72 return "" 

73 if word.startswith("-") and not word.startswith("--"): 

74 word = ("-" + word[1:].replace("y", "")).rstrip("-") if "y" in word else word 

75 return word if len(word) > 1 else "" 

76 return word

79def _given(words: list[str], long: str, short: str | None = None) -> bool: 

80 """Whether `words` pass the flag `long` (or `short`). argparse takes any 

81 unambiguous prefix of a long flag (`--ye` is `--yes`) and `--flag=value`; 

82 the checks below must see what the CLI will.""" 

83 for word in words: 

84 flag = word.partition("=")[0] 

85 if word == short or flag == long: 

86 return True 

87 # Bundled short flags (`-fn5`, `-yy`): over-matches, which only asks or refuses more. 

88 if short and word.startswith("-") and not word.startswith("--") and short[1] in word: 

89 return True 

90 if flag.startswith("--") and len(flag) > 2 and long.startswith(flag): 

91 return True 

92 return False

95def route(line: str) -> tuple[str, str | list[str]]: 

96 """What a typed line does: `("run", argv)` for `kraft <argv>`, `("show", 

97 view)` to open a view instead, or `("say", sentence)` to refuse it. A 

98 leading `/` and a leading `kraft` are dropped, so a sentence the CLI 

99 printed ("kraft item resume a91c") can be pasted back.""" 

100 try: 

101 words = shlex.split(line.strip().removeprefix("/")) 

102 except ValueError as exc: 

103 return "say", f"That line does not parse: {exc}." 

104 if words[:1] == ["kraft"]: 

105 words = words[1:] 

106 head = tuple(words[:2]) 

107 if words in (["addon"], ["admin", "addon"]): 

108 return "show", "addons" 

109 if len(words) == 1 and words[0] in GROUPS: 

110 return "show", GROUPS[words[0]] 

111 if not words or (head == ("admin", "start") and not _given(words, "--detach", "-d")): 

112 return "say", ATTACHED 

113 for prefix, why in REFUSED.items(): 

114 if tuple(words[: len(prefix)]) == prefix: 

115 return "say", why 

116 if head == ("view", "watch"): 

117 return "show", "board" 

118 if head in FOLLOWED and _given(words, "--follow", "-f"): 

119 return "say", FOLLOW_LATER 

120 if head == ("admin", "update"): 

121 # Never `-y`: its own refusals (add-ons it would drop, a 0.x config) still show. 

122 words = [w for w in map(_without_yes, words) if w] 

123 return "run", words

126def ask(argv: list[str]) -> tuple[str, str | None, str] | None: 

127 """What the console asks before running `kraft <argv>`: the sentence, the 

128 word for a stop that ends running agents (None: lists no items), and the 

129 button that goes on. None when it runs without asking.""" 

130 found = next((v for k, v in CONFIRMS.items() if tuple(argv[: len(k)]) == k), None) 

131 if found is None: 

132 return None 

133 sentence, doing, verb = found 

134 if argv[:2] in (["item", "archive"], ["item", "abandon"]) and not _given(argv, "--yes"): 

135 return None # without --yes the CLI only previews; nothing is lost 

136 if argv[:2] == ["admin", "update"] and _given(argv, "--restart"): 

137 doing = "restarting" 

138 return sentence, doing, verb

141class Run: 

142 """One `kraft <argv>` child: its own session, so a stop reaches what it 

143 started (git, a test command) too; no stdin, so nothing waits on a 

144 question; output streamed to `on_line` one line at a time.""" 

145 

146 def __init__(self, argv: list[str], on_line: Callable[[str], None]) -> None: 

147 self.argv = argv 

148 self.on_line = on_line 

149 self.process: asyncio.subprocess.Process | None = None 

150 #: The exit code once it has ended; negative when a signal ended it. 

151 self.code: int | None = None 

152 self._stopped = False 

153 

154 async def finish(self) -> int: 

155 """Start it, pass on its output as it comes, and return its exit code.""" 

156 self.process = await asyncio.create_subprocess_exec( 

157 *COMMAND, 

158 *self.argv, 

159 stdin=asyncio.subprocess.DEVNULL, 

160 stdout=asyncio.subprocess.PIPE, 

161 stderr=asyncio.subprocess.STDOUT, 

162 start_new_session=True, 

163 # Unbuffered, or a Python child's output arrives only when it ends; 

164 # and git must not ask for a password on the console's terminal. 

165 env={**os.environ, "PYTHONUNBUFFERED": "1", "GIT_TERMINAL_PROMPT": "0"}, 

166 ) 

167 assert self.process.stdout is not None 

168 if self._stopped: # asked to stop while it was starting 

169 self.stop() 

170 # Chunks, not readline(): one `--json` line can outgrow its 64 KiB limit. 

171 decoder = codecs.getincrementaldecoder("utf-8")(errors="replace") 

172 pending = "" 

173 while chunk := await self.process.stdout.read(65536): 

174 *lines, pending = (pending + decoder.decode(chunk)).split("\n") 

175 for line in lines: 

176 self.on_line(line) 

177 pending += decoder.decode(b"", final=True) 

178 if pending: 

179 self.on_line(pending) 

180 self.code = await self.process.wait() 

181 return self.code 

182 

183 def stop(self) -> None: 

184 """SIGINT to its process group: the CLI's own clean-up runs (`repo 

185 connect --verify` removes its worktree only on an interrupt). Until 

186 the run has finished, not until the leader has exited: what it started 

187 can outlive it and still hold the output open.""" 

188 self._stopped = True 

189 self._signal(signal.SIGINT) 

190 

191 def _signal(self, sig: int) -> None: 

192 if self.process is not None and self.code is None: 

193 # PermissionError: macOS, for a group whose only member is a zombie. 

194 with contextlib.suppress(ProcessLookupError, PermissionError): 

195 os.killpg(self.process.pid, sig) 

196 

197 def _alive(self) -> bool: 

198 """Whether any member of its group is still there.""" 

199 if self.process is None or self.code is not None: 

200 return False 

201 try: 

202 os.killpg(self.process.pid, 0) 

203 except (ProcessLookupError, PermissionError): 

204 return False 

205 return True 

206 

207 async def end(self) -> None: 

208 """Stop it, and kill its group if it has not stopped in `QUIT_GRACE_S`, 

209 even when the leader is gone and a child of it is not. It polls the 

210 group rather than awaiting `wait()`: on Python 3.12 (before gh-119710) 

211 that also waits for the output pipe to close, which a quitting console 

212 no longer reads.""" 

213 self.stop() 

214 for _ in range(round(QUIT_GRACE_S / 0.05)): 

215 if not self._alive(): 

216 return 

217 await asyncio.sleep(0.05) 

218 self._signal(signal.SIGKILL)
